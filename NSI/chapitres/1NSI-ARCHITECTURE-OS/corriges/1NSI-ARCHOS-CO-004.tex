\begin{corrige}{1NSI-ARCHOS-EX-004}
\begin{python}
import subprocess

subprocess.run(["mkdir", "projet"])
subprocess.run(["touch", "projet/main.py"])
subprocess.run(["touch", "projet/readme.md"])

resultat = subprocess.run(["ls", "projet"], capture_output=True, text=True)
print(resultat.stdout)  # main.py \n readme.md

subprocess.run(["rm", "projet/readme.md"])
resultat2 = subprocess.run(["ls", "projet"], capture_output=True, text=True)
print(resultat2.stdout)  # main.py
\end{python}
Après suppression, seul \lstinline{main.py} reste dans le dossier \lstinline{projet}.
\end{corrige}
