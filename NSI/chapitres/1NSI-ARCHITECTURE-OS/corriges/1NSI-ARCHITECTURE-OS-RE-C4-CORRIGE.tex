\begin{corrige}{1NSI-ARCHITECTURE-OS-RE-C4}
\begin{enumerate}
  \item \lstinline{pwd} affiche \lstinline{/home/eleve}.
        \lstinline{ls} affiche le contenu, dont \lstinline{cours}.
        \lstinline{cd cours} \textbf{n'affiche rien} : elle déplace en silence.
        \lstinline{pwd} affiche alors \lstinline{/home/eleve/cours} — c'est elle qui rend le
        déplacement visible.
  \item \lstinline{mkdir cours/tp}, puis \lstinline{cd cours/tp}, puis \lstinline{ls}, qui
        n'affiche rien puisque le répertoire est vide.
  \item Rien de ce qu'il attend. \lstinline{mkdir cours} tente de \emph{créer} un répertoire
        déjà existant : la commande échoue avec un message d'erreur, et n'affiche aucun
        contenu. Pour lire, il fallait \lstinline{ls cours}.
\end{enumerate}
\end{corrige}
