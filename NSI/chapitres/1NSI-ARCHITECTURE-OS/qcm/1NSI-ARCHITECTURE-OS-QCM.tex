\section*{QCM d'auto-évaluation --- Architecture de von Neumann et systèmes d'exploitation}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte. Entourez la lettre correspondante.}
\end{center}

\bigskip

\subsection*{C1 --- Architecture de von Neumann}

\begin{enumerate}

\item \textbf{[Q1]} Dans l'architecture de von Neumann, instructions et données sont stockées :
  \begin{enumerate}[label=\Alph*.]
    \item dans deux mémoires physiquement séparées.
    \item dans le même espace mémoire.
    \item uniquement dans les registres du processeur.
    \item uniquement sur le disque dur.
  \end{enumerate}

\bigskip

\subsection*{C2 --- Exécution d'instructions}

\item \textbf{[Q2]} Les instructions d'un programme machine s'exécutent :
  \begin{enumerate}[label=\Alph*.]
    \item dans un ordre aléatoire.
    \item toutes en même temps.
    \item une par une, dans l'ordre du programme (sauf branchement).
    \item dans l'ordre inverse du programme.
  \end{enumerate}

\bigskip

\subsection*{C3 --- Systèmes d'exploitation}

\item \textbf{[Q3]} La gestion des processus par le système d'exploitation permet notamment de :
  \begin{enumerate}[label=\Alph*.]
    \item donner l'illusion d'une exécution simultanée de plusieurs programmes sur un seul cœur.
    \item empêcher tout programme de s'exécuter.
    \item remplacer le processeur.
    \item afficher uniquement des fenêtres graphiques.
  \end{enumerate}

\bigskip

\subsection*{C4 --- Ligne de commande}

\item \textbf{[Q4]} La commande qui liste le contenu d'un dossier est :
  \begin{enumerate}[label=\Alph*.]
    \item \lstinline{cd}
    \item \lstinline{ls}
    \item \lstinline{mkdir}
    \item \lstinline{rm}
  \end{enumerate}

\bigskip

\subsection*{C5 --- Droits et permissions}

\item \textbf{[Q5]} Dans la chaîne de permissions \lstinline{rwxr-x---}, le \textbf{groupe} peut :
  \begin{enumerate}[label=\Alph*.]
    \item lire et écrire, mais pas exécuter.
    \item lire et exécuter, mais pas écrire.
    \item tout faire (lecture, écriture, exécution).
    \item rien faire du tout.
  \end{enumerate}

\bigskip

\item \textbf{[Q6]} La notation octale \lstinline{644} correspond aux permissions :
  \begin{enumerate}[label=\Alph*.]
    \item \lstinline{rw-r--r--}
    \item \lstinline{rwxrwxrwx}
    \item \lstinline{r--r--r--}
    \item \lstinline{rw-rw-rw-}
  \end{enumerate}

\end{enumerate}
