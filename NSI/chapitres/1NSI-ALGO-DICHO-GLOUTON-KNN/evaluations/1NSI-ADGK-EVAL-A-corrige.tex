\begin{corrige}{1NSI-ADGK-EVAL-A-EX1}
\textbf{1.} Itération 1 : \lstinline{inf=0}, \lstinline{sup=7}, \lstinline{milieu=3}
(\lstinline{t[3]=21<27}). Itération 2 : \lstinline{inf=4}, \lstinline{sup=7},
\lstinline{milieu=5} (\lstinline{t[5]=33>27}). Itération 3 : \lstinline{inf=4},
\lstinline{sup=4}, \lstinline{milieu=4} (\lstinline{t[4]=27}, trouvé, indice $4$).

\textbf{2.} $V$ vaut successivement $7-0=7$, $7-4=3$, $4-4=0$ : la suite $7,3,0$ est
strictement décroissante.

\textbf{3.} $V$ est un entier positif ou nul qui décroît strictement : une telle suite est
nécessairement finie (elle ne peut pas décroître indéfiniment en restant positive), donc
la boucle termine forcément après un nombre fini d'itérations.
\end{corrige}

\begin{corrige}{1NSI-ADGK-EVAL-A-EX2}
\textbf{1.} Pour $47$ : $20$ (reste $27$), $20$ (reste $7$), $5$ (reste $2$), $2$ (reste
$0$). Rendu : \lstinline{[20, 20, 5, 2]}.

\textbf{2.} \lstinline{k_plus_proches_voisins(animaux, (5.5, 31), 3)} renvoie
\lstinline{"Chat"} : cet animal est proche en masse et en taille du groupe des chats.

\textbf{3.} Non, la méthode gloutonne n'est pas optimale pour \textbf{tout} système de
pièces (seulement pour des systèmes ayant certaines propriétés, comme le système en euros) :
avec $\{4,3,1\}$ par exemple, rendre $6$ de façon gloutonne utilise $3$ pièces
($4+1+1$) alors que $3+3$ n'en utilise que $2$.
\end{corrige}
