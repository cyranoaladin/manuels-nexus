\subsection*{Rappel -- Derivation locale : nombre derive et tangente}

\textbf{Nombre derive :} $f'(a) = \lim_{h \to 0} \dfrac{f(a+h) - f(a)}{h}$.

\textbf{Tangente :} equation $y = f(a) + f'(a)(x - a)$.

\begin{exercice}{1NSI-ADGK-FR-R2-EX1}{1}{8}
Soit $f(x) = x^2 + 3x - 1$.
\begin{enumerate}
  \item Calculer $f'(1)$ par la formule du nombre derive.
  \item Donner l'equation de la tangente a la courbe en $x = 1$.
  \item En quel point la tangente est-elle horizontale ?
\end{enumerate}
\end{exercice}
