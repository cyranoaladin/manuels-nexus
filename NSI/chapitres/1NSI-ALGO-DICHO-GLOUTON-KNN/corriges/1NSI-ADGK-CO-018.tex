\begin{corrige}{1NSI-ADGK-EX-018}

\textbf{Question 1.} La base de la boite est un carre de cote $(20 - 2x)$ et la hauteur est $x$.

\commentaireMarge{Base $(20-2x)^2$, hauteur $x$, donc $V = x(20-2x)^2$.}

\[ V(x) = x(20 - 2x)^2 \quad \text{pour } x \in [0, 10]. \]

\textbf{Question 2.} $V'(x) = (20-2x)^2 + x \times 2(20-2x)(-2) = (20-2x)\bigl[(20-2x) - 4x\bigr] = (20-2x)(20-6x)$.

$V'(x) = 0 \iff x = 10$ ou $x = \dfrac{10}{3}$ (dans $[0,10]$).

Le maximum est atteint en $x = \dfrac{10}{3}$ cm et vaut :
\[ V\!\left(\dfrac{10}{3}\right) = \dfrac{10}{3} \times \left(\dfrac{40}{3}\right)^2 = \dfrac{16000}{27} \approx 593 \text{ cm}^3. \]

\end{corrige}
