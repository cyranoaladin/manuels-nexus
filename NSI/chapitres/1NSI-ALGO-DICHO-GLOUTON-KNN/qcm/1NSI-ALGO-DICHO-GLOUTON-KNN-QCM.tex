\subsection*{C1 --- Dichotomie}

\begin{enumerate}

\item \textbf{[Q1]} La recherche dichotomique nécessite que le tableau soit :
  \begin{enumerate}[label=\Alph*.]
    \item de taille paire.
    \item trié.
    \item composé uniquement d'entiers.
    \item de taille inférieure à $100$.
  \end{enumerate}

\bigskip

\item \textbf{[Q2]} Le variant de boucle $V=\text{borne\_sup}-\text{borne\_inf}$ de la recherche dichotomique sert à prouver :
  \begin{enumerate}[label=\Alph*.]
    \item que l'algorithme trouve toujours la valeur cherchée.
    \item que la boucle termine en un nombre fini d'itérations.
    \item que le tableau est trié.
    \item que le coût est linéaire.
  \end{enumerate}

\bigskip

\subsection*{C2 --- Algorithmes gloutons}

\item \textbf{[Q3]} Un algorithme glouton fait, à chaque étape, le choix :
  \begin{enumerate}[label=\Alph*.]
    \item globalement optimal, garanti par calcul exhaustif.
    \item localement optimal, sans revenir en arrière.
    \item aléatoire.
    \item identique au choix précédent.
  \end{enumerate}

\bigskip

\item \textbf{[Q4]} Avec le système de pièces $\{4, 3, 1\}$, rendre $6$ de façon gloutonne donne :
  \begin{enumerate}[label=\Alph*.]
    \item la solution optimale à $2$ pièces.
    \item une solution à $3$ pièces, non optimale.
    \item une erreur, car $6$ n'est pas atteignable.
    \item une solution à $6$ pièces.
  \end{enumerate}

\bigskip

\subsection*{C3 --- k plus proches voisins}

\item \textbf{[Q5]} L'algorithme des k plus proches voisins prédit la classe d'un nouvel élément à partir de :
  \begin{enumerate}[label=\Alph*.]
    \item la classe majoritaire de ses $k$ plus proches voisins déjà classés.
    \item la classe du point le plus éloigné.
    \item une valeur aléatoire.
    \item la moyenne de toutes les classes existantes.
  \end{enumerate}

\bigskip

\item \textbf{[Q6]} Choisir un $k$ égal à la taille totale du jeu de données (tous les points) :
  \begin{enumerate}[label=\Alph*.]
    \item est toujours la meilleure stratégie.
    \item fait perdre le sens de la méthode, basée sur la proximité locale.
    \item accélère toujours le calcul.
    \item est obligatoire pour que l'algorithme fonctionne.
  \end{enumerate}

\end{enumerate}
