\begin{exercice}{1NSI-ADGK-EX-001}{1}{12}

\lstinline{[2, 5, 8, 12, 16, 23, 38, 45, 56, 72]} (indices $0$ à $9$), pour rechercher la
valeur $23$.
\begin{enumerate}
  \item Pour chaque itération de la boucle \lstinline{while}, donner les valeurs de \lstinline{borne_inf}, \lstinline{borne_sup}, \lstinline{milieu} et \lstinline{tableau_trie[milieu]}.

\end{enumerate}
