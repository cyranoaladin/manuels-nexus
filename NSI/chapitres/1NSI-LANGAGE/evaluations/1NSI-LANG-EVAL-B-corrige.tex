\begin{corrige}{1NSI-LANG-EVAL-B-EX1}
\textbf{1.} Précondition : \lstinline{valeurs} et \lstinline{poids} ont la même longueur,
et la somme des poids est strictement positive.

\textbf{2.}
\begin{python}
def moyenne_ponderee(valeurs, poids):
    assert len(valeurs) == len(poids)
    assert sum(poids) > 0
    total = 0
    for i in range(len(valeurs)):
        total = total + valeurs[i] * poids[i]
    return total / sum(poids)
\end{python}

\textbf{3.} $\dfrac{8\times1+12\times1+16\times2}{1+1+2}=\dfrac{8+12+32}{4}=\dfrac{52}{4}=13$.
\end{corrige}

\begin{corrige}{1NSI-LANG-EVAL-B-EX2}
\textbf{1.} \lstinline{produit_negatifs_bugue([-2, -3, -4])} renvoie $0$. Le résultat
attendu est $-24$ ($(-2)\times(-3)\times(-4)$).

\textbf{2.} L'accumulateur \lstinline{prod} est initialisé à $0$ : dès la première
multiplication (\lstinline{prod *= x}), $0\times x=0$ quel que soit $x$, donc
\lstinline{prod} reste $0$ pour toujours. Il faut initialiser un accumulateur de
\textbf{produit} à $1$ (l'élément neutre de la multiplication), pas à $0$ :
\begin{python}
def produit_negatifs(liste):
    prod = 1
    for x in liste:
        if x < 0:
            prod *= x
    return prod
\end{python}

\textbf{3.} \lstinline{import math} puis \lstinline{math.gcd(60, 48)}, qui renvoie $12$.
\end{corrige}

\textbf{Exercice 3} --- \textit{D'un langage à l'autre, et le contrat d'une fonction} \hfill (10 points)

\begin{enumerate}
  \item (3 pts) \textbf{Noyau commun} : l'affectation \lstinline{n = 0}, la boucle bornée, la
        conditionnelle, l'appel de fonction et le renvoi d'une valeur.
        \textbf{Propre à ce langage} : \lstinline{function}, les déclarations \lstinline{let},
        les accolades, les points-virgules et l'opérateur \lstinline{===}.
  \item (3 pts)
        \begin{python}
def compte(t, x):
    n = 0
    for v in t:
        if v == x:
            n = n + 1
    return n
        \end{python}
        Sur \lstinline{[3, 7, 3, 3, 9]} et \lstinline{x = 3}, elle renvoie \lstinline{3}.
  \item (2 pts) \lstinline{compte(t, x)} prend un tableau et une valeur, et renvoie un entier :
        le nombre d'occurrences de \lstinline{x} dans \lstinline{t}. Rien sur le parcours.
  \item (2 pts) Postcondition : « le résultat est un entier compris entre $0$ et
        $\operatorname{len}(t)$, égal au nombre d'éléments de \lstinline{t} égaux à
        \lstinline{x} ». On peut ajouter que \lstinline{t} n'est pas modifié.
        « les éléments doivent être comparables à \lstinline{x} » porte sur ce que l'appelant
        fournit : c'est une \textbf{pré}condition.
\end{enumerate}
