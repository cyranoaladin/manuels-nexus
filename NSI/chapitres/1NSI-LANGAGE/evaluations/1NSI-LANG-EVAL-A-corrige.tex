\begin{corrige}{1NSI-LANG-EVAL-A-EX1}
\textbf{1.} Précondition : \lstinline{valeurs} et \lstinline{poids} ont la même longueur,
et la somme des poids est strictement positive (pour ne pas diviser par zéro).

\textbf{2.}
\begin{python}
def moyenne_ponderee(valeurs, poids):
    assert len(valeurs) == len(poids)
    assert sum(poids) > 0
    total = 0
    for i in range(len(valeurs)):
        total = total + valeurs[i] * poids[i]
    return total / sum(poids)
\end{python}

\textbf{3.} $\dfrac{12\times1+15\times2+8\times1}{1+2+1}=\dfrac{12+30+8}{4}=\dfrac{50}{4}=12{,}5$.
\end{corrige}

\begin{corrige}{1NSI-LANG-EVAL-A-EX2}
\textbf{1.} \lstinline{somme_positifs_bugue([3, -2, 5])} renvoie $3$. Le résultat attendu
est $8$ ($3+5$, les deux éléments strictement positifs).

\textbf{2.} L'erreur est dans \lstinline{range(len(liste) - 1)} : cette boucle parcourt les
indices de $0$ à \lstinline{len(liste) - 2}, donc \textbf{exclut le dernier élément} de la
liste (indice \lstinline{len(liste) - 1}). Version corrigée :
\begin{python}
def somme_positifs(liste):
    total = 0
    for i in range(len(liste)):
        if liste[i] > 0:
            total += liste[i]
    return total
\end{python}

\textbf{3.} \lstinline{import math} puis \lstinline{math.gcd(84, 30)}, qui renvoie $6$.
\end{corrige}

\textbf{Exercice 3} --- \textit{D'un langage à l'autre, et le contrat d'une fonction} \hfill (10 points)

\begin{enumerate}
  \item (3 pts) \textbf{Noyau commun} : l'affectation \lstinline{m = 0}, la boucle bornée sur
        les indices, la conditionnelle, l'appel de fonction et le renvoi d'une valeur.
        \textbf{Propre à ce langage} : le mot-clé \lstinline{function}, les déclarations
        \lstinline{let}, les accolades et les points-virgules.
  \item (3 pts)
        \begin{python}
def indice_max(t):
    m = 0
    for i in range(len(t)):
        if t[i] > t[m]:
            m = i
    return m
        \end{python}
        Sur \lstinline{[4, 11, 2, 11]} elle renvoie \lstinline{1} : en cas d'égalité, c'est le
        premier indice atteint qui est conservé, car la comparaison est stricte.
  \item (2 pts) \lstinline{indice_max(t)} prend un tableau de nombres et renvoie un entier :
        l'indice d'un plus grand élément. Rien sur la boucle ni sur le parcours.
  \item (2 pts) Postcondition : « le résultat est un indice valide, et aucun élément du tableau
        n'est strictement supérieur à \lstinline{t[resultat]} ». On peut ajouter que
        \lstinline{t} n'est pas modifié.
        « le tableau doit être non vide » contraint ce que l'appelant fournit : c'est une
        \textbf{pré}condition, à sa charge, et non une garantie de la fonction.
\end{enumerate}
