\begin{corrige}{1NSI-LANG-EX-005}
\textbf{1.} Le module de la bibliothèque standard est \lstinline{statistics}. La fonction
qui calcule une moyenne est \lstinline{statistics.mean}.

\textbf{2.}
\begin{python}
import statistics

notes = [10, 12, 14, 20]
moyenne = statistics.mean(notes)     # 14
mediane = statistics.median(notes)   # 13.0
\end{python}

\textbf{3.} La moyenne obtenue est $14$, qui est bien comprise entre $0$ et $20$ : la
postcondition est respectée.
