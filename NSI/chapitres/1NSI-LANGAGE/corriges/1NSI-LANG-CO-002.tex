\begin{corrige}{1NSI-LANG-EX-002}
\textbf{1.}
\begin{python}
def factorielle(n):
    resultat = 1
    for i in range(1, n + 1):
        resultat = resultat * i
    return resultat
\end{python}

\textbf{2.} \textbf{Traits communs} : une fonction nommée \lstinline{factorielle} prenant
un paramètre \lstinline{n} et renvoyant un résultat ; une boucle bornée qui parcourt les
entiers de $1$ à $n$ en accumulant un produit. \textbf{Traits particuliers} au langage de
style C : le type de chaque variable est déclaré explicitement (\lstinline{int}) ; les
blocs d'instructions sont délimités par des accolades \lstinline|{}| (et non par
l'indentation comme en Python) ; chaque instruction se termine par un point-virgule.

\textbf{3.} $5!=1\times2\times3\times4\times5=120$. Par convention, $0!=1$ (produit vide,
la boucle \lstinline{for i in range(1, 1)} ne s'exécute jamais et \lstinline{resultat}
reste égal à $1$).
\end{corrige}
