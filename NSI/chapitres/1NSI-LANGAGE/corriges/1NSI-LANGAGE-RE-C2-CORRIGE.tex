\begin{corrige}{1NSI-LANGAGE-RE-C2}
\begin{enumerate}
  \item \textbf{Noyau commun} : la définition d'une fonction avec un paramètre, la
        conditionnelle, le calcul du reste, le renvoi d'une valeur booléenne.
        \textbf{Propre à ce langage} : le mot-clé \lstinline{function}, les accolades, les
        points-virgules, l'opérateur \lstinline{===}, et l'écriture \lstinline{true}
        en minuscules.
  \item En Python :
        \begin{python}
def pair(n):
    return n % 2 == 0
        \end{python}
  \item L'écriture ne détermine pas la puissance du langage. Les deux disposent des mêmes
        constructions élémentaires et permettent d'exprimer les mêmes algorithmes. Ce qui
        diffère, ce sont la syntaxe, le typage et les bibliothèques — pas ce qu'il est
        possible de calculer.
\end{enumerate}
\end{corrige}
