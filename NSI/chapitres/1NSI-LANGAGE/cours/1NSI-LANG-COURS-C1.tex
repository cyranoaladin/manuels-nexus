\section{Constructions élémentaires}

\margeAppui{Malgré leur grande diversité, presque tous les langages de programmation
partagent le même petit noyau de constructions de base.}

\definition[D1]{
  Les \textbf{constructions élémentaires} d'un langage de programmation impératif sont :
  la \textbf{séquence} (exécuter des instructions les unes après les autres), l'\textbf{affectation}
  (donner une valeur à une variable), la \textbf{conditionnelle} (\lstinline{if}/\lstinline{else}),
  les \textbf{boucles} (\textbf{bornées}, \lstinline{for}, dont le nombre de répétitions est
  connu à l'avance ; \textbf{non bornées}, \lstinline{while}, qui se répètent tant qu'une
  condition est vraie), et l'\textbf{appel de fonction}.
}

\exemple{La fonction \lstinline{somme_chiffres} illustre ces cinq constructions :}

\begin{codereference}{Somme des chiffres d'un entier (toutes les constructions de base)}
\begin{python}
def somme_chiffres(n):
    """Renvoie la somme des chiffres de l'entier positif n."""
    total = 0                  # affectation
    while n > 0:                # boucle NON bornee (while)
        chiffre = n % 10
        total = total + chiffre # sequence d'instructions
        n = n // 10
    return total

for valeur in range(3):         # boucle bornee (for)
    print(valeur, somme_chiffres(valeur * 100 + 34))  # appel de fonction
\end{python}
\end{codereference}

\propriete[Conditionnelle]{
  L'instruction \lstinline{if ... elif ... else ...} exécute un bloc d'instructions
  différent selon qu'une (ou plusieurs) condition(s) booléenne(s) est/sont vraie(s).
}

\begin{python}
def est_pair(n):
    if n % 2 == 0:
        return True
    else:
        return False
\end{python}

\erreurFrequente{Confondre boucle \textbf{bornée} (\lstinline{for}, nombre d'itérations
connu à l'avance, par exemple \lstinline{for i in range(10)}) et boucle \textbf{non bornée}
(\lstinline{while}, le nombre d'itérations dépend de l'évolution d'une condition et n'est
pas toujours connu à l'avance). Utiliser une boucle \lstinline{while} sans faire évoluer la
condition qu'elle teste crée une \textbf{boucle infinie}.}
