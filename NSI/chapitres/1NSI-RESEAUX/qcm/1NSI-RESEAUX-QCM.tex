\section*{QCM d'auto-évaluation --- Réseaux : transmission de données}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte. Entourez la lettre correspondante.}
\end{center}

\bigskip

\subsection*{C1 --- Paquets et encapsulation}

\begin{enumerate}

\item \textbf{[Q1]} Dans l'en-tête d'un paquet, le numéro de paquet sert principalement à :
  \begin{enumerate}[label=\Alph*.]
    \item chiffrer les données transmises.
    \item réordonner les paquets à l'arrivée, quel que soit leur ordre de réception.
    \item accélérer la transmission.
    \item indiquer la taille du disque du destinataire.
  \end{enumerate}

\bigskip

\subsection*{C2 --- Protocole du bit alterné}

\item \textbf{[Q2]} Si l'émetteur ne reçoit pas l'accusé de réception attendu, il :
  \begin{enumerate}[label=\Alph*.]
    \item abandonne définitivement l'envoi du message.
    \item retransmet le paquet avec le même bit.
    \item retransmet le paquet avec le bit opposé.
    \item envoie directement le paquet suivant.
  \end{enumerate}

\bigskip

\subsection*{C3 --- Simulation de réseau}

\item \textbf{[Q3]} Pour éviter qu'une recherche de chemin dans un réseau contenant un cycle ne boucle indéfiniment, il faut :
  \begin{enumerate}[label=\Alph*.]
    \item interdire tout cycle dans le réseau.
    \item marquer les sommets déjà visités.
    \item limiter le réseau à deux machines.
    \item ne jamais utiliser de fonction récursive.
  \end{enumerate}

\bigskip

\subsection*{C4 --- Capteurs et actionneurs}

\item \textbf{[Q4]} Un capteur de température et un relais de chauffage jouent respectivement les rôles de :
  \begin{enumerate}[label=\Alph*.]
    \item actionneur puis capteur.
    \item capteur puis actionneur.
    \item deux capteurs.
    \item deux actionneurs.
  \end{enumerate}

\bigskip

\subsection*{C5 --- IHM programmée}

\item \textbf{[Q5]} Programmer une IHM à partir d'un cahier des charges nécessite en particulier de :
  \begin{enumerate}[label=\Alph*.]
    \item tester uniquement le cas de démonstration prévu.
    \item tester aussi les cas limites (valeurs au seuil, changements d'état).
    \item éviter tout test, le code étant supposé correct.
    \item ignorer les exigences non précisées explicitement.
  \end{enumerate}

\bigskip

\item \textbf{[Q6]} Le principe d'hystérésis (deux seuils différents pour activer/désactiver un actionneur) sert à :
  \begin{enumerate}[label=\Alph*.]
    \item rendre le système plus lent.
    \item éviter des activations/désactivations trop fréquentes autour d'un seul seuil.
    \item économiser de la mémoire.
    \item remplacer le capteur par un actionneur.
  \end{enumerate}

\end{enumerate}
