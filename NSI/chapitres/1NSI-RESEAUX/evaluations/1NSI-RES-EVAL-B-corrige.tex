\begin{corrige}{1NSI-RES-EVAL-B-EX1}
\textbf{1.} $3$ paquets sont créés : \lstinline{"PAQUE"} (numéro 0), \lstinline{"TSRES"}
(numéro 1), \lstinline{"EAU"} (numéro 2).

\textbf{2.} Tentative $1$ : envoi de \lstinline{"M"} avec bit $0$ — réussit. Tentative
$2$ : envoi de \lstinline{"N"} avec bit $1$ — échoue. Tentative $3$ : retransmission de
\lstinline{"N"} avec bit $1$ — réussit. Tentative $4$ : envoi de \lstinline{"O"} avec bit
$0$ — réussit. Messages reçus : \lstinline{[(0, "M"), (1, "N"), (0, "O")]}.

\textbf{3.} Les paquets sont envoyés \textbf{un par un} : l'émetteur attend toujours l'ACK
avant d'envoyer le suivant. Un seul bit suffit donc à distinguer \og un nouveau paquet \fg{}
(bit différent du précédent) d'\og une retransmission \fg{} (même bit que le paquet
précédent, non encore confirmé).
\end{corrige}

\begin{corrige}{1NSI-RES-EVAL-B-EX3}
\textbf{1.} L'etat retient le mode courant ; la decision consulte d'abord ce mode, et ne
revient au capteur que si aucun forcage n'est actif.

\begin{python}
COMMANDES = {"AUTO": None, "ON": True, "OFF": False}
etat = {"override_manuel": None}


def traiter_commande(commande):
    commande = commande.strip().upper()
    if commande not in COMMANDES:
        raise ValueError("commande attendue : AUTO, ON ou OFF")
    etat["override_manuel"] = COMMANDES[commande]


def ventilation_active(co2, seuil=800):
    if etat["override_manuel"] is not None:
        return etat["override_manuel"]
    return co2 > seuil
\end{python}

\textbf{2.} Trois cas de test suffisent a couvrir les trois modes :
\begin{itemize}
  \item apres \lstinline{traiter_commande("AUTO")}, \lstinline{ventilation_active(1000)}
  vaut \lstinline{True} et \lstinline{ventilation_active(500)} vaut \lstinline{False} :
  la decision revient bien au capteur, de part et d'autre du seuil ;
  \item apres \lstinline{traiter_commande("ON")}, \lstinline{ventilation_active(500)} vaut
  \lstinline{True} : le forcage l'emporte sur un taux bas ;
  \item apres \lstinline{traiter_commande("OFF")}, \lstinline{ventilation_active(1000)}
  vaut \lstinline{False} : l'interdiction l'emporte sur un taux eleve.
\end{itemize}
On y ajoutera utilement le refus de \lstinline{traiter_commande("MAX")}, qui verifie la
derniere ligne du cahier des charges.
\end{corrige}
