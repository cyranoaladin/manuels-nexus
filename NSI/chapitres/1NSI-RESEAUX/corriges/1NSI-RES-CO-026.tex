\begin{corrige}{1NSI-RES-EX-002}
\textbf{1.} Tentative $1$ : envoi de \lstinline{"X"} avec bit $0$ — \textbf{échoue}
(perdue). Tentative $2$ : retransmission de \lstinline{"X"} avec bit $0$ — \textbf{réussit}.
Tentative $3$ : envoi de \lstinline{"Y"} avec bit $1$ — \textbf{réussit}. Tentative $4$ :
envoi de \lstinline{"Z"} avec bit $0$ — \textbf{échoue} (perdue). Tentative $5$ :
retransmission de \lstinline{"Z"} avec bit $0$ — \textbf{réussit}.

\textbf{2.} Le destinataire reçoit finalement : \lstinline{[(0, "X"), (1, "Y"), (0, "Z")]}.

\textbf{3.} $5$ tentatives ont été nécessaires pour $3$ messages : ce nombre est
\textbf{supérieur} au nombre de messages, car $2$ tentatives ont échoué et ont dû être
retransmises (une pour \lstinline{"X"}, une pour \lstinline{"Z"}).
\end{corrige}
