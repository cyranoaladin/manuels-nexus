\begin{corrige}{1NSI-RES-EX-005}
\textbf{1.}
\begin{python}
etat = {"pompe": False}

def irrigation_hysteresis(humidite, seuil_bas=30, seuil_haut=60):
    if etat["pompe"]:
        if humidite > seuil_haut:
            etat["pompe"] = False
    else:
        if humidite < seuil_bas:
            etat["pompe"] = True
    return etat["pompe"]
\end{python}

\textbf{2.} La séquence des états est \lstinline{[True, True, True, False, False]}.

\textbf{3.} Oui, la pompe est active à la mesure \lstinline{35}. Elle a été
\textbf{déclenchée} à la mesure précédente ($20\,\%<30\,\%$), et le cahier des charges
précise qu'elle ne s'\textbf{arrête} que lorsque l'humidité dépasse $60\,\%$ : à
$35\,\%$, ce seuil haut n'est pas encore atteint, donc la pompe continue de fonctionner
même si l'humidité n'est plus sous le seuil bas.
\end{corrige}
