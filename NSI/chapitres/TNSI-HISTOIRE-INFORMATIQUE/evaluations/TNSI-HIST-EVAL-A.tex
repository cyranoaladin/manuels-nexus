\section*{Évaluation A --- Histoire de l'informatique}
\textit{Durée : 30 minutes. Documents interdits.}

\baremeIndicatif{Ex. 1 : 10 pts (repères chronologiques) ; Ex. 2 : 10 pts (évolution logiciel/matériel)}

\bigskip

\textbf{Exercice 1} --- \textit{Repères chronologiques} \hfill (10 points)

\begin{enumerate}
  \item (4 pts) Associer chacun des événements suivants à son année :
    présentation de la machine universelle par Turing ; exécution du premier
    programme du Manchester Baby ; mise en service d'ARPANET.
  \item (3 pts) Expliquer en une phrase pourquoi 1936 est une date charnière
    dans les repères du programme d'informatique.
  \item (3 pts) Citer un exemple de diversification des formes d'ordinateurs
    depuis les années 1980.
\end{enumerate}

\bigskip

\textbf{Exercice 2} --- \textit{Évolution logiciel/matériel} \hfill (10 points)

\begin{enumerate}
  \item (5 pts) À partir d'un exemple ancien et d'un exemple récent du cours,
    expliquer ce qui est commun et ce qui a évolué dans les relations entre
    programmes et matériel.
  \item (5 pts) Expliquer le rôle d'un système d'exploitation dans cette
    évolution et citer une contrainte matérielle ou de compatibilité qui
    demeure.
\end{enumerate}
