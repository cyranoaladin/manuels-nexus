\section*{Évaluation A --- Histoire de l'informatique}
\textit{Durée : 30 minutes. Documents interdits.}

\baremeIndicatif{Ex. 1 : 10 pts (repères chronologiques) ; Ex. 2 : 10 pts (évolution logiciel/matériel)}

\bigskip

\textbf{Exercice 1} --- \textit{Repères chronologiques} \hfill (10 points)

\begin{enumerate}
  \item (4 pts) Associer chacun des événements suivants a son année : machine universelle de Turing ; premiers ordinateurs a programme enregistre ; mise en service d'ARPANET.
  \item (3 pts) Expliquer en une phrase pourquoi 1936 est consideree comme une date charniere dans l'histoire de l'informatique.
  \item (3 pts) Citer un exemple de diversification des formes d'ordinateurs depuis les années 1980.
\end{enumerate}

\bigskip

\textbf{Exercice 2} --- \textit{Évolution logiciel/matériel} \hfill (10 points)

\begin{enumerate}
  \item (5 pts) Expliquer, avec un exemple, comment le role du matériel a evolue depuis les debuts de l'informatique.
  \item (5 pts) Expliquer le role d'un système d'exploitation dans cette evolution.
\end{enumerate}
