\begin{corrige}{TNSI-LANG-EX-002}
\begin{python}
import math

print(math.hypot(3, 4))  # 5.0
print(math.gcd(48, 18))  # 6

try:
    math.sqrt("4")
except TypeError as erreur:
    print(erreur)  # "must be real number, not str"
\end{python}
\lstinline{math.sqrt} attend un nombre (entier ou flottant), pas une chaîne de caractères : appeler \lstinline{math.sqrt("4")} lève une \lstinline{TypeError}, même si \lstinline{"4"} \og ressemble \fg{} à un nombre. Il faudrait d'abord convertir explicitement avec \lstinline{math.sqrt(int("4"))} ou \lstinline{math.sqrt(float("4"))}.
\end{corrige}
