\subsection*{Rappel -- Derivation : derivees de reference et regles}

\textbf{Derivees usuelles :} $(x^n)' = n x^{n-1}$, $(\mathrm{e}^x)' = \mathrm{e}^x$, $(\ln x)' = \dfrac{1}{x}$, $(\sqrt{x})' = \dfrac{1}{2\sqrt{x}}$.

\textbf{Regles :} $(u+v)' = u' + v'$, $(uv)' = u'v + uv'$, $\left(\dfrac{u}{v}\right)' = \dfrac{u'v - uv'}{v^2}$.

\begin{exercice}{1NSI-ADGK-FR-R1-EX1}{1}{8}
Calculer les derivees des fonctions suivantes :
\begin{enumerate}
  \item $f(x) = x^4 - 3x^2 + 5$
  \item $g(x) = \dfrac{\mathrm{e}^x}{x}$ sur $\mathbb{R}^*$
  \item $h(x) = x \ln x$ sur $]0, +\infty[$
\end{enumerate}
\end{exercice}
