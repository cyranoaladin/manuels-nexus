\begin{fichemethode}{M5}{Écrire le tri par sélection}
\textbf{Quand l'utiliser ?} Quand tu dois trier un tableau en plaçant directement chaque
élément à sa position définitive : on cherche le plus petit de la zone non triée et on
l'amène en tête de cette zone.

\textbf{Pas à pas :}
\begin{enumerate}
  \item Pour chaque indice $i$ de $0$ à $n-2$, la zone non triée est
    \lstinline{tableau[i:n]}.
  \item Initialise \lstinline{indice_min = i} \textbf{avant} la boucle interne — c'est le
    candidat minimum de la zone.
  \item Parcours $j$ de $i+1$ à $n-1$ : si \lstinline{tableau[j] < tableau[indice_min]},
    alors \lstinline{indice_min = j}.
  \item À la sortie de la boucle interne, \lstinline{indice_min} désigne le minimum de
    toute la zone. Échange-le avec la case $i$.
\end{enumerate}

\textbf{Exemple :}
\begin{python}
def tri_selection(tableau):
    tableau = tableau[:]
    n = len(tableau)
    for i in range(n - 1):
        indice_min = i
        for j in range(i + 1, n):
            if tableau[j] < tableau[indice_min]:
                indice_min = j
        tableau[i], tableau[indice_min] = tableau[indice_min], tableau[i]
    return tableau

tri_selection([9, 1, 5, 3])   # renvoie [1, 3, 5, 9]
\end{python}

\textbf{Pièges :}
\begin{itemize}
  \item Placer \lstinline{indice_min = i} \textbf{à l'intérieur} de la boucle sur $j$ : le
    candidat est réinitialisé à chaque tour, et l'algorithme ne retient plus le minimum
    global de la zone. Le tableau renvoyé n'est alors pas trié.
  \item Faire démarrer la boucle interne à $i$ au lieu de $i+1$ : la comparaison de
    \lstinline{tableau[i]} avec lui-même est inutile, sans être fausse.
  \item Écrire l'échange en trois lignes sans variable temporaire
    (\lstinline{tableau[i] = tableau[indice_min]} puis l'inverse) : la première affectation
    écrase la valeur dont la seconde a besoin.
  \item Comparer \lstinline{tableau[j] < tableau[i]} au lieu de
    \lstinline{tableau[indice_min]} : on compare alors au premier élément de la zone, pas
    au meilleur candidat courant.
\end{itemize}

\textbf{Vérifier :} teste sur un tableau vide, un singleton, un tableau déjà trié, un
tableau trié à l'envers, un tableau avec doublons ; compare à \lstinline{sorted(tableau)}.
