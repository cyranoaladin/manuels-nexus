\begin{corrige}{1NSI-ADGK-EX-014}

\textbf{Question 1.} $f'(x) = 2x\,\mathrm{e}^{-x} - x^2\,\mathrm{e}^{-x} = x(2-x)\,\mathrm{e}^{-x}$.

\commentaireMarge{Factoriser $f'(x)$ par $x\,\mathrm{e}^{-x}$.}

Comme $\mathrm{e}^{-x} > 0$, le signe de $f'$ est celui de $x(2-x)$ :
$f' > 0$ sur $[0,2]$, $f' < 0$ sur $\left]-\infty,0\right[ \cup \left]2,+\infty\right[$.

\textbf{Question 2.} $\lim_{x \to +\infty} x^2\,\mathrm{e}^{-x} = 0$ (croissances compareees) et $\lim_{x \to -\infty} x^2\,\mathrm{e}^{-x} = +\infty$.

\textbf{Question 3.} $f$ admet un minimum local $f(0) = 0$ et un maximum local $f(2) = 4\mathrm{e}^{-2}$.

\end{corrige}
