\begin{exercice}{1NSI-APT-EX-016}{2}{20}

  \item Calculer $f'(x)$ et étudier son signe.
  \item Déterminer les limites aux bornes du domaine de définition.
  \item Dresser le tableau de variations complet.
\end{enumerate}
