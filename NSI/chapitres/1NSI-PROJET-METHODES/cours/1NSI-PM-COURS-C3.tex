\section{Déboguer méthodiquement}

\definition[D3]{
  \textbf{Déboguer} un programme, c'est identifier puis corriger la cause d'un
  comportement incorrect (plantage, résultat faux). Une démarche \textbf{méthodique} vaut
  mieux qu'une correction \og au hasard \fg{}.
}

\exemple{Le programme suivant plante lorsqu'on l'exécute :}

\begin{python}
def moyenne_ponderee_bugue(valeurs, poids):
    total = 0
    for i in range(len(valeurs)):
        total += valeurs[i] * poids[i]
    return total / sum(poids)

moyenne_ponderee_bugue([12, 15, 18], [1, 2])
\end{python}

\propriete[Lire un message d'erreur (traceback)]{
  Un \textbf{traceback} Python indique, de haut en bas, la \textbf{pile des appels} qui a
  mené à l'erreur, et se termine par le \textbf{type} de l'erreur
  (\lstinline{IndexError}) et un \textbf{message} explicatif. La \textbf{dernière ligne de
  code} indiquée (ici, \lstinline{total += valeurs[i] * poids[i]}) est l'endroit précis où
  l'erreur s'est produite — c'est le premier endroit à examiner.
}

\propriete[Démarche méthodique de débogage]{
  \begin{enumerate}
    \item Lire \textbf{entièrement} le message d'erreur : type d'erreur, ligne concernée.
    \item Identifier la \textbf{cause} : ici, \lstinline{poids} a seulement $2$ éléments, mais la boucle tente d'accéder à \lstinline{poids[2]} (car \lstinline{valeurs} en a $3$).
    \item Formuler une \textbf{hypothèse} : les deux listes doivent avoir la même longueur.
    \item \textbf{Vérifier} l'hypothèse, par exemple avec un \lstinline{print(len(valeurs), len(poids))} ou une précondition \lstinline{assert}.
    \item \textbf{Corriger} : ajouter une précondition, ou corriger l'appel fautif.
  \end{enumerate}
}

\begin{codereference}{Corriger avec une précondition explicite}
\begin{python}
def moyenne_ponderee(valeurs, poids):
    assert len(valeurs) == len(poids), "valeurs et poids doivent avoir la meme longueur"
    total = 0
    for i in range(len(valeurs)):
        total += valeurs[i] * poids[i]
    return total / sum(poids)
\end{python}
\end{codereference}

\propriete[Rôle du \lstinline{print} et de \lstinline{assert} en débogage]{
  Insérer temporairement des \lstinline{print(...)} pour afficher l'état de variables clés
  à des points précis du programme est une technique simple et efficace pour localiser un
  bug. Les \lstinline{assert} permettent, eux, de \textbf{documenter} et
  \textbf{vérifier automatiquement} des hypothèses sur le programme, à chaque exécution.
}

\erreurFrequente{Corriger un bug en modifiant du code \textbf{au hasard}, sans avoir
d'abord compris sa cause exacte. Cette approche peut sembler « résoudre » le problème par
chance sur un cas, tout en laissant le bug intact (ou en en créant un nouveau) dans
d'autres cas.}

\alamachine{Provoque volontairement une \lstinline{TypeError} en appelant
\lstinline{moyenne_ponderee(["a", "b"], [1, 2])} (des chaînes au lieu de nombres). Lis le
traceback obtenu, identifie la ligne et le type d'erreur, puis explique en une phrase la
cause du problème.}
