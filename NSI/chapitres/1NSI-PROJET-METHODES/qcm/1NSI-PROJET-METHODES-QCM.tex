\subsection*{C1 --- Histoire de l'informatique}

\begin{enumerate}

\item \textbf{[Q1]} Le concept de machine universelle, capable en principe d'exécuter n'importe quel algorithme, est introduit par :
  \begin{enumerate}[label=\Alph*.]
    \item Blaise Pascal en 1642.
    \item Alan Turing en 1936.
    \item Guido van Rossum en 1991.
    \item Tim Berners-Lee en 1989.
  \end{enumerate}

\bigskip

\subsection*{C2 --- Démarche de projet}

\item \textbf{[Q2]} Le programme officiel réserve à la conception de projets, en classe de Première :
  \begin{enumerate}[label=\Alph*.]
    \item moins d'un dixième de l'horaire annuel.
    \item au moins un quart de l'horaire annuel.
    \item la totalité de l'horaire annuel.
    \item aucune heure dédiée.
  \end{enumerate}

\bigskip

\item \textbf{[Q3]} Découper un projet en jalons permet notamment de :
  \begin{enumerate}[label=\Alph*.]
    \item éviter tout travail en équipe.
    \item suivre la progression et ajuster les objectifs si nécessaire.
    \item garantir qu'aucun bug ne subsistera.
    \item remplacer le cahier des charges.
  \end{enumerate}

\bigskip

\subsection*{C3 --- Débogage}

\item \textbf{[Q4]} Face à un message d'erreur (traceback) Python, la première chose à faire est de :
  \begin{enumerate}[label=\Alph*.]
    \item corriger au hasard une ligne du programme.
    \item lire entièrement le message pour identifier le type d'erreur et la ligne concernée.
    \item ignorer le message et relancer le programme.
    \item supprimer toute la fonction concernée.
  \end{enumerate}

\bigskip

\subsection*{C4 --- Documentation et oral}

\item \textbf{[Q5]} Une bonne docstring de fonction précise notamment :
  \begin{enumerate}[label=\Alph*.]
    \item le nom de l'auteur du code uniquement.
    \item le rôle de la fonction, ses préconditions et ses postconditions.
    \item la date de la dernière modification uniquement.
    \item rien : le code se suffit toujours à lui-même.
  \end{enumerate}

\bigskip

\item \textbf{[Q6]} Une présentation orale efficace d'un projet informatique privilégie :
  \begin{enumerate}[label=\Alph*.]
    \item la lecture du code source ligne par ligne.
    \item le raisonnement : problème traité, choix effectués, difficultés surmontées.
    \item l'énumération de tous les noms de variables utilisées.
    \item la récitation de la documentation officielle de Python.
  \end{enumerate}

\end{enumerate}
