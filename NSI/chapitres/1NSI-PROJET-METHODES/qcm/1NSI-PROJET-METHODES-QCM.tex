\item[] \textbf{Capacité C2}

\item \textbf{[Q2]} Le programme officiel réserve à la conception et à l'élaboration de projets :
  \begin{enumerate}[label=\Alph*.]
    \item moins d'un dixième de l'horaire annuel.

  \end{enumerate}

\item \textbf{[Q3]} Découper un projet en jalons permet notamment :
  \begin{enumerate}[label=\Alph*.]
    \item d'éviter tout travail en équipe.
    \item de garantir qu'aucun bug ne subsistera.
    \item de suivre la progression et d'ajuster les objectifs en cours de route.
    \item de se dispenser d'un cahier des charges.
  \end{enumerate}

\item[] \textbf{Capacité C3}

\item \textbf{[Q4]} Devant un message d'erreur Python, la première chose à faire est :
  \begin{enumerate}[label=\Alph*.]
    \item de corriger au hasard une ligne du programme.
    \item d'ignorer le message et de relancer le programme.
    \item de supprimer la fonction mise en cause.
    \item de lire le message en entier pour identifier le type d'erreur et la ligne concernée.

\item[] \textbf{Capacité C4}

\item \textbf{[Q5]} Une docstring de fonction bien rédigée précise notamment :
  \begin{enumerate}[label=\Alph*.]
    \item le rôle de la fonction, ses préconditions et ses postconditions.
    \item le seul nom de l'auteur du code.
    \item la seule date de la dernière modification.
    \item rien du tout, le code se suffisant à lui-même.
  \end{enumerate}

\item \textbf{[Q6]} Une présentation orale efficace d'un projet informatique met en avant :
  \begin{enumerate}[label=\Alph*.]
    \item la lecture du code source, ligne par ligne.
    \item le raisonnement suivi : problème traité, choix effectués, difficultés surmontées.
    \item l'énumération de tous les noms de variables employés.
    \item la récitation de la documentation officielle de Python.
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des réponses erronées}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{B} — Pascal construit une machine à calculer, capable d'additionner et de soustraire. Elle exécute une opération fixée par sa mécanique, non un programme quelconque. \emph{Renvoi : C1, jalons de l'histoire de l'informatique.}
    \item \textbf{C} — Van Rossum crée le langage Python, plus d'un demi-siècle après la formulation du modèle théorique. \emph{Renvoi : C1, jalons de l'histoire de l'informatique.}
    \item \textbf{D} — Berners-Lee conçoit le Web, une application des réseaux, et non le modèle de calcul qui les précède. \emph{Renvoi : C1, jalons de l'histoire de l'informatique.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — L'élève sous-estime nettement la place du projet, que le programme érige en modalité de travail majeure et non en complément. \emph{Renvoi : C2, place du projet dans le programme.}
    \item \textbf{C} — Le projet mobilise des notions qui doivent d'abord être enseignées ; il ne remplace pas le reste du programme. \emph{Renvoi : C2, place du projet dans le programme.}
    \item \textbf{D} — C'est l'inverse : le programme lui réserve explicitement un quart au moins de l'horaire. \emph{Renvoi : C2, place du projet dans le programme.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — Les jalons servent au contraire à coordonner le travail collectif, en fixant des points de rendez-vous communs. \emph{Renvoi : C2, jalons d'un projet.}
    \item \textbf{B} — Un jalon organise le travail ; il ne teste rien par lui-même. Seuls des critères de validation exécutables peuvent détecter des erreurs. \emph{Renvoi : C2, jalons et critères de validation.}
    \item \textbf{D} — Les jalons découpent le cahier des charges dans le temps ; ils le supposent donc écrit, loin de le remplacer. \emph{Renvoi : C2, cahier des charges et jalons.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — Modifier sans diagnostic peut faire disparaître le symptôme en laissant la cause, ou introduire une seconde erreur. \emph{Renvoi : C3, lecture d'un message d'erreur.}
    \item \textbf{B} — Une erreur est déterministe : relancer le même code sur les mêmes données produira exactement le même message. \emph{Renvoi : C3, lecture d'un message d'erreur.}
    \item \textbf{C} — Supprimer le code fautif supprime aussi la fonctionnalité. Le message désigne un défaut à corriger, pas un morceau à retirer. \emph{Renvoi : C3, démarche de débogage.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{B} — L'auteur relève du suivi de version. La docstring s'adresse à qui veut UTILISER la fonction, et doit donc énoncer son contrat. \emph{Renvoi : C4, documentation d'une fonction.}
    \item \textbf{C} — La date est déjà tenue par l'outil de gestion de versions, et n'apprend rien sur le comportement de la fonction. \emph{Renvoi : C4, documentation d'une fonction.}
    \item \textbf{D} — Le code dit COMMENT la fonction procède, jamais ce qu'elle promet ni ce qu'elle exige. C'est exactement ce que la docstring apporte. \emph{Renvoi : C4, documentation d'une fonction.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{A} — Le code est disponible à la lecture. L'oral doit apporter ce qu'il ne montre pas : les intentions et les arbitrages. \emph{Renvoi : C4, présentation orale d'un projet.}
    \item \textbf{C} — Une énumération de détails d'implémentation n'expose ni le problème ni la solution retenue. \emph{Renvoi : C4, présentation orale d'un projet.}
    \item \textbf{D} — La documentation d'un langage n'est pas le projet. L'oral porte sur le travail conduit par l'élève. \emph{Renvoi : C4, présentation orale d'un projet.}
  \end{itemize}
\end{itemize}
