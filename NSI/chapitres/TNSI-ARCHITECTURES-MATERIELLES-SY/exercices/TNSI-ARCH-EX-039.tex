\begin{python}
reseau = {
    "A": {"B": 2, "C": 10},
    "B": {"A": 2, "C": 2},
    "C": {"A": 10, "B": 2},
}
\end{python}

  \item En utilisant \lstinline{plus_court_cout} du cours, quel est le coût de la route effectivement la moins coûteuse de \texttt{A} vers \texttt{C} ? Ce n'est pas la route directe : pourquoi ?
\end{enumerate}
\end{exercice}
