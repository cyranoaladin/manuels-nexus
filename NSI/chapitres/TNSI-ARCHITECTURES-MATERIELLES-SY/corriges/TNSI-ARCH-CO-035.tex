\begin{corrige}{TNSI-ARCH-EX-003}
\textbf{Q1.} RIP choisirait la route directe \texttt{A-C} : elle ne compte qu'un seul saut, quel que soit son coût. \lstinline{bfs_sauts(reseau, "A", "C")} renvoie $1$.

\textbf{Q2.} \lstinline{plus_court_cout(reseau, "A", "C")} renvoie $4$ : la route la moins coûteuse est \texttt{A-B-C} (coût $2+2=4$), moins chère que la route directe \texttt{A-C} (coût $10$), bien qu'elle traverse un routeur de plus. OSPF choisirait donc \texttt{A-B-C}, RIP choisirait \texttt{A-C} : les deux protocoles divergent car ils n'optimisent pas le même critère.
\end{corrige}
