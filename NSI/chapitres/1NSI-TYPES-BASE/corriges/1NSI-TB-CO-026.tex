\begin{corrige}{1NSI-TB-EX-002}
\textbf{1.} $2^8=256\leqslant300<512=2^9$, donc $300$ nécessite $9$ bits.

\textbf{2.} $20=16+4=10100_2$, soit sur 8 bits : $00010100$.

\textbf{3.} $12=1100_2$, soit sur 8 bits : $00001100$. En inversant chaque bit :
$11110011$. En ajoutant $1$ : $11110100$. Donc $-12$ s'écrit $11110100$ en complément à 2
sur 8 bits.
\end{corrige}
