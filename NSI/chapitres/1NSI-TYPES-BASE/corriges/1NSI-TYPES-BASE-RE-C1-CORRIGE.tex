\begin{corrige}{1NSI-TYPES-BASE-RE-C1}
\begin{enumerate}
  \item Poids $8\,4\,2\,1$ sur $1\,1\,0\,1$ : $8+4+1 = \mathbf{13}$.
        Poids $16\,8\,4\,2\,1$ sur $1\,0\,0\,1\,0$ : $16+2 = \mathbf{18}$.
        Poids $32\,16\,8\,4\,2\,1$ sur $1\,0\,0\,0\,0\,1$ : $32+1 = \mathbf{33}$.
  \item $45$ : $32$ tient, reste $13$ ; $16$ ne tient pas ; $8$ tient, reste $5$ ;
        $4$ tient, reste $1$ ; $2$ ne tient pas ; $1$ tient, reste $0$.
        D'où $45 = 101101_2$.
  \item $32+8+4+1 = 45$. La vérification retombe juste : c'est elle qui valide la conversion,
        pas la relecture de la méthode.
\end{enumerate}
\end{corrige}
