\section{Représentation binaire des entiers relatifs}

\definition[D2]{
  Un entier écrit sur $n$ \textbf{bits} peut représenter $2^n$ valeurs distinctes. Pour un
  entier \textbf{positif}, ces valeurs vont de $0$ à $2^n-1$ : sur $8$ bits, de $0$ à $255$.
}

\propriete[Nombre de bits nécessaires]{
  Le nombre de bits nécessaires pour écrire un entier positif $n$ (avec $n\geqslant1$) est
  $\lfloor \log_2(n)\rfloor + 1$. En pratique, on utilise des tailles standard : $8$, $16$,
  $32$ ou $64$ bits (Python, lui, ajuste automatiquement la taille de ses entiers, sans
  limite fixe).
}

\exemple{$200$ s'écrit sur $8$ bits ($200 < 256=2^8$), mais $200+150=350$ nécessite $9$
bits ($350 \geqslant 256$ mais $350 < 512=2^9$) : une addition peut faire « déborder » le
nombre de bits disponibles.}

\definition[D3]{
  Le \textbf{complément à 2} est la méthode standard pour représenter les entiers
  \textbf{relatifs} (positifs et négatifs) en binaire, sur un nombre fixe $n$ de bits. Pour
  représenter $-x$ (avec $x>0$) : on écrit $x$ en binaire sur $n$ bits, on inverse chaque
  bit (0 devient 1, 1 devient 0), puis on ajoute 1 au résultat.
}

\exemple{Sur $8$ bits, pour représenter $-5$ : $5$ s'écrit \lstinline{00000101} ; en
inversant chaque bit on obtient \lstinline{11111010} ; en ajoutant 1 on obtient
\lstinline{11111011}, qui est la représentation de $-5$ en complément à 2 sur 8 bits.}

\begin{codereference}{Complément à 2 sur $n$ bits}
\begin{python}
def complement_a_2(x, n):
    """Renvoie l'ecriture sur n bits de l'entier relatif x (complement a 2).

    Preconditions : -2**(n-1) <= x < 2**(n-1).
    """
    if x < 0:
        x = (1 << n) + x  # equivaut a l'inversion des bits puis +1
    return format(x, f'0{n}b')

print(complement_a_2(5, 8))    # 00000101
print(complement_a_2(-5, 8))   # 11111011
print(complement_a_2(-1, 8))   # 11111111
\end{python}
\end{codereference}

\propriete[Intérêt du complément à 2]{
  Le complément à 2 permet d'additionner deux entiers relatifs \textbf{avec le même circuit
  binaire} que pour des entiers positifs, sans traitement particulier du signe : le bit de
  poids fort indique le signe (0 = positif, 1 = négatif), et l'addition « déborde »
  naturellement de façon cohérente.
}

\erreurFrequente{Croire qu'un nombre négatif se code en changeant simplement le bit de
poids fort (« signe + valeur absolue »). Ce n'est \textbf{pas} le complément à 2 : cette
méthode naïve poserait des problèmes pour l'addition (deux écritures de 0) et n'est pas
celle utilisée par les machines réelles.}
