\begin{corrige}{1NSI-TB-EVAL-B-EX1}
\textbf{1.} Divisions par 2 : $150=2\times75+0$, $75=2\times37+1$, $37=2\times18+1$,
$18=2\times9+0$, $9=2\times4+1$, $4=2\times2+0$, $2=2\times1+0$, $1=2\times0+1$. Restes de
bas en haut : $150=10010110_2$. En base 16 : $150=16\times9+6$, soit $150=96_{16}$.

\textbf{2.} $2^7=128\leqslant150<256=2^8$, donc $150$ nécessite $8$ bits.

\textbf{3.} $20=00010100_2$ sur 8 bits. Inversion des bits : $11101011$. Ajout de $1$ :
$11101100$. Donc $-20$ s'écrit $11101100$.
\end{corrige}

\begin{corrige}{1NSI-TB-EVAL-B-EX2}
\textbf{1.} Le test renvoie \lstinline{False} : \lstinline{0.2} n'a pas d'écriture binaire
finie, et l'accumulation de trois additions produit une valeur
(\lstinline{0.6000000000000001}) légèrement différente de la représentation flottante de
\lstinline{0.6}. Une réécriture correcte : \lstinline{abs((0.2+0.2+0.2) - 0.6) < 1e-9}.

\textbf{2.}
\begin{center}
\begin{tabular}{ccc|c}
  $a$ & $b$ & $c$ & résultat \\\hline

  V & F & F & F \\
  F & V & V & V \\
  F & V & F & F \\
  F & F & V & F \\
  F & F & F & F \\

L'expression est vraie pour $3$ combinaisons sur $8$.

\textbf{3.} Non. En UTF-8, \lstinline{"vélo"} occupe $5$ octets (le « é » occupe deux
octets en UTF-8, les trois autres lettres un octet chacune). En ISO-8859-1, il occupe
$4$ octets (chaque caractère, y compris « é », est codé sur un seul octet dans cet
encodage).
\end{corrige}
