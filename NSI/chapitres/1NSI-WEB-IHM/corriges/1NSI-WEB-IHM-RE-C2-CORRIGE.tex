\begin{corrige}{1NSI-WEB-IHM-RE-C2}
\begin{enumerate}
  \item \textbf{HTML} : le libellé est le contenu de la balise \lstinline{<button>}.
  \item \textbf{CSS} : le survol est un état d'apparence, décrit sans aucun code.
  \item \textbf{Gestionnaire d'événement}, sur \lstinline{click} : il faut modifier une donnée
        (le compteur) et réafficher le résultat. Ni HTML ni CSS ne savent compter.
  \item \textbf{Gestionnaire d'événement}, sur \lstinline{input} : la vérification a lieu à
        chaque frappe. Sur \lstinline{change}, le message n'apparaîtrait qu'après la sortie du
        champ, ce qui ne correspond pas à l'énoncé.
\end{enumerate}
\end{corrige}
