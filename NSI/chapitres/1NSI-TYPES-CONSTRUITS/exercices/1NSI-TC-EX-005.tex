\begin{exercice}{1NSI-TC-EX-005}{1}{5}

\begin{python}
releves = [("Paris", 32.5), ("Lyon", 35.1), ("Lille", 28.7)]
for ville, temp in releves:
    if temp > 30:
        print(ville)
\end{python}
\begin{enumerate}
  \item Donner, sans exécuter le programme, ce qu'affiche ce code.
  \item Quel est le type de chaque élément de la liste \lstinline{releves} ?
  \item Donner la valeur de \lstinline{releves[1][0]}.
\end{enumerate}
\end{exercice}
