def fusionner_tout(table1, table2, cle, defauts):
    index2 = {ligne[cle]: ligne for ligne in table2}
    fusion = []
    for ligne1 in table1:
        if ligne1[cle] in index2:
            complement = {k: v for k, v in index2[ligne1[cle]].items() if k != cle}
        else:
            complement = defauts
        fusion.append({**ligne1, **complement})
    return fusion
