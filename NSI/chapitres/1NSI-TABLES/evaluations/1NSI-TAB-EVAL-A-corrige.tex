\textbf{1.} Avant conversion, \lstinline{films[0]["annee"]} serait une chaîne de
caractères (\lstinline{str}), par exemple \lstinline{"2014"}.
\begin{python}
for ligne in films:
    ligne["annee"] = int(ligne["annee"])
    ligne["note"] = float(ligne["note"])
\end{python}

\textbf{2.}
\begin{python}
resultat = [f for f in films if f["genre"] == "SF" and f["note"] >= 8.5]
\end{python}

\textbf{3.} $2$ films remplissent ce critère : \textbf{Interstellar} (note $8{,}6$) et
\textbf{Inception} (note $8{,}8$).
\end{corrige}
