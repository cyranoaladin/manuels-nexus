\begin{fichemethode}{M3}{Trier une table suivant une colonne}
\textbf{Quand l'utiliser ?} Pour classer, pour trouver les extrêmes, ou pour préparer une
présentation. Trier n'ajoute aucune information : cela réorganise, et cela seul.

\textbf{Pas à pas :}
\begin{enumerate}
  \item \textbf{Choisis la colonne} et vérifie son type. Une colonne numérique restée en
    chaînes se trie dans l'ordre du dictionnaire : $\text{« }100\text{ »}$ avant
    $\text{« }9\text{ »}$.
  \item \textbf{Choisis le sens.} Croissant par défaut ; décroissant avec
    \lstinline{reverse=True}.
  \item \textbf{Choisis entre \lstinline{sorted} et \lstinline{.sort()}.}
    \lstinline{sorted(table, ...)} renvoie une \textbf{nouvelle} table et laisse l'originale
    intacte ; \lstinline{table.sort(...)} réordonne sur place et \textbf{ne renvoie rien}.
    Préfère \lstinline{sorted} : tu gardes l'ordre d'origine si tu en as besoin ensuite.
  \item Passe la clé sous forme de fonction : \lstinline{key=lambda ligne: ligne["points"]}.
  \item \textbf{Pour trier sur deux colonnes}, renvoie un \textbf{tuple} :
    \lstinline{key=lambda ligne: (ligne["classe"], -ligne["points"])} classe par classe croissante, puis
    par points décroissants à l'intérieur de chaque classe.
\end{enumerate}

\textbf{Exemple :}
\begin{python}
# du meilleur au moins bon, sans toucher a la table d'origine
classement = sorted(table, key=lambda ligne: ligne["points"], reverse=True)

# par classe, puis du meilleur au moins bon dans chaque classe
par_classe = sorted(table, key=lambda ligne: (ligne["classe"], -ligne["points"]))
\end{python}

\textbf{Pièges :}
\begin{itemize}
  \item \textbf{Écrire \lstinline{table = table.sort(...)}}. La méthode renvoie
    \lstinline{None} : la table est perdue. C'est l'erreur classique, et elle ne provoque
    une panne que plus loin dans le programme.
  \item Trier des nombres restés en chaînes. Le programme ne signale rien ; seul le
    résultat est faux. Si $100$ apparaît avant $17$, le diagnostic est immédiat.
  \item Croire qu'un tri décroissant s'obtient en inversant la liste après coup : à valeurs
    égales, l'ordre relatif serait inversé lui aussi. Utilise \lstinline{reverse=True}.
  \item Oublier que le tri est \textbf{stable} : deux lignes de même clé gardent leur ordre
    d'origine. C'est une garantie utile, pas un hasard — elle permet de trier en deux passes.
  \item \textbf{Appeler \lstinline{sorted(table, ...)} sans garder le résultat.}
    \lstinline{sorted} ne modifie pas la table : elle en renvoie une nouvelle. Lire ensuite
    \lstinline{table[0]} en la croyant triée donne la première ligne d'origine.
\end{itemize}

\textbf{Vérifier :} contrôle que la table triée a exactement le \textbf{même nombre de
lignes} que l'originale — trier ne perd ni ne crée rien — puis relis les deux ou trois
premières lignes et la dernière.
