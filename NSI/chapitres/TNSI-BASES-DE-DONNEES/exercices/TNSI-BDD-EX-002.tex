\begin{exercice}{TNSI-BDD-EX-002}{1}{15}
On dispose de la table suivante :
\begin{sql}
CREATE TABLE Film(id INTEGER PRIMARY KEY, titre TEXT, realisateur TEXT, duree INTEGER);
INSERT INTO Film VALUES
  (1, 'Le Voyage de Chihiro', 'Miyazaki', 125),
  (2, 'Princesse Mononoke', 'Miyazaki', 134),
  (3, 'Vice-versa', 'Docter', 95);
\end{sql}
Écrire les requêtes SQL permettant d'afficher :
\begin{enumerate}
  \item les titres des films réalisés par Miyazaki ;
  \item les titres des films dont la durée dépasse $100$ minutes.
\end{enumerate}
\end{exercice}
