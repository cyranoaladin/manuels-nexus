\subsection*{C03B/C03C --- SELECT, WHERE}

\item \textbf{[Q4]} La requête \texttt{SELECT nom FROM Client WHERE ville = 'Lyon';} affiche :
  \begin{enumerate}[label=\Alph*.]
    \item toutes les colonnes des clients de Lyon.
    \item uniquement le nom des clients habitant à Lyon.
    \item le nombre de clients de Lyon.
    \item toutes les villes des clients.
  \end{enumerate}

\bigskip

\item \textbf{[Q5]} Écrire \texttt{WHERE ville = Lyon} (sans apostrophes) au lieu de \texttt{WHERE ville = 'Lyon'} :
  \begin{enumerate}[label=\Alph*.]
    \item ne change rien au résultat.
    \item est une erreur : \texttt{Lyon} est interprété comme un nom de colonne.
    \item rend la requête plus rapide.
    \item est obligatoire pour les textes courts.
  \end{enumerate}

\bigskip

\subsection*{C03D --- JOIN}

\item \textbf{[Q6]} La clause \texttt{JOIN ... ON A.x = B.y} permet de :
  \begin{enumerate}[label=\Alph*.]
    \item trier les résultats de la table \texttt{A}.
    \item combiner les tuples de \texttt{A} et \texttt{B} dont les valeurs de \texttt{x} et \texttt{y} coïncident.
    \item supprimer les doublons de \texttt{A}.
    \item renommer la colonne \texttt{x} en \texttt{y}.
  \end{enumerate}

\end{enumerate}
