class Rectangle:
    def __init__(self, largeur, hauteur):
        self.largeur = largeur
        self.hauteur = hauteur

    def aire(self):
        return self.largeur * self.hauteur

    def perimetre(self):
        return 2 * (self.largeur + self.hauteur)
\end{python}

\textbf{Q1.} Specification par l'interface, sans aucun code :
\begin{itemize}
  \item \lstinline{est_vide()} : sans parametre, renvoie un booleen indiquant si la pile ne
  contient aucun element ; aucune precondition ;
  \item \lstinline{empiler(valeur)} : prend la valeur a placer au sommet, ne renvoie rien,
  modifie la pile ; aucune precondition ;
  \item \lstinline{depiler()} : sans parametre, retire et renvoie la valeur du sommet ;
  \textbf{precondition} : la pile n'est pas vide.
\end{itemize}
Principe LIFO : le dernier element empile est le premier depile.

\textbf{Q2.} Après avoir empile $5,3,8,1$, la pile est $[5,3,8,1]$ (du bas vers le sommet). Depiler trois fois retire $1$, puis $8$, puis $3$ (LIFO) : il reste $[5]$.

\textbf{Q3.} Deux implementations de la meme interface :

\begin{python}
class PileListe:
    def __init__(self):
        self._elements = []

    def est_vide(self):
        return len(self._elements) == 0

    def empiler(self, valeur):
        self._elements.append(valeur)

    def depiler(self):
        assert not self.est_vide(), "depiler sur une pile vide"
        return self._elements.pop()


class PileChainee:
    def __init__(self):
        self._sommet = None

    def est_vide(self):
        return self._sommet is None

    def empiler(self, valeur):
        self._sommet = (valeur, self._sommet)

    def depiler(self):
        assert not self.est_vide(), "depiler sur une pile vide"
        valeur, suivante = self._sommet
        self._sommet = suivante
        return valeur
\end{python}
