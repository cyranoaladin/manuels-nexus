\begin{python}
class Compte:
    def __init__(self, titulaire, solde=0):
        self.titulaire = titulaire
        self.solde = solde

    def deposer(self, montant):
        self.solde = self.solde + montant

    def retirer(self, montant):
        if montant > self.solde:
            raise ValueError("solde insuffisant")
        self.solde = self.solde - montant
\end{python}

Test : \lstinline{c = Compte("Alice", 100)} ; après \lstinline{c.deposer(50)}, \lstinline{c.solde} vaut $150$ ; après \lstinline{c.retirer(30)}, \lstinline{c.solde} vaut $120$.
\end{corrige}
