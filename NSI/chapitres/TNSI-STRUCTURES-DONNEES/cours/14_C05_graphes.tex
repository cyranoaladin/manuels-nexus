\section{Graphes : modélisation et représentations}

% ============================================================
% STRATE 1 — Modélisation et deux representations
% ============================================================

\definition[1]{
Un \textbf{graphe} est constitue de \textbf{sommets} (ou noeuds) relies par des \textbf{arêtes}. Un graphe est \textbf{oriente} si ses arêtes ont un sens (representees par des fleches), \textbf{non oriente} sinon.
}

\margeAppui{
On modélisé par un graphe toute situation faite d'éléments en relation : reseau routier (sommets = villes, arêtes = routes), reseau social (sommets = personnes, arêtes = relations), pages web (sommets = pages, arêtes orientees = liens hypertexte).
}

\propriete[Représentation par matrice d'adjacence]{
Pour un graphe a $n$ sommets numerotes de $0$ a $n-1$, la \textbf{matrice d'adjacence} est un tableau $n \times n$ ou la case $(i,j)$ vaut $1$ (ou \lstinline{True}) s'il existe une arête de $i$ vers $j$, $0$ sinon.
}

\textbf{Exemple.} Graphe oriente a $3$ sommets : $0\to1$, $0\to2$, $1\to2$.

\begin{python}
matrice = [
    [0, 1, 1],
    [0, 0, 1],
    [0, 0, 0],
]
\end{python}

\propriete[Représentation par listes de successeurs]{
Pour chaque sommet, on stocke la \textbf{liste des sommets accessibles directement} (ses successeurs), par exemple sous forme de dictionnaire.
}

\textbf{Exemple.} Le même graphe ($0\to1$, $0\to2$, $1\to2$) représente par listes de successeurs :
\begin{python}
successeurs = {
    0: [1, 2],
    1: [2],
    2: [],
}
\end{python}

\textbf{Conversion.} Pour passer de la matrice aux listes de successeurs, on parcourt chaque ligne $i$ et on retient les indices $j$ ou \lstinline{matrice[i][j] == 1} :
\begin{python}
def matrice_vers_listes(matrice):
    n = len(matrice)
    return {i: [j for j in range(n) if matrice[i][j] == 1] for i in range(n)}
\end{python}

\propriete[Comparaison]{
La \textbf{matrice d'adjacence} permet de tester en temps constant l'existence d'une arête $(i,j)$, mais occupe une memoire en $n^2$ même si le graphe est peu dense (\textbf{creux}). Les \textbf{listes de successeurs} occupent une memoire proportionnelle au nombre d'arêtes reellement presentes, plus adaptees aux graphes creux, mais tester l'existence d'une arête précise peut necessiter de parcourir une liste.
}

% ============================================================
% STRATE 2 — Erreurs frequentes
% ============================================================

\erreurFrequente{
\textbf{Oublier qu'un sommet sans successeur doit quand même apparaitre.} Dans la représentation par listes de successeurs, un sommet isole ou puits (sans arête sortante) doit être present dans le dictionnaire avec une liste \textbf{vide}, pas absent du dictionnaire : sinon, un parcours du graphe risque d'ignorer ce sommet.
}
