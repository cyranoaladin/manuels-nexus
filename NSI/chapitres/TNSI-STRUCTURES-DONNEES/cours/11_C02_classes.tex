\section{Définir et utiliser une classe en Python}

% ============================================================
% STRATE 1 — Syntaxe et exemple
% ============================================================

\definition[1]{
Une \textbf{classe} est un modele qui decrit la structure (\textbf{attributs}, l'etat) et le comportement (\textbf{methodes}) des objets qui en sont des \textbf{instances}. En Python, une classe se definit avec le mot-cle \lstinline{class}, et le constructeur \lstinline{__init__} initialise les attributs d'une nouvelle instance.
}

\textbf{Exemple.}
\begin{python}
class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def distance_origine(self):
        return (self.x ** 2 + self.y ** 2) ** 0.5

    def translater(self, dx, dy):
        self.x = self.x + dx
        self.y = self.y + dy
\end{python}

\textbf{Utilisation :}
\begin{python}
p = Point(3, 4)
print(p.x, p.y)               # acces aux attributs
print(p.distance_origine())   # appel de methode : 5.0
p.translater(1, 1)
print(p.x, p.y)                # 4 5
\end{python}

\propriete[Vocabulaire]{
$\bullet$ \textbf{Attribut} : donnee associee a un objet (etat), accedee par \lstinline{objet.attribut}.

$\bullet$ \textbf{Methode} : fonction associee a une classe, appelee par \lstinline{objet.methode(...)}. Le premier parametre \lstinline{self} designe l'objet lui-meme (fourni automatiquement lors de l'appel).

$\bullet$ \textbf{Instance} : un objet particulier cree a partir d'une classe (par exemple \lstinline{p} est une instance de \lstinline{Point}).
}

% ============================================================
% STRATE 2 — Erreurs frequentes
% ============================================================

\erreurFrequente{
\textbf{Oublier \lstinline{self} dans la definition d'une methode.} Toute methode d'instance doit avoir \lstinline{self} comme premier parametre (meme si on ne l'appelle jamais explicitement lors de l'appel \lstinline{objet.methode()}) : Python le fournit automatiquement.
}
