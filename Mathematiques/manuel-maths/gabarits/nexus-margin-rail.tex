\RequirePackage{luatexbase}

% Un changement de couleur ecrit une paire push/pop sur la pile de couleur du
% moteur. `\color` pousse la sienne sur-le-champ et confie la depile a
% `\aftergroup` : la depile part donc APRES le groupe. Composee dans une boite
% capturee, la poussee suit la note dans son Form XObject tandis que la depile
% reste sur la page -- « pop empty color page stack », une fois par note. Le
% noyau LaTeX fournit deja le groupe qui referme la couleur AVANT la fin de la
% boite ; il est ici nomme pour rester lisible sous n'importe quel catcode.
\begingroup\makeatletter
  \global\let\nxMargeCouleurDebut\color@begingroup
  \global\let\nxMargeCouleurFin\color@endgroup
\endgroup

\newcounter{nxMarginGlobalOrder}
\newbox\nxMarginCaptureBox
\newbox\nxMarginReportDecorationBox

\newcommand{\nxMarginEightDigits}[1]{%
  \ifnum#1<10000000 0\fi
  \ifnum#1<1000000 0\fi
  \ifnum#1<100000 0\fi
  \ifnum#1<10000 0\fi
  \ifnum#1<1000 0\fi
  \ifnum#1<100 0\fi
  \ifnum#1<10 0\fi
  \number#1}
\newcommand{\nxMarginVariant}{professeur}
\newcommand{\nxMarginStableId}[2]{%
  nxm:\nxMarginVariant:#1:\nxMarginEightDigits{#2}}

% ======================================================================
%  REPÈRE DU COMPOSITEUR DE MARGES
% ======================================================================
% Le compositeur travaille dans UN seul repère : la PAGE COMPOSÉE, c'est-à-dire
% le format fini, dont le coin est l'origine du TrimBox. C'est déjà le repère
% de \paperwidth/\paperheight, celui du nœud « current page » de TikZ, celui de
% \put dans « shipout/foreground » — donc celui où le compositeur dessine — et,
% depuis la correction des liens de marge, celui où le ledger compare les
% annotations. Le fond perdu est une extension technique du support : il
% n'est PAS l'origine logique du contenu imprimé.
%
% Conséquence pratique : les quatre longueurs ci-dessous sont l'unique
% expression de la géométrie du rail, et elles sont dans ce repère. Aucun
% \hoffset, aucun \voffset : les ajouter produirait un contenu logique
% translaté du fond perdu (559 403 sp mesurés le 2026-09-03).
%
% \nxMarginReserveRect attend ce même repère.
\newcommand{\nxMargeRailGaucheImpaire}{%
  \dimexpr1in+\oddsidemargin+\textwidth+\marginparsep\relax}
\newcommand{\nxMargeRailGauchePaire}{%
  \dimexpr1in+\evensidemargin-\marginparsep-\marginparwidth\relax}
\newcommand{\nxMargeRailHaut}{%
  \dimexpr1in+\topmargin+\headheight+\headsep\relax}
\newcommand{\nxMargeRailBas}{%
  \dimexpr1in+\topmargin+\headheight+\headsep+\textheight\relax}

\directlua{%
  local path = kpse.find_file("nexus-margin-shipout.lua", "lua")
    or "gabarits/nexus-margin-shipout.lua"
  local chunk, load_error = loadfile(path)
  if not chunk then
    error("cannot load nexus-margin-shipout.lua: " .. tostring(load_error))
  end
  nexus_margin = chunk()
}

% La configuration est prise dans « begindocument/end », et non dans
% \AtBeginDocument. La classe agrandit le support et pose \hoffset/\voffset
% dans « begindocument/end » PRÉCISÉMENT parce que geometry réaffirme
% \pagewidth au sein de \AtBeginDocument. Lire la géométrie de page trop tôt
% donnait un rail calculé avec \hoffset = 0 pt, donc dans le repère du format
% fini, tandis que tout obstacle déclaré dans le corps du document arrivait
% avec \hoffset = 3 mm, donc dans le repère du support : 559 403 sp d'écart,
% exactement le fond perdu. Le repère est désormais déclaré et vérifié.
\AddToHook{begindocument/end}{%
  \ifnxVersionProfesseur
    \gdef\nxMarginVariant{professeur}%
  \else
    \gdef\nxMarginVariant{eleve}%
  \fi
  \setbox\nxMarginReportDecorationBox=\vbox{%
    \hsize=\marginparwidth
    \parindent=0pt
    \nxMargeCouleurDebut
    \sffamily\fontsize{6.5}{8}\selectfont
    \color{encre!65}%
    \noindent\fboxsep=2pt\fboxrule=.4pt
    \fcolorbox{encre!25}{white}{suite de la page 0000}%
    \par\vskip3pt
    \nxMargeCouleurFin}%
  \typeout{NEXUS-MARGIN-REPORT-DECORATION-SP:%
    \number\dimexpr\ht\nxMarginReportDecorationBox+\dp\nxMarginReportDecorationBox\relax}%
  % Tout ce qui suit est dans le repère de la page composée : page_width_sp et
  % page_height_sp sont le format fini, et le rail vient des quatre longueurs
  % canoniques déclarées plus haut.
  %
  % trim_origin_* et support_offset_* ne sont PAS des coordonnées : ils
  % transportent le repère lui-même. trim_origin_* est le coin de la page
  % composée dans le support, tel que la classe le DÉCLARE ; support_offset_*
  % est le décalage réellement en vigueur dans les registres de TeX. Le
  % compositeur refuse de démarrer s'ils ne coïncident pas — c'est-à-dire si
  % les coordonnées qu'il reçoit ne sont pas dans le repère qu'il croit.
  \directlua{nexus_margin.configure({
    variant = "\luaescapestring{\nxMarginVariant}",
    page_width_sp = \number\paperwidth,
    page_height_sp = \number\paperheight,
    trim_origin_x_sp = \number\nxOrigineTrimX,
    trim_origin_y_sp = \number\nxOrigineTrimY,
    support_offset_x_sp = \number\hoffset,
    support_offset_y_sp = \number\voffset,
    rail_width_sp = \number\marginparwidth,
    odd_rail_left_sp = \number\nxMargeRailGaucheImpaire,
    even_rail_left_sp = \number\nxMargeRailGauchePaire,
    rail_top_sp = \number\nxMargeRailHaut,
    rail_bottom_sp = \number\nxMargeRailBas,
    report_box_number = \number\nxMarginReportDecorationBox,
    report_decoration_height_sp = \number\dimexpr
      \ht\nxMarginReportDecorationBox+\dp\nxMarginReportDecorationBox\relax
  })}%
}

\newcommand{\nxMarginAnchorMaterial}[1]{%
  \savepos
  \directlua{nexus_margin.write_anchor_whatsit(
    "\luaescapestring{#1}")}%
  \directlua{nexus_margin.write_anchor_marked_content(
    "\luaescapestring{#1}")}%
  \edef\nxMarginAnchorLua{%
    nexus_margin.resolve_anchor("\luaescapestring{#1}")}%
  \expandafter\latelua\expandafter{\nxMarginAnchorLua}%
}

\AddToHook{shipout/foreground}{%
  \directlua{nexus_margin.render_foreground()}%
}

\newcommand{\nxMarginDeferredAnchor}[1]{%
  \ifhmode
    \vadjust{\hbox to 0pt{\nxMarginAnchorMaterial{#1}\hss}}%
  \else
    \nxMarginAnchorMaterial{#1}%
  \fi
}

\newcommand{\nxMarginReserveMaterial}[5]{%
  \directlua{nexus_margin.write_obstacle_whatsit(
    "\luaescapestring{#1}", #2, #3, #4, #5)}%
}

\NewDocumentCommand\nxMarginReserveRect{m m m m m}{%
  \edef\nxMarginReserveCall{%
    \noexpand\nxMarginReserveMaterial
      {#1}%
      {\number\dimexpr#2\relax}%
      {\number\dimexpr#3\relax}%
      {\number\dimexpr#4\relax}%
      {\number\dimexpr#5\relax}}%
  \ifhmode
    \vadjust{\nxMarginReserveCall}%
  \else
    \nxMarginReserveCall
  \fi
}

\NewDocumentCommand\nxMarginRailNote{m +m}{%
  \stepcounter{nxMarginGlobalOrder}%
  \edef\nxMarginCurrentId{%
    \nxMarginStableId{#1}{\arabic{nxMarginGlobalOrder}}}%
  \setbox\nxMarginCaptureBox=\vbox{%
    \hsize=\marginparwidth
    \linewidth=\hsize
    % Le rail est une colonne etroite : la composer au fer a gauche, comme la
    % classe le declare deja pour les notes de marge exterieure
    % (\raggedrightmarginnote). Justifiee, la moindre formule en ligne y
    % produit un debordement horizontal, que le gate anti-collision rejette.
    \raggedright
    \emergencystretch=\hsize
    % La couleur se referme DANS la boite : sans cela sa depile resterait sur
    % la page pendant que sa poussee partirait dans le Form XObject.
    \nxMargeCouleurDebut
    #2%
    \nxMargeCouleurFin}%
  \directlua{nexus_margin.capture_box(
    "\luaescapestring{\nxMarginCurrentId}",
    "\luaescapestring{#1}",
    \number\value{nxMarginGlobalOrder},
    \number\nxMarginCaptureBox)}%
  \edef\nxMarginDeferredCall{%
    \noexpand\nxMarginDeferredAnchor{\nxMarginCurrentId}}%
  \nxMarginDeferredCall
}

\AddToHook{enddocument/afterlastpage}{%
  \directlua{nexus_margin.finalize()}%
