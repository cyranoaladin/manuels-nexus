\rubrique{Cours}
\section{Sens de variation d'une suite}

\begin{nxdef}[D1]
Une suite $(u_n)$ est \textbf{croissante} si, pour tout entier naturel $n$,
$u_{n+1} \geqslant u_n$.
\end{nxdef}

\exemple{La suite définie par $u_n = 3 \times 0{,}5^n + 1$ est décroissante et
semble se rapprocher de $1$ : le nuage de points le montre immédiatement.}

\begin{nxfigure}{Premiers termes de la suite $u_n = 3\times 0{,}5^n + 1$}
  \figSuite{3*0.5^x + 1}{8}
\end{nxfigure}

\erreurFrequente{Confondre $u_{n+1}$ (le terme suivant) et $u_n + 1$ (le terme
courant augmenté de 1). Sur le nuage de points, $u_{n+1}$ est l'ordonnée du
point \emph{d'abscisse} $n+1$.}

\debutrubrique{Exercices}

\begin{exercice}{1SPE-SUITES-EX-901}{1}{10}
On considère la suite $(u_n)$ définie par $u_0 = 5$ et, pour tout entier
naturel $n$, $u_{n+1} = \frac{1}{2}u_n + 3$.
\begin{enumerate}
  \item Calculer $u_1$, $u_2$ et $u_3$.
  \item Représenter les quatre premiers termes sur un nuage de points.
  \item Conjecturer le sens de variation de $(u_n)$.
\end{enumerate}
\end{exercice}

\begin{exercice}{1SPE-SUITES-EX-902}{2}{15}
Une usine produit $10\,000$ pièces la première année. Chaque année, la
production diminue de $10\,\%$.
\begin{enumerate}
  \item Exprimer $u_{n+1}$ en fonction de $u_n$. Quelle est la nature de la
    suite $(u_n)$ ?
  \item Exprimer $u_n$ en fonction de $n$, puis calculer $u_5$ à l'unité.
\end{enumerate}
\end{exercice}
