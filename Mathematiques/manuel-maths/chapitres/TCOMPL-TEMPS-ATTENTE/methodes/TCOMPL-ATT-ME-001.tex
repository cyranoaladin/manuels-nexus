\begin{fichemethode}{M1}{Reconnaitre et definir une loi geometrique}

\quandUtiliser{%
  Utiliser cette methode lorsqu'une experience de Bernoulli (succes de
  probabilite $p$) est REPETEE de facon independante et que la variable
  etudiee est le RANG du premier succes : il s'agit de justifier que $X$ suit
  la loi geometrique de parametre $p$ et d'en donner les proprietes.
}

\pasApas{%
  \begin{enumerate}
    \item Verifier les trois conditions du modele : epreuves de Bernoulli
      identiques, independantes, et $X$ = rang du premier succes.
    \item Enoncer la loi : pour tout entier $k \geq 1$,
      $P(X = k) = (1-p)^{k-1}\,p$ ($k-1$ echecs puis un succes).
    \item Retenir la queue de distribution : $P(X > n) = (1-p)^n$ (les $n$
      premieres epreuves sont toutes des echecs).
    \item Donner l'esperance : $E(X) = \dfrac{1}{p}$ — le rang moyen du premier
      succes.
    \item Enoncer l'absence de memoire : pour tous entiers $m, n \geq 0$,
      $P_{(X > m)}(X > m+n) = P(X > n)$ — le passe ne change pas l'attente
      restante (exploitation : fiche M3).
  \end{enumerate}
}

\exempleRedige{%
  On lance un de equilibre jusqu'a obtenir un 6 ; $X$ est le rang du premier 6.
  Justifier la loi de $X$ et donner $E(X)$.

  \medskip
  \commentaireMarge{Trois conditions a citer explicitement.}%
  Chaque lancer est une epreuve de Bernoulli de parametre $p = \dfrac{1}{6}$
  (succes : « obtenir 6 »), les lancers sont independants et identiques, et $X$
  est le rang du premier succes : $X$ suit la loi geometrique de parametre
  $\dfrac{1}{6}$.

  \medskip
  Donc $P(X = k) = \left(\dfrac{5}{6}\right)^{k-1}\dfrac{1}{6}$ pour $k \geq 1$,
  et $E(X) = \dfrac{1}{p} = 6$ : en moyenne, le premier 6 arrive au sixieme
  lancer.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piege 1.} Confondre avec la loi binomiale : la binomiale
      compte les succes en $n$ epreuves FIXEES, la geometrique attend le
      PREMIER succes (nombre d'epreuves non borne).
    \item \textbf{Piege 2.} Ecrire $P(X = k) = p^{k-1}(1-p)$ : l'exposant
      $k-1$ porte sur l'ECHEC $(1-p)$.
    \item \textbf{Piege 3.} Oublier de justifier l'independance des epreuves
      avant de conclure sur la loi.
  \end{itemize}
}

\verifier{%
  Controler que les $P(X = k)$ somment a 1 (serie geometrique), que
  $E(X) = \frac{1}{p}$ est coherent (plus $p$ est petit, plus l'attente est
  longue), et tester $P(X > 0) = 1$.
}

\end{fichemethode}
