\begin{fichemethode}{M6}{Calculer l'espérance d'une loi exponentielle par une intégrale}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande de CALCULER
  $E(T) = \displaystyle\lim_{A \to +\infty}\int_0^A t\,\lambda e^{-\lambda t}\,dt$
  pour une loi exponentielle, ou de vérifier la formule $E(T) = \frac{1}{\lambda}$.
}

\pasApas{%
  \begin{enumerate}
    \item Écrire l'espérance comme intégrale impropre :
      $E(T) = \displaystyle\lim_{A \to +\infty} \int_0^A t\,\lambda
      e^{-\lambda t}\,dt$.
    \item Chercher une primitive de $t \mapsto t\,\lambda e^{-\lambda t}$ :
      poser $G(t) = -\left(t + \dfrac{1}{\lambda}\right) e^{-\lambda t}$ et
      VERIFIER en derivant que $G'(t) = t\,\lambda e^{-\lambda t}$.
    \item Calculer $\displaystyle\int_0^A = G(A) - G(0)
      = \dfrac{1}{\lambda} - \left(A + \dfrac{1}{\lambda}\right)e^{-\lambda A}$.
    \item Passer a la limite : $\left(A + \frac{1}{\lambda}\right)
      e^{-\lambda A} \to 0$ quand $A \to +\infty$ (croissance comparee), donc
      $E(T) = \dfrac{1}{\lambda}$.
    \item Interpréter : duree moyenne d'attente, coherente avec le paramètre.
  \end{enumerate}
}

\exempleRedige{%
  Calculer l'espérance de $T$ de loi exponentielle de paramètre
  $\lambda = \dfrac{1}{2}$.

  \medskip
  \commentaireMarge{Primitive proposee, controlee par dérivation.}%
  Posons $G(t) = -(t + 2)\,e^{-t/2}$. Alors
  $G'(t) = -e^{-t/2} + \dfrac{1}{2}(t+2)e^{-t/2} = \dfrac{t}{2}\,e^{-t/2}$,
  qui est bien la fonction a integrer.

  \medskip
  \commentaireMarge{Croissance comparee : $A e^{-A/2} \to 0$.}%
  $\displaystyle\int_0^A \dfrac{t}{2}e^{-t/2}\,dt = G(A) - G(0)
  = 2 - (A+2)e^{-A/2} \xrightarrow[A \to +\infty]{} 2$.

  \medskip
  Donc $E(T) = 2 = \dfrac{1}{\lambda}$ : l'attente moyenne est de 2 unités de
  temps.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Integrer $\lambda e^{-\lambda t}$ (la densite) au
      lieu de $t\,\lambda e^{-\lambda t}$ : l'espérance porte sur $t\,f(t)$.
    \item \textbf{Piège 2.} Oublier le terme $\frac{1}{\lambda}$ dans la
      primitive $G$ : contrôler SYSTEMATIQUEMENT en derivant.
    \item \textbf{Piège 3.} Affirmer $(A + \frac{1}{\lambda})e^{-\lambda A} \to 0$
      sans invoquer la croissance comparee (l'exponentielle l'emporte sur le
      polynôme).
  \end{itemize}
}

\verifier{%
  Dériver la primitive $G$ pour retrouver exactement $t\,\lambda e^{-\lambda t}$,
  contrôler $E(T) = \frac{1}{\lambda} > 0$, et confronter a une simulation ou a
  l'ordre de grandeur du contexte.
}
