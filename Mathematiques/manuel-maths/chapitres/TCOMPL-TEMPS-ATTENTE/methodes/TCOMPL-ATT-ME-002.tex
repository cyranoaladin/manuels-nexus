\begin{fichemethode}{M2}{Calculer des probabilités pour une loi géométrique}

\quandUtiliser{%
  Utiliser cette méthode lorsque $X$ suit une loi géométrique de paramètre $p$
  et que l'énoncé demande $P(X = k)$, $P(X \leq n)$, $P(X > n)$ ou une
  probabilité du type « entre $a$ et $b$ ».
}

\pasApas{%
  \begin{enumerate}
    \item Valeur exacte : $P(X = k) = (1-p)^{k-1}\,p$.
    \item Queue : $P(X > n) = (1-p)^n$ — a privilegier, c'est la formule la
      plus simple.
    \item Complementaire : $P(X \leq n) = 1 - P(X > n) = 1 - (1-p)^n$.
    \item Intervalle : $P(a \leq X \leq b) = P(X > a-1) - P(X > b)
      = (1-p)^{a-1} - (1-p)^{b}$.
    \item Donner la valeur exacte (fraction ou puissance) puis un arrondi, et
      interpréter dans le contexte.
  \end{enumerate}
}

\exempleRedige{%
  $X$ est le rang du premier 6 au de ($p = \frac{1}{6}$). Calculer $P(X = 3)$,
  $P(X \leq 3)$ et $P(X > 4)$.

  \medskip
  \commentaireMarge{Deux echecs puis un succès.}%
  $P(X = 3) = \left(\dfrac{5}{6}\right)^{2} \times \dfrac{1}{6}
  = \dfrac{25}{216} \approx 0{,}116$.

  \medskip
  \commentaireMarge{Passer par l'événement contraire.}%
  $P(X \leq 3) = 1 - \left(\dfrac{5}{6}\right)^{3} = \dfrac{91}{216}
  \approx 0{,}421$.

  \medskip
  $P(X > 4) = \left(\dfrac{5}{6}\right)^{4} = \dfrac{625}{1296}
  \approx 0{,}482$ : pres d'une fois sur deux, le premier 6 n'est toujours pas
  sorti après 4 lancers.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Decaler l'exposant : $P(X > n) = (1-p)^n$ mais
      $P(X = k)$ contient $(1-p)^{k-1}$.
    \item \textbf{Piège 2.} Confondre $P(X \leq n)$ et $P(X < n)$ : pour une
      variable entiere, $P(X < n) = P(X \leq n-1)$.
    \item \textbf{Piège 3.} Sommer terme a terme $P(X=1) + \dots + P(X=n)$ sans
      reconnaitre que le complementaire $(1-p)^n$ donne le résultat en une
      ligne.
  \end{itemize}
}

\verifier{%
  Contrôler que chaque probabilité est dans $[0\,;\,1]$, que
  $P(X \leq n) + P(X > n) = 1$, et retrouver un résultat par l'autre chemin
  (somme directe contre complementaire).
}

\sEntrainer{\refExos{M2}}

\end{fichemethode}
