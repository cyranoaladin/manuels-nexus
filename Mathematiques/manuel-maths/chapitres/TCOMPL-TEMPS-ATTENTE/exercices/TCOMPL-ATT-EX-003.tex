\begin{exercice}{TCOMPL-ATT-EX-003}{1}{15}
Le temps d'attente $X$ (en minutes) à un guichet suit la loi exponentielle de paramètre $\lambda=0{,}25$, de densité $f(x)=0{,}25\,\mathrm{e}^{-0{,}25x}$ pour $x\geq0$.
\begin{enumerate}
  \item Vérifier que $f$ est bien une densité de probabilité.
  \item Calculer $P(X\leq2)$, la probabilité d'attendre moins de $2$ minutes.
\end{enumerate}
\end{exercice}
