\begin{exercice}{TCOMPL-ATT-EX-004}{2}{15}

\begin{enumerate}
  \item Écrire l'intégrale donnant $E(X)$.
  \item Calculer $E(X)$ (on admet que $\displaystyle\int_0^{+\infty} x\,\lambda\mathrm{e}^{-\lambda x}\,\mathrm{d}x = \dfrac{1}{\lambda}$) et interpréter.
\end{enumerate}
