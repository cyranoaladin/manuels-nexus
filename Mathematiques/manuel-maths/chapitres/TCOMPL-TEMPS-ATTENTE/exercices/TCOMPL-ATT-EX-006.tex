\begin{exercice}{TCOMPL-ATT-EX-006}{2}{15}
Soit $X$ géométrique de paramètre $p$.
\begin{enumerate}
  \item Montrer que $P(X>n)=(1-p)^n$ pour tout entier $n\geq0$.
  \item En déduire que $P_{X>n}(X>n+k)=P(X>k)$ : c'est l'absence de mémoire.
  \item Une machine tombe en panne chaque jour avec probabilité $0{,}05$.
        Elle a déjà fonctionné $30$ jours. Quelle est la probabilité qu'elle
        tienne $10$ jours de plus ?
\end{enumerate}
\end{exercice}
