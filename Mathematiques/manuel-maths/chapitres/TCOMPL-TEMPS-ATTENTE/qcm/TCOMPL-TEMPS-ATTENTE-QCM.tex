\subsection*{C1/C2 --- Loi géométrique}

\begin{enumerate}

\item \textbf{[Q1]} Pour $X\sim\mathcal{G}(p)$, $P(X=k)$ (pour $k\geq1$) vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $(1-p)^{k}p$.
    \item $(1-p)^{k-1}p$.
    \item $p^k(1-p)$.
    \item $kp$.
  \end{enumerate}

\bigskip

\item \textbf{[Q2]} L'espérance d'une loi géométrique de paramètre $p$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $p$.
    \item $1-p$.
    \item $\dfrac{1}{p}$.
    \item $np$.
  \end{enumerate}

\bigskip

\subsection*{C3 --- Absence de mémoire}

\item \textbf{[Q3]} L'absence de mémoire signifie que :
  \begin{enumerate}[label=\Alph*.]
    \item le passé influence toujours l'avenir.
    \item la probabilité conditionnelle du temps d'attente restant ne dépend pas du temps déjà écoulé.
    \item la loi géométrique n'a pas d'espérance.
    \item le succès devient certain après plusieurs échecs.
  \end{enumerate}

\bigskip

\subsection*{C4/C5 --- Loi exponentielle, densité}

\item \textbf{[Q4]} Une fonction $f$ est une densité de probabilité sur $[0,+\infty[$ si :
  \begin{enumerate}[label=\Alph*.]
    \item $f$ est positive et son intégrale sur $[0,+\infty[$ vaut $1$.
    \item $f$ est négative.
    \item $f(0)=1$.
    \item $f$ est croissante.
  \end{enumerate}

\bigskip

\item \textbf{[Q5]} Pour une loi à densité, $P(X=x_0)$ pour une valeur précise $x_0$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $1$.
    \item $f(x_0)$.
    \item $0$.
    \item impossible à déterminer.
  \end{enumerate}

\bigskip

\subsection*{C6 --- Espérance}

\item \textbf{[Q6]} L'espérance d'une loi exponentielle de paramètre $\lambda$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $\lambda$.
    \item $\dfrac{1}{\lambda}$.
    \item $\lambda^2$.
    \item $1-\lambda$.
  \end{enumerate}

\end{enumerate}
