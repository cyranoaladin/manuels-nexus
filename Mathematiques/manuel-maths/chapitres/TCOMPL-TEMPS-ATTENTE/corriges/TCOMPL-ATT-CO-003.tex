\begin{corrige}{TCOMPL-ATT-EX-003}
\textbf{1.} $f(x)=0{,}25\,\mathrm{e}^{-0{,}25x} \geq 0$ pour tout $x\geq0$, et :
\[
  \int_0^{+\infty} 0{,}25\,\mathrm{e}^{-0{,}25x}\,\mathrm{d}x = \Big[-\mathrm{e}^{-0{,}25x}\Big]_0^{+\infty} = 0-(-1) = 1.
\]
$f$ est donc bien une densité de probabilité.

\textbf{2.} $P(X\leq2) = 1-\mathrm{e}^{-0{,}25\times2} = 1-\mathrm{e}^{-0{,}5} \approx 0{,}3935$.
\end{corrige}
