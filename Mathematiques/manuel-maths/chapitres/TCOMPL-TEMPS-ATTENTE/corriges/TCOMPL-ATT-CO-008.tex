\begin{corrige}{TCOMPL-ATT-EX-008}
\textbf{1.} $E(T)=\displaystyle\int_0^{+\infty}t\lambda\mathrm{e}^{-\lambda t}
\,\mathrm{d}t$.

\textbf{2.} Avec $u=t$ et $v'=\lambda\mathrm{e}^{-\lambda t}$ :
$E(T)=\left[-t\mathrm{e}^{-\lambda t}\right]_0^{+\infty}
+\int_0^{+\infty}\mathrm{e}^{-\lambda t}\,\mathrm{d}t
=0+\dfrac{1}{\lambda}=\dfrac{1}{\lambda}$.

\textbf{3.} Pour $\lambda=0{,}5$, $E(T)=2$ : la durée moyenne d'attente est de
deux unités de temps. Plus $\lambda$ est grand, plus l'événement survient
tôt en moyenne.
\end{corrige}
