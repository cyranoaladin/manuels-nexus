\begin{corrige}{TCOMPL-ATT-EX-002}
\textbf{1.} Par l'absence de mémoire, sachant $4$ échecs déjà observés, la probabilité d'attendre encore $6$ tentatives est \textbf{la même} que la probabilité initiale d'attendre $6$ tentatives dès le départ : $P_{X>4}(X>4+6) = P(X>6)$.

\textbf{2.} $P(X>6) = (1-0{,}25)^6 = 0{,}75^6 \approx 0{,}178$.
\end{corrige}
