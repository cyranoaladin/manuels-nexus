\begin{corrige}{TCOMPL-ATT-EV-A-EX1}
\textbf{1.} Chaque partie est une épreuve de Bernoulli indépendante de succès $p=\frac{1}{8}$ : $X$, rang de la première réussite, suit $\mathcal{G}\!\left(\frac{1}{8}\right)$.

\textbf{2.} $P(X=5) = \left(\frac{7}{8}\right)^4\times\frac{1}{8} = \frac{2401}{32768} \approx 0{,}0733$.

\textbf{3.} $E(X) = 8$ : il faut en moyenne $8$ parties pour obtenir l'objet rare.
\end{corrige}

\begin{corrige}{TCOMPL-ATT-EV-A-EX2}
\textbf{1.} Pour $X\sim\mathcal{G}(p)$, $P_{X>n}(X>n+k)=P(X>k)$ pour tous $n,k\geq0$ : le passé n'influence pas l'avenir.

\textbf{2.} Par l'absence de mémoire, la probabilité d'attendre encore $5$ parties sachant $3$ échecs déjà observés est \textbf{la même} que la probabilité initiale d'attendre $5$ parties : $P_{X>3}(X>8) = P(X>5)$.

\textbf{3.} $P(X>5) = \left(\frac{7}{8}\right)^5 \approx 0{,}513$.
\end{corrige}
