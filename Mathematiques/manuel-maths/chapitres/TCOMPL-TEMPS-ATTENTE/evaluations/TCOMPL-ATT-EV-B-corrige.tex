\begin{corrige}{TCOMPL-ATT-EV-B-EX1}
\textbf{1.} $f(x)=\frac{1}{15}\mathrm{e}^{-x/15}\geq0$ pour tout $x\geq0$, et $\displaystyle\int_0^{+\infty}\frac{1}{15}\mathrm{e}^{-x/15}\,\mathrm{d}x = \Big[-\mathrm{e}^{-x/15}\Big]_0^{+\infty} = 0-(-1)=1$. $f$ est bien une densité.

\textbf{2.} $P(X\leq10) = 1-\mathrm{e}^{-10/15} = 1-\mathrm{e}^{-2/3} \approx 0{,}4866$.

\textbf{3.} $E(X) = \displaystyle\int_0^{+\infty} x\times\frac{1}{15}\mathrm{e}^{-x/15}\,\mathrm{d}x = \dfrac{1}{1/15} = 15$ minutes.

\textbf{4.} $P_{X>s}(X>s+t) = P(X>t)$ pour tous $s,t\geq0$ : le temps déjà écoulé sans nouveau client ne change rien à la loi du temps d'attente restant. C'est réaliste ici car les arrivées de clients sont indépendantes les unes des autres, sans effet de \og fatigue \fg{} ou d'\og imminence \fg{} après une longue attente.
\end{corrige}

\begin{corrige}{TCOMPL-ATT-EV-B-EX2}
\textbf{1.} La densité est constante sur $[0\,;20]$ et nulle ailleurs ; pour que son intégrale vaille $1$, elle vaut $\dfrac{1}{20-0}=\dfrac1{20}$. Pour $t\in[0\,;20]$ :
\[
  F(t)=P(T\leq t)=\int_0^t \frac1{20}\,\mathrm{d}u=\frac{t}{20}.
\]

\textbf{2.} $P(T\leq5)=F(5)=\dfrac{5}{20}=\dfrac14=0{,}25$ et
$P(8\leq T\leq12)=F(12)-F(8)=\dfrac{12-8}{20}=\dfrac15=0{,}2$.

\textbf{3.} $E(T)=\dfrac{a+b}{2}=\dfrac{0+20}{2}=10$ minutes et
$V(T)=\dfrac{(b-a)^2}{12}=\dfrac{400}{12}=\dfrac{100}{3}\approx33{,}3$.
Un voyageur qui arrive au hasard attend en moyenne $10$ minutes, soit la moitié
de l'intervalle entre deux bus : aucune heure d'arrivée n'est privilégiée.
\end{corrige}
