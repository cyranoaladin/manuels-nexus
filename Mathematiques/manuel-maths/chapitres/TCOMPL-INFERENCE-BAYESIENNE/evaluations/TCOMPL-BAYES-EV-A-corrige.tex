\begin{corrige}{TCOMPL-BAYES-EV-A-EX1}
\textbf{1.} $P_B(A) = \dfrac{P_A(B)\times P(A)}{P(B)}$.

\textbf{2.} $P(B) = P(A)\times P_A(B) + P(\bar A)\times P_{\bar A}(B) = 0{,}03\times0{,}85+0{,}97\times0{,}04 = 0{,}0255+0{,}0388 = 0{,}0643$.

\textbf{3.} $P_B(A) = \dfrac{0{,}85\times0{,}03}{0{,}0643} = \dfrac{0{,}0255}{0{,}0643} \approx 0{,}397$.

\textbf{4.} $P_A(B)=0{,}85$ mesure la fiabilité du test chez les personnes \textbf{déjà connues} porteuses ; $P_B(A)\approx0{,}397$ mesure la probabilité d'être réellement porteur \textbf{après} un résultat positif, dans toute la population -- cette dernière est plombée par le nombre de faux positifs, très important car peu de gens sont réellement porteurs ($3\%$ seulement).
\end{corrige}
