\section*{\textcolor{chapcolor}{\MakeUppercase{Inference bayesienne et conditionnement}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q1]} La formule de Bayes s'ecrit $P(A|B) =$
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{P(B|A)P(A)}{P(B)}$
    \item $\frac{P(A|B)P(B)}{P(A)}$
    \item $P(A) P(B)$
    \item $P(A \cap B)$
  \end{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q2]} Si A et B sont independants, alors $P(A \cap B) =$
  \begin{enumerate}[label=\Alph*.]
    \item $P(A) + P(B)$
    \item $P(A|B)$
    \item $P(A) \times P(B)$
    \item $P(A)/P(B)$
  \end{enumerate}

\item \textbf{[Q3]} La formule des probabilites totales pour une partition $(A_k)$ s'ecrit $P(B) =$
  \begin{enumerate}[label=\Alph*.]
    \item $\sum P(A_k)$
    \item $\sum P(B|A_k) P(A_k)$
    \item $\prod P(B|A_k)$
    \item $P(B|A_1)$
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q4]} En inference bayesienne, la probabilite $P(A)$ avant de connaitre B s'appelle :
  \begin{enumerate}[label=\Alph*.]
    \item La probabilite a posteriori
    \item La vraisemblance
    \item La valeur p
    \item La probabilite a priori
  \end{enumerate}

\item[] \textbf{Capacite C4}

\item \textbf{[Q5]} Un test de sensibilite 95\% et specificite 90\% sur une maladie rare (1\%) donne une VPP :
  \begin{enumerate}[label=\Alph*.]
    \item Faible (inférieure à 10\%) car la maladie est rare
    \item Tres forte (95\%)
    \item Egale a 50\%
    \item Exactement 90\%
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des reponses erronees}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{B} — Tautologie sans inversion. La formule de Bayes est P(A|B)=P(B|A)P(A)/P(B), lorsque P(B)>0. \emph{Renvoi : C2.}
    \item \textbf{C} — Independance non supposee. La formule de Bayes est P(A|B)=P(B|A)P(A)/P(B), lorsque P(B)>0. \emph{Renvoi : C2.}
    \item \textbf{D} — Oubli de la division par P(B). La formule de Bayes est P(A|B)=P(B|A)P(A)/P(B), lorsque P(B)>0. \emph{Renvoi : C2.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — Formule d'evenements disjoints. L'indépendance donne P(A inter B)=P(A)P(B). \emph{Renvoi : C1.}
    \item \textbf{B} — Probabilite conditionnelle. L'indépendance donne P(A inter B)=P(A)P(B). \emph{Renvoi : C1.}
    \item \textbf{D} — Rapport incorrect. L'indépendance donne P(A inter B)=P(A)P(B). \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — La somme des P(A\_k) vaut 1. Pour une partition (A\_k), la formule totale est P(B)=somme\_k P(B|A\_k)P(A\_k). \emph{Renvoi : C1.}
    \item \textbf{C} — Produit au lieu d'une somme. Pour une partition (A\_k), la formule totale est P(B)=somme\_k P(B|A\_k)P(A\_k). \emph{Renvoi : C1.}
    \item \textbf{D} — Seul le premier terme considere. Pour une partition (A\_k), la formule totale est P(B)=somme\_k P(B|A\_k)P(A\_k). \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — P(A|B) est la probabilite a posteriori. La probabilité attribuée à A avant l'observation de B est la probabilité a priori. \emph{Renvoi : C3.}
    \item \textbf{B} — P(B|A) est la vraisemblance. La probabilité attribuée à A avant l'observation de B est la probabilité a priori. \emph{Renvoi : C3.}
    \item \textbf{C} — Terme de test d'hypothese. La probabilité attribuée à A avant l'observation de B est la probabilité a priori. \emph{Renvoi : C3.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{B} — Oubli de l'effet du biais de rarete (paradoxe du faux positif). La VPP vaut 0,95*0,01/(0,95*0,01+0,10*0,99), soit environ 8,8 \%, donc moins de 10 \%. \emph{Renvoi : C4.}
    \item \textbf{C} — Valeur arbitraire. La VPP vaut 0,95*0,01/(0,95*0,01+0,10*0,99), soit environ 8,8 \%, donc moins de 10 \%. \emph{Renvoi : C4.}
    \item \textbf{D} — Confusion avec la specificite. La VPP vaut 0,95*0,01/(0,95*0,01+0,10*0,99), soit environ 8,8 \%, donc moins de 10 \%. \emph{Renvoi : C4.}
  \end{itemize}
\end{itemize}
