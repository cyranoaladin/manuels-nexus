\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} La formule de Bayes s'ecrit $P(A|B) =$
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{P(B|A)P(A)}{P(B)}$
    \item $\frac{P(A|B)P(B)}{P(A)}$
    \item $P(A) P(B)$
    \item $P(A \cap B)$
  \end{enumerate}

\item \textbf{[Q2]} Si A et B sont independants, alors $P(A \cap B) =$
  \begin{enumerate}[label=\Alph*.]
    \item $P(A) + P(B)$
    \item $P(A|B)$
    \item $P(A) \times P(B)$
    \item $P(A)/P(B)$
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q3]} La formule des probabilites totales pour une partition $(A_k)$ s'ecrit $P(B) =$
  \begin{enumerate}[label=\Alph*.]
    \item $\sum P(A_k)$
    \item $\sum P(B|A_k) P(A_k)$
    \item $\prod P(B|A_k)$
    \item $P(B|A_1)$
  \end{enumerate}

\item \textbf{[Q4]} En inference bayesienne, la probabilite $P(A)$ avant de connaitre B s'appelle :
  \begin{enumerate}[label=\Alph*.]
    \item La probabilite a posteriori
    \item La vraisemblance
    \item La valeur p
    \item La probabilite a priori
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q5]} Un test de sensibilite 95% et specificite 90% sur une maladie rare (1%) donne une VPP :
  \begin{enumerate}[label=\Alph*.]
    \item Faible (inférieure à 10%) car la maladie est rare
    \item Tres forte (95%)
    \item Egale a 50%
    \item Exactement 90%
  \end{enumerate}

\end{enumerate}
