\section*{\textcolor{chapcolor}{\MakeUppercase{Inference bayesienne et conditionnement}}}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q1]} La formule de Bayes s'ecrit $P(A|B) =$
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{P(B|A)P(A)}{P(B)}$
    \item $\frac{P(A|B)P(B)}{P(A)}$
    \item $P(A) P(B)$
    \item $P(A \cap B)$
  \end{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q2]} Si A et B sont indépendants, alors $P(A \cap B) =$
  \begin{enumerate}[label=\Alph*.]
    \item $P(A) + P(B)$
    \item $P(A|B)$
    \item $P(A) \times P(B)$
    \item $P(A)/P(B)$
  \end{enumerate}

\item \textbf{[Q3]} La formule des probabilités totales pour une partition $(A_k)$ s'ecrit $P(B) =$
  \begin{enumerate}[label=\Alph*.]
    \item $\sum P(B|A_k) P(A_k)$
    \item $\sum P(A_k)$
    \item $\prod P(B|A_k)$
    \item $P(B|A_1)$
  \end{enumerate}

\item[] \textbf{Capacité C3}

\item \textbf{[Q4]} En inference bayesienne, la probabilité $P(A)$ avant de connaitre B s'appelle :
  \begin{enumerate}[label=\Alph*.]
    \item La probabilité a posteriori
    \item La probabilité a priori
    \item La valeur p
    \item La vraisemblance
  \end{enumerate}

\item[] \textbf{Capacité C4}

\item \textbf{[Q5]} Un test de sensibilite 95\% et specificite 90\% sur une maladie rare (1\%) donne une VPP :
  \begin{enumerate}[label=\Alph*.]
    \item Exactement 90\%
    \item Très forte (95\%)
    \item Égale a 50\%
    \item Faible (inférieure à 10\%) car la maladie est rare
  \end{enumerate}

\item[] \textbf{Capacité C5}

\item \textbf{[Q6]} Un test de dépistage a une sensibilité de $0{,}99$ et une spécificité de $0{,}99$. Lorsque la prévalence de la maladie passe de $10\%$ à $0{,}1\%$, sa valeur prédictive positive :
  \begin{enumerate}[label=\Alph*.]
    \item reste inchangée, car elle ne dépend que des qualités du test
    \item chute fortement, car les faux positifs deviennent majoritaires parmi les tests positifs
    \item augmente, car la maladie est devenue plus rare
    \item devient nulle
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des réponses erronées}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{B} — Tautologie sans inversion. La formule de Bayes est P(A|B)=P(B|A)P(A)/P(B), lorsque P(B)>0. \emph{Renvoi : C2.}
    \item \textbf{C} — Indépendance non supposee. La formule de Bayes est P(A|B)=P(B|A)P(A)/P(B), lorsque P(B)>0. \emph{Renvoi : C2.}
    \item \textbf{D} — Oubli de la division par P(B). La formule de Bayes est P(A|B)=P(B|A)P(A)/P(B), lorsque P(B)>0. \emph{Renvoi : C2.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — Formule d'événements disjoints. L'indépendance donne P(A inter B)=P(A)P(B). \emph{Renvoi : C1.}
    \item \textbf{B} — Probabilité conditionnelle. L'indépendance donne P(A inter B)=P(A)P(B). \emph{Renvoi : C1.}
    \item \textbf{D} — Rapport incorrect. L'indépendance donne P(A inter B)=P(A)P(B). \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{B} — La somme des P(A\_k) vaut 1. Pour une partition (A\_k), la formule totale est P(B)=somme\_k P(B|A\_k)P(A\_k). \emph{Renvoi : C1.}
    \item \textbf{C} — Produit au lieu d'une somme. Pour une partition (A\_k), la formule totale est P(B)=somme\_k P(B|A\_k)P(A\_k). \emph{Renvoi : C1.}
    \item \textbf{D} — Seul le premier terme considere. Pour une partition (A\_k), la formule totale est P(B)=somme\_k P(B|A\_k)P(A\_k). \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — P(A|B) est la probabilité a posteriori. La probabilité attribuée à A avant l'observation de B est la probabilité a priori. \emph{Renvoi : C3.}
    \item \textbf{C} — Terme de test d'hypothèse. La probabilité attribuée à A avant l'observation de B est la probabilité a priori. \emph{Renvoi : C3.}
    \item \textbf{D} — P(B|A) est la vraisemblance. La probabilité attribuée à A avant l'observation de B est la probabilité a priori. \emph{Renvoi : C3.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — Confusion avec la specificite. La VPP vaut 0,95*0,01/(0,95*0,01+0,10*0,99), soit environ 8,8 \%, donc moins de 10 \%. \emph{Renvoi : C4.}
    \item \textbf{B} — Oubli de l'effet du biais de rarete (paradoxe du faux positif). La VPP vaut 0,95*0,01/(0,95*0,01+0,10*0,99), soit environ 8,8 \%, donc moins de 10 \%. \emph{Renvoi : C4.}
    \item \textbf{C} — Valeur arbitraire. La VPP vaut 0,95*0,01/(0,95*0,01+0,10*0,99), soit environ 8,8 \%, donc moins de 10 \%. \emph{Renvoi : C4.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{A} — La formule de Bayes fait intervenir la prévalence : $\mathrm{VPP}=\frac{se\,p}{se\,p+(1-sp)(1-p)}$. Avec $p=0{,}1$ elle vaut $\frac{0{,}099}{0{,}108}\approx0{,}92$ ; avec $p=0{,}001$ elle vaut $\frac{0{,}00099}{0{,}01098}\approx0{,}09$. Sensibilité et spécificité seules ne déterminent pas la VPP. \emph{Renvoi : C5.}
    \item \textbf{C} — C'est l'inverse : plus la maladie est rare, plus les $1\%$ de faux positifs prélevés sur une population saine très nombreuse écrasent les vrais positifs. La VPP passe de $0{,}92$ à $0{,}09$. \emph{Renvoi : C5.}
    \item \textbf{D} — Elle diminue fortement mais reste strictement positive : $\frac{0{,}00099}{0{,}01098}\approx0{,}09$, soit environ un test positif sur onze qui correspond à un vrai malade. \emph{Renvoi : C5.}
  \end{itemize}
\end{itemize}
