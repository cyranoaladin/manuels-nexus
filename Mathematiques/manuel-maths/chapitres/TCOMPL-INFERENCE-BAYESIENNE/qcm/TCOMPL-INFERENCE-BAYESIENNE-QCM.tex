\subsection*{C2/C3 --- Formule de Bayes}

\begin{enumerate}

\item \textbf{[Q1]} La formule de Bayes s'écrit :
  \begin{enumerate}[label=\Alph*.]
    \item $P_B(A) = P_A(B)$.
    \item $P_B(A) = \dfrac{P_A(B)\times P(A)}{P(B)}$.
    \item $P_B(A) = P(A)+P(B)$.
    \item $P_B(A) = P(A)\times P(B)$.
    \end{enumerate}

\bigskip

\item \textbf{[Q2]} $P_B(A)$ et $P_A(B)$ sont :
  \begin{enumerate}[label=\Alph*.]
    \item toujours égales.
    \item en général différentes.
    \item toujours de somme $1$.
    \item toujours de produit $1$.
  \end{enumerate}

\subsection*{C1 --- Probabilités conditionnelles}

\item \textbf{[Q3]} La probabilité a priori d'un événement $A$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $P(A)$, avant toute observation.

\subsection*{C4 --- Tests de dépistage}

\item \textbf{[Q4]} La sensibilité d'un test mesure :
  \begin{enumerate}[label=\Alph*.]
    \item la probabilité d'être malade sachant le test positif.
    \item la probabilité que le test soit positif sachant que le patient est malade.
    \item la proportion de malades dans la population.
    \item la probabilité que le test soit négatif sachant le patient sain.
  \end{enumerate}

\bigskip

\subsection*{C5 --- VPP et prévalence}

\item \textbf{[Q5]} Pour un test de sensibilité et spécificité fixées, la valeur prédictive positive :
  \begin{enumerate}[label=\Alph*.]
    \item ne dépend pas de la prévalence.
    \item est une fonction croissante de la prévalence.
    \item est une fonction décroissante de la prévalence.
    \item vaut toujours $1$.
  \end{enumerate}

\bigskip

\item \textbf{[Q6]} Un test très sensible et spécifique appliqué à une maladie très rare donne typiquement :
  \begin{enumerate}[label=\Alph*.]
    \item une VPP très élevée, proche de $1$.
    \item une VPP potentiellement faible, à cause du grand nombre de faux positifs en valeur absolue.
    \item une VPP toujours égale à la sensibilité.
    \item aucun faux positif possible.
  \end{enumerate}

\end{enumerate}
