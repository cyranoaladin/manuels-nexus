\begin{corrige}{TCOMPL-BAYES-EX-001}
\textbf{1.} Par la formule des probabilités totales (système complet $A$, $\bar A$) :
\[
  P(B) = P(A)\times P_A(B) + P(\bar A)\times P_{\bar A}(B) = 0{,}05\times0{,}8 + 0{,}95\times0{,}1 = 0{,}04+0{,}095 = 0{,}135 = \frac{27}{200}.
\]

\textbf{2.} Par la formule de Bayes :
\[
  P_B(A) = \frac{P_A(B)\times P(A)}{P(B)} = \frac{0{,}8\times0{,}05}{0{,}135} = \frac{0{,}04}{0{,}135} = \frac{8}{27} \approx 0{,}296.
\]
Un individu testé positif a donc environ $29{,}6\%$ de chances de porter réellement le gène.
\end{corrige}
