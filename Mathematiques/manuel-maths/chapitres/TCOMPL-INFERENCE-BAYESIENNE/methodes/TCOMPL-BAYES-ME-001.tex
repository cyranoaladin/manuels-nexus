\begin{fichemethode}{M1}{Calculer avec des probabilités conditionnelles et un arbre pondere}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé fournit des probabilités
  conditionnelles ($P_A(B)$, « sachant que... ») et demande de calculer la
  probabilité d'une intersection ou d'un événement global : arbre pondere et
  formule des probabilités totales.
}

\pasApas{%
  \begin{enumerate}
    \item Nommer les événements et construire l'arbre : premier niveau $A$,
      $\overline{A}$ ; second niveau $B$, $\overline{B}$ conditionne par la
      branche précédente.
    \item Reporter les poids : $P(A)$ sur la première branche, $P_A(B)$ sur la
      branche suivante — chaque nœud a des branches de somme 1.
    \item Intersection : multiplier le long du chemin,
      $P(A \cap B) = P(A) \times P_A(B)$.
    \item Probabilités totales : sommer tous les chemins menant a $B$,
      $P(B) = P(A)\,P_A(B) + P(\overline{A})\,P_{\overline{A}}(B)$.
    \item Contrôler : chaque résultat est dans $[0\,;\,1]$ et les branches issues
      d'un même nœud totalisent 1.
  \end{enumerate}
}

\exempleRedige{%
  On donne $P(A) = 0{,}3$, $P_A(B) = 0{,}6$ et $P_{\overline{A}}(B) = 0{,}2$.
  Calculer $P(A \cap B)$ puis $P(B)$.

  \medskip
  \commentaireMarge{Multiplier le long d'un chemin de l'arbre.}%
  $P(A \cap B) = P(A) \times P_A(B) = 0{,}3 \times 0{,}6 = 0{,}18$.

  \medskip
  \commentaireMarge{Sommer les deux chemins menant a $B$.}%
  $P(B) = P(A)\,P_A(B) + P(\overline{A})\,P_{\overline{A}}(B)
  = 0{,}3 \times 0{,}6 + 0{,}7 \times 0{,}2 = 0{,}18 + 0{,}14 = 0{,}32$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Confondre $P(A \cap B)$ (chemin complet) et $P_A(B)$
      (poids d'une seule branche).
    \item \textbf{Piège 2.} Oublier un chemin dans la formule des probabilités
      totales : $B$ peut se realiser via $A$ ET via $\overline{A}$.
    \item \textbf{Piège 3.} Écrire $P(\overline{A}) = P(A)$ : c'est
      $1 - P(A)$.
  \end{itemize}
}

\verifier{%
  Vérifier que les branches issues de chaque nœud somment a 1, que
  $P(A \cap B) \leq \min\bigl(P(A), P(B)\bigr)$, et que $P(B)$ est bien entre les
  deux probabilités conditionnelles utilisees.
}

\sEntrainer{\refExos{M1}}

\end{fichemethode}
