\begin{fichemethode}{M1}{Reconnaître et utiliser une loi binomiale}

\quandUtiliser{%
  Utiliser cette méthode lorsqu'une situation repete une même epreuve a deux issues, de facon indépendante.
}

\pasApas{%
  \begin{enumerate}
    \item \textbf{Vérifier les conditions du schema de Bernoulli} : $n$ repetitions indépendantes d'une même epreuve a deux issues (succès/echec).
    \item \textbf{Identifier les paramètres} $n$ (nombre de repetitions) et $p$ (probabilité de succès a chaque epreuve).
    \item \textbf{Écrire} $X \sim \mathcal{B}(n,p)$, ou $X$ compte le nombre de succès.
    \item \textbf{Calculer} $P(X=k) = \dbinom{n}{k}p^k(1-p)^{n-k}$, ou une probabilité cumulee, a l'aide de la calculatrice/d'un algorithme si nécessaire.
    \item \textbf{Pour un seuil} : chercher $k$ par balayage (calculatrice/algorithme) jusqu'a atteindre la condition demandee.
  \end{enumerate}
}

\end{fichemethode}
