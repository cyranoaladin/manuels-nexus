\begin{exercice}{TSPE-PROBA-EX-008}{3}{18}
$X \sim \mathcal{B}(40,\,0{,}25)$.
\begin{enumerate}
  \item En redigeant la demonstration du cours (decomposition en variables de Bernoulli, linearite de l'esperance), etablir la valeur de $E(X)$.
  \item De meme, en utilisant l'additivite de la variance (justifier l'independance), etablir la valeur de $V(X)$.
  \item Application numerique : calculer $E(X)$ et $V(X)$.
\end{enumerate}
\end{exercice}
