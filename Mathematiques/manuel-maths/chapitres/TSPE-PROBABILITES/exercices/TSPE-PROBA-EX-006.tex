\begin{exercice}{TSPE-PROBA-EX-006}{2}{12}
On lance $2$ des equilibres a $6$ faces. $X$ est la somme des points obtenus.
\begin{enumerate}
  \item Ecrire $X$ comme somme de deux variables aleatoires plus simples $X_1$ et $X_2$ que l'on precisera.
  \item Calculer $E(X_1)$.
  \item En deduire $E(X)$ en utilisant la linearite de l'esperance.
\end{enumerate}
\end{exercice}
