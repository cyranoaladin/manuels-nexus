\begin{corrige}{TSPE-PROBA-EX-003}

\textbf{Q1.} Les $10$ tirs sont des repetitions independantes d'une meme epreuve a deux issues (touche/rate), de probabilite de succes constante $0{,}3$ : c'est un schema de Bernoulli, donc $X \sim \mathcal{B}(10,\,0{,}3)$.

\textbf{Q2.} $P(X=0) = \dbinom{10}{0}(0{,}3)^0(0{,}7)^{10} = (0{,}7)^{10} \approx 0{,}0282$.

\textbf{Q3.} $P(X\geq 1) = 1-P(X=0) \approx 1-0{,}0282 = \boxed{0{,}9718}$.

\end{corrige}
