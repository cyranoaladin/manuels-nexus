\begin{corrige}{TSPE-PROBA-EX-006}

\textbf{Q1.} $X = X_1+X_2$, ou $X_1$ (resp. $X_2$) est le resultat du premier (resp. second) de, chacun suivant la loi uniforme sur $\{1,2,3,4,5,6\}$.

\textbf{Q2.} $E(X_1) = \dfrac{1+2+3+4+5+6}{6} = \dfrac{21}{6} = \dfrac{7}{2}$.

\textbf{Q3.} Par linearite de l'esperance : $E(X) = E(X_1)+E(X_2) = \dfrac{7}{2}+\dfrac{7}{2} = \boxed{7}$.

\end{corrige}
