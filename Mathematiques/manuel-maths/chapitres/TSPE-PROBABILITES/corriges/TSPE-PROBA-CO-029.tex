\begin{corrige}{TSPE-PROBA-EX-002}

\textbf{Q1.} Les $8$ lancers sont des repetitions independantes d'une meme epreuve a deux issues (Face/Pile) : c'est un schema de Bernoulli. $X$ suit donc la loi binomiale $\mathcal{B}(8,\,0{,}5)$ (avec $p=0{,}5$ la probabilite d'obtenir Face a un lancer).

\textbf{Q2.} $P(X=3) = \dbinom{8}{3}\times(0{,}5)^3\times(0{,}5)^5$.

\end{corrige}
