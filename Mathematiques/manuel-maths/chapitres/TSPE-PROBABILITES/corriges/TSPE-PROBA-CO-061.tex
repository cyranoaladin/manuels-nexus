\begin{corrige}{TSPE-PROBA-EX-061}
\textbf{1.} Le prélèvement est assimilé à $n$ épreuves de Bernoulli
indépendantes de paramètre $p=0{,}04$, et $X$ compte les succès :
$X\sim\mathcal{B}(n\,;\,0{,}04)$.

\textbf{2.} $P(X=0)=0{,}96^{50}\approx0{,}1299$.

\textbf{3.} $P(X\geq1)=1-0{,}96^{\,n}>0{,}95$ équivaut à
$0{,}96^{\,n}<0{,}05$, soit $n\ln0{,}96<\ln0{,}05$. Comme $\ln0{,}96<0$, on
obtient $n>\dfrac{\ln0{,}05}{\ln0{,}96}\approx73{,}4$ : le plus petit entier
est $n=74$. Vérification : $0{,}96^{73}\approx0{,}0508>0{,}05$ et
$0{,}96^{74}\approx0{,}0488<0{,}05$.
\end{corrige}
