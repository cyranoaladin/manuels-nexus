\section*{\textcolor{chapcolor}{\MakeUppercase{QCM d'auto-évaluation -- Probabilités}}}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacité C5}

\item \textbf{[Q1]} Si X suit B(n,p), alors P(X=k) vaut :
  \begin{enumerate}[label=\Alph*.]
    \item C(n,k) p\textasciicircum{}k (1-p)\textasciicircum{}(n-k)
    \item p\textasciicircum{}k (1-p)\textasciicircum{}(n-k)
    \item C(n,k) p\textasciicircum{}k
    \item np\textasciicircum{}k
  \end{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q2]} Un schema de Bernoulli suppose que les epreuves sont :
  \begin{enumerate}[label=\Alph*.]
    \item dependantes et identiques
    \item indépendantes et identiques
    \item indépendantes et différentes
    \item toutes de probabilité 1/2
  \end{enumerate}

\item[] \textbf{Capacité C7}

\item \textbf{[Q3]} La linéarité de l'espérance E(X+Y)=E(X)+E(Y) est valable :
  \begin{enumerate}[label=\Alph*.]
    \item toujours, sans condition d'independence
    \item uniquement si X et Y sont indépendantes
    \item uniquement si X=Y
    \item jamais
  \end{enumerate}

\item[] \textbf{Capacité C8}

\item \textbf{[Q4]} L'additivite de la variance V(X+Y)=V(X)+V(Y) est garantie lorsque :
  \begin{enumerate}[label=\Alph*.]
    \item que X=Y
    \item que X et Y soient indépendantes
    \item rien de particulier
    \item que E(X)=E(Y)
  \end{enumerate}

\item[] \textbf{Capacité C9}

\item \textbf{[Q5]} Pour X \textasciitilde{} B(n,p), V(X) vaut :
  \begin{enumerate}[label=\Alph*.]
    \item np
    \item n(1-p)
    \item np(1-p)
    \item p(1-p)
  \end{enumerate}

\item[] \textbf{Capacité C10}

\item \textbf{[Q6]} L'inégalité de Bienayme-Tchebychev majoré :
  \begin{enumerate}[label=\Alph*.]
    \item P(Y=mu)
    \item E(Y)
    \item V(Y)
    \item P(|Y-mu|>=delta)
  \end{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q7]} On répète trois fois, de façon indépendante, une épreuve dont la probabilité de succès est $0{,}2$. La probabilité d'obtenir au moins un succès vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $0{,}2$
    \item $0{,}6$
    \item $0{,}488$
    \item $0{,}512$
  \end{enumerate}

\item[] \textbf{Capacité C3}

\item \textbf{[Q8]} Une machine produit $2\%$ de pièces défectueuses, indépendamment les unes des autres. Le plus petit nombre $n$ de pièces à prélever pour que la probabilité d'en obtenir au moins une défectueuse atteigne $0{,}5$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $n=25$
    \item $n=50$
    \item $n=2$
    \item $n=35$
  \end{enumerate}

\item[] \textbf{Capacité C4}

\item \textbf{[Q9]} Pour $X\sim\mathcal B(50;0{,}3)$, déterminer le plus petit intervalle $[a;b]$ tel que $P(a\leq X\leq b)\geq0{,}95$ se fait :
  \begin{enumerate}[label=\Alph*.]
    \item en cumulant numériquement les $P(X=k)$ autour de l'espérance, à la calculatrice ou en Python
    \item en résolvant une équation du second degré
    \item en prenant $[0;50]$, le seul intervalle sûr
    \item en prenant $E(X)\pm\sigma$
  \end{enumerate}

\item[] \textbf{Capacité C6}

\item \textbf{[Q10]} Pour calculer l'espérance du nombre $X$ de faces obtenues en dix lancers d'une pièce équilibrée, la décomposition utile est :
  \begin{enumerate}[label=\Alph*.]
    \item $X=10X_1$
    \item $X=X_1+\dots+X_{10}$, où $X_i$ vaut $1$ si le $i$-ème lancer donne face et $0$ sinon
    \item $X=\max(X_1;\dots;X_{10})$
    \item aucune décomposition n'est possible : il faut la loi complète de $X$
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des réponses erronées}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{B} — Cette expression donne la probabilité d'une configuration précise de $k$ succès ; elle oublie les $C(n,k)$ positions possibles des succès. \emph{Renvoi : C5.}
    \item \textbf{C} — Le facteur $(1-p)^{n-k}$ manque : les $n-k$ echecs de chaque configuration doivent eux aussi contribuer a sa probabilité. \emph{Renvoi : C5.}
    \item \textbf{D} — Le produit $np^k$ ne compte ni les choix des $k$ succès ni la probabilité des $n-k$ echecs ; ce n'est pas une probabilité binomiale. \emph{Renvoi : C5.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — La dépendance contredit la définition du schema : la probabilité d'un succès doit rester $p$ d'une epreuve a l'autre. \emph{Renvoi : M1, hypothèses du schéma de Bernoulli.}
    \item \textbf{C} — Des epreuves différentes peuvent avoir des probabilités de succès distinctes ; un schema de Bernoulli repete la même epreuve. \emph{Renvoi : M1, hypothèses du schéma de Bernoulli.}
    \item \textbf{D} — La probabilité de succès est un réel $p$ quelconque de $[0,1]$ ; la valeur $1/2$ n'est imposee que pour une piece equilibree. \emph{Renvoi : M1, hypothèses du schéma de Bernoulli.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{B} — L'indépendance est requise pour additionner les variances, pas pour la linéarité de l'espérance de variables integrables. \emph{Renvoi : C7, linéarité de l'espérance.}
    \item \textbf{C} — L'égalité $X=Y$ n'est pas nécessaire : la somme des esperances vaut l'espérance de la somme pour deux variables quelconques. \emph{Renvoi : C7, linéarité de l'espérance.}
    \item \textbf{D} — La propriété est toujours valable ; par exemple pour deux constantes $X=1$ et $Y=2$, les deux membres valent $3$. \emph{Renvoi : C7, linéarité de l'espérance.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — Si $X=Y$, alors $V(X+Y)=V(2X)=4V(X)$, tandis que $V(X)+V(Y)=2V(X)$ : l'égalité n'est donc pas garantie. \emph{Renvoi : C8, variance d'une somme et indépendance.}
    \item \textbf{C} — Sans hypothèse, la formule comporte le terme $2\operatorname{Cov}(X,Y)$ : l'indépendance permet precisement d'annuler cette covariance. \emph{Renvoi : C8, variance d'une somme et indépendance.}
    \item \textbf{D} — L'égalité des esperances ne renseigne pas sur la covariance de $X$ et $Y$ : elle ne suffit donc pas a rendre les variances additives. \emph{Renvoi : C8, variance d'une somme et indépendance.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — Le produit $np$ est l'espérance de la loi binomiale ; la variance comporte aussi le facteur de dispersion $1-p$. \emph{Renvoi : C9, espérance et variance de la loi binomiale.}
    \item \textbf{B} — L'expression $n(1-p)$ oublie le facteur $p$ associe aux succès ; chaque indicatrice de Bernoulli a pour variance $p(1-p)$. \emph{Renvoi : C9, espérance et variance de la loi binomiale.}
    \item \textbf{D} — Le produit $p(1-p)$ est la variance d'une seule epreuve de Bernoulli ; la somme de $n$ epreuves indépendantes apporte le facteur $n$. \emph{Renvoi : C9, espérance et variance de la loi binomiale.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{A} — Bienayme-Tchebychev concerne l'événement $|Y-\mu|\geq\delta$, pas la probabilité ponctuelle que $Y$ soit exactement égal a $\mu$. \emph{Renvoi : C10, inégalité de Bienaymé-Tchebychev.}
    \item \textbf{B} — L'espérance $E(Y)$ est le centre de l'événement de deviation ; l'inégalité ne cherche pas a majorer cette valeur. \emph{Renvoi : C10, inégalité de Bienaymé-Tchebychev.}
    \item \textbf{C} — La variance intervient dans le majorant $V(Y)/\delta^2$, mais la quantité majorée est la probabilité d'un ecart a la moyenne. \emph{Renvoi : C10, inégalité de Bienaymé-Tchebychev.}
  \end{itemize}
  \item \textbf{Q7}
  \begin{itemize}
    \item \textbf{A} — $0{,}2$ est la probabilité de succès d'\textbf{une} épreuve. Sur trois répétitions, on passe par l'événement contraire : $1-0{,}8^3=0{,}488$. \emph{Renvoi : C1.}
    \item \textbf{B} — $3\times0{,}2=0{,}6$ additionne les probabilités de trois événements qui ne sont pas incompatibles : les succès peuvent se produire ensemble, et la somme surestime. \emph{Renvoi : C1.}
    \item \textbf{D} — $0{,}512=0{,}8^3$ est la probabilité de n'obtenir \textbf{aucun} succès. La réponse cherchée en est le complément : $1-0{,}512=0{,}488$. \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q8}
  \begin{itemize}
    \item \textbf{A} — Avec $n=25$, $1-0{,}98^{25}\approx0{,}397$, encore inférieur à $0{,}5$. Il faut résoudre $0{,}98^{\,n}\leq0{,}5$, soit $n\geq\tfrac{\ln0{,}5}{\ln0{,}98}\approx34{,}3$. \emph{Renvoi : C3.}
    \item \textbf{B} — $n=50$ convient ($\approx0{,}636$) mais ce n'est pas le plus petit : dès $n=35$ la probabilité vaut environ $0{,}507$. \emph{Renvoi : C3.}
    \item \textbf{C} — Avec $n=2$, $1-0{,}98^2\approx0{,}040$ : la probabilité croît lentement, il faut des dizaines de pièces pour atteindre une chance sur deux. \emph{Renvoi : C3.}
  \end{itemize}
  \item \textbf{Q9}
  \begin{itemize}
    \item \textbf{B} — $P(X=k)=\binom{50}{k}0{,}3^k0{,}7^{50-k}$ n'est pas polynomiale du second degré en $k$ : aucune résolution algébrique fermée ne donne les bornes, on les cherche numériquement. \emph{Renvoi : C4.}
    \item \textbf{C} — $[0;50]$ a bien une probabilité de $1$, mais l'énoncé demande le \textbf{plus petit} intervalle : c'est la minimalité qui fait l'intérêt de la recherche. \emph{Renvoi : C4.}
    \item \textbf{D} — Un intervalle d'un écart-type autour de la moyenne couvre environ $68\%$ des cas dans le modèle normal, très loin des $95\%$ demandés. \emph{Renvoi : C4.}
  \end{itemize}
  \item \textbf{Q10}
  \begin{itemize}
    \item \textbf{A} — Cette écriture ferait de $X$ une variable ne prenant que les valeurs $0$ et $10$ : elle suppose les dix lancers identiques au premier, alors qu'ils sont indépendants et peuvent différer. \emph{Renvoi : C6.}
    \item \textbf{C} — Le maximum vaut $1$ dès qu'une face apparaît : il détecte l'existence d'un succès, il n'en compte pas le nombre. \emph{Renvoi : C6.}
    \item \textbf{D} — C'est justement ce que la linéarité de l'espérance permet d'éviter : $E(X)=\sum E(X_i)=10\times\tfrac12=5$, sans écrire un seul coefficient binomial. \emph{Renvoi : C6.}
  \end{itemize}
\end{itemize}
