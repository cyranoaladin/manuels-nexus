  \begin{enumerate}[label=\Alph*.]

    \item independantes et identiques

  \end{enumerate}

\item[] \textbf{Capacite C7}

\item \textbf{[Q3]} La linearite de l'esperance E(X+Y)=E(X)+E(Y) est valable :
  \begin{enumerate}[label=\Alph*.]
    \item uniquement si X et Y sont independantes
    \item toujours, sans condition d'independence
    \item uniquement si X=Y
    \item jamais
  \end{enumerate}

\item \textbf{[Q4]} L'additivite de la variance V(X+Y)=V(X)+V(Y) necessite :
  \begin{enumerate}[label=\Alph*.]
    \item que X et Y soient independantes
    \item que X=Y
    \item rien de particulier
    \item que E(X)=E(Y)
  \end{enumerate}

\item[] \textbf{Capacite C8}

\item \textbf{[Q5]} Pour X ~ B(n,p), V(X) vaut :

    \item n(1-p)
    \item p(1-p)
  \end{enumerate}

\item[] \textbf{Capacite C10}

\item \textbf{[Q6]} L'inégalité de Bienayme-Tchebychev majore :

    \item P(|Y-mu|>=delta)
    \item E(Y)
    \item V(Y)
    \item P(Y=mu)
  \end{enumerate}
