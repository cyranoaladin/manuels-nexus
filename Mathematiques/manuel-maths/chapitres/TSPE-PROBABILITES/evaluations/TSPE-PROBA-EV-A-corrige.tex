\textbf{Q1.} Premier tirage : $P(R)=\dfrac{5}{8}$, $P(B)=\dfrac{3}{8}$. Deuxieme tirage (sans remise) : si $R$ au premier tirage, $P_R(B)=\dfrac{3}{7}$ ; si $B$, $P_B(R)=\dfrac{5}{7}$.

\textbf{Q2.} $P(\text{couleurs differentes}) = P(R_1)\times P_{R_1}(B_2) + P(B_1)\times P_{B_1}(R_2) = \dfrac{5}{8}\times\dfrac{3}{7}+\dfrac{3}{8}\times\dfrac{5}{7} = \dfrac{15}{56}+\dfrac{15}{56} = \boxed{\dfrac{15}{28}}$.
\end{corrige}
