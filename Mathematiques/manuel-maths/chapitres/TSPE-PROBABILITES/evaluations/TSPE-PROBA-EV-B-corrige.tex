\begin{corrige}{TSPE-PROBA-EV-B-EX1}
\textbf{Q1.} $X \sim \mathcal{B}(15,\,0{,}1)$.

\textbf{Q2.} $P(X=0) = (0{,}9)^{15} \approx 0{,}2059$. $P(X\geq1) = 1-P(X=0) \approx \boxed{0{,}7941}$.
\end{corrige}

\begin{corrige}{TSPE-PROBA-EV-B-EX2}
\textbf{Q1.} $E(Y_1)=p=0{,}3$. $V(Y_1)=p(1-p)=0{,}3\times0{,}7=0{,}21$.

\textbf{Q2.} Chaque $Y_i$ vaut $1$ si la machine $i$ tombe en panne et $0$ sinon. La somme $Y_1+\cdots+Y_4$ ajoute donc un $1$ par machine en panne et un $0$ par machine intacte : elle compte exactement le nombre de machines tombees en panne, c'est-a-dire $X$. D'ou $X=Y_1+\cdots+Y_4$.

\textbf{Q3.} Par linéarité de l'espérance (valable \textbf{sans} hypothèse d'indépendance) : $E(X)=E(Y_1)+\cdots+E(Y_4) = 4\times0{,}3 = \boxed{1{,}2}$.

\textbf{Q4.} Les $4$ machines sont indépendantes (donne dans l'énoncé), donc l'additivite de la variance s'applique : $V(Y_1+\cdots+Y_4) = 4\times0{,}21 = \boxed{0{,}84}$.
\end{corrige}

\begin{corrige}{TSPE-PROBA-EV-B-EX3}
On veut $\dfrac{1}{4n\delta^2} \leq \alpha$, soit $n \geq \dfrac{1}{4\delta^2\alpha} = \dfrac{1}{4\times(0{,}1)^2\times0{,}05} = \dfrac{1}{0{,}002} = 500$.

Un échantillon de taille $\boxed{n=500}$ suffit.
\end{corrige}
