\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} Le taux d'variation de la fonction $f$ entre $a$ et $b$ ($a 
eq b$) est égal à :
  \begin{enumerate}[label=\Alph*.]
    \item $\dfrac{f(b)}{b}$
    \item $\dfrac{f(b) - f(a)}{b - a}$
    \item $f(b) - f(a)$
    \item $\dfrac{b - a}{f(b) - f(a)}$
  \end{enumerate}

\item \textbf{[Q2]} Pour $h 
eq 0$, le taux d'accroissement de $f$ en $a$ s'écrit :
  \begin{enumerate}[label=\Alph*.]
    \item $f(a+h) - f(a)$
    \item $\dfrac{f(a+h) - f(a)}{h}$
    \item $\dfrac{h}{f(a+h) - f(a)}$
    \item $h \times [f(a+h) - f(a)]$
  \end{enumerate}

\item \textbf{[Q3]} Si $d(t)$ désigne la position d'un mobile à l'instant $t$, la vitesse moyenne entre $t_1$ et $t_2$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $d(t_2) - d(t_1)$
    \item $d'(t_1)$
    \item $\dfrac{d(t_2) - d(t_1)}{t_2 - t_1}$
    \item $d''(t_1)$
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q4]} Par définition, le nombre dérivé $f'(a)$ est la limite quand $h \to 0$ de :
  \begin{enumerate}[label=\Alph*.]
    \item $\dfrac{f(a+h) - f(a)}{h}$
    \item $\dfrac{f(x) - f(a)}{x}$
    \item $f(a)$
    \item $f(a+h)$
  \end{enumerate}

\item \textbf{[Q5]} Pour $f(x) = x^2$, le taux d'accroissement en $a=2$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $4 + h$
    \item $4 + h^2$
    \item $4$
    \item $4h$
  \end{enumerate}

\item \textbf{[Q6]} La vitesse instantanée à l'instant $t=2$ correspond à :
  \begin{enumerate}[label=\Alph*.]
    \item $d(2)$
    \item $d'(2)$
    \item $\dfrac{d(2)}{2}$
    \item $d''(2)$
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q7]} La pente de la droite passant par $A(x_A, y_A)$ et $B(x_B, y_B)$ ($x_A 
eq x_B$) vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $\dfrac{y_B}{x_B}$
    \item $\dfrac{y_B - y_A}{x_B - x_A}$
    \item $y_B - y_A$
    \item $\dfrac{x_B - x_A}{y_B - y_A}$
  \end{enumerate}

\item \textbf{[Q8]} La courbe représentative de $f$ admet une tangente horizontale au point d'abscisse $a$ si :
  \begin{enumerate}[label=\Alph*.]
    \item $f(a) = 0$
    \item $f'(a) = 0$
    \item $f'(a) = 1$
    \item $f(a) = f'(a)$
  \end{enumerate}

\item \textbf{[Q9]} Géométriquement, le nombre dérivé $f'(a)$ représente :
  \begin{enumerate}[label=\Alph*.]
    \item l'ordonnée à l'origine de la tangente
    \item l'abscisse du point de contact
    \item le coefficient directeur de la tangente en $a$
    \item la longueur du segment sécant
  \end{enumerate}

\item[] \textbf{Capacite C4}

\item \textbf{[Q10]} L'équation réduite de la tangente à la courbe de $f$ au point d'abscisse $a$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $y = f'(a)x + f(a)$
    \item $y = f'(a)(x - a) + f(a)$
    \item $y = f(a)(x - a) + f'(a)$
    \item $y = f'(x)(x - a) + f(a)$
  \end{enumerate}

\item \textbf{[Q11]} Si $f(1)=3$ et $f'(1)=-2$, l'équation de la tangente en $a=1$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $y = -2x + 3$
    \item $y = -2x + 5$
    \item $y = 3x - 2$
    \item $y = -2x - 1$
  \end{enumerate}

\item \textbf{[Q12]} Une droite d'équation $y = mx + p$ passe par $(a, f(a))$ avec pour pente $f'(a)$. Son ordonnée à l'origine $p$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $f(0)$
    \item $f(a) - a f'(a)$
    \item $0$
    \item $f'(a)$
  \end{enumerate}

\item[] \textbf{Capacite C5}

\item \textbf{[Q13]} L'approximation affine de $f(a+h)$ pour $h$ voisin de 0 est donnée par :
  \begin{enumerate}[label=\Alph*.]
    \item $f(a+h) \approx f'(a) h$
    \item $f(a+h) \approx f(a) + f'(a) h$
    \item $f(a+h) \approx f'(a) + f(a) h$
    \item $f(a+h) \approx (f(a) + f'(a)) h$
  \end{enumerate}

\item \textbf{[Q14]} L'approximation affine locale est d'autant plus précise que :
  \begin{enumerate}[label=\Alph*.]
    \item $|h|$ est grand
    \item $|h|$ est très proche de 0
    \item $f'(a) = 0$
    \item $a = 0$
  \end{enumerate}

\item \textbf{[Q15]} Pour $f(x)=\sqrt{x}$, comme $f(4)=2$ et $f'(4)=\dfrac{1}{4}$, une valeur approchée de $\sqrt{4{,}06}$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $2{,}06$
    \item $2{,}015$
    \item $0{,}015$
    \item $2{,}15$
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des reponses erronees}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{A} — Quotient $f(b)/b$ au lieu de $(f(b)-f(a))/(b-a)$. \emph{Renvoi : M1.}
    \item \textbf{C} — Numérateur seul, dénominateur oublié. \emph{Renvoi : M1.}
    \item \textbf{D} — Quotient inversé. \emph{Renvoi : M1.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — Accroissement de $f$ seul, sans diviser par $h$. \emph{Renvoi : M1.}
    \item \textbf{C} — Quotient inversé. \emph{Renvoi : M1.}
    \item \textbf{D} — Produit au lieu d'un quotient. \emph{Renvoi : M1.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — Distance parcourue confondue avec vitesse moyenne. \emph{Renvoi : M1, R5.}
    \item \textbf{B} — Vitesse en un instant confondue avec vitesse moyenne. \emph{Renvoi : M1, M2.}
    \item \textbf{D} — Accélération confondue avec vitesse. \emph{Renvoi : M2.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{B} — Pente de sécante confondue avec pente de tangente. \emph{Renvoi : M2.}
    \item \textbf{C} — $f(a)$ est l'ordonnée, pas la pente. \emph{Renvoi : M2.}
    \item \textbf{D} — Le nombre dérivé n'est pas une valeur de $f$. \emph{Renvoi : M2.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{B} — Développement incorrect de $(2+h)^2$. \emph{Renvoi : R2.}
    \item \textbf{C} — Erreur d'identification du terme 4. \emph{Renvoi : R2.}
    \item \textbf{D} — Produit $4h$ au lieu de somme $4+h$. \emph{Renvoi : R2.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{A} — $d(2)$ est la position, pas la vitesse. \emph{Renvoi : M2.}
    \item \textbf{C} — Vitesse moyenne confondue avec vitesse instantanée. \emph{Renvoi : M1, M2.}
    \item \textbf{D} — Accélération = dérivée de la vitesse. \emph{Renvoi : M2.}
  \end{itemize}
  \item \textbf{Q7}
  \begin{itemize}
    \item \textbf{A} — Quotient $y_B/x_B$ au lieu de pente. \emph{Renvoi : R3.}
    \item \textbf{C} — Différence d'ordonnées sans diviser par $\Delta x$. \emph{Renvoi : R3.}
    \item \textbf{D} — Quotient inversé. \emph{Renvoi : R3.}
  \end{itemize}
  \item \textbf{Q8}
  \begin{itemize}
    \item \textbf{A} — $f(a)=0$ signifie point sur l'axe, pas tangente horizontale. \emph{Renvoi : M3.}
    \item \textbf{C} — Pente 1 = droite à 45°, pas horizontale. \emph{Renvoi : R3.}
    \item \textbf{D} — Aucune raison d'égalité $f(a) = f'(a)$. \emph{Renvoi : M3.}
  \end{itemize}
  \item \textbf{Q9}
  \begin{itemize}
    \item \textbf{A} — Sans la pente, la tangente n'est pas déterminée. \emph{Renvoi : M3.}
    \item \textbf{B} — Sans le point de contact, la tangente n'est pas positionnée. \emph{Renvoi : M3, M4.}
    \item \textbf{D} — La tangente n'est pas une sécante. \emph{Renvoi : M3.}
  \end{itemize}
  \item \textbf{Q10}
  \begin{itemize}
    \item \textbf{A} — Ordonnée à l'origine confondue avec $f(a)$ sauf si $a=0$. \emph{Renvoi : M4.}
    \item \textbf{C} — Rôles de $f(a)$ et $f'(a)$ inversés. \emph{Renvoi : M4.}
    \item \textbf{D} — $f'(x)$ varie avec $x$, la pente est $f'(a)$. \emph{Renvoi : M4.}
  \end{itemize}
  \item \textbf{Q11}
  \begin{itemize}
    \item \textbf{A} — $f'(a)x + f(a)$ au lieu de $f'(a)(x-a)+f(a)$. \emph{Renvoi : M4.}
    \item \textbf{C} — Rôles de $f(a)$ et $f'(a)$ inversés. \emph{Renvoi : M4.}
    \item \textbf{D} — Ordonnée à l'origine mal calculée. \emph{Renvoi : M4.}
  \end{itemize}
  \item \textbf{Q12}
  \begin{itemize}
    \item \textbf{A} — $y(0)=f(0)$ n'est vrai que si $a=0$. \emph{Renvoi : M4.}
    \item \textbf{C} — La tangente passe par $(a, f(a))$, pas par $(a, 0)$. \emph{Renvoi : M4.}
    \item \textbf{D} — $y(a)$ doit valoir $f(a)$, pas $f'(a)$. \emph{Renvoi : M4.}
  \end{itemize}
  \item \textbf{Q13}
  \begin{itemize}
    \item \textbf{A} — Terme $f(a)$ oublié. \emph{Renvoi : M5.}
    \item \textbf{C} — Rôles de $f(a)$ et $f'(a)$ inversés. \emph{Renvoi : M5.}
    \item \textbf{D} — Somme $f(a)+f'(a)$ multipliée par $h$. \emph{Renvoi : M5.}
  \end{itemize}
  \item \textbf{Q14}
  \begin{itemize}
    \item \textbf{A} — Plus $h$ est grand, plus l'approximation est mauvaise. \emph{Renvoi : M5.}
    \item \textbf{C} — $f'(a)=0$ donne $f(a+h) \approx f(a)$, pas meilleur. \emph{Renvoi : M5.}
    \item \textbf{D} — La qualité ne dépend pas de $a=0$ mais de $|h|$. \emph{Renvoi : M5.}
  \end{itemize}
  \item \textbf{Q15}
  \begin{itemize}
    \item \textbf{A} — $h=0{,}06$ ajouté à $\sqrt{4}$ sans multiplier par $f'(4)$. \emph{Renvoi : M5.}
    \item \textbf{C} — Produit $h \times f'(a)$ sans ajouter $f(a)$. \emph{Renvoi : M5.}
    \item \textbf{D} — Erreur de placement de la virgule. \emph{Renvoi : M5.}
  \end{itemize}
\end{itemize}
