\begin{corrige}{1SPE-DERLOCAL-EX-032}
\begin{enumerate}
\item Le taux de variation de $f$ entre $x=1$ et $x=5$ est :
\[
\tau = \frac{f(5) - f(1)}{5 - 1} = \frac{26 - 2}{4} = \frac{24}{4} = \boxed{6}
\]

\item Le taux de variation de $f$ entre $x=3$ et $x=7$ est :
\[
\tau = \frac{f(7) - f(3)}{7 - 3} = \frac{50 - 10}{4} = \frac{40}{4} = \boxed{10}
\]
\end{enumerate}
\end{corrige}
