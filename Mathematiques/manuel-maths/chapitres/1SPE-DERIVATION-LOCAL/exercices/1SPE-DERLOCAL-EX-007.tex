\begin{exercice}{1SPE-DERLOCAL-EX-007}{1}{8}
Soit $f(x)=x^2+2x$.
\begin{enumerate}
  \item Calculer le taux de variation de $f$ entre $3$ et $3+h$ pour $h=1$, $h=0{,}1$ et $h=0{,}01$.
  \item Vers quelle valeur semblent tendre ces taux lorsque $h$ se rapproche de $0$ ?
  \item En déduire $f'(3)$.
\end{enumerate}
\end{exercice}
