\section{Probabilité conditionnelle}

% ============================================================
% STRATE 1 --- Définitions et propriétés fondamentales
% ============================================================

\definition[1]{
Soit $A$ et $B$ deux événements d'un univers $\Omega$, avec $P(A) > 0$.

La \textbf{probabilité conditionnelle de $B$ sachant $A$}, notee $P_A(B)$, est le nombre :
\[
  P_A(B) = \frac{P(A \cap B)}{P(A)}.
\]

On lit : << probabilité de $B$ sachant que $A$ est realise >>.
}

\exemple{
Dans un lycee de $500$ eleves, $200$ sont en Première, et parmi ceux-ci $80$ suivent la specialite Maths.

On note $A$ : << l'élève est en Première >> et $B$ : << l'élève suit Maths >>.

$P(A) = \dfrac{200}{500} = \dfrac{2}{5}$, \quad $P(A \cap B) = \dfrac{80}{500} = \dfrac{4}{25}$.

\[
  P_A(B) = \frac{P(A \cap B)}{P(A)} = \frac{\frac{4}{25}}{\frac{2}{5}} = \frac{4}{25} \times \frac{5}{2} = \frac{2}{5}.
\]

Parmi les élèves de Première, $\dfrac{2}{5}$ suivent Maths.
}

\contreexemple{
Attention : $P_A(B)
\neq P_B(A)$ en general !

Ici, $P_B(A) = \dfrac{P(A \cap B)}{P(B)}$ necessite de connaitre $P(B)$ (la proportion de tous les élèves qui font Maths, pas seulement ceux de Première).
}

% ============================================================
% STRATE 2 --- Propriétés et formule de multiplication
% ============================================================

\propriete[1]{
Si $P(A) > 0$, alors :
\[
  P(A \cap B) = P(A) \times P_A(B).
\]

C'est la \textbf{formule de multiplication} (ou formule des probabilités composees).
}

\exemple{
On tire une carte au hasard dans un jeu de 32 cartes. On note $A$ : << la carte est un coeur >> et $B$ : << la carte est un roi >>.

$P(A) = \dfrac{8}{32} = \dfrac{1}{4}$.

Sachant que la carte est un coeur, la probabilité que ce soit le roi de coeur est $P_A(B) = \dfrac{1}{8}$.

Donc $P(A \cap B) = P(A) \times P_A(B) = \dfrac{1}{4} \times \dfrac{1}{8} = \dfrac{1}{32}$.
}

\propriete[2]{
Si $P(A) > 0$, alors la fonction $B \mapsto P_A(B)$ vérifie les axiomes d'une probabilité :
\begin{itemize}
  \item $P_A(\Omega) = 1$ ;
  \item Pour tout événement $B$, $0 \leq P_A(B) \leq 1$ ;
  \item $P_A(\overline{B}) = 1 - P_A(B)$.
\end{itemize}
}
