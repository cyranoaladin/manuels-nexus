\begin{fichemethode}{M4}{Déterminer si deux événements sont indépendants}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande si deux événements sont indépendants,
  ou lorsque l'on veut savoir si la realisation de l'un influence la probabilité de l'autre.
}

\pasApas{%
  \begin{enumerate}
    \item J'identifie les événements $A$ et $B$.
    \item Je calcule $P(A)$, $P(B)$ et $P(A \cap B)$ (par arbre, tableau ou formule).
    \item Je calcule le produit $P(A) \times P(B)$.
    \item Je compare : si $P(A \cap B) = P(A) \times P(B)$, les événements sont indépendants.
          Sinon, ils ne le sont pas.
    \item \textbf{Méthode alternative (si $P(A) > 0$)} : je compare $P_A(B)$ et $P(B)$.
          Égalité $\Leftrightarrow$ indépendance.
  \end{enumerate}
}

\exempleRedige{%
  On lance deux des equilibres. $A$ : << le premier de donne $6$ >>, $B$ : << la somme vaut $7$ >>.

  \medskip
  \commentaireMarge{Calculer $P(A)$, $P(B)$, $P(A \cap B)$.}%
  $P(A) = \dfrac{1}{6}$, \quad $P(B) = \dfrac{6}{36} = \dfrac{1}{6}$, \quad $A \cap B = \{(6,1)\}$ donc $P(A \cap B) = \dfrac{1}{36}$.

  \commentaireMarge{Comparer.}%
  $P(A) \times P(B) = \dfrac{1}{6} \times \dfrac{1}{6} = \dfrac{1}{36} = P(A \cap B)$.

  Les événements $A$ et $B$ sont indépendants.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Confondre indépendance et incompatibilite. Deux événements
          incompatibles de probabilités non nulles ne sont \textbf{jamais} indépendants.
    \item \textbf{Piège 2.} Affirmer l'indépendance sans calcul, << parce que ca parait logique >>.
    \item \textbf{Piège 3.} Oublier de conclure par une phrase.
  \end{itemize}
}

\verifier{%
  Si $A$ et $B$ sont indépendants, vérifier aussi que $P_A(B) = P(B)$ (ou inversement).
}
