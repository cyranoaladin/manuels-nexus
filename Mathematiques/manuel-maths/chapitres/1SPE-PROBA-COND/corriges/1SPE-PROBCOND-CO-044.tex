\begin{corrige}{1SPE-PROBCOND-EX-044}

\textbf{1.} Arbre des possibilités pour un enfant. Chaque parent $Aa$ transmet $A$ ou $a$ avec la probabilité $\dfrac{1}{2}$, indépendamment de l'autre parent.

\begin{itemize}
  \item Premier niveau (allèle du père) : $A$ de probabilité $\dfrac{1}{2}$, et $a$ de probabilité $\dfrac{1}{2}$.
  \item Sous chaque branche (allèle de la mère) : $A$ de probabilité $\dfrac{1}{2}$, et $a$ de probabilité $\dfrac{1}{2}$.
\end{itemize}

Les quatre feuilles $AA$, $Aa$, $aA$ et $aa$ ont donc chacune la probabilité $\dfrac{1}{4}$.

\medskip

\textbf{2.} $P(aa) = \frac{1}{2}\times\frac{1}{2} = \frac{1}{4}$, $P(Aa) = 2\times\frac{1}{2}\times\frac{1}{2} = \frac{1}{2}$, $P(AA) = \frac{1}{4}$.

\textbf{3.} $P(\text{porteur}|\text{non malade}) = \frac{P(Aa)}{P(Aa)+P(AA)} = \frac{1/2}{1/2+1/4} = \frac{1/2}{3/4} = \frac{2}{3}$.

\end{corrige}
