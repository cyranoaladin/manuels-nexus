\begin{corrige}{1SPE-PROBCOND-EX-034}

\textbf{1.} $A = \{2,4,6\}$, $P(A) = \frac{1}{2}$. $B = \{5,6\}$, $P(B) = \frac{1}{3}$. $A \cap B = \{6\}$, $P(A \cap B) = \frac{1}{6}$.

\textbf{2.} $P(A) \times P(B) = \frac{1}{2} \times \frac{1}{3} = \frac{1}{6} = P(A \cap B)$. Les événements sont indépendants.

\end{corrige}
