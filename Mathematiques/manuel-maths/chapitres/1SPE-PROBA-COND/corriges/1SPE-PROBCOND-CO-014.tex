\begin{corrige}{1SPE-PROBCOND-EX-014}

\textbf{1.} Branches : $A$ ($\frac{2}{5}$) puis $B$ ($\frac{3}{4}$) ou $\overline{B}$ ($\frac{1}{4}$) ; $\overline{A}$ ($\frac{3}{5}$) puis $B$ ($\frac{1}{3}$) ou $\overline{B}$ ($\frac{2}{3}$).

\medskip
\textbf{2.} $P(A \cap B) = \dfrac{2}{5} \times \dfrac{3}{4} = \dfrac{3}{10}$. $P(\overline{A} \cap B) = \dfrac{3}{5} \times \dfrac{1}{3} = \dfrac{1}{5}$.

\end{corrige}
