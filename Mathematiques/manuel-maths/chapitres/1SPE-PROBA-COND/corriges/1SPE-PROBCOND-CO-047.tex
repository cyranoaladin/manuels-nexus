\begin{corrige}{1SPE-PROBCOND-EX-047}

\textbf{1.} On note $D$ « le sportif est dopé » et $T$ « le test est positif ». La sensibilité est $P_D(T) = 0{,}99$ et la spécificité $P_{\overline{D}}(\overline{T}) = 0{,}95$. D'où l'arbre pondéré suivant.

\begin{nxarbreproba}
  \nxarbreracine{o}
  \nxarbrenoeud{d}{o}{$D$}{$0{,}05$}
  \nxarbrefeuille{dt}{d}{$T$}{$0{,}99$}{$T \cap D$}{$0{,}0495$}
  \nxarbrefeuille{dn}{d}{$\overline{T}$}{$0{,}01$}{$\overline{T} \cap D$}{$0{,}0005$}
  \nxarbrenoeud{s}{o}{$\overline{D}$}{$0{,}95$}
  \nxarbrefeuille{st}{s}{$T$}{$0{,}05$}{$T \cap \overline{D}$}{$0{,}0475$}
  \nxarbrefeuille{sn}{s}{$\overline{T}$}{$0{,}95$}{$\overline{T} \cap \overline{D}$}{$0{,}9025$}
\end{nxarbreproba}

Lecture de l'arbre :

\begin{itemize}
  \item Premier niveau : $D$ de probabilité $0{,}05$, et $\overline{D}$ de probabilité $0{,}95$.
  \item Sous $D$ : $T$ de probabilité $0{,}99$, et $\overline{T}$ de probabilité $0{,}01$.
  \item Sous $\overline{D}$ : $T$ de probabilité $0{,}05$, et $\overline{T}$ de probabilité $0{,}95$.
\end{itemize}

\medskip

\textbf{2.} $P(T) = \frac{5}{100}\times\frac{99}{100}+\frac{95}{100}\times\frac{5}{100} = \frac{495+475}{10000} = \frac{970}{10000} = \frac{97}{1000}$.

\textbf{3.} $P_T(D) = \frac{495/10000}{97/1000} = \frac{495}{970} = \frac{99}{194} \approx 51\,\%$.

\textbf{4.} Environ $1$ test positif sur $2$ correspond a un vrai dope. La sanction ne devrait pas être automatique ; un second test de confirmation est recommande.

\end{corrige}
