\begin{corrige}{1SPE-PROBCOND-EX-041}

\textbf{1.} $M$ : malade, $T$ : test positif. $P(M)=\frac{1}{100}$, $P_M(T)=\frac{19}{20}$, $P_{\overline{M}}(T)=\frac{3}{100}$.

\textbf{2.} La sensibilité est $P_M(T) = 0{,}95$ et la spécificité $P_{\overline{M}}(\overline{T}) = 0{,}97$. D'où l'arbre pondéré suivant.

Lecture de l'arbre :

\begin{itemize}
  \item Premier niveau : $M$ de probabilité $0{,}01$, et $\overline{M}$ de probabilité $0{,}99$.
  \item Sous $M$ : $T$ de probabilité $0{,}95$, et $\overline{T}$ de probabilité $0{,}05$.
  \item Sous $\overline{M}$ : $T$ de probabilité $0{,}03$, et $\overline{T}$ de probabilité $0{,}97$.
\end{itemize}

\medskip

\textbf{3.} $P(T) = \frac{1}{100}\times\frac{19}{20}+\frac{99}{100}\times\frac{3}{100} = \frac{19}{2000}+\frac{297}{10000} = \frac{95+297}{10000} = \frac{392}{10000} = \frac{49}{1250}$.

\textbf{4.} $P_T(M) = \frac{19/2000}{49/1250} = \frac{95}{392} \approx 24{,}2\,\%$. Un test positif ne correspond a une maladie réelle que dans $1$ cas sur $4$.
