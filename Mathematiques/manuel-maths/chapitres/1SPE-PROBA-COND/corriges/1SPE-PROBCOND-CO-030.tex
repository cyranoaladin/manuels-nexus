\begin{corrige}{1SPE-PROBCOND-EX-030}

\textbf{1.} $P(D) = \frac{1}{4}\times\frac{1}{100} + \frac{7}{20}\times\frac{2}{100} + \frac{1}{4}\times\frac{3}{100} + \frac{3}{20}\times\frac{5}{100} = \frac{5+14+15+15}{2000} = \frac{49}{2000}$.

\textbf{2.} $P_D(L_2) = \frac{14/2000}{49/2000} = \frac{14}{49} = \frac{2}{7}$, $P_D(L_3) = \frac{15}{49}$, $P_D(L_4) = \frac{15}{49}$. Les lots $L_3$ et $L_4$ sont les plus probables.
