\begin{corrige}{1SPE-PROBCOND-EX-003}

\textbf{1.} $P(F) = \dfrac{120}{200} = \dfrac{3}{5}$ et $P(F \cap S) = \dfrac{45}{200} = \dfrac{9}{40}$.

\medskip
\textbf{2.} $P_F(S) = \dfrac{P(F \cap S)}{P(F)} = \dfrac{\frac{9}{40}}{\frac{3}{5}} = \dfrac{9}{40} \times \dfrac{5}{3} = \dfrac{3}{8}$.

Parmi les filles, $\dfrac{3}{8} = 37{,}5\,\%$ pratiquent un sport.

\end{corrige}
