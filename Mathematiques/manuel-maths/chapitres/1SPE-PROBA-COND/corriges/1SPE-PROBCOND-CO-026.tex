\begin{corrige}{1SPE-PROBCOND-EX-026}

\textbf{1.} $P(M) = \frac{5}{1000}$, $P_M(T) = \frac{98}{100}$, $P_{\overline{M}}(T) = 1 - \frac{96}{100} = \frac{4}{100}$.

\textbf{2.} $P(T) = \frac{5}{1000}\times\frac{98}{100} + \frac{995}{1000}\times\frac{4}{100} = \frac{490}{100000}+\frac{3980}{100000} = \frac{4470}{100000} = \frac{447}{10000}$.

\textbf{3.} $P_T(M) = \frac{490/100000}{447/10000} = \frac{49}{447} \approx 11\,\%$.

\textbf{4.} Malgre la bonne sensibilite, un test positif ne signifie la maladie que dans $11\,\%$ des cas, a cause de la faible prevalence.

\end{corrige}
