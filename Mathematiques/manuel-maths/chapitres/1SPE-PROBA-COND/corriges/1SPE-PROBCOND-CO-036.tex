\begin{corrige}{1SPE-PROBCOND-EX-036}

\textbf{1.} $P(A) = \frac{26}{52} = \frac{1}{2}$, $P(B) = \frac{4}{52} = \frac{1}{13}$, $P(A \cap B) = \frac{2}{52} = \frac{1}{26}$.

\textbf{2.} $P(A) \times P(B) = \frac{1}{26} = P(A \cap B)$. Indépendants.

\textbf{3.} Par propriété, si $A$ et $B$ sont indépendants, $\overline{A}$ et $B$ le sont aussi.
