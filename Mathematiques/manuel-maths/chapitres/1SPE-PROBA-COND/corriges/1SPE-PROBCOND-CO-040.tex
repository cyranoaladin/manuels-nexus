\begin{corrige}{1SPE-PROBCOND-EX-040}

\textbf{1.} $P(3S) = \left(\frac{2}{3}\right)^3 = \frac{8}{27}$.

\textbf{2.} $P(\text{au moins 1}) = 1 - P(0S) = 1 - \left(\frac{1}{3}\right)^3 = 1 - \frac{1}{27} = \frac{26}{27}$.

\textbf{3.} $P(2S) = \binom{3}{2}\left(\frac{2}{3}\right)^2\left(\frac{1}{3}\right) = 3 \times \frac{4}{9} \times \frac{1}{3} = \frac{4}{9}$.

\end{corrige}
