\begin{corrige}{1SPE-PROBCOND-EX-006}

\textbf{1.} $P_H(S) = \dfrac{P(H \cap S)}{P(H)} = \dfrac{36/100}{60/100} = \dfrac{36}{60} = \dfrac{3}{5} = 60\,\%$.

$P_F(S) = \dfrac{P(F \cap S)}{P(F)} = \dfrac{20/100}{40/100} = \dfrac{20}{40} = \dfrac{1}{2} = 50\,\%$.

\medskip
\textbf{2.} Les hommes sont proportionnellement plus sportifs ($60\,\%$) que les femmes ($50\,\%$).

\end{corrige}
