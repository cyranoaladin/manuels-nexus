\begin{corrige}{1SPE-PROBCOND-EX-004}

\textbf{1.} $P(A \cap B) = P(A) \times P_A(B) = \dfrac{1}{4} \times \dfrac{3}{5} = \dfrac{3}{20}$.

\medskip
\textbf{2.} $P_B(A) = \dfrac{P(A \cap B)}{P(B)} = \dfrac{\frac{3}{20}}{\frac{1}{2}} = \dfrac{3}{20} \times 2 = \dfrac{3}{10}$.

\end{corrige}
