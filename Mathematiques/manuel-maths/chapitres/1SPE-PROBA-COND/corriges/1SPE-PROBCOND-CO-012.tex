\begin{corrige}{1SPE-PROBCOND-EX-012}

\textbf{1.} Branches : pile ($\frac{1}{3}$) puis lot ($\frac{3}{4}$) ou pas ($\frac{1}{4}$) ; face ($\frac{2}{3}$) puis lot ($\frac{1}{4}$) ou pas ($\frac{3}{4}$).

\medskip
\textbf{2.} $P(\text{pile} \cap \text{lot}) = \dfrac{1}{3} \times \dfrac{3}{4} = \dfrac{1}{4}$. $P(\text{face} \cap \text{lot}) = \dfrac{2}{3} \times \dfrac{1}{4} = \dfrac{1}{6}$.

\medskip
\textbf{3.} $\dfrac{1}{4} + \dfrac{1}{12} + \dfrac{1}{6} + \dfrac{1}{2} = 1$ \checkmark.

\end{corrige}
