\begin{corrige}{1SPE-PROBCOND-EX-017}

\textbf{1.} Arbre : deux niveaux, chaque niveau P ($\frac{2}{3}$) ou F ($\frac{1}{3}$).

\medskip
\textbf{2.} $P(PP) = \frac{2}{3} \times \frac{2}{3} = \frac{4}{9}$.

\medskip
\textbf{3.} $P(\text{1 pile}) = P(PF) + P(FP) = \frac{2}{3}\times\frac{1}{3} + \frac{1}{3}\times\frac{2}{3} = \frac{4}{9}$.

\end{corrige}
