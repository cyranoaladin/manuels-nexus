\begin{corrige}{1SPE-PROBCOND-EX-028}

\textbf{1.} \textit{Démonstration.} $\{B, \overline{B}\}$ est une partition, donc $A = (A \cap B) \cup (A \cap \overline{B})$ avec événements incompatibles. Par additivite : $P(A) = P(A \cap B) + P(A \cap \overline{B}) = P(B)\,P_B(A) + P(\overline{B})\,P_{\overline{B}}(A)$. $\square$

\textbf{2.} $P(A) = \frac{3}{8}\times\frac{2}{3} + \frac{5}{8}\times\frac{1}{4} = \frac{1}{4}+\frac{5}{32} = \frac{8}{32}+\frac{5}{32} = \frac{13}{32}$.

\end{corrige}
