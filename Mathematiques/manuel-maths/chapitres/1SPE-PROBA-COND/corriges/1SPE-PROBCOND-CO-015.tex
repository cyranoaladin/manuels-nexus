\begin{corrige}{1SPE-PROBCOND-EX-015}

\textbf{1.} On note $R$ « l'élève a révisé » et $S$ « il réussit l'examen ». L'arbre pondéré de la situation est le suivant.

\begin{nxarbreproba}
  \nxarbreracine{o}
  \nxarbrenoeud{r}{o}{$R$}{$\dfrac{3}{5}$}
  \nxarbrefeuille{rs}{r}{$S$}{$\dfrac{9}{10}$}{$S \cap R$}{$\dfrac{27}{50}$}
  \nxarbrefeuille{rn}{r}{$\overline{S}$}{$\dfrac{1}{10}$}{$\overline{S} \cap R$}{$\dfrac{3}{50}$}
  \nxarbrenoeud{n}{o}{$\overline{R}$}{$\dfrac{2}{5}$}
  \nxarbrefeuille{ns}{n}{$S$}{$\dfrac{7}{10}$}{$S \cap \overline{R}$}{$\dfrac{14}{50}$}
  \nxarbrefeuille{nn}{n}{$\overline{S}$}{$\dfrac{3}{10}$}{$\overline{S} \cap \overline{R}$}{$\dfrac{6}{50}$}
\end{nxarbreproba}

Lecture de l'arbre :

\begin{itemize}
  \item Premier niveau : $R$ de probabilité $\dfrac{3}{5}$, et $\overline{R}$ de probabilité $\dfrac{2}{5}$.
  \item Sous $R$ : $S$ de probabilité $\dfrac{9}{10}$, et $\overline{S}$ de probabilité $\dfrac{1}{10}$.
  \item Sous $\overline{R}$ : $S$ de probabilité $\dfrac{7}{10}$, et $\overline{S}$ de probabilité $\dfrac{3}{10}$.
\end{itemize}

Les branches issues d'un même nœud somment bien à $1$.

\medskip

\textbf{2.} Chaque chemin se lit en multipliant les probabilités de ses branches. $P(S \cap R) = \frac{3}{5} \times \frac{9}{10} = \frac{27}{50}$. $P(S \cap \overline{R}) = \frac{2}{5} \times \frac{7}{10} = \frac{14}{50}$. $P(\overline{S} \cap R) = \frac{3}{5} \times \frac{1}{10} = \frac{3}{50}$. $P(\overline{S} \cap \overline{R}) = \frac{2}{5} \times \frac{3}{10} = \frac{6}{50}$.

\medskip
\textbf{3.} $\dfrac{27}{50} + \dfrac{3}{50} + \dfrac{14}{50} + \dfrac{6}{50} = \dfrac{50}{50} = 1$ \checkmark.

\end{corrige}
