\begin{corrige}{1SPE-PROBCOND-EX-021}

\textbf{1.} L'arbre pondéré de la situation est le suivant.

\begin{nxarbreproba}
  \nxarbreracine{o}
  \nxarbrenoeud{b}{o}{$B$}{$\dfrac{2}{5}$}
  \nxarbrefeuille{ba}{b}{$A$}{$\dfrac{3}{4}$}{$A \cap B$}{$\dfrac{3}{10}$}
  \nxarbrefeuille{bn}{b}{$\overline{A}$}{$\dfrac{1}{4}$}{$\overline{A} \cap B$}{$\dfrac{1}{10}$}
  \nxarbrenoeud{c}{o}{$\overline{B}$}{$\dfrac{3}{5}$}
  \nxarbrefeuille{ca}{c}{$A$}{$\dfrac{1}{3}$}{$A \cap \overline{B}$}{$\dfrac{1}{5}$}
  \nxarbrefeuille{cn}{c}{$\overline{A}$}{$\dfrac{2}{3}$}{$\overline{A} \cap \overline{B}$}{$\dfrac{2}{5}$}
\end{nxarbreproba}

Lecture de l'arbre :

\begin{itemize}
  \item Premier niveau : $B$ de probabilité $\dfrac{2}{5}$, et $\overline{B}$ de probabilité $\dfrac{3}{5}$.
  \item Sous $B$ : $A$ de probabilité $\dfrac{3}{4}$, et $\overline{A}$ de probabilité $\dfrac{1}{4}$.
  \item Sous $\overline{B}$ : $A$ de probabilité $\dfrac{1}{3}$, et $\overline{A}$ de probabilité $\dfrac{2}{3}$.
\end{itemize}

\medskip

\textbf{2.} $P(A) = P(B)\,P_B(A) + P(\overline{B})\,P_{\overline{B}}(A) = \frac{2}{5}\times\frac{3}{4} + \frac{3}{5}\times\frac{1}{3} = \frac{3}{10} + \frac{1}{5} = \frac{1}{2}$.

\end{corrige}
