\begin{corrige}{1SPE-PROBCOND-EX-019}

\textbf{1.} On note $R_i$ « la $i$-ième boule est rouge ». Le sac contient $5$ rouges et $3$ bleues, et les tirages sont sans remise. L'énoncé ne demande de développer que la branche rouge aux niveaux $2$ et $3$ : l'arbre est donc volontairement partiel.

\begin{nxarbreproba}
  \nxarbreracine{o}
  \nxarbrenoeud{a}{o}{$R_1$}{$\dfrac{5}{8}$}
  \nxarbrenoeud{ab}{a}{$R_2$}{$\dfrac{4}{7}$}
  \nxarbrefeuille{abc}{ab}{$R_3$}{$\dfrac{3}{6}$}{$R_1 \cap R_2 \cap R_3$}{$\dfrac{5}{28}$}
  \nxarbrefeuille{b}{o}{$\overline{R_1}$}{$\dfrac{3}{8}$}{}{}
\end{nxarbreproba}

Lecture de l'arbre :

\begin{itemize}
  \item Premier niveau : $R_1$ de probabilité $\dfrac{5}{8}$, et $\overline{R_1}$ de probabilité $\dfrac{3}{8}$.
  \item Sous $R_1$ (il reste $4$ rouges sur $7$ boules) : $R_2$ de probabilité $\dfrac{4}{7}$.
  \item Sous $R_1 \cap R_2$ (il reste $3$ rouges sur $6$ boules) : $R_3$ de probabilité $\dfrac{3}{6} = \dfrac{1}{2}$.
\end{itemize}

\medskip

\textbf{2.} $P(3R) = \frac{5}{8} \times \frac{4}{7} \times \frac{3}{6} = \frac{60}{336} = \frac{5}{28}$.

\medskip
\textbf{3.} $P(0R) = P(B_1 \cap B_2 \cap B_3) = \frac{3}{8} \times \frac{2}{7} \times \frac{1}{6} = \frac{6}{336} = \frac{1}{56}$.

\end{corrige}
