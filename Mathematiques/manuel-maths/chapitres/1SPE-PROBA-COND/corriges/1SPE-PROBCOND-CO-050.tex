\begin{corrige}{1SPE-PROBCOND-EX-050}

\textbf{1.} $P_T(M) = \dfrac{p \times \frac{95}{100}}{p \times \frac{95}{100} + (1-p) \times \frac{10}{100}} = \dfrac{95p}{95p + 10(1-p)} = \dfrac{95p}{85p + 10}$.

\textbf{2.} $P_T(M) \geq \frac{1}{2} \Leftrightarrow \frac{95p}{85p+10} \geq \frac{1}{2} \Leftrightarrow 190p \geq 85p + 10 \Leftrightarrow 105p \geq 10 \Leftrightarrow p \geq \frac{2}{21}$.

\textbf{3.} La prevalence doit être d'au moins $\frac{2}{21} \approx 9{,}5\,\%$ pour qu'un test positif corresponde a plus d'une chance sur deux d'être malade.

\end{corrige}
