\begin{corrige}{1SPE-PROBCOND-EX-008}

\textbf{1.} $P(C) = \dfrac{8}{32} = \dfrac{1}{4}$, $P(F) = \dfrac{12}{32} = \dfrac{3}{8}$, $P(C \cap F) = \dfrac{3}{32}$ (roi, dame, valet de coeur).

\medskip
\textbf{2.} $P_C(F) = \dfrac{P(C \cap F)}{P(C)} = \dfrac{\frac{3}{32}}{\frac{1}{4}} = \dfrac{3}{32} \times 4 = \dfrac{3}{8}$.

Parmi les coeurs, $\dfrac{3}{8}$ sont des figures.

\medskip
\textbf{3.} $P_F(C) = \dfrac{P(C \cap F)}{P(F)} = \dfrac{\frac{3}{32}}{\frac{3}{8}} = \dfrac{3}{32} \times \dfrac{8}{3} = \dfrac{1}{4}$.

\end{corrige}
