\begin{corrige}{1SPE-PROBCOND-EX-002}

\textbf{1.} $P_B(A) = \dfrac{P(A \cap B)}{P(B)} = \dfrac{\frac{3}{20}}{\frac{2}{5}} = \dfrac{3}{20} \times \dfrac{5}{2} = \dfrac{3}{8}$.

\medskip
\textbf{2.} $P_B(\overline{A}) = 1 - P_B(A) = 1 - \dfrac{3}{8} = \dfrac{5}{8}$.

\end{corrige}
