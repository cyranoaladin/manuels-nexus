\begin{corrige}{1SPE-PROBCOND-EX-045}

\textbf{1.} Arbre pondéré. On note $P_1$ « il pleut aujourd'hui » et $P_2$ « il pleut demain ».

\begin{itemize}
  \item Premier niveau : $P_1$ de probabilité $\dfrac{2}{5}$, et $\overline{P_1}$ de probabilité $\dfrac{3}{5}$.
  \item Sous $P_1$ : $P_2$ de probabilité $\dfrac{7}{10}$, et $\overline{P_2}$ de probabilité $\dfrac{3}{10}$.
  \item Sous $\overline{P_1}$ : $P_2$ de probabilité $\dfrac{3}{10}$, et $\overline{P_2}$ de probabilité $\dfrac{7}{10}$.
\end{itemize}

\medskip

\textbf{2.} $P(\text{pluie demain}) = \frac{2}{5}\times\frac{7}{10}+\frac{3}{5}\times\frac{3}{10} = \frac{14}{50}+\frac{9}{50} = \frac{23}{50} = 46\,\%$.

\end{corrige}
