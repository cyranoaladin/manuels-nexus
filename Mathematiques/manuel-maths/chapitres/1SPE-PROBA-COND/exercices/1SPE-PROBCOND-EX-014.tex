\begin{exercice}{1SPE-PROBCOND-EX-014}{1}{8}

\begin{enumerate}
  \item Construire l'arbre pondere.
  \item Calculer $P(A \cap B)$ et $P(\overline{A} \cap B)$.
\end{enumerate}
\end{exercice}
