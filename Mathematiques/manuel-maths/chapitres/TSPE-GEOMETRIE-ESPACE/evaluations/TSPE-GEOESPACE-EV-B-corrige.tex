\begin{corrige}{TSPE-GEOESPACE-EV-B-EX1}

\textbf{1.} Les trois vecteurs $\vec{AB}$, $\vec{AD}$ et $\vec{AE}$ portent trois arêtes
du cube issues du même sommet $A$ : ils ne sont pas coplanaires, puisque $E$ n'appartient
pas au plan $(ABD)$. Trois vecteurs non coplanaires de l'espace en forment une base. Le
déterminant le confirme : $\det\left(\vec{AB},\vec{AD},\vec{AE}\right)=1\neq0$.

En allant de $A$ à $G$ on parcourt une arête dans chacune des trois directions, donc
$\vec{AG}=\vec{AB}+\vec{AD}+\vec{AE}$ : les coordonnées de $\vec{AG}$ dans cette base sont
$(1\,;1\,;1)$.

\textbf{2.} $\vec{AF}=\vec{AB}+\vec{AE}$, de coordonnées $(1\,;0\,;1)$ : on va de $A$ à $B$,
puis on monte d'une arête.

Pour $\vec{BH}$, on passe par $A$ : $\vec{BH}=\vec{BA}+\vec{AH}=-\vec{AB}+\left(\vec{AD}+\vec{AE}\right)$,
soit $\vec{BH}=-\vec{AB}+\vec{AD}+\vec{AE}$, de coordonnées $(-1\,;1\,;1)$.

\textbf{3.} $\vec{w}=\vec{AB}+\tfrac12\vec{AD}$ a pour coordonnées
$\left(1\,;\tfrac12\,;0\right)$ : le point $M$ est le \textbf{milieu de $[BC]$}.
Graphiquement, on reporte $\vec{AB}$ depuis $A$, ce qui donne $B$ ; puis, depuis $B$, on
reporte la moitié de $\vec{AD}$ — c'est-à-dire qu'on avance d'une demi-arête dans la
direction de $\vec{AD}$. La construction d'une combinaison linéaire est toujours cette mise
bout à bout des vecteurs, chacun multiplié par son coefficient.

\end{corrige}

\begin{corrige}{TSPE-GEOESPACE-EV-B-EX2}

\textbf{1.} Le plan $(BDE)$ admet $\vec{n}(1;1;1)$ pour vecteur normal et passe par
$B(1;0;0)$ : son équation est $x+y+z-1=0$ (on vérifie que $D$ et $E$ la satisfont).
La droite $(AG)$ est dirigée par $\vec{AG}(1;1;1)$, qui est \textbf{colinéaire à $\vec{n}$} :
la droite est donc \textbf{orthogonale au plan}. Elle n'est pas parallèle à celui-ci et le
coupe en un unique point. En remplaçant $M(t;t;t)$ dans l'équation : $3t-1=0$, donc
$t=\tfrac13$ et le point d'intersection est $\left(\tfrac13;\tfrac13;\tfrac13\right)$.

\textbf{2.} $\vec{AC}(1;1;0)$, $\vec{AF}(1;0;1)$ et $\vec{AH}(0;1;1)$. Leur déterminant vaut
\[
  \det\left(\vec{AC},\vec{AF},\vec{AH}\right)
  = 1\times(0\times1-1\times1) - 1\times(1\times1-1\times0) + 0
  = -1-1 = -2 \neq 0 .
\]
Les trois vecteurs ne sont donc pas coplanaires : les points $A$, $C$, $F$ et $H$
\textbf{ne sont pas coplanaires}. Ils forment les sommets d'un tétraèdre régulier inscrit
dans le cube.

\textbf{3.} $\vec{EG}=(1;1;1)-(0;0;1)=(1;1;0)=\vec{AC}$. Les deux vecteurs sont
\textbf{égaux}, donc colinéaires. Un quadrilatère dont deux côtés opposés sont portés par
des vecteurs égaux est un \textbf{parallélogramme} : $ACGE$ en est un.

\end{corrige}

\begin{corrige}{TSPE-GEOESPACE-EV-B-EX3}

\textbf{1.} Soit $M(x;y;z)$. Le point $M$ appartient au plan passant par $A_0$ et de vecteur
normal $\vec{n}$ si et seulement si $\vec{A_0M}$ est orthogonal à $\vec{n}$, c'est-à-dire
$\vec{n}\cdot\vec{A_0M}=0$. En coordonnées :
\[
  a(x-x_0)+b(y-y_0)+c(z-z_0)=0
  \iff ax+by+cz-\left(ax_0+by_0+cz_0\right)=0 .
\]
C'est bien une équation de la forme $ax+by+cz+d=0$ avec $d=-(ax_0+by_0+cz_0)$.
Réciproquement, tout point vérifiant cette équation donne $\vec{n}\cdot\vec{A_0M}=0$, donc
appartient au plan : l'équivalence est établie.

Pour le plan $(BDE)$, avec $\vec{n}(1;1;1)$ et $B(1;0;0)$ :
$d=-(1\times1+1\times0+1\times0)=-1$, donc $(BDE):\ x+y+z-1=0$.

\textbf{2.} Le projeté orthogonal $H'$ de $A$ sur $(BDE)$ est le point du plan tel que
$\vec{AH'}$ soit colinéaire à $\vec{n}$ : on cherche donc $H'=A+t\vec{n}=(t;t;t)$ vérifiant
l'équation du plan. On obtient $3t-1=0$, soit $t=\tfrac13$ et
\[
  H'\left(\tfrac13\,;\tfrac13\,;\tfrac13\right).
\]
On retrouve le point d'intersection de $(AG)$ et de $(BDE)$ de l'exercice $2$, ce qui est
cohérent : $(AG)$ est précisément la perpendiculaire au plan issue de $A$.

\end{corrige}

\begin{corrige}{TSPE-GEOESPACE-EV-B-EX4}

\textbf{1.} $\vec{AC}(1;1;0)$ et $\vec{AF}(1;0;1)$ ont pour normes $\sqrt2$ et pour produit
scalaire $\vec{AC}\cdot\vec{AF}=1$. L'aire du triangle vaut
\[
  \mathcal{A}=\frac12\sqrt{\|\vec{AC}\|^2\|\vec{AF}\|^2-\left(\vec{AC}\cdot\vec{AF}\right)^2}
  =\frac12\sqrt{2\times2-1}=\frac{\sqrt3}{2}.
\]
(De façon équivalente : $\cos\widehat{CAF}=\tfrac12$, l'angle vaut $60^\circ$, et
$\mathcal{A}=\tfrac12\times\sqrt2\times\sqrt2\times\sin60^\circ=\tfrac{\sqrt3}{2}$.)

Le volume du tétraèdre $ACFH$ est le sixième de la valeur absolue du déterminant calculé à
l'exercice $2$ :
\[
  V=\frac{\left|\det\left(\vec{AC},\vec{AF},\vec{AH}\right)\right|}{6}=\frac{2}{6}=\frac13 .
\]
Le tétraèdre occupe donc le tiers du cube, dont le volume vaut $1$.

\textbf{2.} $\vec{MA}\cdot\vec{MG}=0$ signifie que le triangle $AMG$ est rectangle en $M$,
ou que $M$ est confondu avec $A$ ou $G$. L'ensemble $\mathcal{E}$ est donc la
\textbf{sphère de diamètre $[AG]$} : son centre est le milieu
$\Omega\left(\tfrac12;\tfrac12;\tfrac12\right)$ — le centre du cube — et son rayon vaut
$\dfrac{AG}{2}=\dfrac{\sqrt3}{2}$.

\textbf{3.} Pour $B$ : $\vec{BA}=(-1;0;0)$ et $\vec{BG}=(0;1;1)$, donc
$\vec{BA}\cdot\vec{BG}=0$ : le point $B$ \textbf{appartient} à $\mathcal{E}$ (le triangle
$ABG$ est rectangle en $B$).

Pour le centre $\Omega$ du cube : $\vec{\Omega A}$ et $\vec{\Omega G}$ sont opposés, et
$\vec{\Omega A}\cdot\vec{\Omega G}=-\dfrac34\neq0$. Le centre \textbf{n'appartient pas} à
$\mathcal{E}$ — ce qui est attendu : il en est le centre, pas un point de la sphère.

\end{corrige}
