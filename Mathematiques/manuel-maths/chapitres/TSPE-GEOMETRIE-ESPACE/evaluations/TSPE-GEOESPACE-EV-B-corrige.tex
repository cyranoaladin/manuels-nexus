\textbf{Exercice 1 — C12.}

\textbf{1.} $x=2+t$, $y=1-2t$, $z=3t$, avec $t\in\mathbb{R}$.
\hfill\textit{(2 pts)}

\textbf{2.} Pour $M(4;-3;6)$~: $4=2+t$ donne $t=2$~; on vérifie
$y=1-4=-3$ et $z=6$. Les trois équations sont satisfaites par la même valeur,
donc $M\in\mathcal{D}$.
Pour $N(4;-3;5)$~: $t=2$ conviendrait pour $x$ et $y$, mais la troisième
équation donnerait $5=6$. Donc $N\notin\mathcal{D}$. \hfill\textit{(3 pts)}

\textbf{Exercice 2 — C13.}

\textbf{1.} $\vec{AM}\cdot\vec{n}=0$ s'écrit $3(x-1)-1(y-2)+2(z+1)=0$, soit
$3x-3-y+2+2z+2=0$ et enfin $3x-y+2z+1=0$. Contrôle~:
$3\times1-2+2\times(-1)+1=0$. \hfill\textit{(3 pts)}

\textbf{2.} $\vec{n}\cdot\vec{v}=3\times1+(-1)\times1+2\times(-1)=0$~: oui,
$\vec{v}$ dirige $\mathcal{P}$. \hfill\textit{(2 pts)}

\textbf{Exercice 3 — C7 et C8.}

\textbf{1.} $\vec{AB}(2;-1;2)$ et $\vec{AC}(1;2;-1)$, donc
$\vec{AB}\cdot\vec{AC}=2-2-2=-2\neq0$. Le triangle n'est pas rectangle en
$A$. \hfill\textit{(3 pts)}

\textbf{2.} $\vec{n}(2;-1;2)$ et $\lVert\vec{n}\rVert=\sqrt{4+1+4}=3$, donc
$d(O,\mathcal{Q})=\dfrac{\lvert 0-0+0-9\rvert}{3}=\dfrac{9}{3}=3$.
\hfill\textit{(3 pts)}

\medskip
\textbf{Exercice 4 — C15 et C11.}

\textbf{1.} En reportant~: $t+(-t)+1-3=0$, soit $-2=0$. Le système n'a aucune
solution. Ce n'est pas une erreur~: $\Delta$ est \emph{strictement parallèle}
à $\mathcal{R}$. On le confirme par $\vec{n}\cdot\vec{u}=1-1+0=0$ et par le
fait que $(0;0;1)$ ne vérifie pas l'équation du plan. \hfill\textit{(2 pts)}

\textbf{2.} $H$ et $N$ appartiennent à $\mathcal{S}$, donc $\vec{HN}$ est un
vecteur de $\mathcal{S}$. Comme $(MH)$ est orthogonale à $\mathcal{S}$, on a
$\vec{MH}\cdot\vec{HN}=0$~: le triangle $MHN$ est rectangle en $H$. Par
Pythagore, $MN^2=MH^2+HN^2$. Or $N\neq H$ donc $HN^2>0$, d'où $MN^2>MH^2$ et,
les longueurs étant positives, $MN>MH$. \hfill\textit{(2 pts)}

\end{corrige}
