\begin{evaluation}{TSPE-GEOESPACE-EV-A}{55}

\begin{exercice}{TSPE-GEOESPACE-EV-A-EX1}{2}{12}
\textbf{Exercice 1} (5 points) — \textit{Capacité C12}

\medskip

Soit $\mathcal{D}$ la droite passant par $A(1;0;3)$ et de vecteur directeur
$\vec{u}\begin{pmatrix}2\\1\\-1\end{pmatrix}$.
\begin{enumerate}
  \item (2 pts) Écrire une représentation paramétrique de $\mathcal{D}$.
  \item (3 pts) Le point $M(5;2;1)$ appartient-il à $\mathcal{D}$ ? Et le
        point $N(5;2;0)$ ? Justifier dans les deux cas.
\end{enumerate}
\end{exercice}

\begin{exercice}{TSPE-GEOESPACE-EV-A-EX2}{2}{12}
\textbf{Exercice 2} (5 points) — \textit{Capacité C13}

\medskip

\begin{enumerate}
  \item (3 pts) Déterminer une équation cartésienne du plan $\mathcal{P}$
        passant par $A(2;-1;3)$ et de vecteur normal
        $\vec{n}\begin{pmatrix}1\\4\\-2\end{pmatrix}$, en détaillant le calcul.
  \item (2 pts) Le vecteur $\vec{v}(2;0;1)$ dirige-t-il $\mathcal{P}$ ?
        Justifier.
\end{enumerate}
\end{exercice}

\begin{exercice}{TSPE-GEOESPACE-EV-A-EX3}{2}{16}
\textbf{Exercice 3} (6 points) — \textit{Capacités C7 et C8}

\medskip

Dans un repère orthonormé, $A(1;2;0)$, $B(3;1;2)$ et $C(0;4;1)$.
\begin{enumerate}
  \item (3 pts) Calculer $\vec{AB}\cdot\vec{AC}$. Le triangle $ABC$ est-il
        rectangle en $A$ ? Justifier.
  \item (3 pts) Calculer la distance du point $O(0;0;0)$ au plan
        $\mathcal{Q}:\ x+2y-2z-6=0$.
\end{enumerate}
\end{exercice}

\begin{exercice}{TSPE-GEOESPACE-EV-A-EX4}{3}{15}
\textbf{Exercice 4} (4 points) — \textit{Capacités C15 et C11}

\medskip

\begin{enumerate}
  \item (2 pts) Étudier l'intersection du plan $\mathcal{R}:\ x+y+z-3=0$ et de
        la droite $\Delta$ définie par $x=t$, $y=-t$, $z=0$. Interpréter
        géométriquement le résultat obtenu.
  \item (2 pts) Soit $H$ le projeté orthogonal d'un point $M$ sur un plan
        $\mathcal{S}$, et $N$ un point de $\mathcal{S}$ distinct de $H$.
        Démontrer que $MN>MH$.
\end{enumerate}
\end{exercice}

\end{evaluation}
