\begin{evaluation}{TSPE-GEOESPACE-EV-B}{55}

\noindent
\textit{Dans tout le sujet, $ABCDEFGH$ est un cube d'arête $1$. On travaille dans le
repère orthonormé $\left(A\,;\vec{AB},\vec{AD},\vec{AE}\right)$, dans lequel
$A(0;0;0)$, $B(1;0;0)$, $C(1;1;0)$, $D(0;1;0)$, $E(0;0;1)$, $F(1;0;1)$,
$G(1;1;1)$ et $H(0;1;1)$.}

\begin{exercice}{TSPE-GEOESPACE-EV-B-EX1}{2}{15}
\textbf{Exercice 1} (5 points) — \textit{Capacités C1, C2, C4, C5}

\medskip

\begin{enumerate}
  \item (1 pt) Justifier que $\left(\vec{AB},\vec{AD},\vec{AE}\right)$ est une base
        de l'espace, puis lire sur la figure les coordonnées de $\vec{AG}$ dans
        cette base.
  \item (2 pts) Décomposer $\vec{AF}$ et $\vec{BH}$ comme combinaisons linéaires de
        $\vec{AB}$, $\vec{AD}$ et $\vec{AE}$.
  \item (2 pts) On pose $\vec{w}=\vec{AB}+\tfrac12\vec{AD}$. Placer sur la figure le
        point $M$ tel que $\vec{AM}=\vec{w}$, et décrire en une phrase la
        construction graphique qui donne $\vec{w}$ à partir de $\vec{AB}$ et
        $\vec{AD}$.
\end{enumerate}
\end{exercice}

\begin{exercice}{TSPE-GEOESPACE-EV-B-EX2}{2}{15}
\textbf{Exercice 2} (5 points) — \textit{Capacités C3, C6}

\medskip

\begin{enumerate}
  \item (2 pts) Déterminer la position relative de la droite $(AG)$ et du plan
        $(BDE)$. On précisera s'il y a intersection, et de quelle nature.
  \item (2 pts) Les points $A$, $C$, $F$ et $H$ sont-ils coplanaires ? Justifier par
        un calcul.
  \item (1 pt) Les vecteurs $\vec{AC}$ et $\vec{EG}$ sont-ils colinéaires ? Qu'en
        déduit-on pour le quadrilatère $ACGE$ ?
\end{enumerate}
\end{exercice}

\begin{exercice}{TSPE-GEOESPACE-EV-B-EX3}{3}{15}
\textbf{Exercice 3} (5 points) — \textit{Capacités C16, C14}

\medskip

\begin{enumerate}
  \item (3 pts) Démontrer que le plan passant par un point $A_0(x_0;y_0;z_0)$ et de
        vecteur normal $\vec{n}(a;b;c)$ non nul a pour équation cartésienne
        $ax+by+cz+d=0$, avec $d=-(ax_0+by_0+cz_0)$. Appliquer au plan $(BDE)$,
        dont on admettra que $\vec{n}(1;1;1)$ est un vecteur normal.
  \item (2 pts) Calculer les coordonnées du projeté orthogonal de $A$ sur le plan
        $(BDE)$.
\end{enumerate}
\end{exercice}

\begin{exercice}{TSPE-GEOESPACE-EV-B-EX4}{3}{15}
\textbf{Exercice 4} (5 points) — \textit{Capacités C9, C10}

\medskip

\begin{enumerate}
  \item (2 pts) Calculer l'aire du triangle $ACF$, puis le volume du tétraèdre
        $ACFH$.
  \item (2 pts) Déterminer l'ensemble $\mathcal{E}$ des points $M$ de l'espace tels
        que $\vec{MA}\cdot\vec{MG}=0$. Préciser sa nature, son centre et son rayon.
  \item (1 pt) Le point $B$ appartient-il à $\mathcal{E}$ ? Et le centre du cube ?
        Justifier par le calcul.
\end{enumerate}
\end{exercice}
