\begin{exercice}{TSPE-GEOESPACE-EX-058}{3}{25}
$ABCDEFGH$ est un cube d'arête $1$. On note $\mathcal{P}$ le plan $(ACE)$.

\emph{Les questions 1 et 2 se traitent sans coordonnées.}
\begin{enumerate}
  \item À l'aide de la relation de Chasles uniquement, montrer que
        $\vec{AG}=\vec{AC}+\vec{AE}$. En déduire que les points $A$, $C$, $G$
        et $E$ sont coplanaires.
  \item On munit désormais l'espace du repère orthonormé
        $(A;\vec{AB},\vec{AD},\vec{AE})$. Montrer que la droite $(BD)$ est
        orthogonale au plan $\mathcal{P}$.
  \item Déterminer la nature du triangle $ACE$, puis calculer son aire.
  \item Déduire de la question 2 la position du projeté orthogonal $K$ de $B$
        sur $\mathcal{P}$, puis la distance de $B$ à $\mathcal{P}$. On
        vérifiera le résultat par la formule de la distance d'un point à un
        plan.
\end{enumerate}
\end{exercice}
