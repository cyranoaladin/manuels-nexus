\begin{corrige}{TSPE-GEOESPACE-EX-007}

\textbf{Q1.} $\begin{cases}x=t\\y=2t\\z=1-t\end{cases}\quad(t\in\mathbb{R})$.

\textbf{Q2.} On substitue dans l'equation de $\mathcal{P}$ :
\[
  2(t)-(2t)+(1-t)-4=0 \iff 2t-2t+1-t-4=0 \iff -t-3=0 \iff t=-3.
\]

\textbf{Q3.} Le point d'intersection a pour coordonnees $\big(-3,\,2\times(-3),\,1-(-3)\big) = (-3,-6,4)$.

Verification : $2\times(-3)-(-6)+4-4 = -6+6+4-4=0$ \checkmark.

\end{corrige}
