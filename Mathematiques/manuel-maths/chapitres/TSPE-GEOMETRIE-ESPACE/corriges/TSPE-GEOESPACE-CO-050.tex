\begin{corrige}{TSPE-GEOESPACE-EX-001}

\textbf{Q1.} $\vec{AB}\begin{pmatrix}1\\2\\2\end{pmatrix}$, $\vec{AC}\begin{pmatrix}2\\-2\\1\end{pmatrix}$.

\textbf{Q2.} $\vec{AB}\cdot\vec{AC} = 1\times2+2\times(-2)+2\times1 = 2-4+2 = 0$.

\textbf{Q3.} Comme $\vec{AB}\cdot\vec{AC}=0$, les vecteurs $\vec{AB}$ et $\vec{AC}$ sont orthogonaux : le triangle $ABC$ est rectangle en $A$.
