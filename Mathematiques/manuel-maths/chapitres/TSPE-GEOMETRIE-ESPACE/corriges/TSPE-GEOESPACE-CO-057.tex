\begin{corrige}{TSPE-GEOESPACE-EX-057}

\textbf{Q1.} $\vec{SB}(2;0;-2)$ donne $SB=\sqrt{4+0+4}=2\sqrt2$. De même
$SC=2\sqrt2$ et $\vec{BC}(-2;2;0)$ donne $BC=2\sqrt2$. Les trois côtés sont
égaux~: le triangle $SBC$ est \emph{équilatéral}.

\textbf{Q2.} $\vec{SB}\cdot\vec{SC}=2\times0+0\times2+(-2)\times(-2)=4$. Donc
\[
  \cos\widehat{BSC}=\frac{4}{2\sqrt2\times2\sqrt2}=\frac{4}{8}=\frac12,
  \qquad \widehat{BSC}=\frac{\pi}{3}.
\]
C'est cohérent avec un triangle équilatéral.

\textbf{Q3.} $\mathcal{A}_{SBC}=\dfrac12\,SB\cdot SC\cdot\sin\widehat{BSC}
=\dfrac12\times2\sqrt2\times2\sqrt2\times\dfrac{\sqrt3}{2}
=\dfrac12\times8\times\dfrac{\sqrt3}{2}=2\sqrt3$.

\textbf{Q4.} Le triangle $ABC$ est rectangle en $A$ avec $AB=AC=2$, donc
$\mathcal{A}_{ABC}=\dfrac12\times2\times2=2$. La hauteur issue de $S$ est
$SA=2$, car $(SA)$ est orthogonale au plan $(ABC)$. Ainsi
\[
  V=\frac13\times2\times2=\frac43 .
\]

\textbf{Q5.} En prenant $SBC$ pour base~:
$V=\dfrac13\times\mathcal{A}_{SBC}\times d\bigl(A,(SBC)\bigr)$, donc
\[
  d\bigl(A,(SBC)\bigr)=\frac{3V}{\mathcal{A}_{SBC}}
  =\frac{3\times\frac43}{2\sqrt3}=\frac{4}{2\sqrt3}
  =\frac{2}{\sqrt3}=\frac{2\sqrt3}{3}.
\]
Le volume, invariant par changement de base, permet ainsi d'obtenir une
distance sans écrire l'équation du plan.

\end{corrige}
