\begin{corrige}{TSPE-GEOESPACE-EX-005}

\textbf{Q1.} $H$ et $N$ appartiennent tous deux a $\mathcal{P}$, donc le vecteur $\vec{HN}$ est un vecteur du plan $\mathcal{P}$.

\textbf{Q2.} Par construction, $(MH)$ est orthogonale a $\mathcal{P}$ : $\vec{MH}$ est donc orthogonal a tout vecteur de $\mathcal{P}$, en particulier a $\vec{HN}$. Ainsi $\vec{MH}\cdot\vec{HN}=0$.

\textbf{Q3.} $\vec{MH}\cdot\vec{HN}=0$ signifie que le triangle $MHN$ est rectangle en $H$. Le theoreme de Pythagore donne $MN^2 = MH^2+HN^2$.

\textbf{Q4.} $N 
eq H$ donc $HN>0$, donc $HN^2>0$. Ainsi $MN^2 = MH^2+HN^2 > MH^2$, et comme $MN,MH\geq0$, on en deduit $MN>MH$.

\textbf{Q5.} Ce resultat montre que $H$ est strictement plus proche de $M$ que tout autre point $N$ de $\mathcal{P}$ : $H$ est donc le point de $\mathcal{P}$ le plus proche de $M$.

\end{corrige}
