\begin{corrige}{TSPE-GEOESPACE-EX-053}

\textbf{Q1.} $\vec{AG}=\vec{AB}+\vec{AD}+\vec{AE}$, donc le point $G$ a pour
coordonnées $(1;1;1)$ dans le repère d'origine $A$.

\textbf{Q2.} $\vec{BG}=\vec{BC}+\vec{CG}=\vec{AD}+\vec{AE}$, soit $(0;1;1)$.
Les coordonnées d'un \emph{point} sont celles du vecteur qui joint l'origine
du repère à ce point~; celles d'un \emph{vecteur} $\vec{BG}$ se calculent par
différence $G-B$. Comme $B$ n'est pas l'origine, les deux triplets diffèrent.

\textbf{Q3.} $\vec{DF}=\vec{DA}+\vec{AB}+\vec{BF}=-\vec{AD}+\vec{AB}+\vec{AE}$,
soit $(1;-1;1)$.

\textbf{Q4.} La base étant orthonormée,
$AG=\sqrt{1^2+1^2+1^2}=\sqrt{3}$.

\textbf{Q5.} Les coordonnées de $G$ restent $(1;1;1)$~: elles expriment $G$ en
fonction des vecteurs de base, et ces vecteurs sont toujours les arêtes du
solide. En revanche la longueur change~: avec $AE=4\,AB$ et des arêtes
orthogonales, $AG=\sqrt{1+1+16}=\sqrt{18}=3\sqrt{2}$. Des coordonnées repèrent
donc une position dans une base~; elles ne mesurent une longueur que si la
base est orthonormée.

\end{corrige}
