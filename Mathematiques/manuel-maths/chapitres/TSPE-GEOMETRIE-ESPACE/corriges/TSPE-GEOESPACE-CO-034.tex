\begin{corrige}{TSPE-GEOESPACE-EX-006}

\textbf{Q1.} $\vec{AB}\begin{pmatrix}-1\\1\\0\end{pmatrix}$, $\vec{AC}\begin{pmatrix}-1\\0\\1\end{pmatrix}$, $\vec{AD}\begin{pmatrix}-1\\-1\\2\end{pmatrix}$.

\textbf{Q2.} Le systeme $a\vec{AB}+b\vec{AC}=\vec{AD}$ s'ecrit :
\[
  \begin{cases} -a-b = -1 \\ a = -1 \\ b = 2 \end{cases}
\]
Les deux dernieres equations donnent $a=-1$, $b=2$. On verifie la premiere : $-(-1)-2 = 1-2=-1$ \checkmark. Le systeme est coherent : $a=-1$, $b=2$.

\textbf{Q3.} $\vec{AD} = -\vec{AB}+2\vec{AC}$ : $\vec{AD}$ est combinaison lineaire de $\vec{AB}$ et $\vec{AC}$, donc $A,B,C,D$ sont coplanaires.

\end{corrige}
