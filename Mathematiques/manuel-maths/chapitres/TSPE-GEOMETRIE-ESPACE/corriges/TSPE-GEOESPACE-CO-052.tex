\begin{corrige}{TSPE-GEOESPACE-EX-052}

\textbf{Q1.} $\vec{AG}=\vec{AB}+\vec{BC}+\vec{CG}=\vec{AB}+\vec{AD}+\vec{AE}$,
soit $(1;1;1)$~: les trois arêtes sont parcourues dans le sens des vecteurs de
base.

\textbf{Q2.} Chemin $E\to A\to C$~: $\vec{EC}=\vec{EA}+\vec{AC}$. Or
$\vec{EA}=-\vec{AE}$ et $\vec{AC}=\vec{AB}+\vec{AD}$, donc
$\vec{EC}=\vec{AB}+\vec{AD}-\vec{AE}$, soit $(1;1;-1)$.

Chemin $H\to D\to A\to B$~: $\vec{HB}=\vec{HD}+\vec{DA}+\vec{AB}$, avec
$\vec{HD}=-\vec{AE}$ et $\vec{DA}=-\vec{AD}$, donc
$\vec{HB}=\vec{AB}-\vec{AD}-\vec{AE}$, soit $(1;-1;-1)$.

\textbf{Q3.} Chemin $D\to A\to B\to F$~:
$\vec{DF}=\vec{DA}+\vec{AB}+\vec{BF}=-\vec{AD}+\vec{AB}+\vec{AE}$, soit
$(1;-1;1)$.

\textbf{Q4.} $I$ est le milieu de $[GH]$, donc
$\vec{AI}=\dfrac{\vec{AG}+\vec{AH}}{2}$. Avec $\vec{AG}(1;1;1)$ et
$\vec{AH}=\vec{AD}+\vec{AE}$ soit $(0;1;1)$, on obtient
$\vec{AI}=\left(\dfrac12;1;1\right)$, c'est-à-dire
$\vec{AI}=\dfrac12\vec{AB}+\vec{AD}+\vec{AE}$.

\textbf{Q5.} Non. Deux vecteurs sont colinéaires si leurs coordonnées sont
proportionnelles. Or les quatre triplets $(1;1;1)$, $(1;1;-1)$, $(1;-1;-1)$ et
$(1;-1;1)$ ont tous une première coordonnée égale à $1$~: un rapport de
proportionnalité vaudrait nécessairement $1$, ce qui imposerait l'égalité des
vecteurs. Comme ils sont deux à deux distincts, aucun couple n'est colinéaire.

\end{corrige}
