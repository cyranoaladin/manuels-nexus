\begin{corrige}{TSPE-GEOESPACE-EX-007}

\textbf{Q1.} $\begin{cases}x=t\\y=2t\\z=1-t\end{cases}\quad(t\in\mathbb{R})$.

\textbf{Q2.} On substitue dans l'équation de $\mathcal{P}$ :
\[
  2(t)-(2t)+(1-t)-4=0 \iff 2t-2t+1-t-4=0 \iff -t-3=0 \iff t=-3.
\]

\textbf{Q3.} Le point d'intersection a pour coordonnées $\big(-3,\,2\times(-3),\,1-(-3)\big) = (-3,-6,4)$.

Vérification : $2\times(-3)-(-6)+4-4 = -6+6+4-4=0$ \checkmark.


\textbf{Q4.} Avec $\mathcal{D}'$~: $2t-2t+0-4=0$, soit $-4=0$. L'équation
obtenue est impossible~: le système n'a aucune solution. Ce n'est pas une
erreur de calcul mais un résultat --- la droite $\mathcal{D}'$ n'a aucun point
commun avec $\mathcal{P}$. On le confirme avec les vecteurs~:
$\vec{n}\cdot\vec{u'}=2\times1+(-1)\times2+1\times0=0$, et le point $(0;0;0)$
de $\mathcal{D}'$ ne vérifie pas l'équation du plan.

\textbf{Q5.} Une valeur unique du paramètre correspond à une droite
\emph{sécante} au plan~: c'est le cas de $\mathcal{D}$, qui le perce en
$(-3;-6;4)$. Une équation impossible correspond à une droite
\emph{strictement parallèle} au plan~: c'est le cas de $\mathcal{D}'$. Le
troisième cas, une identité du type $0=0$, correspondrait à une droite
\emph{incluse} dans le plan.

\end{corrige}
