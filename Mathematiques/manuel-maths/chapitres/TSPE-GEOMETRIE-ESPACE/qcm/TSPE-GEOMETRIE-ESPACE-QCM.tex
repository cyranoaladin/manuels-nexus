\section*{\textcolor{chapcolor}{\MakeUppercase{QCM d'auto-évaluation — Géométrie dans l'espace}}}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacité C3}

\item \textbf{[Q1]} Deux droites de l'espace non paralleles sont :
  \begin{enumerate}[label=\Alph*.]
    \item secantes ou non coplanaires.
    \item toujours non coplanaires.
    \item toujours secantes.
    \item toujours confondues.
  \end{enumerate}

\item[] \textbf{Capacité C7}

\item \textbf{[Q2]} Deux vecteurs $\vec{u}$ et $\vec{v}$ sont orthogonaux si et seulement si :
  \begin{enumerate}[label=\Alph*.]
    \item $\vec{u}\cdot\vec{v}=0$.
    \item $\vec{u}\cdot\vec{v}=1$.
    \item $\vec{u}=\vec{v}$.
    \item $\|\vec{u}\|=\|\vec{v}\|$.
  \end{enumerate}

\item[] \textbf{Capacité C13}

\item \textbf{[Q3]} Le vecteur normal au plan d'équation $2x-3y+z+1=0$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $(2,3,1)$.
    \item $(2,-3,1)$.
    \item $(1,-3,2)$.
    \item $(-2,3,1)$.
  \end{enumerate}

\item[] \textbf{Capacité C12}

\item \textbf{[Q4]} Une représentation parametrique de droite comporte :
  \begin{enumerate}[label=\Alph*.]
    \item trois paramètres.
    \item deux paramètres.
    \item un seul paramètre.
    \item aucun paramètre.
  \end{enumerate}

\item[] \textbf{Capacité C11}

\item \textbf{[Q5]} Le projete orthogonal $H$ d'un point $M$ sur un plan $\mathcal{P}$ vérifie :
  \begin{enumerate}[label=\Alph*.]
    \item $H$ est le point de $\mathcal{P}$ le plus eloigne de $M$.
    \item $H=M$ toujours.
    \item $H$ est un point quelconque de $\mathcal{P}$.
    \item $H$ est le point de $\mathcal{P}$ le plus proche de $M$.
  \end{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q6]} Dans l'espace, les combinaisons linéaires $a\vec{u}+b\vec{v}$ de deux vecteurs non colinéaires décrivent :
  \begin{enumerate}[label=\Alph*.]
    \item tout l'espace.
    \item un plan.
    \item une droite.
    \item un seul point.
  \end{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q7]} $ABCDEFGH$ est un cube. Dans la base $(\vec{AB},\vec{AD},\vec{AE})$, le vecteur $\vec{EC}$ se décompose en :
  \begin{enumerate}[label=\Alph*.]
    \item $\vec{AB}+\vec{AD}+\vec{AE}$
    \item $\vec{AB}-\vec{AD}+\vec{AE}$
    \item $\vec{AB}+\vec{AD}-\vec{AE}$
    \item $-\vec{AB}+\vec{AD}+\vec{AE}$
  \end{enumerate}

\item[] \textbf{Capacité C4}

\item \textbf{[Q8]} Trois vecteurs de l'espace deux à deux non colinéaires :
  \begin{enumerate}[label=\Alph*.]
    \item forment toujours une base.
    \item sont toujours coplanaires.
    \item sont toujours orthogonaux.
    \item peuvent être coplanaires.
  \end{enumerate}

\item[] \textbf{Capacité C5}

\item \textbf{[Q9]} Dans une base \emph{non} orthonormée, connaître les coordonnées $(1;1;1)$ d'un vecteur permet :
  \begin{enumerate}[label=\Alph*.]
    \item de le décomposer sur la base, mais pas de calculer sa norme.
    \item de calculer sa norme, égale à $\sqrt{3}$.
    \item d'affirmer qu'il est unitaire.
    \item d'affirmer qu'il est orthogonal aux vecteurs de base.
  \end{enumerate}

\item[] \textbf{Capacité C6}

\item \textbf{[Q10]} Pour démontrer que quatre points $A$, $B$, $C$, $D$ sont coplanaires, il suffit de vérifier que :
  \begin{enumerate}[label=\Alph*.]
    \item $A$, $B$ et $C$ sont coplanaires.
    \item $\vec{AD}$ est une combinaison linéaire de $\vec{AB}$ et $\vec{AC}$.
    \item $\vec{AB}$ et $\vec{AC}$ sont colinéaires.
    \item les quatre points sont deux à deux distincts.
  \end{enumerate}

\item[] \textbf{Capacité C8}

\item \textbf{[Q11]} Repère orthonormé. La distance du point $M(1;2;3)$ au plan d'équation $x+2y+2z-6=0$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $5$
    \item $\dfrac{5}{9}$
    \item $\dfrac{5}{3}$
    \item $\sqrt{38}$
  \end{enumerate}

\item[] \textbf{Capacité C9}

\item \textbf{[Q12]} Repère orthonormé, $A(0;0;0)$, $B(1;0;0)$ et $C(1;1;0)$. La mesure de l'angle $\widehat{BAC}$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $0$
    \item $\dfrac{\pi}{2}$
    \item $\dfrac{\pi}{3}$
    \item $\dfrac{\pi}{4}$
  \end{enumerate}

\item[] \textbf{Capacité C10}

\item \textbf{[Q13]} Une droite est orthogonale à un plan lorsque son vecteur directeur est orthogonal :
  \begin{enumerate}[label=\Alph*.]
    \item à deux vecteurs non colinéaires du plan.
    \item à une seule droite du plan.
    \item au vecteur normal du plan.
    \item à toutes les droites de l'espace.
  \end{enumerate}

\item[] \textbf{Capacité C14}

\item \textbf{[Q14]} Repère orthonormé. Le projeté orthogonal de $M(3;3;3)$ sur le plan $x+y+z-3=0$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $(3;3;0)$
    \item $(1;1;1)$
    \item $(0;0;0)$
    \item $(3;3;3)$
  \end{enumerate}

\item[] \textbf{Capacité C15}

\item \textbf{[Q15]} En étudiant l'intersection d'une droite et d'un plan, on aboutit à l'équation $0=5$. On conclut que :
  \begin{enumerate}[label=\Alph*.]
    \item il y a une erreur de calcul.
    \item la droite est incluse dans le plan.
    \item la droite est strictement parallèle au plan.
    \item la droite coupe le plan en un point.
  \end{enumerate}

\item[] \textbf{Capacité C16}

\item \textbf{[Q16]} Une équation cartésienne du plan passant par $A(2;-1;3)$ et de vecteur normal $\vec{n}(1;4;-2)$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $x+4y-2z=0$
    \item $x+4y-2z-8=0$
    \item $2x-y+3z+8=0$
    \item $x+4y-2z+8=0$
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des réponses erronées}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{B} — Deux droites non paralleles peuvent être coplanaires et secantes ; elles ne sont donc pas toujours non coplanaires. \emph{Renvoi : C3, positions relatives de deux droites de l'espace.}
    \item \textbf{C} — Dans l'espace, deux droites non paralleles peuvent être gauches : elles ne sont alors pas coplanaires et ne se coupent pas. \emph{Renvoi : C3, positions relatives de deux droites de l'espace.}
    \item \textbf{D} — Des droites confondues ont la même direction et sont donc paralleles ; cette possibilite est exclue par l'hypothèse. \emph{Renvoi : C3, positions relatives de deux droites de l'espace.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{B} — Un produit scalaire égal a $1$ n'impose pas un angle droit ; l'orthogonalite est caracterisee par un produit scalaire nul. \emph{Renvoi : C7, produit scalaire et orthogonalité.}
    \item \textbf{C} — Si $\vec u=\vec v$ et ce vecteur est non nul, alors $\vec u\cdot\vec v=\|\vec u\|^2>0$ : ils ne sont pas orthogonaux. \emph{Renvoi : C7, produit scalaire et orthogonalité.}
    \item \textbf{D} — L'égalité des normes fixe seulement les longueurs ; deux vecteurs de même norme peuvent former n'importe quel angle. \emph{Renvoi : C7, produit scalaire et orthogonalité.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — Le coefficient de $y$ dans l'équation est $-3$ ; remplacer ce coefficient par $3$ ne donne pas un vecteur normal au plan. \emph{Renvoi : M1, vecteur normal et équation cartésienne d'un plan.}
    \item \textbf{C} — Les composantes d'un vecteur normal sont les coefficients de $x$, $y$ et $z$ dans cet ordre ; ici $2$, $-3$ et $1$. \emph{Renvoi : M1, vecteur normal et équation cartésienne d'un plan.}
    \item \textbf{D} — Ce vecteur change les signes des deux premieres composantes seulement ; il n'est donc pas proportionnel a $(2,-3,1)$. \emph{Renvoi : M1, vecteur normal et équation cartésienne d'un plan.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — Trois paramètres indépendants peuvent parcourir l'espace entier ; une droite s'ecrit avec un seul paramètre réel. \emph{Renvoi : C12, représentation paramétrique d'une droite.}
    \item \textbf{B} — Deux paramètres indépendants engendrent en general une surface plane, alors qu'une droite ne possede qu'une direction libre. \emph{Renvoi : C12, représentation paramétrique d'une droite.}
    \item \textbf{D} — Sans paramètre, on ne décrit pas le deplacement depuis un point selon un vecteur directeur ; la représentation ne serait pas parametrique. \emph{Renvoi : C12, représentation paramétrique d'une droite.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — Un plan est non borne, donc il ne possede pas de point le plus eloigne de $M$ ; la projection realise au contraire la distance minimale. \emph{Renvoi : C11, projeté orthogonal et distance à un plan.}
    \item \textbf{B} — On a $H=M$ seulement lorsque $M$ appartient deja au plan ; sinon le projete $H$ est un point distinct situe sur le plan. \emph{Renvoi : C11, projeté orthogonal et distance à un plan.}
    \item \textbf{C} — Le projete n'est pas quelconque : c'est le pied de la perpendiculaire issue de $M$, ce qui assure la distance minimale au plan. \emph{Renvoi : C11, projeté orthogonal et distance à un plan.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{A} — Deux vecteurs n'engendrent qu'un plan : il reste une direction hors de leur portée, et il faut un troisième vecteur non coplanaire pour atteindre tout l'espace. \emph{Renvoi : C1, combinaisons linéaires de vecteurs de l'espace.}
    \item \textbf{C} — Une droite serait engendrée par un seul vecteur ; avec deux vecteurs non colinéaires on obtient une famille à deux paramètres. \emph{Renvoi : M2, passer d'une figure aux coordonnées.}
    \item \textbf{D} — Seule la combinaison nulle donnerait un point ; ici $a$ et $b$ parcourent tous les réels. \emph{Renvoi : C1, combinaisons linéaires de vecteurs de l'espace.}
  \end{itemize}
  \item \textbf{Q7}
  \begin{itemize}
    \item \textbf{A} — C'est la décomposition de $\vec{AG}$ : elle monte l'arête verticale au lieu de la descendre. Le chemin $E\to A$ parcourt $\vec{AE}$ à rebours, d'où le signe moins. \emph{Renvoi : M2, passer d'une figure aux coordonnées.}
    \item \textbf{B} — Le signe négatif a été placé sur $\vec{AD}$ ; or le chemin $E\to A\to C$ parcourt $\vec{AD}$ dans son sens et $\vec{AE}$ à rebours. \emph{Renvoi : C2, décomposer un vecteur sur une figure.}
    \item \textbf{D} — Le signe négatif a été placé sur $\vec{AB}$ ; le trajet horizontal $A\to C$ suit pourtant $\vec{AB}$ puis $\vec{AD}$ dans leur sens. \emph{Renvoi : C2, décomposer un vecteur sur une figure.}
  \end{itemize}
  \item \textbf{Q8}
  \begin{itemize}
    \item \textbf{A} — La condition d'être une base porte sur les trois vecteurs ensemble : $\vec{u}$, $\vec{v}$ et $\vec{u}+\vec{v}$ sont deux à deux non colinéaires et pourtant coplanaires. \emph{Renvoi : C4, reconnaître une base de l'espace.}
    \item \textbf{B} — Ils peuvent l'être sans l'être toujours : $\vec{i}$, $\vec{j}$, $\vec{k}$ sont deux à deux non colinéaires et non coplanaires. \emph{Renvoi : M3, colinéarité, alignement, coplanarité.}
    \item \textbf{C} — La colinéarité et l'orthogonalité sont deux propriétés indépendantes ; ne pas être colinéaires n'impose aucun angle droit. \emph{Renvoi : C7, produit scalaire et orthogonalité.}
  \end{itemize}
  \item \textbf{Q9}
  \begin{itemize}
    \item \textbf{B} — La formule $\sqrt{x^2+y^2+z^2}$ suppose la base orthonormée. Dans un parallélépipède étiré, les mêmes coordonnées donnent une autre longueur. \emph{Renvoi : C5, lire des coordonnées dans une base.}
    \item \textbf{C} — Rien n'indique que le vecteur soit de norme $1$ ; les coordonnées ne mesurent pas de longueur ici. \emph{Renvoi : C5, lire des coordonnées dans une base.}
    \item \textbf{D} — Les coordonnées ne renseignent pas sur les angles tant que la base n'est pas orthonormée. \emph{Renvoi : M2, passer d'une figure aux coordonnées.}
  \end{itemize}
  \item \textbf{Q10}
  \begin{itemize}
    \item \textbf{A} — Trois points sont toujours coplanaires : le vérifier ne prouve rien, c'est le quatrième qui décide. \emph{Renvoi : C6, configurations dans l'espace.}
    \item \textbf{C} — Cela prouverait que $A$, $B$, $C$ sont alignés, ce qui est une autre propriété et ne dit rien de $D$. \emph{Renvoi : M3, colinéarité, alignement, coplanarité.}
    \item \textbf{D} — Quatre points distincts peuvent parfaitement ne pas être coplanaires, comme les sommets d'un tétraèdre. \emph{Renvoi : C6, configurations dans l'espace.}
  \end{itemize}
  \item \textbf{Q11}
  \begin{itemize}
    \item \textbf{A} — C'est le numérateur seul : il manque la division par $\lVert\vec{n}\rVert=\sqrt{1+4+4}=3$. \emph{Renvoi : M6, projeter orthogonalement.}
    \item \textbf{B} — La division a été faite par $a^2+b^2+c^2=9$ au lieu de sa racine carrée $3$. \emph{Renvoi : C8, distance d'un point à un plan.}
    \item \textbf{D} — C'est la distance de $M$ au point $(6;0;0)$ du plan : un point quelconque du plan surestime la distance, que seul le projeté orthogonal réalise. \emph{Renvoi : C8, distance d'un point à un plan.}
  \end{itemize}
  \item \textbf{Q12}
  \begin{itemize}
    \item \textbf{A} — Le produit scalaire $\vec{AB}\cdot\vec{AC}=1$ a été pris pour le cosinus. Il faut diviser par $\lVert\vec{AB}\rVert\lVert\vec{AC}\rVert=\sqrt{2}$. \emph{Renvoi : M5, longueurs, angles et aires.}
    \item \textbf{B} — Un angle droit exigerait un produit scalaire nul ; ici il vaut $1$. \emph{Renvoi : C7, produit scalaire et orthogonalité.}
    \item \textbf{C} — $\cos\theta=\dfrac{1}{\sqrt2}$ et non $\dfrac12$ : la confusion porte sur la valeur remarquable. \emph{Renvoi : M5, longueurs, angles et aires.}
  \end{itemize}
  \item \textbf{Q13}
  \begin{itemize}
    \item \textbf{B} — Le plan possède une infinité de directions : $\vec{u}(0;1;1)$ est orthogonal à l'axe des abscisses sans l'être au plan $z=0$. \emph{Renvoi : C10, orthogonalité d'une droite et d'un plan.}
    \item \textbf{C} — Un vecteur orthogonal au vecteur normal dirige le plan, il ne lui est pas orthogonal. \emph{Renvoi : M7, orthogonalité droite-plan.}
    \item \textbf{D} — Seul le vecteur nul est orthogonal à toutes les directions de l'espace ; un vecteur directeur est non nul. \emph{Renvoi : C10, orthogonalité d'une droite et d'un plan.}
  \end{itemize}
  \item \textbf{Q14}
  \begin{itemize}
    \item \textbf{A} — Annuler la cote ne projette que sur le plan $z=0$ : le point $(3;3;0)$ ne vérifie même pas l'équation du plan. \emph{Renvoi : M6, projeter orthogonalement.}
    \item \textbf{C} — L'origine appartient-elle au plan ? $0+0+0-3=-3\neq0$ : non, ce n'est donc pas un projeté possible. \emph{Renvoi : C14, coordonnées du projeté orthogonal.}
    \item \textbf{D} — C'est le point $M$ lui-même, qui n'appartient pas au plan puisque $9-3=6\neq0$. \emph{Renvoi : C14, coordonnées du projeté orthogonal.}
  \end{itemize}
  \item \textbf{Q15}
  \begin{itemize}
    \item \textbf{A} — Une équation impossible est un résultat, pas un échec : elle traduit exactement l'absence de point commun. \emph{Renvoi : M9, traduire une intersection en système.}
    \item \textbf{B} — L'inclusion se traduirait par une identité du type $0=0$, vraie pour tout paramètre. \emph{Renvoi : C15, traduire et interpréter un système.}
    \item \textbf{D} — Une intersection en un point donnerait une unique valeur du paramètre, pas une impossibilité. \emph{Renvoi : C15, traduire et interpréter un système.}
  \end{itemize}
  \item \textbf{Q16}
  \begin{itemize}
    \item \textbf{A} — C'est l'équation du plan parallèle passant par l'origine : le terme constant a été oublié. Or $A$ ne vérifie pas cette équation, puisque $2-4-6=-8\neq0$. \emph{Renvoi : M1, équation cartésienne d'un plan.}
    \item \textbf{B} — Le signe du terme constant est inversé : $d=-(a x_A+b y_A+c z_A)=-(-8)=8$. \emph{Renvoi : C16, démontrer l'équation cartésienne d'un plan.}
    \item \textbf{C} — Les coordonnées de $A$ ont été prises pour le vecteur normal et inversement : $(a;b;c)$ doit être $\vec{n}$. \emph{Renvoi : C13, vecteur normal et équation cartésienne.}
  \end{itemize}
\end{itemize}
