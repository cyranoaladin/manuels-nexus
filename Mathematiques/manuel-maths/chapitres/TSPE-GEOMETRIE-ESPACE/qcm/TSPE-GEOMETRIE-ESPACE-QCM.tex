\section*{\textcolor{chapcolor}{\MakeUppercase{QCM d'auto-evaluation — Géométrie dans l'espace}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} Deux droites de l'espace non paralleles sont :
  \begin{enumerate}[label=\Alph*.]
    \item toujours secantes.
    \item toujours non coplanaires.
    \item secantes ou non coplanaires.
    \item toujours confondues.
  \end{enumerate}

\item \textbf{[Q2]} Deux vecteurs $\vec{u}$ et $\vec{v}$ sont orthogonaux si et seulement si :
  \begin{enumerate}[label=\Alph*.]
    \item $\vec{u}\cdot\vec{v}=1$.
    \item $\vec{u}\cdot\vec{v}=0$.
    \item $\vec{u}=\vec{v}$.
    \item $\|\vec{u}\|=\|\vec{v}\|$.
  \end{enumerate}

\item \textbf{[Q3]} Le vecteur normal au plan d'equation $2x-3y+z+1=0$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $(2,-3,1)$.
    \item $(2,3,1)$.
    \item $(1,-3,2)$.
    \item $(-2,3,1)$.
  \end{enumerate}

\item \textbf{[Q4]} Une representation parametrique de droite comporte :
  \begin{enumerate}[label=\Alph*.]
    \item un seul parametre.
    \item deux parametres.
    \item trois parametres.
    \item aucun parametre.
  \end{enumerate}

\item \textbf{[Q5]} Le projete orthogonal $H$ d'un point $M$ sur un plan $\mathcal{P}$ verifie :
  \begin{enumerate}[label=\Alph*.]
    \item $H$ est le point de $\mathcal{P}$ le plus eloigne de $M$.
    \item $H$ est le point de $\mathcal{P}$ le plus proche de $M$.
    \item $H$ est un point quelconque de $\mathcal{P}$.
    \item $H=M$ toujours.
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des reponses erronees}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{A} — Deux droites non paralleles peuvent etre non coplanaires ; dans ce cas elles ne sont pas secantes. \emph{Renvoi : C1, positions relatives de deux droites.}
    \item \textbf{B} — Deux droites non paralleles peuvent aussi etre coplanaires et secantes ; elles ne sont donc pas toujours non coplanaires. \emph{Renvoi : C1, positions relatives de deux droites.}
    \item \textbf{D} — Deux droites confondues sont paralleles ; cette option contredit l'hypothese de droites non paralleles. \emph{Renvoi : C1, positions relatives de deux droites.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — Un produit scalaire egal a $1$ ne caracterise pas l'orthogonalite ; il doit etre nul. \emph{Renvoi : C1, produit scalaire et orthogonalite.}
    \item \textbf{C} — Deux vecteurs egaux ne sont orthogonaux que dans le cas particulier du vecteur nul. \emph{Renvoi : C1, produit scalaire et orthogonalite.}
    \item \textbf{D} — L'egalite des normes ne donne aucune orthogonalite ; deux vecteurs de meme norme peuvent former n'importe quel angle. \emph{Renvoi : C1, produit scalaire et orthogonalite.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{B} — Le signe du coefficient de $y$ doit etre negatif : le vecteur normal canonique est $(2,-3,1)$. \emph{Renvoi : C1, equation cartesienne d'un plan.}
    \item \textbf{C} — Les coordonnees d'un vecteur normal sont les coefficients de $x$, $y$ et $z$ dans cet ordre. \emph{Renvoi : C1, equation cartesienne d'un plan.}
    \item \textbf{D} — Le vecteur $(-2,3,1)$ n'est pas proportionnel a $(2,-3,1)$ : le signe de la troisieme coordonnee devrait aussi changer. \emph{Renvoi : C1, equation cartesienne d'un plan.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{B} — Deux parametres independants decrivent en general un plan ; une droite affine n'en requiert qu'un. \emph{Renvoi : C1, representation parametrique d'une droite.}
    \item \textbf{C} — Trois coordonnees ne signifient pas trois parametres : elles dependent toutes du meme parametre reel. \emph{Renvoi : C1, representation parametrique d'une droite.}
    \item \textbf{D} — Sans parametre, on decrirait un point fixe et non l'ensemble des points d'une droite. \emph{Renvoi : C1, representation parametrique d'une droite.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — Le projete orthogonal minimise la distance a $M$ parmi les points du plan ; il n'est pas le plus eloigne. \emph{Renvoi : C1, projection orthogonale sur un plan.}
    \item \textbf{C} — Le point $H$ est impose par la perpendiculaire au plan passant par $M$ ; ce n'est pas un point quelconque. \emph{Renvoi : C1, projection orthogonale sur un plan.}
    \item \textbf{D} — $H=M$ uniquement lorsque $M$ appartient deja au plan ; ce n'est pas vrai en general. \emph{Renvoi : C1, projection orthogonale sur un plan.}
  \end{itemize}
\end{itemize}
