\subsection*{Erreur fréquente}

\erreurFrequente{
Dans $ax+by+cz+d=0$, lire $(a;b;c)$ comme un vecteur \emph{directeur} du plan.
}

\textbf{Pourquoi c'est faux.} $(a;b;c)$ est un vecteur \emph{normal}~: il est
orthogonal au plan, donc il ne s'y trouve pas. Le confondre avec une direction
du plan inverse complètement la géométrie de la situation.

\contreexemple{
Le plan $z=0$ s'écrit $0x+0y+1z+0=0$, de vecteur normal $\vec{n}(0;0;1)$. Ce
vecteur est vertical~: il n'appartient pas au plan horizontal, il lui est
perpendiculaire.
}

\subsection*{À retenir}

Un vecteur $\vec{v}$ dirige le plan si et seulement si
$\vec{n}\cdot\vec{v}=0$. Le point $M(x;y;z)$ appartient au plan si et
seulement si ses coordonnées annulent $ax+by+cz+d$.

\subsection*{Méthode corrective}

\begin{enumerate}
  \item Extraire $\vec{n}(a;b;c)$ de l'équation.
  \item Pour tester si un vecteur dirige le plan, calculer son produit
        scalaire avec $\vec{n}$~: il doit être nul.
  \item Pour tester si un point appartient au plan, remplacer dans l'équation.
  \item Ne jamais utiliser $(a;b;c)$ comme direction dans une représentation
        paramétrique du plan.
\end{enumerate}

\begin{exercice}{TSPE-GEOESPACE-RE-C13-EX1}{2}{12}
Repère orthonormé. Soit $\mathcal{P}$ le plan d'équation $2x-y+3z-5=0$.
\begin{enumerate}
  \item Donner un vecteur normal à $\mathcal{P}$.
  \item Le point $A(1;0;1)$ appartient-il à $\mathcal{P}$ ?
  \item Le vecteur $\vec{v}(1;2;0)$ dirige-t-il $\mathcal{P}$ ?
  \item Le vecteur normal $\vec{n}$ peut-il diriger $\mathcal{P}$ ? Justifier.
\end{enumerate}
\end{exercice}

\begin{corrige}{TSPE-GEOESPACE-RE-C13-EX1}
\begin{enumerate}
  \item $\vec{n}(2;-1;3)$ est normal à $\mathcal{P}$.
  \item $2\times1-0+3\times1-5=2+3-5=0$~: oui, $A\in\mathcal{P}$.
  \item $\vec{n}\cdot\vec{v}=2\times1+(-1)\times2+3\times0=0$~: oui,
        $\vec{v}$ dirige $\mathcal{P}$.
  \item Non. On aurait besoin de $\vec{n}\cdot\vec{n}=0$, or
        $\vec{n}\cdot\vec{n}=4+1+9=14\neq0$. Un vecteur normal non nul n'est
        jamais une direction du plan~: il lui est orthogonal.
\end{enumerate}
\end{corrige}
