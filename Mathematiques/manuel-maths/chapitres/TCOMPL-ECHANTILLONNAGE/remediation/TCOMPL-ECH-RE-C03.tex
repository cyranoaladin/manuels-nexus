\subsection*{Rappel -- Fonction exponentielle : propriétés et dérivée}

\textbf{Propriétés :} $\mathrm{e}^{a+b} = \mathrm{e}^a \mathrm{e}^b$, $\mathrm{e}^0 = 1$, $\mathrm{e}^x > 0$.

\textbf{Dérivée :} $(\mathrm{e}^x)' = \mathrm{e}^x$. \textbf{Limite :} $\lim_{x \to +\infty} \mathrm{e}^x = +\infty$, $\lim_{x \to -\infty} \mathrm{e}^x = 0$.

\begin{exercice}{TCOMPL-ECH-FR-R3-EX1}{1}{8}
\begin{enumerate}
  \item Simplifier $\mathrm{e}^{\ln 3}$ et $\mathrm{e}^{2x + 1} \cdot \mathrm{e}^{-x}$.
  \item Résoudre $\mathrm{e}^{2x} = \mathrm{e}^{x+2}$.
  \item Déterminer $\lim_{x \to +\infty} \dfrac{\mathrm{e}^x}{x^2}$.
\end{enumerate}
\end{exercice}
