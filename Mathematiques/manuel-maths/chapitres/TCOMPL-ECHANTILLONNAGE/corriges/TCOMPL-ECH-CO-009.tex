\begin{corrige}{TCOMPL-ECH-EX-009}
\textbf{1.} Pour $n=1$, $X$ vaut $1$ avec probabilité $p$ et $0$ sinon :
$E(X)=0\times(1-p)+1\times p=p$.

\textbf{2.} Pour $n=2$ : $P(X=0)=(1-p)^2$, $P(X=1)=2p(1-p)$,
$P(X=2)=p^2$. Donc
$E(X)=0+2p(1-p)+2p^2=2p(1-p)+2p^2=2p$.

\textbf{3.} Pour $n=3$ : $E(X)=3p(1-p)^2+2\times3p^2(1-p)+3p^3=3p$ après
développement. Les trois cas donnent $E(X)=np$, formule générale de
l'espérance d'une loi binomiale.
\end{corrige}
