\begin{corrige}{TCOMPL-ECH-EX-004}
\textbf{1.} $P(X=0)=(1-p)^2$, $P(X=1)=\dbinom{2}{1}p(1-p)=2p(1-p)$, $P(X=2)=p^2$.

\textbf{2.}
\[
  E(X) = 0\times(1-p)^2 + 1\times2p(1-p) + 2\times p^2 = 2p(1-p)+2p^2 = 2p-2p^2+2p^2 = 2p.
\]
On retrouve bien $E(X)=np$ avec $n=2$.
\end{corrige}
