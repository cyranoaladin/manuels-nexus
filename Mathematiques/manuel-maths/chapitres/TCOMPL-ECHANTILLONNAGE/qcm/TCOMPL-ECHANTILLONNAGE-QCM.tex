\section*{\textcolor{chapcolor}{\MakeUppercase{Echantillonnage et estimation}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} L'intervalle de confiance usuel au niveau 95% pour une proportion p avec un echantillon de taille n est :
  \begin{enumerate}[label=\Alph*.]
    \item $[f - \frac{1}{\sqrt{n}} ; f + \frac{1}{\sqrt{n}}]$
    \item $[f - \frac{1}{n} ; f + \frac{1}{n}]$
    \item $[f - 1,96 ; f + 1,96]$
    \item $[0 ; 1]$
  \end{enumerate}

\item \textbf{[Q2]} Pour diviser la largeur de l'intervalle de confiance par 2, la taille n de l'echantillon doit etre :
  \begin{enumerate}[label=\Alph*.]
    \item Doublee
    \item Triplee
    \item Multipliee par 4
    \item Divisee par 2
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q3]} La frequence observee f sur un echantillon de taille n suit approximativement :
  \begin{enumerate}[label=\Alph*.]
    \item Une loi uniforme
    \item Une loi normale sous les conditions d'application
    \item Une loi de Poisson
    \item Une loi d'exponentielle
  \end{enumerate}

\item \textbf{[Q4]} Un estimateur sans biais du milieu d'un intervalle de confiance est :
  \begin{enumerate}[label=\Alph*.]
    \item La taille n
    \item La variance
    \item L'ecart-type
    \item La frequence observee f
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q5]} Les conditions standard $n \ge 30$, $nf \ge 5$ et $n(1-f) \ge 5$ permettent d'assurer :
  \begin{enumerate}[label=\Alph*.]
    \item La validite de l'approximation normale de la frequence
    \item Que la taille n est infinie
    \item Que p est exactement egal a f
    \item Que la variance est nulle
  \end{enumerate}

\end{enumerate}
