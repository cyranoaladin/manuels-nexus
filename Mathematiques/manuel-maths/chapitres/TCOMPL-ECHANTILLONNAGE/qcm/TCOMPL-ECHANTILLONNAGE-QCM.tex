\subsection*{C1/C2 --- Bernoulli, binomiale, Pascal}

\begin{enumerate}

\item \textbf{[Q1]} Un schéma de Bernoulli suppose que les répétitions sont :
  \begin{enumerate}[label=\Alph*.]
    \item dépendantes les unes des autres.
    \item indépendantes, avec la même probabilité de succès.
    \item de probabilités toutes différentes.
    \item en nombre variable.
  \end{enumerate}

\bigskip

\item \textbf{[Q2]} $\binom{5}{2}$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $10$.
    \item $25$.
    \item $7$.
    \item $2$.
  \end{enumerate}

\subsection*{C3/C6 --- Calculs, espérance}

\item \textbf{[Q3]} Pour $X\sim\mathcal{B}(n,p)$, l'espérance $E(X)$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $n+p$.
    \item $np$.
    \item $n/p$.
    \item $p^n$.
  \end{enumerate}

\subsection*{C4 --- Intervalle de fluctuation}

\item \textbf{[Q4]} Un intervalle de fluctuation $I$ au seuil $95\%$ vérifie :
  \begin{enumerate}[label=\Alph*.]
    \item $P(X\in I) = 1$.
    \item $P(X\in I) \geq 0{,}95$.
    \item $P(X\in I) \leq 0{,}05$.
    \item $I$ contient toutes les valeurs possibles de $X$.
  \end{enumerate}

\bigskip

\item \textbf{[Q5]} Si une valeur observée est hors de l'intervalle de fluctuation au seuil $95\%$, on peut :
  \begin{enumerate}[label=\Alph*.]
    \item conclure avec certitude absolue que l'hypothèse est fausse.
    \item rejeter l'hypothèse, au risque de $5\%$ de se tromper.
    \item ne rien en conclure.
    \item accepter l'hypothèse.
  \end{enumerate}

\bigskip

\subsection*{C5 --- Simulation}

\item \textbf{[Q6]} Simuler $N=100$ échantillons de taille $n=50$ signifie :
  \begin{enumerate}[label=\Alph*.]
    \item tirer une seule fois $5000$ valeurs.
    \item répéter $100$ fois le tirage de $50$ valeurs, produisant $100$ moyennes.
    \item tirer $50$ valeurs une seule fois.
    \item tirer $100$ valeurs une seule fois.
  \end{enumerate}

\end{enumerate}
