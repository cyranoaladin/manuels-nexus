\begin{exercice}{TCOMPL-ECH-EX-004}{2}{15}
Soit $X \sim \mathcal{B}(2,p)$.
\begin{enumerate}
  \item Écrire $P(X=0)$, $P(X=1)$, $P(X=2)$ en fonction de $p$.
  \item En utilisant la définition $E(X) = \displaystyle\sum_{k=0}^{2} k\,P(X=k)$, démontrer que $E(X)=2p$.
\end{enumerate}
\end{exercice}
