\begin{exercice}{TCOMPL-ECH-EX-038}{2}{15}
\begin{enumerate}
  \item Une variable aléatoire $X$ suit la loi uniforme sur $\{1,2,\ldots,n\}$ et son espérance vaut $10{,}5$. Déterminer $n$.
  \item On lance deux dés équilibrés à six faces et on note $S$ la somme des deux numéros. Un élève affirme : « chaque dé suit une loi uniforme, donc $S$ aussi ». Réfuter cette affirmation en comparant $P(S=2)$ et $P(S=7)$.
  \item Calculer $E(S)$ et vérifier la cohérence avec l'espérance d'un seul dé.
\end{enumerate}
\end{exercice}
