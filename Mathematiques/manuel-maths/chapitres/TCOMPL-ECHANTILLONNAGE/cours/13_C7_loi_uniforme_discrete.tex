\section{La loi uniforme sur $\{1,2,\ldots,n\}$}

% ============================================================
% STRATE 1 — Definition
% ============================================================

\definition[1]{
Soit $n$ un entier naturel non nul. Une variable aleatoire $X$ suit la \textbf{loi uniforme} sur $\{1,2,\ldots,n\}$ lorsque ses $n$ valeurs possibles ont toutes la \textbf{même} probabilité :
\[
  P(X=k) = \frac{1}{n} \qquad \text{pour tout } k \in \{1,2,\ldots,n\}.
\]
}

\margeAppui{
C'est la traduction de l'equiprobabilite : aucune valeur n'est privilegiee. La somme des probabilités vaut bien $n \times \dfrac{1}{n} = 1$, comme l'exige toute loi de probabilité.
}

\exemple{
Le lancer d'un de equilibre a six faces : $X$ est le numero obtenu, $P(X=k) = \dfrac{1}{6}$ pour $k$ de $1$ a $6$.
}

% ============================================================
% STRATE 2 — Esperance
% ============================================================

\propriete[Espérance de la loi uniforme]{
Si $X$ suit la loi uniforme sur $\{1,2,\ldots,n\}$, alors
\[
  E(X) = \frac{n+1}{2}.
\]
}

\demonstration{
Par définition de l'espérance d'une variable aléatoire finie :
\[
  E(X) = \sum_{k=1}^{n} k\,P(X=k) = \sum_{k=1}^{n} k \times \frac{1}{n} = \frac{1}{n}\sum_{k=1}^{n} k .
\]
La somme des $n$ premiers entiers vaut $\displaystyle\sum_{k=1}^{n} k = \frac{n(n+1)}{2}$. Donc
\[
  E(X) = \frac{1}{n} \times \frac{n(n+1)}{2} = \frac{n+1}{2}.
\]
}

\margeAppui{
$\dfrac{n+1}{2}$ est le milieu de $1$ et de $n$ : pour une loi uniforme, l'espérance est le \textbf{centre} de l'intervalle des valeurs, ce qui est coherent avec l'idee de symetrie.
}

\exemple{
Pour le de a six faces, $E(X) = \dfrac{6+1}{2} = 3{,}5$ : sur un grand nombre de lancers, la moyenne des numeros obtenus s'approche de $3{,}5$.
}

\erreurFrequente{
\textbf{Confondre la loi uniforme sur $\{1,\ldots,n\}$ avec la loi uniforme sur $\{0,1,\ldots,n\}$.} La première porte sur $n$ valeurs et a pour espérance $\dfrac{n+1}{2}$ ; la seconde porte sur $n+1$ valeurs et a pour espérance $\dfrac{n}{2}$. Toujours compter les valeurs possibles avant d'appliquer une formule.
}
