\section{Calculer avec la loi binomiale}

% ============================================================
% STRATE 1 — Calcul de probabilites
% ============================================================

\propriete[Calculs usuels]{
Pour $X \sim \mathcal{B}(n,p)$, on calcule à la calculatrice ou en Python :
\begin{itemize}
  \item $P(X=k) = \binom{n}{k}p^k(1-p)^{n-k}$ ;
  \item $P(X\leq k) = \displaystyle\sum_{i=0}^{k} P(X=i)$ (fonction de répartition).
\end{itemize}
}

\exemple{
Soit $X \sim \mathcal{B}(10, 0{,}3)$. On calcule :
\[
  P(X=3) = \binom{10}{3}\times0{,}3^3\times0{,}7^7 \approx 0{,}2668.
\]
}

% ============================================================
% STRATE 2 — Esperance d'une binomiale, n<=3
% ============================================================

\theoreme[Espérance de la loi binomiale]{
Si $X \sim \mathcal{B}(n,p)$, alors $E(X) = np$.
}

\demonstration{
Démontrons le résultat pour $n=3$ (le cas général se traite de façon analogue). La loi de $X$ est donnée par $P(X=k)=\binom{3}{k}p^k(1-p)^{3-k}$ pour $k\in\{0,1,2,3\}$, avec $\binom{3}{0}=1$, $\binom{3}{1}=3$, $\binom{3}{2}=3$, $\binom{3}{3}=1$. Donc :
\[
  E(X) = \sum_{k=0}^{3} k\,P(X=k) = 0 + 1\times3p(1-p)^2 + 2\times3p^2(1-p) + 3\times p^3.
\]
En développant et en factorisant par $3p$ :
\[
  E(X) = 3p\big[(1-p)^2 + 2p(1-p) + p^2\big] = 3p\big[(1-p)+p\big]^2 = 3p \times 1 = 3p.
\]
On retrouve bien $E(X)=np$ avec $n=3$.
}

\erreurFrequente{
\textbf{Confondre $E(X)=np$ avec $P(X=n\times p)$.} L'espérance $np$ n'est \textbf{pas nécessairement un entier}, et ce n'est pas la valeur la plus probable de $X$ en général (même si elle en est souvent proche) : c'est une moyenne théorique, pas une probabilité.
}
