\begin{fichemethode}{M6}{Demontrer l'esperance d'une loi binomiale (n petits)}

\quandUtiliser{%
  Utiliser cette methode lorsque l'enonce demande de DEMONTRER que
  $E(X) = np$ pour $X$ suivant une loi binomiale $\mathcal{B}(n\,;\,p)$ avec
  $n \leq 3$, par le calcul direct de la definition.
}

\pasApas{%
  \begin{enumerate}
    \item Ecrire la loi de $X$ : pour $k = 0, 1, \dots, n$,
      $P(X = k) = \binom{n}{k} p^k (1-p)^{n-k}$.
    \item Ecrire la definition de l'esperance :
      $E(X) = \displaystyle\sum_{k=0}^{n} k\,P(X = k)$ (le terme $k = 0$ est
      nul).
    \item Substituer les valeurs de $n$ : pour $n = 3$,
      $E(X) = 1 \cdot 3p(1-p)^2 + 2 \cdot 3p^2(1-p) + 3 \cdot p^3$.
    \item Developper et regrouper : factoriser par $p$ (ou par $np$) et
      reconnaitre un developpement remarquable.
    \item Conclure $E(X) = np$ et verifier la coherence (esperance d'autant
      plus grande que $n$ ou $p$ augmente).
  \end{enumerate}
}

\exempleRedige{%
  Demontrer que pour $X \sim \mathcal{B}(3\,;\,p)$, $E(X) = 3p$.

  \medskip
  \commentaireMarge{Les coefficients binomiaux de $n = 3$ : 1, 3, 3, 1.}%
  $E(X) = 0 + 1 \times 3p(1-p)^2 + 2 \times 3p^2(1-p) + 3 \times p^3$.

  \medskip
  \commentaireMarge{Developper puis regrouper.}%
  $E(X) = 3p\left[(1-p)^2 + 2p(1-p) + p^2\right]
  = 3p\left[(1-p) + p\right]^2 = 3p \times 1 = 3p$. \hfill$\square$

  \medskip
  On reconnait l'identite $(a+b)^2$ avec $a = 1-p$, $b = p$ : la somme entre
  crochets vaut $\bigl((1-p)+p\bigr)^2 = 1$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piege 1.} Oublier les coefficients binomiaux dans
      $P(X = k)$ ($3p^2(1-p)$ et non $p^2(1-p)$ pour $k = 2$).
    \item \textbf{Piege 2.} Erreur de developpement dans le regroupement : la
      factorisation par $3p$ PUIS l'identite remarquable evitent le
      developpement brutal.
    \item \textbf{Piege 3.} Croire la preuve generale : ce calcul direct ne
      couvre que le $n$ traite — la formule generale $E(X) = np$ est admise
      (ou demontree par linearite en Terminale).
  \end{itemize}
}

\verifier{%
  Controler que les $P(X = k)$ somment a 1, tester la formule obtenue pour
  une valeur numerique de $p$ (ex. $p = \frac{1}{2}$ : $E = 1{,}5$), et
  refaire le calcul pour $n = 2$ ($E = 2p$).
}

\end{fichemethode}
