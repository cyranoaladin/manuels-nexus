\subsection*{Rappel -- Limites de fonctions et croissances comparees}

\textbf{Croissances compareees :} $\lim_{x \to +\infty} \dfrac{\mathrm{e}^x}{x^n} = +\infty$, $\lim_{x \to +\infty} \dfrac{\ln x}{x} = 0$, $\lim_{x \to +\infty} x\,\mathrm{e}^{-x} = 0$.

\begin{exercice}{TCOMPL-INEG-FR-R4-EX1}{1}{10}
Determiner les limites suivantes :
\begin{enumerate}
  \item $\lim_{x \to +\infty} \dfrac{x^3}{\mathrm{e}^x}$
  \item $\lim_{x \to 0^+} x \ln x$
  \item $\lim_{x \to +\infty} \left(x - \ln x\right)$
\end{enumerate}
\end{exercice}
