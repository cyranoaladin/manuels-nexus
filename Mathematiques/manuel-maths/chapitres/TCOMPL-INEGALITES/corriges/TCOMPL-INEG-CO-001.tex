\begin{corrige}{TCOMPL-INEG-EX-001}
\textbf{1.} $L(0) = \dfrac{0+0}{3} = 0$. $L(1) = \dfrac{2+1}{3} = \dfrac{3}{3} = 1$.

\textbf{2.} $L'(x) = \dfrac{4x+1}{3}$, donc $L''(x) = \dfrac{4}{3}$, qui est \textbf{constante et positive} : $L$ est convexe sur $[0,1]$ (et même sur $\mathbb{R}$).
