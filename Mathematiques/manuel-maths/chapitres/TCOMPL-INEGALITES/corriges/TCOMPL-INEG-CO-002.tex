\begin{corrige}{TCOMPL-INEG-EX-002}
\textbf{1.} L'aire cherchée est celle comprise entre la bissectrice $y=x$ et
la courbe de $L$ :
\[
  \mathcal{A} = \int_0^1\big(x - L(x)\big)\,\mathrm{d}x
  = \int_0^1 x\,\mathrm{d}x - \int_0^1 \frac{2x^2+x}{3}\,\mathrm{d}x .
\]
Or $\displaystyle\int_0^1 x\,\mathrm{d}x = \frac{1}{2}$ et
$\displaystyle\int_0^1 \frac{2x^2+x}{3}\,\mathrm{d}x
= \frac{1}{3}\left[\frac{2x^3}{3}+\frac{x^2}{2}\right]_0^1
= \frac{1}{3}\left(\frac{2}{3}+\frac{1}{2}\right) = \frac{7}{18}$.
Donc $\mathcal{A} = \dfrac{1}{2}-\dfrac{7}{18} = \dfrac{1}{9}$.

\textbf{2.} L'indice de Gini est le double de cette aire :
\[
  G = 2\mathcal{A} = \frac{2}{9} \approx 0{,}22.
\]
On retrouve la forme équivalente
$G = 1 - 2\displaystyle\int_0^1 L(x)\,\mathrm{d}x = 1 - \frac{7}{9} = \frac{2}{9}$.
\end{corrige}
