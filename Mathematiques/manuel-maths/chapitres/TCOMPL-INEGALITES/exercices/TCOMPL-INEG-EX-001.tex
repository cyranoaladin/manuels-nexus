\begin{exercice}{TCOMPL-INEG-EX-001}{1}{10}
On modélise une courbe de Lorenz par $L(x) = \dfrac{2x^2+x}{3}$ sur $[0,1]$.
\begin{enumerate}
  \item Vérifier que $L(0)=0$ et $L(1)=1$.
  \item Calculer $L''(x)$ et en déduire que $L$ est bien convexe sur $[0,1]$.
\end{enumerate}
\end{exercice}
