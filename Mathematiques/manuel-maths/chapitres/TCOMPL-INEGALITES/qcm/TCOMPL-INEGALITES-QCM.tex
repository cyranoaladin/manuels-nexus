\section*{\textcolor{chapcolor}{\MakeUppercase{Inegalites et approximations}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} L'inegalite de Bienayme-Tchebycheff s'ecrit $P(|X - \mu| \ge a) \le$
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{V(X)}{a}$
    \item $\frac{V(X)}{a^2}$
    \item $a^2 V(X)$
    \item $\frac{\sigma}{a}$
  \end{enumerate}

\item \textbf{[Q2]} L'inegalite de Markov s'applique a une variable aleatoire X a valeurs :
  \begin{enumerate}[label=\Alph*.]
    \item Quelconques
    \item Strictement negatives
    \item Positives ou nulles
    \item Entieres relatives
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q3]} L'inegalite de Markov affirme pour $X \ge 0$ et $a > 0$ que $P(X \ge a) \le$
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{E(X)}{a}$
    \item $\frac{E(X)}{a^2}$
    \item $\frac{V(X)}{a}$
    \item $a E(X)$
  \end{enumerate}

\item \textbf{[Q4]} La loi forte des grands nombres stipule que la moyenne d'un echantillon converge vers :
  \begin{enumerate}[label=\Alph*.]
    \item L'ecart-type
    \item La variance
    \item 0
    \item L'esperance mathématique
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q5]} Si $V(X) = 4$, la probabilite $P(|X - E(X)| \ge 4)$ est majoree par :
  \begin{enumerate}[label=\Alph*.]
    \item 1
    \item 1/4
    \item 1/2
    \item 1/16
  \end{enumerate}

\end{enumerate}
