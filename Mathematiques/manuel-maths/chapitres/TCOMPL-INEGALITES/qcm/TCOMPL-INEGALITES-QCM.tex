\subsection*{C1/C2 --- Courbe de Lorenz}

\begin{enumerate}

\item \textbf{[Q1]} Une courbe de Lorenz $L$ vérifie toujours :
  \begin{enumerate}[label=\Alph*.]
    \item $L(0)=1$ et $L(1)=0$.
    \item $L(0)=0$ et $L(1)=1$.
    \item $L(0)=L(1)$.
    \item $L(x)>x$ pour tout $x$.
  \end{enumerate}

\bigskip

\item \textbf{[Q2]} La fonction modélisant une courbe de Lorenz doit être :
  \begin{enumerate}[label=\Alph*.]
    \item décroissante et concave.
    \item croissante et convexe.
    \item constante.
    \item croissante et concave.
  \end{enumerate}

\bigskip

\subsection*{C3 --- Indice de Gini}

\item \textbf{[Q3]} Un indice de Gini égal à $0$ correspond à :
  \begin{enumerate}[label=\Alph*.]
    \item une inégalité maximale.
    \item une égalité parfaite.
    \item une erreur de calcul.
    \item une répartition impossible.
  \end{enumerate}

\bigskip

\item \textbf{[Q4]} Entre deux répartitions dont les courbes de Lorenz sont $L_1$ et $L_2$ avec $L_1(x) > L_2(x)$ pour tout $x\in\,]0,1[$, la répartition la plus inégalitaire est celle de :
  \begin{enumerate}[label=\Alph*.]
    \item indice $G_1$ (associé à $L_1$).
    \item indice $G_2$ (associé à $L_2$).
    \item elles sont forcément égales.
    \item impossible à dire sans calcul.
  \end{enumerate}

\bigskip

\subsection*{C5 --- Calcul par intégrale}

\item \textbf{[Q5]} La formule $G = 1 - 2\displaystyle\int_0^1 L(x)\,\mathrm{d}x$ découle du fait que :
  \begin{enumerate}[label=\Alph*.]
    \item l'aire sous la bissectrice sur $[0,1]$ vaut $1$.
    \item l'aire sous la bissectrice sur $[0,1]$ vaut $\dfrac{1}{2}$.
    \item $L$ est toujours égale à $x$.
    \item l'intégrale de $L$ vaut toujours $1$.
  \end{enumerate}

\bigskip

\item \textbf{[Q6]} Pour $L(x)=x$ (égalité parfaite), $\displaystyle\int_0^1 L(x)\,\mathrm{d}x$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $1$.
    \item $\dfrac{1}{2}$.
    \item $0$.
    \item $2$.
  \end{enumerate}

\end{enumerate}
