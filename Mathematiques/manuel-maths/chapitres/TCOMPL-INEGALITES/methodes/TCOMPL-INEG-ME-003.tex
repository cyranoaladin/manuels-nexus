\begin{fichemethode}{M3}{Calculer et interpréter un indice de Gini}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande l'indice de Gini d'une
  repartition, a partir de l'aire entre la courbe de Lorenz et la bissectrice.
}

\pasApas{%
  \begin{enumerate}
    \item Noter $\mathcal{A}$ l'aire du domaine compris entre la bissectrice
      $y = x$ et la courbe de Lorenz sur $[0\,;\,1]$.
    \item Definir l'indice de Gini : $G = 2\mathcal{A}$ (l'aire sous la
      bissectrice valant $\frac{1}{2}$, on normalise pour que $G \in [0\,;\,1]$).
    \item Calculer $\mathcal{A}$ : par intégrale si $L$ est modélisée
      (fiche M5), ou par decoupage en trapezes si la courbe est donnee par
      points.
    \item Conclure $G = 2\mathcal{A}$.
    \item Interpréter : $G = 0$ égalité parfaite ($L$ = bissectrice) ;
      $G$ proche de 1 = repartition très inegalitaire ; comparer deux
      repartitions par leurs indices.
  \end{enumerate}
}

\exempleRedige{%
  La courbe de Lorenz d'une repartition est modélisée par $L(x) = x^2$.
  Calculer l'indice de Gini et interpréter.

  \medskip
  \commentaireMarge{$G = 2\mathcal{A}$, aire entre $y=x$ et $L$.}%
  $\mathcal{A} = \displaystyle\int_0^1 \bigl(x - x^2\bigr)\,dx
  = \left[\dfrac{x^2}{2} - \dfrac{x^3}{3}\right]_0^1 = \dfrac{1}{6}$,
  donc $G = 2 \times \dfrac{1}{6} = \dfrac{1}{3} \approx 0{,}33$.

  \medskip
  Interprétation : $G = 0{,}33$ traduit une inégalité moderee — plus proche de
  l'égalité parfaite ($0$) que d'une concentration extreme ($1$).
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Oublier le facteur 2 : $G = 2\mathcal{A}$, pas
      $\mathcal{A}$.
    \item \textbf{Piège 2.} Integrer $L(x) - x$ (négatif) : l'aire est
      $\int_0^1 (x - L(x))\,dx$ car la bissectrice est AU-DESSUS.
    \item \textbf{Piège 3.} Comparer des indices de Gini calcules sur des
      grandeurs différentes sans le signaler (revenus vs patrimoines).
  \end{itemize}
}

\verifier{%
  Contrôler $0 \leq G \leq 1$, tester les cas limites ($L(x) = x$ donne $G=0$),
  et vérifier la cohérence avec l'allure de la courbe (plus creusee = $G$ plus
  grand).
}

\end{fichemethode}
