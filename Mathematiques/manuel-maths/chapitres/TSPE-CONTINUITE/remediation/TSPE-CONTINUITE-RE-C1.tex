\begin{exercice}{TSPE-CONTIN-RE-C1-EX1}{1}{10}
Soit $f(x) = x^3 + 2x - 5$ sur $[1, 2]$.
\begin{enumerate}
  \item Justifier la continuite de $f$ sur $[1,2]$, calculer $f(1)$ et $f(2)$.
  \item Etudier le signe de $f'$ sur $[1,2]$.
  \item Justifier que l'equation $f(x)=0$ admet une unique solution dans $[1,2]$.
\end{enumerate}
\end{exercice}

\begin{corrige}{TSPE-CONTIN-RE-C1-EX1}
\textbf{a.} $f$ polynome, continue sur $\mathbb{R}$. $f(1) = 1+2-5=-2$, $f(2)=8+4-5=7$.

\textbf{b.} $f'(x) = 3x^2+2 > 0$ pour tout $x$ : $f$ strictement croissante.

\textbf{c.} $f$ continue et strictement croissante sur $[1,2]$, $0$ compris entre $f(1)=-2$ et $f(2)=7$ : par le TVI (cas monotone), unique solution dans $[1,2]$.
\end{corrige}
