\begin{fichemethode}{M2}{Étudier une suite définie par $u_{n+1}=f(u_n)$}

\quandUtiliser{%
  Utiliser cette méthode lorsque $(u_n)$ est définie par une relation de récurrence $u_{n+1} = f(u_n)$ avec $f$ continue.
}

\pasApas{%
  \begin{enumerate}
    \item \textbf{Stabilite :} montrer par récurrence que $u_n \in I$ pour tout $n$, ou $I$ est un intervalle tel que $f(I) \subset I$.
    \item \textbf{Monotonie :} étudier le signe de $f(x)-x$ sur $I$, ou comparer directement $u_{n+1}$ et $u_n$ par récurrence.
    \item \textbf{Convergence :} si $(u_n)$ est monotone et bornee (consequence de l'étape 1), conclure a la convergence vers un $\ell \in I$.
    \item \textbf{Identification de la limite :} $f$ etant continue, $\ell$ vérifie $f(\ell)=\ell$. Résoudre cette équation dans $I$ pour déterminer $\ell$.
  \end{enumerate}
}

\end{fichemethode}
