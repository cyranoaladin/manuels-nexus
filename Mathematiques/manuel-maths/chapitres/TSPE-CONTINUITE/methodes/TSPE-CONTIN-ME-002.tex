\begin{fichemethode}{M2}{Étudier une suite définie par $u_{n+1}=f(u_n)$}

\quandUtiliser{%
  Utiliser cette methode lorsque $(u_n)$ est definie par une relation de recurrence $u_{n+1} = f(u_n)$ avec $f$ continue.
}

\pasApas{%
  \begin{enumerate}
    \item \textbf{Stabilite :} montrer par recurrence que $u_n \in I$ pour tout $n$, ou $I$ est un intervalle tel que $f(I) \subset I$.
    \item \textbf{Monotonie :} etudier le signe de $f(x)-x$ sur $I$, ou comparer directement $u_{n+1}$ et $u_n$ par recurrence.
    \item \textbf{Convergence :} si $(u_n)$ est monotone et bornee (consequence de l'etape 1), conclure a la convergence vers un $\ell \in I$.
    \item \textbf{Identification de la limite :} $f$ etant continue, $\ell$ verifie $f(\ell)=\ell$. Resoudre cette equation dans $I$ pour determiner $\ell$.
  \end{enumerate}
}

\end{fichemethode}
