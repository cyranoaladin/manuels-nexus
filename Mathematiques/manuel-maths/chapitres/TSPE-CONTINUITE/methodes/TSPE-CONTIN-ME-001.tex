\begin{fichemethode}{M1}{Appliquer le théorème des valeurs intermédiaires}

\quandUtiliser{%
  Utiliser cette methode lorsque l'enonce demande de montrer l'existence (et eventuellement l'unicite) d'une solution a une equation $f(x) = k$, sans la calculer explicitement.
}

\pasApas{%
  \begin{enumerate}
    \item \textbf{Justifier la continuite} de $f$ sur l'intervalle $[a,b]$ considere (somme/produit/quotient/composee de fonctions usuelles continues).
    \item \textbf{Calculer ou encadrer} $f(a)$ et $f(b)$, et verifier que $k$ est compris entre les deux.
    \item \textbf{Conclure a l'existence} d'un $c \in [a,b]$ tel que $f(c)=k$ par le TVI.
    \item \textbf{Pour l'unicite} : etudier le signe de $f'$ pour justifier la stricte monotonie de $f$ sur $[a,b]$, puis conclure par le corollaire du TVI.
  \end{enumerate}
}

\end{fichemethode}
