\begin{exercice}{TSPE-CONTIN-EV-B-EX1}{2}{25}

\textbf{Exercice 1} (10 points) — \textit{Capacite C1}

\medskip

Soit $f(x) = \ln(x) + 2x - 5$ sur $[1, 3]$.
\begin{enumerate}
  \item (3 pts) Calculer $f'(x)$ et justifier que $f$ est strictement croissante sur $[1,3]$.
  \item (2 pts) Calculer $f(1)$. Que vaut $f(3)$ (arrondir a $10^{-2}$) ?
  \item (5 pts) Justifier qu'il existe un unique reel $\alpha \in [1,3]$ tel que $f(\alpha) = 0$.
\end{enumerate}
\end{exercice}

%===============================================================

\begin{exercice}{TSPE-CONTIN-EV-B-EX2}{2}{25}

\textbf{Exercice 2} (10 points) — \textit{Capacite C2}

\medskip

Soit $(u_n)$ definie par $u_0 = 1$ et $u_{n+1} = \sqrt{3u_n}$. On pose $f(x) = \sqrt{3x}$.
\begin{enumerate}
  \item (3 pts) Montrer par recurrence que $1 \leq u_n \leq 3$ pour tout $n$.
  \item (3 pts) Montrer que $(u_n)$ est croissante.
  \item (4 pts) Justifier que $(u_n)$ converge vers un reel $\ell$, et calculer $\ell$.
\end{enumerate}
\end{exercice}
