\begin{corrige}{TSPE-CONTIN-EX-017}

\textbf{Q1.} $f(x) = x^2 + x^{-1}$, donc $f'(x) = 2x - \dfrac{1}{x^2}$.

\textbf{Q2.} Pour $x \in [1\,;2]$ : $2x \geq 2$ et $\dfrac{1}{x^2} \leq 1$ (car $x^2 \geq 1$). Donc $f'(x) \geq 2 - 1 = 1 > 0$.

\textbf{Q3.} $f(1) = 1 + 1 = 2$ et $f(2) = 4 + \dfrac{1}{2} = 4{,}5$.

\textbf{Q4.} $f$ est continue et strictement croissante sur $[1\,;2]$, et $3$ est compris entre $f(1) = 2$ et $f(2) = 4{,}5$ : l'equation $f(x) = 3$ admet une unique solution $\alpha$ dans $[1\,;2]$. Comme $f(1{,}5) = 2{,}25 + \dfrac{2}{3} \approx 2{,}9167 < 3$ et $f(1{,}6) = 2{,}56 + 0{,}625 = 3{,}185 > 3$, on a $1{,}5 < \alpha < 1{,}6$.
\end{corrige}
