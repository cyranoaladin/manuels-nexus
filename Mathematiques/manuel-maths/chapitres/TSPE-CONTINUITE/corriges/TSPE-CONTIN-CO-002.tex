\begin{corrige}{TSPE-CONTIN-EX-002}

\textbf{Q1.} $f$ est la somme d'une fonction exponentielle et d'une fonction affine, toutes deux continues sur $\mathbb{R}$ : $f$ est donc continue sur $[0,2]$.

\textbf{Q2.} $f(0) = \mathrm{e}^0 - 0 - 1 = 1 - 1 = 0$. Donc $x=0$ est deja une solution (evidente) de $f(x)=0$.

\textbf{Q3.} $f(2) = \mathrm{e}^2 - 5 \approx 2{,}39$. Comme $f$ est continue sur $[0,2]$ et que $2$ est compris entre $f(0)=0$ et $f(2)\approx 2{,}39$, le theoreme des valeurs intermediaires assure l'existence d'au moins un $c \in [0,2]$ tel que $f(c) = 2$.

\end{corrige}
