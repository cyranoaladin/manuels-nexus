\textbf{Q1.} $f'(x) = 5x^4 + 1$. Comme $x^4 \geq 0$ pour tout $x$, on a $f'(x) \geq 1 > 0$ : $f$ est strictement croissante sur $\mathbb{R}$.

\textbf{Q2.} $\lim_{x \to -\infty} f(x) = -\infty$ et $\lim_{x \to +\infty} f(x) = +\infty$ (terme de plus haut degre $x^5$). $f$ est continue et strictement croissante sur $\mathbb{R}$, et $0$ est compris entre ces deux limites. Par le TVI (cas strictement monotone, applique sur un intervalle $[-M, M]$ suffisamment grand), l'equation $f(x) = 0$ admet une unique solution reelle $\alpha$.

\textbf{Q3.} $f(0) = -1 < 0$ et $f(1) = 1 > 0$. Comme $f$ est strictement croissante, $\alpha$ est le seul reel ou $f$ s'annule ; puisque $f(0)<0<f(1)$, on a $\alpha \in [0,1]$.

def f(x):
    return x**5 + x - 1

def dichotomie(a, b, precision):
    while b - a > precision:
        m = (a + b) / 2
        if f(a) * f(m) <= 0:
            b = m
        else:
            a = m
    return (a, b)

encadrement = dichotomie(0, 1, 1e-3)
print(encadrement)
