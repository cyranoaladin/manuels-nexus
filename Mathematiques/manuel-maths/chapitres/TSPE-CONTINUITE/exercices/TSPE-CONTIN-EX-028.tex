\begin{exercice}{TSPE-CONTIN-EX-028}{2}{15}
Soit $g$ la fonction definie sur $[0\,;4]$ par $g(x) = x^3 - 6x^2 + 9x$.
\begin{enumerate}
  \item Montrer que $g'(x) = 3(x-1)(x-3)$ et dresser le tableau de variations de $g$ sur $[0\,;4]$.
  \item Calculer $g(0)$, $g(1)$, $g(3)$ et $g(4)$.
  \item Expliquer pourquoi le theoreme des valeurs intermediaires ne peut pas s'appliquer directement a $g$ sur $[0\,;4]$ pour conclure a l'unicite d'une solution.
  \item Determiner le nombre de solutions de l'equation $g(x) = 2$ sur $[0\,;4]$, en traitant chaque intervalle de monotonie.
  \item Verifier en remarquant que $2$ est solution evidente de $g(x) = 2$, puis en factorisant $g(x) - 2$.
\end{enumerate}
\end{exercice}
