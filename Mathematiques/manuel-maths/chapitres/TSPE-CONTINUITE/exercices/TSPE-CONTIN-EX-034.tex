\begin{exercice}{TSPE-CONTIN-EX-034}{1}{12}
On considere la suite $(u_n)$ definie par $u_0 = 0$ et, pour tout entier $n \geq 0$,
\[ u_{n+1} = \sqrt{u_n + 2}. \]
On pose $f(x) = \sqrt{x+2}$.
\begin{enumerate}
  \item Calculer $u_1$, $u_2$ et $u_3$ a $10^{-3}$ pres.
  \item Montrer que si $x \in [0\,;2]$, alors $f(x) \in [0\,;2]$ : l'intervalle $[0\,;2]$ est stable par $f$.
  \item Demontrer par recurrence que, pour tout $n$, $0 \leq u_n \leq 2$.
  \item Etudier le signe de $f(x) - x$ sur $[0\,;2]$ et en deduire que $(u_n)$ est croissante.
  \item Justifier que $(u_n)$ converge, puis determiner sa limite.
\end{enumerate}
\end{exercice}
