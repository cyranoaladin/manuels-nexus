\begin{exercice}{TSPE-CONTIN-EX-012}{1}{8}
Soit $f$ la fonction definie sur $[1\,;2]$ par $f(x) = x^5 + x - 3$.
\begin{enumerate}
  \item Montrer que $f$ est strictement croissante sur $[1\,;2]$.
  \item Calculer $f(1)$ et $f(2)$.
  \item En deduire que l'equation $f(x) = 0$ admet une unique solution $\alpha$ dans $[1\,;2]$.
  \item Encadrer $\alpha$ a $0{,}1$ pres.
\end{enumerate}
\end{exercice}
