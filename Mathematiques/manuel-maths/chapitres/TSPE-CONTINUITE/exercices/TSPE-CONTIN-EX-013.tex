\begin{exercice}{TSPE-CONTIN-EX-013}{1}{10}
Soit $f$ la fonction definie sur $[2\,;3]$ par $f(x) = 2x^3 - 9x^2 + 12x$.
\begin{enumerate}
  \item Calculer $f'(x)$ et verifier que $f'(x) = 6(x-1)(x-2)$.
  \item Etudier le signe de $f'(x)$ sur $[2\,;3]$ et en deduire le sens de variation de $f$.
  \item Calculer $f(2)$ et $f(3)$.
  \item Montrer que l'equation $f(x) = 6$ admet une unique solution $\alpha$ dans $[2\,;3]$, puis encadrer $\alpha$ a $0{,}1$ pres.
\end{enumerate}
\end{exercice}
