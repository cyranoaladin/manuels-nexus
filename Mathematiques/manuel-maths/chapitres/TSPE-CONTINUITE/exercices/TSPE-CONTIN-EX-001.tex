\begin{exercice}{TSPE-CONTIN-EX-001}{1}{8}
Soit $f(x) = x^3 - 3x + 1$ sur $[0, 2]$.
\begin{enumerate}
  \item Justifier que $f$ est continue sur $[0, 2]$.
  \item Calculer $f(0)$ et $f(2)$.
  \item Justifier que l'equation $f(x) = 2$ admet au moins une solution dans $[0, 2]$.
\end{enumerate}
\end{exercice}
