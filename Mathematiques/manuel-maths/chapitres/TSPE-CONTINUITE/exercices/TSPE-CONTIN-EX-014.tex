\begin{exercice}{TSPE-CONTIN-EX-014}{1}{10}
Soit $f$ la fonction definie sur $[1\,;4]$ par $f(x) = \sqrt{x} + x - 3$.
\begin{enumerate}
  \item Calculer $f'(x)$ et justifier que $f'(x) > 0$ sur $[1\,;4]$.
  \item Calculer $f(1)$ et $f(4)$.
  \item Montrer que l'equation $f(x) = 0$ admet une unique solution $\alpha$ dans $[1\,;4]$.
  \item Montrer que $1{,}6 < \alpha < 1{,}7$.
\end{enumerate}
\end{exercice}
