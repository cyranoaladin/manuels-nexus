\begin{exercice}{TSPE-CONTIN-EX-007}{2}{15}
Soit $(u_n)$ definie par $u_0 = 0$ et $u_{n+1} = \dfrac{1}{3}u_n + 2$. On pose $f(x) = \dfrac{1}{3}x + 2$.
\begin{enumerate}
  \item Montrer par recurrence que $0 \leq u_n \leq 3$ pour tout $n$.
  \item Montrer que $(u_n)$ est croissante (on pourra etudier le signe de $u_{n+1}-u_n$ par recurrence, en commencant par verifier $u_1 \geq u_0$).
  \item Justifier que $(u_n)$ converge vers un reel $\ell$, et calculer $\ell$ en utilisant la continuite de $f$.
  \item Retrouver ce resultat en montrant que la suite $(v_n)$ definie par $v_n = u_n - 3$ est geometrique.
\end{enumerate}
\end{exercice}
