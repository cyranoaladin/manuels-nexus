\begin{exercice}{TSPE-CONTIN-EX-046}{2}{15}
On considere de nouveau $f(x) = \sqrt{x+2}$ sur $[0\,;3]$ et une suite $(u_n)$ verifiant $u_{n+1} = f(u_n)$.

\begin{enumerate}
  \item Calculer $f'(x)$ et montrer que, pour tout $x \in [0\,;3]$, $0 < f'(x) \leq \dfrac{1}{2\sqrt2}$.
  \item En admettant l'inegalite des accroissements finis, justifier que pour tous $a, b$ de $[0\,;3]$ :
  \[ \left|f(b) - f(a)\right| \leq \frac{1}{2\sqrt2}\,\left|b - a\right|. \]
  \item En prenant $a = 2$ (point fixe) et $b = u_n$, montrer que $\left|u_{n+1} - 2\right| \leq \dfrac{1}{2\sqrt2}\left|u_n - 2\right|$.
  \item En deduire que $\left|u_n - 2\right| \leq \left(\dfrac{1}{2\sqrt2}\right)^{n}\left|u_0 - 2\right|$, puis que $(u_n)$ converge vers $2$ quel que soit $u_0 \in [0\,;3]$.
  \item Combien d'iterations suffisent pour garantir $\left|u_n - 2\right| < 10^{-6}$ en partant de $u_0 = 0$ ?
\end{enumerate}
\end{exercice}
