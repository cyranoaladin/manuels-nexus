\begin{exercice}{TSPE-CONTIN-EX-002}{1}{8}
Soit $f(x) = \mathrm{e}^x - 2x - 1$ sur $[0, 2]$.
\begin{enumerate}
  \item Justifier que $f$ est continue sur $[0, 2]$.
  \item Calculer $f(0)$. Que peut-on en deduire sur l'equation $f(x) = 0$ ?
  \item Montrer que l'equation $f(x) = 2$ admet au moins une solution dans $[0, 2]$.
\end{enumerate}
\end{exercice}
