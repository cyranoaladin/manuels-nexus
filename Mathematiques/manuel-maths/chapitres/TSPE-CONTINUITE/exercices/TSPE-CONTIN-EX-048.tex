\begin{exercice}{TSPE-CONTIN-EX-048}{3}{20}
Soit $(u_n)$ definie par $u_0 = 1$ et $u_{n+1} = \dfrac{1}{1 + u_n}$.
\begin{enumerate}
  \item Calculer $u_1$ a $u_6$ sous forme de fractions irreductibles. Reconnaitre les numerateurs et denominateurs obtenus.
  \item Montrer que tous les termes appartiennent a $\left[\dfrac12\,;1\right]$.
  \item Resoudre $f(x) = x$ sur $\left]0\,;+\infty\right[$, ou $f(x) = \dfrac{1}{1+x}$. On note $\varphi$ la solution.
  \item Montrer que la suite n'est pas monotone, mais que les sous-suites $(u_{2n})$ et $(u_{2n+1})$ le sont, en precisant leur sens de variation.
  \item Montrer que $\left|u_{n+1} - \varphi\right| \leq \dfrac{2}{3}\left|u_n - \varphi\right|$, puis conclure que $(u_n)$ converge vers $\varphi$.
\end{enumerate}
\end{exercice}
