\begin{exercice}{TSPE-CONTIN-EX-025}{2}{15}
Soit $f$ la fonction definie sur $[-2\,;2]$ par $f(x) = x^3 - 3x$, et soit $m$ un reel.
\begin{enumerate}
  \item Montrer que $f'(x) = 3(x-1)(x+1)$ et dresser le tableau de variations de $f$ sur $[-2\,;2]$.
  \item Calculer $f(-2)$, $f(-1)$, $f(1)$ et $f(2)$.
  \item Discuter, selon la valeur de $m$, le nombre de solutions de l'equation $f(x) = m$ sur $[-2\,;2]$.
  \item Verifier la reponse pour $m = 0$ en resolvant l'equation exactement.
\end{enumerate}
\end{exercice}
