\begin{exercice}{TSPE-CONTIN-EX-011}{1}{8}
Soit $f$ la fonction definie sur $[-1\,;0]$ par $f(x) = \mathrm{e}^{x} + x$.
\begin{enumerate}
  \item Justifier que $f$ est derivable et calculer $f'(x)$.
  \item Determiner le sens de variation de $f$ sur $[-1\,;0]$.
  \item Montrer que l'equation $f(x) = 0$ admet une unique solution $\alpha$ dans $[-1\,;0]$.
  \item Montrer que $-0{,}6 < \alpha < -0{,}5$.
\end{enumerate}
\end{exercice}
