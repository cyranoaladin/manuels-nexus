\begin{exercice}{TSPE-CONTIN-EX-044}{2}{15}
Soit $(u_n)$ definie par $u_0 = 0$ et $u_{n+1} = \dfrac{u_n^2}{4} + 1$, et soit $f(x) = \dfrac{x^2}{4} + 1$.
\begin{enumerate}
  \item Calculer $u_1$, $u_2$, $u_3$ et $u_4$.
  \item Resoudre $f(x) = x$. Que remarque-t-on sur la nature de la racine ?
  \item Montrer que $[0\,;2]$ est stable par $f$, puis que $0 \leq u_n \leq 2$ pour tout $n$.
  \item Montrer que $f(x) - x = \dfrac{(x-2)^2}{4}$ et en deduire le sens de variation de $(u_n)$.
  \item Determiner la limite de $(u_n)$. Comparer la vitesse de convergence a celle de l'exercice \textsc{ex-041} et proposer une explication a partir de Q4.
\end{enumerate}
\end{exercice}
