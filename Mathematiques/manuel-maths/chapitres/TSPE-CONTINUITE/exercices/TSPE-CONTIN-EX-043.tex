\begin{exercice}{TSPE-CONTIN-EX-043}{2}{18}
On definit $(u_n)$ par $u_0 = 2$ et $u_{n+1} = \dfrac{1}{2}\left(u_n + \dfrac{3}{u_n}\right)$. Cette suite est l'\emph{algorithme de Heron} pour approcher $\sqrt3$.
\begin{enumerate}
  \item Calculer $u_1$, $u_2$ et $u_3$ a $10^{-7}$ pres. Comparer a $\sqrt3 \approx 1{,}7320508$.
  \item Resoudre $f(x) = x$ sur $\left]0\,;+\infty\right[$, ou $f(x) = \dfrac{1}{2}\left(x + \dfrac{3}{x}\right)$.
  \item Montrer que, pour tout $x > 0$, $f(x) - \sqrt3 = \dfrac{\left(x - \sqrt3\right)^2}{2x}$. En deduire que $u_n \geq \sqrt3$ pour tout $n \geq 1$.
  \item Montrer que $f(x) - x = \dfrac{3 - x^2}{2x}$, puis que $(u_n)$ est decroissante a partir du rang $1$.
  \item Conclure sur la convergence et la limite de $(u_n)$.
\end{enumerate}
\end{exercice}
