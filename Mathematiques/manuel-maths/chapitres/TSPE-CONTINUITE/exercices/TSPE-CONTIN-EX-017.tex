\begin{exercice}{TSPE-CONTIN-EX-017}{1}{10}
Soit $f$ la fonction definie sur $[1\,;2]$ par $f(x) = x^2 + \dfrac{1}{x}$.
\begin{enumerate}
  \item Calculer $f'(x)$.
  \item Montrer que $f'(x) > 0$ pour tout $x \in [1\,;2]$.
  \item Calculer $f(1)$ et $f(2)$.
  \item Montrer que l'equation $f(x) = 3$ admet une unique solution $\alpha$ dans $[1\,;2]$, puis encadrer $\alpha$ a $0{,}1$ pres.
\end{enumerate}
\end{exercice}
