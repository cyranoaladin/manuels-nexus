\begin{exercice}{TSPE-CONTIN-EX-003}{2}{12}
Soit $f(x) = x^3 - 3x + 3$ sur $[-3, -1]$.
\begin{enumerate}
  \item Calculer $f'(x)$ et montrer que $f'(x) > 0$ pour tout $x \in \left]-3, -1\right[$. Que vaut $f'(-1)$ ?
  \item En deduire que $f$ est strictement croissante sur $[-3, -1]$ tout entier, bornes comprises.
  \item Calculer $f(-3)$ et $f(-1)$.
  \item Montrer que l'equation $f(x) = 1$ admet une unique solution $\alpha$ dans $[-3, -1]$.
\end{enumerate}
\end{exercice}
