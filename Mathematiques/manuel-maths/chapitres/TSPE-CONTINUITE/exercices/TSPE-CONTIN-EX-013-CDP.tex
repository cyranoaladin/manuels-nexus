\begin{coupdepouce}{TSPE-CONTIN-EX-013}
\coupDePouce{1}{Pour Q1, developper $6(x-1)(x-2)$ et comparer avec $f'(x)$.}
\coupDePouce{2}{En $x=2$ la derivee s'annule : cela n'empeche pas la stricte croissance, car c'est une borne isolee.}
\coupDePouce{3}{Ici le nombre vise n'est pas $0$ mais $6$ : c'est $6$ qu'il faut situer entre $f(2)$ et $f(3)$.}
\end{coupdepouce}
