\begin{exercice}{TSPE-CONTIN-EX-015}{1}{10}
Soit $f$ la fonction definie sur $[0\,;3]$ par $f(x) = \dfrac{x}{x+1}$.
\begin{enumerate}
  \item Montrer que $f'(x) = \dfrac{1}{(x+1)^2}$, puis donner le sens de variation de $f$ sur $[0\,;3]$.
  \item Calculer $f(0)$ et $f(3)$.
  \item Montrer que l'equation $f(x) = 0{,}6$ admet une unique solution $\alpha$ dans $[0\,;3]$.
  \item Resoudre algebriquement $f(x) = 0{,}6$ et donner la valeur exacte de $\alpha$.
\end{enumerate}
\end{exercice}
