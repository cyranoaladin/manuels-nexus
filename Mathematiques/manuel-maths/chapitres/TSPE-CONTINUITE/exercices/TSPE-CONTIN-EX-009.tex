\begin{exercice}{TSPE-CONTIN-EX-009}{1}{8}
Soit $f$ la fonction definie sur $[0\,;1]$ par $f(x) = x^3 + x - 1$.
\begin{enumerate}
  \item Calculer $f'(x)$ et determiner son signe sur $[0\,;1]$.
  \item Calculer $f(0)$ et $f(1)$.
  \item Montrer que l'equation $f(x) = 0$ admet une unique solution $\alpha$ dans $[0\,;1]$.
  \item Justifier que $0{,}6 < \alpha < 0{,}7$.
\end{enumerate}
\end{exercice}
