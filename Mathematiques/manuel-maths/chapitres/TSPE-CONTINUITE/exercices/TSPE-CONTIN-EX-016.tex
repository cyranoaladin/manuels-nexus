\begin{exercice}{TSPE-CONTIN-EX-016}{1}{10}
Soit $f$ la fonction definie sur $[0\,;1]$ par $f(x) = x\,\mathrm{e}^{-x}$.
\begin{enumerate}
  \item Montrer que $f'(x) = (1-x)\,\mathrm{e}^{-x}$.
  \item Etudier le signe de $f'(x)$ sur $[0\,;1]$ et en deduire le sens de variation de $f$.
  \item Calculer $f(0)$ et donner une valeur approchee de $f(1)$ a $10^{-3}$ pres.
  \item Montrer que l'equation $f(x) = 0{,}2$ admet une unique solution $\alpha$ dans $[0\,;1]$, puis encadrer $\alpha$ a $0{,}1$ pres.
\end{enumerate}
\end{exercice}
