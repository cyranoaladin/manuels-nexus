\begin{exercice}{TSPE-CONTIN-EX-010}{1}{8}
Soit $f$ la fonction definie sur $[0\,;1]$ par $f(x) = x^3 - 3x^2 + 1$.
\begin{enumerate}
  \item Calculer $f'(x)$ et le factoriser.
  \item Montrer que $f'(x) < 0$ pour tout $x \in \left]0\,;1\right[$.
  \item Montrer que l'equation $f(x) = 0$ admet une unique solution $\alpha$ dans $[0\,;1]$.
  \item Donner un encadrement de $\alpha$ d'amplitude $0{,}1$.
\end{enumerate}
\end{exercice}
