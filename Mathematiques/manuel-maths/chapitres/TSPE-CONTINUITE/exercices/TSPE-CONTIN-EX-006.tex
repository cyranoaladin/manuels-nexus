\begin{exercice}{TSPE-CONTIN-EX-006}{1}{12}
Soit $(u_n)$ la suite definie par $u_0 = 0$ et $u_{n+1} = \sqrt{u_n + 2}$.
\begin{enumerate}
  \item Justifier que la fonction $f(x) = \sqrt{x+2}$ est continue sur $[0, 2]$ et que $f([0,2]) \subset [0,2]$.
  \item On admet que $(u_n)$ est croissante et majoree par $2$. Justifier qu'elle converge vers un reel $\ell \in [0, 2]$.
  \item Montrer que $\ell$ verifie $\ell^2 - \ell - 2 = 0$, puis determiner $\ell$.
\end{enumerate}
\end{exercice}
