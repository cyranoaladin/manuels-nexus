\subsection*{C1 — Théorème des valeurs intermédiaires}

\begin{enumerate}

\item \textbf{[Q1]} Soit $f$ continue sur $[0,5]$ avec $f(0) = -2$ et $f(5) = 3$. Que peut-on affirmer ?
  \begin{enumerate}[label=\Alph*.]
    \item $f$ s'annule au moins une fois sur $[0,5]$.
    \item $f$ s'annule exactement une fois sur $[0,5]$.
    \item $f$ ne s'annule jamais sur $[0,5]$.
    \item $f$ est strictement croissante sur $[0,5]$.
  \end{enumerate}

\bigskip

\item \textbf{[Q2]} Pour garantir l'\emph{unicite} de la solution de $f(x)=k$ donnee par le TVI, il faut en plus supposer que $f$ est :
  \begin{enumerate}[label=\Alph*.]
    \item derivable.
    \item strictement monotone.
    \item positive.
    \item bornee.
  \end{enumerate}

\bigskip

\item \textbf{[Q3]} Le TVI s'applique a une fonction $f$ sur $[a,b]$ a condition que :
  \begin{enumerate}[label=\Alph*.]
    \item $f$ soit derivable sur $[a,b]$.
    \item $f$ soit continue sur $[a,b]$.
    \item $f$ soit positive sur $[a,b]$.
    \item $f(a) = f(b)$.
  \end{enumerate}

\bigskip

\subsection*{C2 — Suites récurrentes et continuité}

\item \textbf{[Q4]} Soit $f$ continue sur $I$, $f(I) \subset I$, et $(u_n)$ definie par $u_{n+1}=f(u_n)$ avec $u_0 \in I$. Si $(u_n)$ converge vers $\ell \in I$, alors :
  \begin{enumerate}[label=\Alph*.]
    \item $\ell = u_0$.
    \item $f(\ell) = \ell$.
    \item $f'(\ell) = 0$.
    \item $\ell$ n'est pas determine.
  \end{enumerate}

\bigskip

\item \textbf{[Q5]} Pour appliquer le resultat de la question precedente, quelle hypothese sur $f$ est indispensable ?
  \begin{enumerate}[label=\Alph*.]
    \item $f$ derivable en $\ell$.
    \item $f$ continue en $\ell$.
    \item $f$ strictement monotone.
    \item $f$ positive.
  \end{enumerate}

\bigskip

\item \textbf{[Q6]} Que peut-on dire d'une suite $(u_n)$ telle que $\ell = f(\ell)$ admet deux solutions dans $I$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $(u_n)$ converge forcement vers les deux limites a la fois.
    \item $(u_n)$ ne peut pas converger.
    \item Si $(u_n)$ converge, sa limite est l'une des deux solutions, mais on ne sait pas laquelle sans etude complementaire.
    \item $(u_n)$ converge forcement vers la plus grande des deux solutions.
  \end{enumerate}

\end{enumerate}
