\section*{\textcolor{chapcolor}{\MakeUppercase{Continuite et théorème des valeurs intermediaires — QCM d'auto-évaluation}}}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q1]} Soit $f$ continue sur $[0\,;5]$ avec $f(0) = -2$ et $f(5) = 3$. Que peut-on affirmer ?
  \begin{enumerate}[label=\Alph*.]
    \item $f$ s'annule au moins une fois sur $[0\,;5]$.
    \item $f$ s'annule exactement une fois sur $[0\,;5]$.
    \item $f$ ne s'annule jamais sur $[0\,;5]$.
    \item $f$ est strictement croissante sur $[0\,;5]$.
  \end{enumerate}

\item \textbf{[Q2]} Pour garantir l'\emph{unicite} de la solution de $f(x) = k$ donnee par le TVI, il faut supposer de plus que $f$ est :
  \begin{enumerate}[label=\Alph*.]
    \item dérivable.
    \item strictement monotone.
    \item positive.
    \item bornee.
  \end{enumerate}

\item \textbf{[Q3]} Soit $f$ continue et strictement décroissante sur $[1\,;4]$, avec $f(1) = 7$ et $f(4) = -2$. L'équation $f(x) = 3$ :
  \begin{enumerate}[label=\Alph*.]
    \item n'a aucune solution.
    \item a exactement deux solutions.
    \item a exactement une solution.
    \item a une infinite de solutions.
  \end{enumerate}

\item \textbf{[Q4]} Une fonction $f$ continue sur $[0\,;6]$ est strictement croissante de $-1$ a $4$, puis strictement décroissante de $4$ a $2$. Le nombre de solutions de $f(x) = 3$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $2$
    \item $1$
    \item $0$
    \item $3$
  \end{enumerate}

\item \textbf{[Q5]} Même fonction qu'a la question précédente. Le nombre de solutions de $f(x) = 5$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $1$
    \item $0$
    \item $2$
    \item impossible a déterminer.
  \end{enumerate}

\item \textbf{[Q6]} On sait que $f$ est continue et strictement croissante sur $[2\,;3]$, et que $f(2{,}4) < 0 < f(2{,}5)$. Alors la solution $\alpha$ de $f(x) = 0$ vérifie :
  \begin{enumerate}[label=\Alph*.]
    \item $\alpha < 2{,}4$
    \item $\alpha > 2{,}5$
    \item $2{,}4 < \alpha < 2{,}5$
    \item $\alpha = 2{,}45$
  \end{enumerate}

\item \textbf{[Q7]} Une dichotomie part de l'intervalle $[0\,;1]$. Combien d'étapes faut-il pour obtenir une amplitude strictement inferieure a $0{,}01$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $5$
    \item $100$
    \item $10$
    \item $7$
  \end{enumerate}

\item \textbf{[Q8]} Parmi ces affirmations sur le théorème des valeurs intermediaires, laquelle est \textbf{fausse} ?
  \begin{enumerate}[label=\Alph*.]
    \item Il fournit une valeur approchee de la solution.
    \item Il prouve l'existence d'une solution.
    \item Il necessite la continuite de la fonction.
    \item Il s'applique sur un intervalle.
  \end{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q9]} Soit $(u_n)$ définie par $u_{n+1} = f(u_n)$ avec $f$ continue. Si $(u_n)$ converge vers $\ell$, alors :
  \begin{enumerate}[label=\Alph*.]
    \item $f(\ell) = 0$
    \item $f(\ell) = \ell$
    \item $\ell = 0$
    \item $f$ est croissante.
  \end{enumerate}

\item \textbf{[Q10]} Pour montrer qu'une suite $u_{n+1} = f(u_n)$ reste dans un intervalle $I$, on démontré d'abord que :
  \begin{enumerate}[label=\Alph*.]
    \item $f$ est croissante sur $I$.
    \item $f$ est positive sur $I$.
    \item $f(I) \subset I$.
    \item $u_0 = 0$.
  \end{enumerate}

\item \textbf{[Q11]} Une suite croissante et majorée :
  \begin{enumerate}[label=\Alph*.]
    \item converge vers son majorant.
    \item est necessairement géométrique.
    \item peut diverger vers $+\infty$.
    \item converge, sans qu'on sache a priori vers quoi.
  \end{enumerate}

\item \textbf{[Q12]} La fonction $f$ admet exactement deux points fixes, $0$ et $2$. La suite $u_{n+1} = f(u_n)$ avec $u_0 = 1$ converge. Sa limite :
  \begin{enumerate}[label=\Alph*.]
    \item vaut $0$ ou $2$, et il faut un argument supplementaire pour trancher.
    \item vaut forcement $2$.
    \item vaut forcement $0$.
    \item vaut $1$.
  \end{enumerate}

\item \textbf{[Q13]} Une suite $u_{n+1} = f(u_n)$ dont les termes encadrent alternativement la limite :
  \begin{enumerate}[label=\Alph*.]
    \item ne peut pas converger.
    \item peut converger, ce que montre par exemple une majoration $\left|u_{n+1} - \ell\right| \leq k\left|u_n - \ell\right|$ avec $k < 1$.
    \item converge si $f$ est croissante.
    \item converge par le théorème de la limite monotone.
  \end{enumerate}

\item \textbf{[Q14]} Si $\left|u_{n+1} - \ell\right| \leq \dfrac{1}{2}\left|u_n - \ell\right|$ pour tout $n$, et $\left|u_0 - \ell\right| = 4$, alors $\left|u_5 - \ell\right|$ est majoré par :
  \begin{enumerate}[label=\Alph*.]
    \item $2$
    \item $0{,}5$
    \item $0{,}125$
    \item $0$
  \end{enumerate}

\item \textbf{[Q15]} Un algorithme cherche le premier $n$ tel que $\left|u_n - \ell\right| < 10^{-6}$ par une boucle \verb|while|. Ce programme :
  \begin{enumerate}[label=\Alph*.]
    \item se termine toujours.
    \item ne se termine jamais.
    \item se termine si $(u_n)$ est croissante.
    \item se termine si l'on a prealablement démontré que $(u_n)$ converge vers $\ell$.
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des réponses erronées}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{B} — Confusion entre le TVI et son corollaire : l'unicite exige la stricte monotonie, qui n'est pas supposee ici. \emph{Renvoi : M1, étape 4.}
    \item \textbf{C} — L'élève n'utilise pas le changement de signe entre les images des bornes. \emph{Renvoi : M1, étape 2.}
    \item \textbf{D} — L'élève deduit la monotonie du seul fait que $f(5) > f(0)$. \emph{Renvoi : RE-C1.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — La derivabilite seule n'assure pas la stricte monotonie : il faut encore étudier le signe de $f'$ pour obtenir l'unicite. \emph{Renvoi : M1, étape 4.}
    \item \textbf{C} — Le signe de $f$ n'intervient nulle part dans l'énoncé du théorème. \emph{Renvoi : cours C1.}
    \item \textbf{D} — Une fonction continue sur un segment est automatiquement bornee : l'hypothèse serait redondante. \emph{Renvoi : cours C1.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — L'élève vérifie si $3$ est compris entre les bornes dans le mauvais ordre : $-2 < 3 < 7$ convient. \emph{Renvoi : M1, étape 2.}
    \item \textbf{B} — Sur un intervalle de stricte monotonie, une valeur ne peut être atteinte qu'une fois. \emph{Renvoi : M4, étape 3.}
    \item \textbf{D} — Une infinite de solutions exigerait que $f$ reste égale a $3$ sur un intervalle ; cela contredit la stricte décroissance supposee. \emph{Renvoi : RE-C1.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{B} — Une seule branche a été examinee : il faut traiter les deux morceaux monotones. \emph{Renvoi : M4, étape 3.}
    \item \textbf{C} — L'élève conclut trop vite parce que $f$ n'est pas monotone sur l'intervalle entier. \emph{Renvoi : M4, étape 1.}
    \item \textbf{D} — Il n'y a que deux branches monotones, donc au plus deux solutions. \emph{Renvoi : M4, vérification.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — L'élève applique le TVI sans vérifier que $5$ depasse le maximum de $f$, qui vaut $4$. \emph{Renvoi : M4, étape 3.}
    \item \textbf{C} — Compter deux solutions suppose un passage au niveau $5$ sur chacune des deux branches ; or les deux culminent au maximum $4$. \emph{Renvoi : M4, étape 3.}
    \item \textbf{D} — Le tableau de variations suffit à conclure : connaissant le maximum et les limites aux bornes, chaque branche monotone porte un nombre déterminé de solutions de $f(x)=5$. \emph{Renvoi : M4, vérification.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{A} — Sens de lecture inverse : pour une fonction croissante, $f$ est négative avant $\alpha$. \emph{Renvoi : M3, piège 1.}
    \item \textbf{B} — Comme $f(2{,}5)>0$ et que $f$ est croissante, la racine a deja été franchie avant $2{,}5$ ; elle ne peut donc pas être a droite. \emph{Renvoi : M3, piège 1.}
    \item \textbf{D} — L'encadrement ne donne pas le milieu : la solution n'est pas necessairement au centre. \emph{Renvoi : M3, étape 4.}
  \end{itemize}
  \item \textbf{Q7}
  \begin{itemize}
    \item \textbf{A} — $2^{-5} = 0{,}03125$ reste superieur a $0{,}01$. \emph{Renvoi : M3, étape 3.}
    \item \textbf{B} — L'élève raisonne comme pour un balayage de pas $0{,}01$, non comme pour une division par deux. \emph{Renvoi : M3, piège 3.}
    \item \textbf{C} — Confusion avec la precision $10^{-3}$, qui demande effectivement $10$ étapes. \emph{Renvoi : M3, étape 3.}
  \end{itemize}
  \item \textbf{Q8}
  \begin{itemize}
    \item \textbf{B} — Le TVI affirme qu'une valeur intermediaire est atteinte par une fonction continue ; l'existence d'une solution est donc bien sa conclusion. \emph{Renvoi : cours C1.}
    \item \textbf{C} — La continuite est bien l'hypothèse centrale, sans laquelle le théorème est faux. \emph{Renvoi : cours C1.}
    \item \textbf{D} — Le TVI compare les images des bornes d'un intervalle et garantit les valeurs intermediaires sur cet intervalle ; cette affirmation est vraie. \emph{Renvoi : cours C1.}
  \end{itemize}
  \item \textbf{Q9}
  \begin{itemize}
    \item \textbf{A} — Confusion entre point fixe et racine : passer à la limite dans $u_{n+1}=f(u_n)$ donne $\ell=f(\ell)$, et non $f(\ell)=0$, qui caractériserait un zéro de $f$. \emph{Renvoi : M2, étape finale.}
    \item \textbf{C} — Le passage a la limite donne seulement $f(\ell)=\ell$ ; aucune hypothèse n'impose que l'unique point fixe considere soit $0$. \emph{Renvoi : RE-C2.}
    \item \textbf{D} — La monotonie de $f$ n'est ni supposée ni déductible : la convergence de $(u_n)$ n'impose que la continuité utilisée au passage à la limite, et une $f$ non monotone peut engendrer une suite convergente. \emph{Renvoi : cours C2.}
  \end{itemize}
  \item \textbf{Q10}
  \begin{itemize}
    \item \textbf{A} — La croissance sert au sens de variation, pas a la stabilite de l'intervalle. \emph{Renvoi : M2, étape 1.}
    \item \textbf{B} — Le signe de $f$ n'assure pas que les termes restent dans $I$. \emph{Renvoi : M2, étape 1.}
    \item \textbf{D} — La valeur initiale doit appartenir a $I$, mais elle n'a aucune raison d'être nulle. \emph{Renvoi : M2, étape 1.}
  \end{itemize}
  \item \textbf{Q11}
  \begin{itemize}
    \item \textbf{A} — Erreur classique : la limite est le plus petit majorant, pas n'importe quel majorant. \emph{Renvoi : RE-C2.}
    \item \textbf{B} — Être croissante et majorée est un comportement, pas une forme : la suite $u_n = 1 - 1/n$ est croissante et majorée sans être géométrique, aucun quotient $u_{n+1}/u_n$ n'y étant constant. \emph{Renvoi : RE-C2.}
    \item \textbf{C} — Diverger vers $+\infty$ signifie depasser tout réel, ce qui est impossible pour une suite majorée par un réel fixe. \emph{Renvoi : cours C2.}
  \end{itemize}
  \item \textbf{Q12}
  \begin{itemize}
    \item \textbf{B} — Même erreur : le choix du point fixe demande un encadrement ou une monotonie. \emph{Renvoi : M2, étape finale.}
    \item \textbf{C} — L'équation $f(\ell) = \ell$ donne des candidats, pas la reponse. \emph{Renvoi : M2, étape finale.}
    \item \textbf{D} — $1$ n'est pas un point fixe : la limite vérifie necessairement $f(\ell) = \ell$. \emph{Renvoi : cours C2.}
  \end{itemize}
  \item \textbf{Q13}
  \begin{itemize}
    \item \textbf{A} — L'alternance n'empeche pas la convergence, elle empeche seulement d'invoquer la monotonie. \emph{Renvoi : M5, quand l'utiliser.}
    \item \textbf{C} — La croissance de $f$ ne garantit ni que la suite est bornee ni qu'elle converge ; il faut un encadrement ou une contraction supplementaire. \emph{Renvoi : cours C2.}
    \item \textbf{D} — Ce théorème exige la monotonie, precisement ce dont on ne dispose pas ici. \emph{Renvoi : M5, quand l'utiliser.}
  \end{itemize}
  \item \textbf{Q14}
  \begin{itemize}
    \item \textbf{A} — Une seule application de l'inégalité donne $|u_1-\ell|\leq 2$ ; il faut l'itérer cinq fois pour atteindre $|u_5-\ell|\leq 4/2^5 = 1/8$. \emph{Renvoi : M5, étape 4.}
    \item \textbf{B} — La valeur $0{,}5$ correspond a $4(1/2)^3$ et n'applique la contraction que trois fois ; au rang $5$, il faut calculer $4(1/2)^5$. \emph{Renvoi : M5, étape 4.}
    \item \textbf{D} — La majoration tend vers zero mais ne l'atteint a aucun rang fini. \emph{Renvoi : M5, étape 5.}
  \end{itemize}
  \item \textbf{Q15}
  \begin{itemize}
    \item \textbf{A} — Sans convergence demontree, la condition d'arret peut n'être jamais satisfaite. \emph{Renvoi : M3, piège 2.}
    \item \textbf{B} — Le programme se termine bien des lors que la convergence est acquise. \emph{Renvoi : M3, piège 2.}
    \item \textbf{C} — Une suite croissante peut diverger vers $+\infty$ et ne jamais approcher $\ell$. \emph{Renvoi : M3, piège 2.}
  \end{itemize}
\end{itemize}
