\subsection*{C2 — Principes et formules}

\begin{enumerate}

\item \textbf{[Q1]} Le nombre de facons de ranger $5$ livres distincts sur une etagere est :
  \begin{enumerate}[label=\Alph*.]
    \item $5$.
    \item $5^5$.
    \item $5! = 120$.
    \item $\dbinom{5}{1}$.
  \end{enumerate}

\bigskip

\item \textbf{[Q2]} Pour choisir $3$ personnes parmi $10$ sans distinguer de role, on calcule :
  \begin{enumerate}[label=\Alph*.]
    \item $10 \times 9 \times 8$.
    \item $\dbinom{10}{3}$.
    \item $10^3$.
    \item $3!$.
  \end{enumerate}

\item \textbf{[Q3]} $\displaystyle\sum_{k=0}^{n}\dbinom{n}{k}$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $n^2$.
    \item $2n$.
    \item $2^n$.
    \item $n!$.
  \end{enumerate}

\subsection*{C4 — Relation de Pascal}

\item \textbf{[Q4]} La relation de Pascal s'ecrit :
  \begin{enumerate}[label=\Alph*.]
    \item $\dbinom{n}{k} = \dbinom{n-1}{k-1} + \dbinom{n-1}{k}$.
    \item $\dbinom{n}{k} = \dbinom{n-1}{k} \times \dbinom{n-1}{k-1}$.
    \item $\dbinom{n}{k} = \dbinom{n}{k-1} + \dbinom{n}{k+1}$.
    \item $\dbinom{n}{k} = \dbinom{n-1}{k}$.
  \end{enumerate}

\bigskip

\item \textbf{[Q5]} Dans la preuve combinatoire de la relation de Pascal, on partitionne les parties a $k$ elements selon :
  \begin{enumerate}[label=\Alph*.]
    \item leur taille.
    \item si elles contiennent un element fixe ou non.
    \item leur ordre.
    \item leur complementaire.
  \end{enumerate}

\end{enumerate}
