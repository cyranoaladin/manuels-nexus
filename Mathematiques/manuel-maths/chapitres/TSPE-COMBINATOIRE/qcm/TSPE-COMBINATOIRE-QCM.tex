\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} Le nombre de k-listes d'un ensemble a n elements est :
  \begin{enumerate}[label=\Alph*.]
    \item $n^k$
    \item $k^n$
    \item $n!$
    \item $\binom{n}{k}$
  \end{enumerate}

\item \textbf{[Q2]} Le nombre d'arrangements sans repetition $A_n^k$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $n^k$
    \item $\frac{n!}{k!}$
    \item $\frac{n!}{(n-k)!}$
    \item $\binom{n}{k}$
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q3]} Le nombre de combinaisons $\binom{5}{2}$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item 20
    \item 10
    \item 5
    \item 25
  \end{enumerate}

\item \textbf{[Q4]} Propriete de symetrie des coefficients binomiaux :
  \begin{enumerate}[label=\Alph*.]
    \item $\binom{n}{k} = \binom{n}{n-k}$
    \item $\binom{n}{k} = \binom{n-1}{k}$
    \item $\binom{n}{k} = \binom{k}{n}$
    \item $\binom{n}{k} = \binom{n}{k+1}$
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q5]} Le triangle de Pascal donne $\binom{n}{k} + \binom{n}{k+1} =$
  \begin{enumerate}[label=\Alph*.]
    \item $\binom{n}{k+2}$
    \item $\binom{n+2}{k}$
    \item $\binom{n+1}{k}$
    \item $\binom{n+1}{k+1}$
  \end{enumerate}

\end{enumerate}
