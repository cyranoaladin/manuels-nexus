\begin{exercice}{TSPE-COMBI-RE-C2-EX1}{1}{8}
Combien y a-t-il de facons de choisir $2$ delegues (sans role distinct) dans une classe de $28$ eleves ?
\end{exercice}

% BEGIN-VERIFY
% from sympy import binomial
% assert binomial(28, 2) == 378
% END-VERIFY

\begin{corrige}{TSPE-COMBI-RE-C2-EX1}
Choix sans ordre de $2$ elements parmi $28$ : $\dbinom{28}{2} = \dfrac{28\times27}{2} = 378$.
\end{corrige}
