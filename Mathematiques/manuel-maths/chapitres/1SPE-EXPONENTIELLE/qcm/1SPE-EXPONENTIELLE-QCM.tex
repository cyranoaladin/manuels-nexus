\section*{\textcolor{chapcolor}{\MakeUppercase{Fonction exponentielle — QCM d'auto-evaluation}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} Parmi les proprietes suivantes, laquelle caracterise la fonction exponentielle ?
  \begin{enumerate}[label=\Alph*.]
    \item C'est la seule fonction derivable sur $\mathbb{R}$ telle que $f'(x) = f(x)$ pour tout $x$ et $f(1) = \mathrm{e}$.
    \item C'est la seule fonction derivable sur $\mathbb{R}$ telle que $f'(x) = f(x)$ pour tout $x$ et $f(0) = 1$.
    \item C'est la seule fonction derivable sur $\mathbb{R}$ telle que $f'(x) = f(x)$ pour tout $x$ et $f(0) = 0$.
    \item C'est la seule fonction derivable sur $\mathbb{R}$ telle que $f'(x) = x f(x)$ pour tout $x$ et $f(0) = 1$.
  \end{enumerate}

\item \textbf{[Q2]} Quelle est la valeur de $\mathrm{e}^0$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $0$
    \item $\mathrm{e}$
    \item $1$
    \item La valeur n'est pas definie.
  \end{enumerate}

\item \textbf{[Q3]} Pour tout reel $x$, on a :
  \begin{enumerate}[label=\Alph*.]
    \item $\mathrm{e}^x > 0$
    \item $\mathrm{e}^x \geqslant 0$ (avec $\mathrm{e}^x = 0$ possible)
    \item $\mathrm{e}^x$ peut etre negatif si $x < 0$
    \item $\mathrm{e}^x > 1$ pour tout $x \in \mathbb{R}$
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q4]} Simplifier $\mathrm{e}^3 \times \mathrm{e}^5$.
  \begin{enumerate}[label=\Alph*.]
    \item $\mathrm{e}^{15}$
    \item $\mathrm{e}^8$
    \item $2\mathrm{e}^8$
    \item $\mathrm{e}^{35}$
  \end{enumerate}

\item \textbf{[Q5]} Simplifier $\dfrac{\mathrm{e}^{7}}{\mathrm{e}^{3}}$.
  \begin{enumerate}[label=\Alph*.]
    \item $\mathrm{e}^{7/3}$
    \item $\mathrm{e}^{-21}$
    \item $\mathrm{e}^{4}$
    \item $\mathrm{e}^{10}$
  \end{enumerate}

\item \textbf{[Q6]} Simplifier $(\mathrm{e}^{2})^3 \times \mathrm{e}^{-4}$.
  \begin{enumerate}[label=\Alph*.]
    \item $\mathrm{e}^{-2}$
    \item $\mathrm{e}^{1}$
    \item $\mathrm{e}^{2}$
    \item $\mathrm{e}^{10}$
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q7]} La fonction exponentielle est :
  \begin{enumerate}[label=\Alph*.]
    \item strictement decroissante sur $\mathbb{R}$
    \item strictement croissante sur $\mathbb{R}$
    \item croissante sur $]0\,;+\infty[$ et decroissante sur $]-\infty\,;0[$
    \item constante sur $\mathbb{R}$
  \end{enumerate}

\item \textbf{[Q8]} Quelle est la limite de $\mathrm{e}^x$ lorsque $x \to +\infty$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $0$
    \item $1$
    \item $\mathrm{e}$
    \item $+\infty$
  \end{enumerate}

\item \textbf{[Q9]} Sachant que la fonction exponentielle est strictement croissante, comparer $\mathrm{e}^{2{,}5}$ et $\mathrm{e}^{\sqrt{7}}$.
  \begin{enumerate}[label=\Alph*.]
    \item $\mathrm{e}^{2{,}5} < \mathrm{e}^{\sqrt{7}}$
    \item $\mathrm{e}^{2{,}5} > \mathrm{e}^{\sqrt{7}}$
    \item $\mathrm{e}^{2{,}5} = \mathrm{e}^{\sqrt{7}}$
    \item On ne peut pas comparer sans calculatrice.
  \end{enumerate}

\item[] \textbf{Capacite C4}

\item \textbf{[Q10]} Soit $f(x) = \mathrm{e}^{3x+1}$. Quelle est $f'(x)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\mathrm{e}^{3x+1}$
    \item $(3x+1)\,\mathrm{e}^{3x+1}$
    \item $3\,\mathrm{e}^{3x+1}$
    \item $3\,\mathrm{e}^{3}$
  \end{enumerate}

\item \textbf{[Q11]} Soit $g(x) = \mathrm{e}^{x^2}$. Quelle est $g'(x)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\mathrm{e}^{x^2}$
    \item $x^2\,\mathrm{e}^{x^2}$
    \item $2x\,\mathrm{e}^{x^2}$
    \item $2x\,\mathrm{e}^{2x}$
  \end{enumerate}

\item \textbf{[Q12]} Soit $h(x) = x\,\mathrm{e}^{x}$. Quelle est $h'(x)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\mathrm{e}^x$
    \item $x\,\mathrm{e}^x$
    \item $(1+x)\,\mathrm{e}^x$
    \item $(x-1)\,\mathrm{e}^x$
  \end{enumerate}

\item[] \textbf{Capacite C5}

\item \textbf{[Q13]} Resoudre $\mathrm{e}^{x} = \mathrm{e}^{5}$.
  \begin{enumerate}[label=\Alph*.]
    \item $x = \mathrm{e}^5$
    \item $x = 5$
    \item $x = \ln 5$
    \item Pas de solution.
  \end{enumerate}

\item \textbf{[Q14]} Resoudre l'inequation $\mathrm{e}^{2x-1} \geqslant 1$.
  \begin{enumerate}[label=\Alph*.]
    \item $x \geqslant 1$
    \item $x \geqslant \dfrac{1}{2}$
    \item $x \leqslant \dfrac{1}{2}$
    \item $x \geqslant 0$
  \end{enumerate}

\item \textbf{[Q15]} Resoudre $\mathrm{e}^{x^2 - 4} = 1$.
  \begin{enumerate}[label=\Alph*.]
    \item $x = 4$
    \item $x = 2$
    \item $x = -2$ ou $x = 2$
    \item $x = 0$
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des reponses erronees}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{A} — La condition initiale correcte est $f(0) = 1$, pas $f(1) = \mathrm{e}$. L'eleve confond la condition qui definit l'exponentielle avec une consequence ($\exp(1) = \mathrm{e}$). \emph{Renvoi : M1, etape 1.}
    \item \textbf{C} — Avec $f(0) = 0$, la seule solution de $f' = f$ est la fonction nulle $f = 0$, pas l'exponentielle. La condition initiale $f(0) = 1$ est essentielle. \emph{Renvoi : M1, etape 1.}
    \item \textbf{D} — L'equation differentielle de l'exponentielle est $f' = f$, pas $f'(x) = xf(x)$. Confusion entre deux equations differentielles distinctes. \emph{Renvoi : M1, etape 1.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — L'eleve confond l'exposant $0$ avec le resultat : $\mathrm{e}^0 = 1$, pas $0$. Confusion entre puissance nulle et resultat nul. \emph{Renvoi : M1, propriete fondamentale.}
    \item \textbf{B} — L'eleve confond $\mathrm{e}^0$ et $\mathrm{e}^1$. On a $\mathrm{e}^1 = \mathrm{e}$ mais $\mathrm{e}^0 = 1$. \emph{Renvoi : M1, propriete fondamentale.}
    \item \textbf{D} — L'exponentielle est definie sur $\mathbb{R}$ entier, donc $\mathrm{e}^0$ est bien defini et vaut $1$. \emph{Renvoi : M1, etape 1.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{B} — L'exponentielle est strictement positive : $\mathrm{e}^x > 0$ pour tout $x$, elle n'atteint jamais $0$. \emph{Renvoi : M1, etape 2.}
    \item \textbf{C} — Pour $x < 0$, $\mathrm{e}^x$ est positif (par exemple $\mathrm{e}^{-1} = 1/\mathrm{e} \approx 0{,}37 > 0$). L'eleve confond exposant negatif et resultat negatif. \emph{Renvoi : M1, etape 2.}
    \item \textbf{D} — Pour $x < 0$, $\mathrm{e}^x < 1$ (par exemple $\mathrm{e}^{-1} \approx 0{,}37 < 1$). La propriete $\mathrm{e}^x > 1$ n'est vraie que pour $x > 0$. \emph{Renvoi : M1, etape 2 ; M3.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — L'eleve a multiplie les exposants ($3 \times 5 = 15$) au lieu de les additionner ($3 + 5 = 8$). Confusion entre $\mathrm{e}^a \times \mathrm{e}^b = \mathrm{e}^{a+b}$ et $(\mathrm{e}^a)^b = \mathrm{e}^{ab}$. \emph{Renvoi : M2, regle 1.}
    \item \textbf{C} — L'eleve a additionne les coefficients devant $\mathrm{e}$ (comme $1 + 1 = 2$) en traitant $\mathrm{e}^3$ et $\mathrm{e}^5$ comme des termes au lieu de facteurs. \emph{Renvoi : M2, regle 1 ; R4.}
    \item \textbf{D} — L'eleve a concatene les chiffres $3$ et $5$ au lieu de les additionner. \emph{Renvoi : M2, regle 1 ; R4.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — L'eleve a divise les exposants ($7/3$) au lieu de les soustraire ($7 - 3 = 4$). Confusion entre quotient d'exponentielles et quotient d'exposants. \emph{Renvoi : M2, regle 2 ; R4.}
    \item \textbf{B} — L'eleve a multiplie les exposants avec un signe negatif ($-7 \times 3 = -21$) au lieu de soustraire. \emph{Renvoi : M2, regle 2 ; R4.}
    \item \textbf{D} — L'eleve a additionne les exposants ($7 + 3 = 10$) au lieu de les soustraire. Confusion entre produit et quotient d'exponentielles. \emph{Renvoi : M2, regle 2.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{A} — L'eleve a calcule $(\mathrm{e}^2)^3 = \mathrm{e}^{2+3} = \mathrm{e}^5$ (addition au lieu de multiplication) puis $5 + (-4) = 1$... Non, plus probablement $\mathrm{e}^{2} \times \mathrm{e}^{-4} = \mathrm{e}^{-2}$ en oubliant la puissance 3. \emph{Renvoi : M2, regle 3 ; R4.}
    \item \textbf{B} — L'eleve a calcule $(\mathrm{e}^2)^3 = \mathrm{e}^{2+3} = \mathrm{e}^{5}$ (addition au lieu de multiplication des exposants) puis $5 - 4 = 1$. \emph{Renvoi : M2, regle 3 ; R4.}
    \item \textbf{D} — L'eleve a calcule $(\mathrm{e}^2)^3 = \mathrm{e}^6$ correctement, puis a additionne $6 + 4 = 10$ au lieu de soustraire $6 - 4 = 2$. \emph{Renvoi : M2, regles 2-3.}
  \end{itemize}
  \item \textbf{Q7}
  \begin{itemize}
    \item \textbf{A} — L'eleve confond avec la fonction $x \mapsto \mathrm{e}^{-x}$ qui est decroissante. L'exponentielle $x \mapsto \mathrm{e}^x$ est strictement croissante car $\exp'(x) = \exp(x) > 0$. \emph{Renvoi : M3, etape 1.}
    \item \textbf{C} — L'exponentielle n'a pas de minimum en $x = 0$. Elle est strictement croissante sur $\mathbb{R}$ tout entier, sans changement de variation. \emph{Renvoi : M3, etape 1.}
    \item \textbf{D} — La derivee de l'exponentielle est $\exp(x) > 0$, donc la fonction est strictement croissante, certainement pas constante. \emph{Renvoi : M3, etape 1.}
  \end{itemize}
  \item \textbf{Q8}
  \begin{itemize}
    \item \textbf{A} — C'est la limite en $-\infty$ qui vaut $0$ ($\lim_{x \to -\infty} \mathrm{e}^x = 0$). Confusion entre les deux limites aux infinis. \emph{Renvoi : M3, etape 2.}
    \item \textbf{B} — L'eleve confond la valeur en $0$ ($\mathrm{e}^0 = 1$) avec la limite en $+\infty$. \emph{Renvoi : M3, etape 2.}
    \item \textbf{C} — L'eleve confond la valeur en $1$ ($\mathrm{e}^1 = \mathrm{e}$) avec la limite en $+\infty$. \emph{Renvoi : M3, etape 2.}
  \end{itemize}
  \item \textbf{Q9}
  \begin{itemize}
    \item \textbf{B} — L'eleve a mal compare les exposants. Or $2{,}5^2 = 6{,}25 < 7$ donc $2{,}5 < \sqrt{7}$, et par croissance de l'exponentielle, $\mathrm{e}^{2{,}5} < \mathrm{e}^{\sqrt{7}}$. \emph{Renvoi : M3, etape 3.}
    \item \textbf{C} — $2{,}5 
eq \sqrt{7}$ car $2{,}5^2 = 6{,}25 
eq 7$. Comme les exposants sont differents et l'exponentielle est injective, les images sont differentes. \emph{Renvoi : M3 ; R3.}
    \item \textbf{D} — On peut comparer sans calculatrice en comparant les exposants : $2{,}5^2 = 6{,}25 < 7$ donc $2{,}5 < \sqrt{7}$, et on utilise la croissance de $\exp$. \emph{Renvoi : M3, etape 3.}
  \end{itemize}
  \item \textbf{Q10}
  \begin{itemize}
    \item \textbf{A} — L'eleve a oublie de multiplier par $u'(x) = 3$. La formule $(\mathrm{e}^u)' = u' \mathrm{e}^u$ necessite le facteur $u'$. \emph{Renvoi : M4, etape 2.}
    \item \textbf{B} — L'eleve a confondu $u'(x)$ avec $u(x)$ : il a ecrit $u(x) \cdot \mathrm{e}^{u(x)}$ au lieu de $u'(x) \cdot \mathrm{e}^{u(x)}$. \emph{Renvoi : M4, etape 1 ; R1.}
    \item \textbf{D} — La derivee de $u(x) = 3x + 1$ est bien $3$, mais l'exponentielle reste $\mathrm{e}^{3x+1}$ (pas $\mathrm{e}^3$). \emph{Renvoi : M4, etape 2.}
  \end{itemize}
  \item \textbf{Q11}
  \begin{itemize}
    \item \textbf{A} — L'eleve a oublie le facteur $u'(x) = 2x$ dans la formule $(\mathrm{e}^u)' = u' \mathrm{e}^u$. \emph{Renvoi : M4, etape 2.}
    \item \textbf{B} — L'eleve a confondu $u'(x) = 2x$ avec $u(x) = x^2$. Il a ecrit $u \cdot \mathrm{e}^u$ au lieu de $u' \cdot \mathrm{e}^u$. \emph{Renvoi : M4, etape 1 ; R1.}
    \item \textbf{D} — La derivee $u'(x) = 2x$ est correcte, mais l'exposant reste $x^2$ et non $2x$. L'eleve a remplace l'exposant par la derivee. \emph{Renvoi : M4, etape 2.}
  \end{itemize}
  \item \textbf{Q12}
  \begin{itemize}
    \item \textbf{A} — L'eleve a derive uniquement $x$ (donnant $1$) et multiplie par $\mathrm{e}^x$, en oubliant la derivee de $\mathrm{e}^x$ multipliee par $x$ : formule du produit $(uv)' = u'v + uv'$ non appliquee. \emph{Renvoi : M4 ; R1.}
    \item \textbf{B} — L'eleve a oublie le terme $u'v = 1 \cdot \mathrm{e}^x$ dans la formule du produit. Il n'a garde que $uv' = x \cdot \mathrm{e}^x$. \emph{Renvoi : M4 ; R1.}
    \item \textbf{D} — L'eleve a obtenu $x - 1$ au lieu de $x + 1$. Erreur de signe dans l'application de la formule du produit. \emph{Renvoi : M4 ; R1.}
  \end{itemize}
  \item \textbf{Q13}
  \begin{itemize}
    \item \textbf{A} — L'eleve a confondu l'egalite des exponentielles avec la valeur de l'exponentielle : $\mathrm{e}^x = \mathrm{e}^5$ implique $x = 5$ (injectivite), pas $x = \mathrm{e}^5$. \emph{Renvoi : M5, etape 1.}
    \item \textbf{C} — Le logarithme neperien n'est pas necessaire ici. L'injectivite de l'exponentielle donne directement $x = 5$. $\ln 5$ serait la solution de $\mathrm{e}^x = 5$. \emph{Renvoi : M5, etape 1.}
    \item \textbf{D} — L'equation admet bien une solution ($x = 5$). L'eleve a peut-etre cru que les exponentielles ne pouvaient pas etre egales. \emph{Renvoi : M5, etape 1.}
  \end{itemize}
  \item \textbf{Q14}
  \begin{itemize}
    \item \textbf{A} — L'eleve a resolu $2x - 1 \geqslant 1$ au lieu de $2x - 1 \geqslant 0$ (oubli que $1 = \mathrm{e}^0$, pas $\mathrm{e}^1$). \emph{Renvoi : M5, etape 2.}
    \item \textbf{C} — L'eleve a inverse le sens de l'inequation. L'exponentielle etant strictement croissante, elle conserve le sens des inegalites. \emph{Renvoi : M5, etape 2 ; M3.}
    \item \textbf{D} — L'eleve a ecrit $\mathrm{e}^{2x-1} \geqslant \mathrm{e}^0$ puis $2x \geqslant 0$ en oubliant le $-1$ dans l'exposant. \emph{Renvoi : M5, etape 2.}
  \end{itemize}
  \item \textbf{Q15}
  \begin{itemize}
    \item \textbf{A} — L'eleve a resolu $x^2 = 4$ mais a donne $x = 4$ au lieu de $x = \pm 2$. Confusion entre $x^2 = 4$ et $x = 4$. \emph{Renvoi : M5, etape 1 ; R3.}
    \item \textbf{B} — L'eleve a oublie la solution $x = -2$. L'equation $x^2 = 4$ admet deux solutions : $x = 2$ et $x = -2$. \emph{Renvoi : M5, etape 1 ; R3.}
    \item \textbf{D} — L'eleve a teste $x = 0$ : $\mathrm{e}^{0 - 4} = \mathrm{e}^{-4} 
eq 1$. Il faut $x^2 - 4 = 0$, pas $x = 0$. \emph{Renvoi : M5, etape 1.}
  \end{itemize}
\end{itemize}
