\begin{exercice}{1SPE-EXPO-EX-045}{3}{15}
Une quantité est modélisée en temps continu par
\[Q(t)=120\,\mathrm{e}^{0{,}05t},\qquad t\geq 0.\]
On observe cette quantité à chaque unité de temps et l'on pose $u_n=Q(n)$.

\begin{enumerate}
\item Exprimer $u_n$ en fonction de $n$.
\item Montrer que $(u_n)$ est géométrique et préciser sa raison.
\item Calculer $Q(5)$, puis $u_5$, au dixième près. Commenter le résultat.
\item Interpréter le coefficient $120$ et la raison $\mathrm{e}^{0{,}05}$ dans le contexte du modèle.
\end{enumerate}
\end{exercice}
