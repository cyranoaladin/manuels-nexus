\begin{center}\textbf{Approfondissement — Vers la Terminale}\end{center}
\begin{coupdepouce}{1SPE-EXPO-EX-034}
\coupDePouce{1}{Pour $f(x) = \mathrm{e}^{x^2}$, identifier $u(x) = x^2$. Quelle est la dérivée de $x^2$ ?}
\coupDePouce{2}{Appliquer la formule : $f'(x) = u'(x) \times \mathrm{e}^{u(x)} = \ldots \times \mathrm{e}^{x^2}$.}
