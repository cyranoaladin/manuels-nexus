\begin{exercice}{1SPE-EXPO-EX-042}{1}{8}
Une quantité est modélisée par $Q(t)=A\,\mathrm{e}^{at}$, où $A>0$ et $t\geq 0$.

\begin{enumerate}
\item Exprimer la valeur initiale $Q(0)$.
\item Calculer le quotient $\dfrac{Q(t+1)}{Q(t)}$.
\item Expliquer le lien entre ce quotient et la raison de la suite $(Q(n))_{n\in\mathbb N}$.
\end{enumerate}
\end{exercice}
