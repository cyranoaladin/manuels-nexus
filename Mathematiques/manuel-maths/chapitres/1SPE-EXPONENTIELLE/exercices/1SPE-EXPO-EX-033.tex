\begin{exercice}{1SPE-EXPO-EX-033}{1}{7}

\begin{enumerate}
\item $f(x) = \mathrm{e}^{x} + x$
\item $g(x) = \mathrm{e}^{-x} + \mathrm{e}^{x}$
\item $h(x) = x - \mathrm{e}^{x}$
\end{enumerate}
