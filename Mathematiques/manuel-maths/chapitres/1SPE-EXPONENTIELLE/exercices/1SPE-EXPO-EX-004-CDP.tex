\begin{coupdepouce}{1SPE-EXPO-EX-004}
\coupDePouce{1}{Revoir les trois propriétés fondamentales de l'exponentielle : $\exp' = \exp$, $\exp(0) = 1$ et $\exp(x) > 0$ pour tout réel $x$.}
\coupDePouce{2}{Attention à la nuance entre $\geqslant 0$ (positif ou nul) et $> 0$ (strictement positif). L'exponentielle peut-elle valoir exactement $0$ ?}
\end{coupdepouce}
