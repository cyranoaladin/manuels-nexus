\begin{exercice}{1SPE-EXPO-EX-026}{2}{10}
On considère la fonction $f$ définie sur $\mathbb{R}$ par $f(x) = \mathrm{e}^{2x-1}$.
\begin{enumerate}
  \item Écrire $f(x)=\mathrm{e}^{-1}\mathrm{e}^{2x}$, puis calculer $f'(x)$.
  \item En déduire le sens de variation de $f$ sur $\mathbb{R}$.
  \item Calculer $f(-1)$, $f(0)$ et $f(1)$, puis placer les points correspondants.
  \item Esquisser la courbe de $f$ en utilisant sa croissance et ces trois points.
\end{enumerate}
\end{exercice}
