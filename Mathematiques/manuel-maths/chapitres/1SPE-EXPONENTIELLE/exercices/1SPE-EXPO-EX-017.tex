\begin{exercice}{1SPE-EXPO-EX-017}{2}{10}
\begin{enumerate}
  \item Développer et simplifier $A = \left(\mathrm{e}^{2x} - 1\right)\left(\mathrm{e}^{2x} + 1\right)$.
  \item Développer et simplifier $B = \left(\mathrm{e}^x + \mathrm{e}^{-x}\right)^2$.
  \item Sachant que $\mathrm{e}^x = 2$, calculer la valeur exacte de $\mathrm{e}^{3x} + \mathrm{e}^{-3x}$.
  \item Simplifier $C = \dfrac{\mathrm{e}^{3x} \times \mathrm{e}^{-x}}{\left(\mathrm{e}^x\right)^2}$.
\end{enumerate}
\end{exercice}
