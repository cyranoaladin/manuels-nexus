\begin{corrige}{1SPE-EXPO-EX-042}
\begin{enumerate}
\item $Q(0)=A\,\mathrm{e}^{0}=A$ : le coefficient $A$ est la valeur initiale.
\item
\[\frac{Q(t+1)}{Q(t)}=\frac{A\,\mathrm{e}^{a(t+1)}}{A\,\mathrm{e}^{at}}=\mathrm{e}^{a}.\]
\item Pour tout entier naturel $n$, $Q(n+1)=\mathrm{e}^{a}Q(n)$. La suite $(Q(n))$ est donc géométrique de raison $\mathrm{e}^{a}$.
\end{enumerate}
