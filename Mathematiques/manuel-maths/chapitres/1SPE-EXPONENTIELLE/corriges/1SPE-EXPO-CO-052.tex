\begin{corrige}{1SPE-EXPO-EX-052}
\begin{enumerate}
\item Une baisse de $18\,\%$ correspond au coefficient
  $1 - \dfrac{18}{100} = 0{,}82$. Au bout de $3$ ans :
  \[
    18\,000 \times 0{,}82^3 = 9\,924{,}624 \approx 9\,925 \text{ euros.}
  \]

\item Remonter d'une année sur la précédente revient à multiplier par
  l'inverse :
  \[
    \frac{1}{0{,}82} \approx 1{,}219\,512 .
  \]
  Le taux réciproque est donc d'environ $+21{,}95\,\%$, et non $+18\,\%$.

\item Une baisse de $18\,\%$ suivie d'une hausse de $18\,\%$ donne
  \[
    0{,}82 \times 1{,}18 = 0{,}967\,6 < 1 .
  \]
  Il manque $3{,}24\,\%$ pour revenir à la valeur de départ : la hausse porte
  sur une valeur déjà diminuée, donc plus petite.

\item $V(1) = 18\,000\,\mathrm{e}^{k}$ doit valoir $18\,000 \times 0{,}82$,
  donc $\mathrm{e}^{k} = 0{,}82$ et $k = \ln(0{,}82) \approx -0{,}1985$.
  Alors
  \[
    V(3) = 18\,000\,\mathrm{e}^{3k} = 18\,000 \times 0{,}82^3 = 9\,924{,}62,
  \]
  ce qui redonne bien la question 1. Le signe négatif de $k$ traduit la
  décroissance.

\item Le coefficient réciproque est $\dfrac{1}{\mathrm{e}^{k}} =
  \mathrm{e}^{-k} \approx 1{,}2195$ : changer le sens d'une évolution
  exponentielle revient à changer le signe de $k$.
\end{enumerate}
\end{corrige}
