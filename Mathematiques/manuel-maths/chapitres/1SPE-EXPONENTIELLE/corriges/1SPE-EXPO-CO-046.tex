\begin{corrige}{1SPE-EXPO-EX-046}
\begin{enumerate}
\item On calcule :
\[P(0) = 1000\,\mathrm{e}^{0{,}05 \times 0} = 1000\,\mathrm{e}^0 = 1000.\]
\textbf{Interprétation :} au temps initial ($t = 0$), la population compte $1000$ bactéries.

\item $P(20)=1000\mathrm{e}\approx\boxed{2718}$ et $P(40)=1000\mathrm{e}^{2}\approx\boxed{7389}$.

\item $P(t+10)=1000\mathrm{e}^{0{,}05t+0{,}5}=\mathrm{e}^{0{,}5}P(t)$. Toutes les dix heures, la population est donc multipliée par le même coefficient $\mathrm{e}^{0{,}5}\approx1{,}65$.

\item $P'(t)=50\mathrm{e}^{0{,}05t}>0$ pour tout $t\geqslant0$. Le modèle décrit donc une population croissante.
\end{enumerate}
\end{corrige}
