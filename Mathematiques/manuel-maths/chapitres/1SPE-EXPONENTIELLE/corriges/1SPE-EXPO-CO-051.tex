\begin{corrige}{1SPE-EXPO-EX-051}
\begin{enumerate}
\item Une hausse de $8\,\%$ correspond au coefficient $1 + \dfrac{8}{100} = 1{,}08$ ;
  une baisse de $5\,\%$ au coefficient $1 - \dfrac{5}{100} = 0{,}95$.

\item Les évolutions successives se composent en \textbf{multipliant} les
  coefficients :
  \[
    1{,}08 \times 0{,}95 = 1{,}026.
  \]
  Le taux d'évolution équivalent est donc $1{,}026 - 1 = 0{,}026$, soit
  $+2{,}6\,\%$.

\item Les taux d'évolution ne s'additionnent pas : la baisse de $5\,\%$ porte
  sur une valeur déjà augmentée, donc plus grande. Elle retire davantage que
  $5\,\%$ de la valeur initiale, et le gain net ($+2{,}6\,\%$) est inférieur à
  la somme naïve $8 - 5 = 3$.

\item Après $3$ périodes, le coefficient vaut $1{,}026^3$. En posant
  $k = \ln(1{,}026) \approx 0{,}0257$, on a $1{,}026 = \mathrm{e}^{k}$, donc
  \[
    1{,}026^3 = \mathrm{e}^{3k} \approx \mathrm{e}^{0{,}0771}.
  \]
  Le coefficient après $t$ périodes s'écrit $\mathrm{e}^{kt}$ avec
  $k \approx 0{,}0257$ : une évolution à taux constant est bien une croissance
  exponentielle.
\end{enumerate}
\end{corrige}
