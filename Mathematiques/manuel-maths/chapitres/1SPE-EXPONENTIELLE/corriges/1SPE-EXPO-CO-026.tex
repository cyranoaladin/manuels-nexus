\begin{corrige}{1SPE-EXPO-EX-026}
\textbf{1.} Par la propriété $\mathrm{e}^{a+b}=\mathrm{e}^a\mathrm{e}^b$,
\[f(x)=\mathrm{e}^{-1}\mathrm{e}^{2x}.\]
La dérivée de $x\mapsto\mathrm{e}^{2x}$ vaut $2\mathrm{e}^{2x}$, donc
\[f'(x)=\mathrm{e}^{-1}\times 2\mathrm{e}^{2x}=2\mathrm{e}^{2x-1}.\]

Comme $\mathrm{e}^{2x-1} > 0$ pour tout réel $x$, on a $f'(x) = 2\,\mathrm{e}^{2x-1} > 0$ pour tout réel $x$.

\textbf{2.} Puisque $f'(x) > 0$ pour tout $x \in \mathbb{R}$, la fonction $f$ est \textbf{strictement croissante} sur $\mathbb{R}$.

\textbf{3.} On obtient
\[f(-1)=\mathrm{e}^{-3},\qquad f(0)=\mathrm{e}^{-1},\qquad f(1)=\mathrm{e}.\]

\textbf{4.} La courbe passe par $(-1;\mathrm{e}^{-3})$, $(0;\mathrm{e}^{-1})$ et $(1;\mathrm{e})$ et elle est strictement croissante sur $\mathbb{R}$.
\end{corrige}
