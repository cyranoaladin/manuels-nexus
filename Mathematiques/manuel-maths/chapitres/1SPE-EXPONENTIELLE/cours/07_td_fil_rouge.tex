\section*{TD fil rouge — Des intérêts composés au nombre $\mathrm{e}$}
\textit{Durée : 50 minutes — Toutes les capacités C1 à C5 sont mobilisées.}

\subsection*{Partie A — Une approximation numérique de $\mathrm{e}$}
Pour $n\geqslant1$, on pose $a_n=\left(1+\dfrac1n\right)^n$.
\begin{enumerate}
\item Calculer $a_n$ pour $n=1$, $2$, $4$, $12$, $100$ et $1\,000$.
\item Observer les valeurs et conjecturer un nombre vers lequel elles semblent se stabiliser.
\item Comparer la conjecture à la valeur affichée par la calculatrice pour $\mathrm{e}$.
\end{enumerate}

\subsection*{Partie B — Propriétés algébriques}
Un capital est modélisé par $C(t)=1000\mathrm{e}^{0{,}05t}$.
\begin{enumerate}[resume]
\item Calculer $C(0)$, $C(5)$ et $C(10)$, arrondis à l'euro.
\item Montrer que $C(t+1)=\mathrm{e}^{0{,}05}C(t)$.
\item Plus généralement, simplifier $C(t+h)/C(t)$ et interpréter le résultat.
\end{enumerate}

\subsection*{Partie C — Dérivée, variation et courbe}
\begin{enumerate}[resume]
\item Calculer $C'(t)$ et justifier que $C$ est croissante.
\item Tracer sur calculatrice les courbes de $t\mapsto\mathrm{e}^{0{,}05t}$ et $t\mapsto\mathrm{e}^{-0{,}05t}$ pour $0\leqslant t\leqslant20$.
\item Décrire le rôle du signe du coefficient placé devant $t$ dans l'exposant.
\end{enumerate}

\subsection*{Partie D — Synthèse du modèle}
\begin{enumerate}[resume]
\item Comparer $C(10)$ et $C(20)$, puis calculer $C(20)/C(10)$.
\item Expliquer pourquoi une variation par facteur constant pendant des durées égales conduit à un modèle exponentiel.
\item Rédiger une synthèse distinguant l'observation numérique de $\mathrm{e}$, les propriétés algébriques et l'usage du modèle.
\end{enumerate}
