\section{Signe, variations et courbe de la fonction exponentielle}

\margeAppui{La fonction exponentielle est strictement positive et croissante sur $\mathbb{R}$.}

\propriete[Signe et variations]{
Pour tout réel $x$, $\mathrm{e}^x>0$. Comme
$(\mathrm{e}^x)'=\mathrm{e}^x>0$, la fonction exponentielle est strictement
croissante sur $\mathbb{R}$.
}

\propriete[Conservation de l'ordre]{
Pour tous réels $a$ et $b$ :
\[
\mathrm{e}^a<\mathrm{e}^b\iff a<b,
\qquad
\mathrm{e}^a=\mathrm{e}^b\iff a=b.
\]
En particulier, $\mathrm{e}^x<1$ si $x<0$, $\mathrm{e}^0=1$ et
$\mathrm{e}^x>1$ si $x>0$.
}

\propriete[Courbe représentative]{
La courbe passe par $(0\,;\,1)$ et par $(1\,;\,\mathrm{e})$. Elle reste
strictement au-dessus de l'axe des abscisses et monte lorsque l'abscisse
augmente.
}

\exemple{
Comparer $\mathrm{e}^{2{,}5}$ et $\mathrm{e}^{\sqrt7}$ sans calculatrice.
Comme $2{,}5^2=6{,}25<7$, on a $2{,}5<\sqrt7$, donc
$\mathrm{e}^{2{,}5}<\mathrm{e}^{\sqrt7}$.
}

\propriete[Lien avec les suites géométriques]{
Pour un pas de temps $h$ fixé et $u_n=A\mathrm{e}^{knh}$,
\[
\frac{u_{n+1}}{u_n}=\mathrm{e}^{kh}.
\]
Les valeurs d'un modèle exponentiel observées à intervalles de temps réguliers
forment donc une suite géométrique de raison $\mathrm{e}^{kh}$.
}
