\section{Algorithmique : construire et approcher l'exponentielle}

\margeAppui{
Le programme cite deux \textbf{exemples d'algorithme} pour cette partie : la
construction de l'exponentielle par la methode d'Euler, et l'approximation de
$\mathrm{e}$ par la suite $\left(\left(1+\frac{1}{n}\right)^n\right)$. Il les
cite, il ne les impose pas.
}

\textbf{Le probleme.} La fonction exponentielle est definie par une propriete
--- l'unique fonction derivable verifiant $f' = f$ et $f(0) = 1$ --- et non
par une formule de calcul. Comment tracer sa courbe, et comment obtenir des
valeurs de $\mathrm{e}$, a partir de cette seule propriete ?

\propriete[Methode d'Euler]{
Sur un petit intervalle, on remplace la courbe par sa tangente :
$f(x+h) \approx f(x) + h\,f'(x)$. Comme $f' = f$, cela donne
$f(x+h) \approx f(x) + h\,f(x)$.

En partant de $x_0 = 0$ et $y_0 = 1$, on construit donc, pour un pas $h > 0$ :
\[
  x_{k+1} = x_k + h,
  \qquad
  y_{k+1} = y_k + h\,y_k = (1+h)\,y_k .
\]
Les points $(x_k\,;\,y_k)$ approchent la courbe de l'exponentielle. Entree :
le pas $h$ et le nombre de pas. Sortie : la liste des points.
}

\begin{python}
def euler_exponentielle(pas, nombre_de_pas):
    """Points (x ; y) approchant la courbe de exp, par la methode d'Euler."""
    if pas <= 0 or nombre_de_pas < 0:
        raise ValueError("le pas doit etre strictement positif")
    x, y = 0.0, 1.0
    points = [(x, y)]
    for _ in range(nombre_de_pas):
        y = y + pas * y          # car f' = f
        x = x + pas
        points.append((x, y))
    return points

def approche_de_e(n):
    """Valeur approchee de e par la suite (1 + 1/n)^n."""
    if n <= 0:
        raise ValueError("n doit etre strictement positif")
    return (1 + 1 / n) ** n
\end{python}

\exemple{
\textbf{Euler avec dix pas de $0{,}1$.} \code{euler\_exponentielle(0.1, 10)}
s'arrete en $x = 1$ avec $y = 2{,}593\,742\ldots$, alors que
$\mathrm{e} = 2{,}718\,281\ldots$ : l'ecart vaut $0{,}125$. En divisant le pas
par dix, il tombe a $0{,}013$ ; en le divisant encore par dix, a $0{,}0014$.
L'ecart est a peu pres proportionnel au pas.

\medskip
\textbf{La suite $a_n$.} \code{approche\_de\_e(10)} renvoie $2{,}593\,742$,
\code{approche\_de\_e(1000)} renvoie $2{,}716\,924$.
}

\propriete[Les deux algorithmes n'en font qu'un]{
Prenons pour Euler le pas $h = \dfrac{1}{n}$ et faisons exactement $n$ pas. La
relation $y_{k+1} = (1+h)y_k$ donne, en partant de $y_0 = 1$ :
\[
  y_n = (1+h)^n = \left(1 + \frac{1}{n}\right)^{\!n} = a_n .
\]
La valeur qu'Euler produit en $x = 1$ \emph{est} le terme $a_n$. Les deux
exemples du programme sont donc le meme calcul, lu de deux facons : une
construction geometrique pas a pas, et une suite numerique.
}

\propriete[Travail demande]{%
\begin{enumerate}
  \item Verifier a la main que $y_1 = 1 + h$ et $y_2 = (1+h)^2$.
  \item Faire tourner Euler avec $h = 0{,}5$ puis $h = 0{,}25$ jusqu'a
        $x = 1$, et comparer les ecarts a $\mathrm{e}$.
  \item Verifier numeriquement que $y_n = a_n$ pour $h = 1/n$.
  \item L'ecart a $\mathrm{e}$ diminue-t-il assez vite pour obtenir six
        decimales ? Estimer le $n$ necessaire.
\end{enumerate}
}

\exemple{
\textbf{Reponse a la question 4.} L'ecart etant a peu pres proportionnel a
$1/n$, il faudrait $n$ de l'ordre du million pour six decimales. C'est
beaucoup : cette suite \emph{converge} vers $\mathrm{e}$, mais lentement. La
methode d'Euler est un outil de \emph{construction}, pas de calcul precis.
}

\erreurFrequente{
\textbf{Prendre la construction pour une demonstration.} La methode d'Euler
\emph{trace} une courbe approchee ; elle ne demontre ni l'existence ni
l'unicite de la fonction exponentielle. Le tableau de valeurs de $a_n$ ne
demontre pas davantage que la suite converge : il le suggere.
}

\erreurFrequente{
\textbf{Croire qu'un pas plus petit donne toujours mieux.} L'ecart de methode
diminue avec $h$, mais le nombre d'operations augmente d'autant, et les
erreurs d'arrondi s'accumulent. Passe un certain point, reduire le pas ne
gagne plus rien.
}
