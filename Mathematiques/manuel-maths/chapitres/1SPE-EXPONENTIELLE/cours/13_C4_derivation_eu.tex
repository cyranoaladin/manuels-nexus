\section{Dérivée de \texorpdfstring{$t\mapsto\mathrm{e}^{at}$}{t ↦ eᵃᵗ}}

\theoreme[Dérivée d'une exponentielle à exposant linéaire]{
Pour tout réel $a$, la fonction $f:t\mapsto\mathrm{e}^{at}$ est dérivable
sur $\mathbb{R}$ et
\[
f'(t)=a\,\mathrm{e}^{at}.
\]
}

\exemple{
Si $f(t)=\mathrm{e}^{3t}$, alors $f'(t)=3\mathrm{e}^{3t}$.
Si $g(t)=\mathrm{e}^{-0{,}2t}$, alors
$g'(t)=-0{,}2\mathrm{e}^{-0{,}2t}$.
}

\propriete[Présence d'un terme constant]{
La propriété algébrique donne
$\mathrm{e}^{at+b}=\mathrm{e}^b\mathrm{e}^{at}$. Le facteur
$\mathrm{e}^b$ est constant, donc
\[
(\mathrm{e}^{at+b})'=a\mathrm{e}^{at+b}.
\]
}

\erreurFrequente{
Le coefficient $a$ multiplie la fonction dérivée ; l'exposant $at$ reste
inchangé. Pour $a<0$, il ne faut pas perdre le signe négatif.
}
