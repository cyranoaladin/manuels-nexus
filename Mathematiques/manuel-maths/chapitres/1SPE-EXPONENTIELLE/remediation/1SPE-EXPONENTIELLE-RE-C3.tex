\subsection*{Remédiation C3 — Signe, croissance et courbe}

\begin{exercice}{1SPE-EXPO-RE-C3-EX1}{1}{6}
Répondre par vrai ou faux en justifiant.
\begin{enumerate}
\item $\mathrm{e}^x>0$ pour tout réel $x$.
\item Si $a<b$, alors $\mathrm{e}^a<\mathrm{e}^b$.
\item $\mathrm{e}^{-2}>\mathrm{e}^{-1}$.
\end{enumerate}
\end{exercice}
\coupDePouce{1}{La fonction exponentielle est strictement positive et strictement croissante.}
\begin{corrige}{1SPE-EXPO-RE-C3-EX1}
Les deux premières affirmations sont vraies. La troisième est fausse car $-2<-1$, donc $\mathrm{e}^{-2}<\mathrm{e}^{-1}$.
\end{corrige}

\begin{exercice}{1SPE-EXPO-RE-C3-EX2}{1}{7}
Calculer $\mathrm{e}^{-1}$, $\mathrm{e}^{0}$ et $\mathrm{e}$ à $10^{-2}$ près, placer les points correspondants, puis tracer une courbe compatible avec la croissance de l'exponentielle.
\end{exercice}
\begin{corrige}{1SPE-EXPO-RE-C3-EX2}
$\mathrm{e}^{-1}\approx0{,}37$, $\mathrm{e}^{0}=1$ et $\mathrm{e}\approx2{,}72$. La courbe passe par ces trois points et monte de gauche à droite.
\end{corrige}

\begin{exercice}{1SPE-EXPO-RE-C3-EX3}{2}{8}
Soit $f(x)=\mathrm{e}^x-x-1$.
\begin{enumerate}
\item Calculer $f(0)$ et $f'(x)$.
\item Étudier le signe de $f'$ puis les variations de $f$.
\item En déduire que $\mathrm{e}^x\geqslant x+1$ pour tout réel $x$.
\end{enumerate}
\end{exercice}
\begin{corrige}{1SPE-EXPO-RE-C3-EX3}
$f(0)=0$ et $f'(x)=\mathrm{e}^x-1$. Le signe de $f'$ est celui de $x$ : $f$ décroît jusqu'à $0$, puis croît. Son minimum vaut $0$, donc $\mathrm{e}^x-x-1\geqslant0$.
\end{corrige}
