\subsection*{Remédiation C4 — Dériver $t\mapsto\mathrm{e}^{at}$}

\begin{exercice}{1SPE-EXPO-RE-C4-EX1}{1}{6}
Dériver $f(x)=\mathrm{e}^{5x}$, $g(x)=\mathrm{e}^{-2x}$ et $h(x)=3\mathrm{e}^{4x}$.
\end{exercice}
\coupDePouce{1}{Utiliser $(\mathrm{e}^{ax})'=a\mathrm{e}^{ax}$.}
\begin{corrige}{1SPE-EXPO-RE-C4-EX1}
$f'(x)=5\mathrm{e}^{5x}$, $g'(x)=-2\mathrm{e}^{-2x}$ et $h'(x)=12\mathrm{e}^{4x}$.
\end{corrige}

\begin{exercice}{1SPE-EXPO-RE-C4-EX2}{2}{8}
On pose $p(x)=(2x+1)\mathrm{e}^{-x}$.
\begin{enumerate}
\item Calculer $p'(x)$ et la factoriser.
\item Étudier son signe puis les variations de $p$.
\end{enumerate}
\end{exercice}
\begin{corrige}{1SPE-EXPO-RE-C4-EX2}
$p'(x)=2\mathrm{e}^{-x}-(2x+1)\mathrm{e}^{-x}=(1-2x)\mathrm{e}^{-x}$. Comme $\mathrm{e}^{-x}>0$, $p'$ a le signe de $1-2x$ : $p$ croît jusqu'à $1/2$, puis décroît.
\end{corrige}
