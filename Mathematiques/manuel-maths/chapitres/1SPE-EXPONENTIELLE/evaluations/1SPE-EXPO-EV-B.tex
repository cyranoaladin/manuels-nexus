\section*{Évaluation B — Fonction exponentielle}
\textit{Durée : 55 minutes — 20 points — Calculatrice autorisée.}

\textbf{Exercice 1 — 5 points — C1, C2}
\begin{enumerate}
\item La fonction $h$ définie sur $\mathbb{R}$ par $h(x)=2\mathrm{e}^{x}$ vérifie-t-elle $h'=h$ ? Vérifie-t-elle $h(0)=1$ ? Conclure. \hfill(2 pts)
\item Simplifier $\mathrm{e}^{3x+2}\mathrm{e}^{-x-2}$ et $\dfrac{\mathrm{e}^{-2x}}{\mathrm{e}^{x}}$. \hfill(3 pts)
\end{enumerate}

\textbf{Exercice 2 — 5 points — C3, C4}
On pose $g(x)=\mathrm{e}^{0{,}3x}$.
\begin{enumerate}
\item Calculer $g'(x)$ et justifier le sens de variation de $g$. \hfill(3 pts)
\item Calculer $g(0)$ et $g(4)$ à $10^{-2}$ près, puis esquisser la courbe pour $x\geqslant0$. \hfill(2 pts)
\end{enumerate}

\textbf{Exercice 3 — 5 points — C2, C3}
Une suite géométrique vérifie $v_0=300$ et $v_{n+1}=\mathrm{e}^{-0{,}2}v_n$.
\begin{enumerate}
\item Exprimer $v_n$ en fonction de $n$. \hfill(2 pts)
\item Calculer $v_3$ et relier les points $(n;v_n)$ à la courbe de $x\mapsto300\mathrm{e}^{-0{,}2x}$. \hfill(3 pts)
\end{enumerate}

\textbf{Exercice 4 — 5 points — C5}
La concentration d'un médicament est modélisée par $C(t)=80\mathrm{e}^{-0{,}15t}$.
\begin{enumerate}
\item Calculer et interpréter $C(0)$ et $C(8)$. \hfill(2 pts)
\item Montrer que $C(t+2)=\mathrm{e}^{-0{,}3}C(t)$ et interpréter. \hfill(2 pts)
\item Dire si le modèle décrit une croissance ou une décroissance et justifier. \hfill(1 pt)
\end{enumerate}
