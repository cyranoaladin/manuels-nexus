\section*{Évaluation A — Fonction exponentielle}
\textit{Durée : 55 minutes — 20 points — Calculatrice autorisée.}

\textbf{Exercice 1 — 5 points — C1, C2}
\begin{enumerate}
\item Rappeler l'équation différentielle et la condition initiale qui définissent la fonction exponentielle. \hfill(2 pts)
\item Simplifier $\mathrm{e}^{2x-1}\mathrm{e}^{1-x}$ et $\dfrac{\mathrm{e}^{3x}}{\mathrm{e}^{x}}$. \hfill(3 pts)
\end{enumerate}

\textbf{Exercice 2 — 5 points — C3, C4}
On pose $f(x)=\mathrm{e}^{-0{,}4x}$.
\begin{enumerate}
\item Calculer $f'(x)$ et justifier le sens de variation de $f$. \hfill(3 pts)
\item Calculer $f(0)$ et $f(5)$ à $10^{-2}$ près, puis esquisser la courbe pour $x\geqslant0$. \hfill(2 pts)
\end{enumerate}

\textbf{Exercice 3 — 5 points — C2, C3}
Une suite géométrique vérifie $u_0=120$ et $u_{n+1}=\mathrm{e}^{0{,}1}u_n$.
\begin{enumerate}
\item Exprimer $u_n$ en fonction de $n$. \hfill(2 pts)
\item Calculer $u_5$ et relier les points $(n;u_n)$ à la courbe de $x\mapsto120\mathrm{e}^{0{,}1x}$. \hfill(3 pts)
\end{enumerate}

\textbf{Exercice 4 — 5 points — C5}
La population d'une culture est modélisée par $P(t)=2500\mathrm{e}^{0{,}06t}$.
\begin{enumerate}
\item Calculer et interpréter $P(0)$ et $P(12)$. \hfill(2 pts)
\item Montrer que $P(t+4)=\mathrm{e}^{0{,}24}P(t)$ et interpréter. \hfill(2 pts)
\item Dire si le modèle décrit une croissance ou une décroissance et justifier. \hfill(1 pt)
\end{enumerate}
