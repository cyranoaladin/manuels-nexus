\begin{fichemethode}{M4}{Construire un modèle exponentiel}
\quandUtiliser{J'utilise cette méthode lorsqu'une grandeur évolue continûment avec une croissance ou une décroissance exponentielle.}
\pasApas{\begin{enumerate}
  \item J'identifie la valeur initiale $Q_0=Q(0)$.
  \item Je détermine le paramètre $a$ et son signe.
  \item J'écris $Q(t)=Q_0\mathrm{e}^{at}$ avec les unités.
  \item Je vérifie la valeur initiale et le sens de variation attendu.
\end{enumerate}}
\exempleRedige{Une substance a une masse initiale de $120$~g et suit le
modèle $M(t)=120\mathrm{e}^{-0{,}08t}$. Le signe négatif indique une
décroissance et $M(0)=120$~g.}
\pieges{\begin{itemize}
  \item Confondre $Q_0$ et le paramètre $a$.
  \item Oublier le signe négatif dans un modèle de décroissance.
  \item Oublier les unités de la grandeur et du temps.
\end{itemize}}
\verifier{Je remplace $t$ par $0$ et je contrôle que le modèle redonne $Q_0$.}
\sEntrainer{\refExos{M4}}
\end{fichemethode}
