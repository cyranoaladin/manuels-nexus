\begin{fichemethode}{M3}{Dériver $t\mapsto\mathrm{e}^{at}$}
\quandUtiliser{J'utilise cette méthode lorsque l'exposant est le produit d'un réel constant $a$ par la variable $t$.}
\pasApas{\begin{enumerate}
  \item J'identifie le coefficient réel $a$ dans l'exposant $at$.
  \item J'écris directement $(\mathrm{e}^{at})'=a\mathrm{e}^{at}$.
  \item Je conserve le signe de $a$ et l'exposant $at$ inchangé.
\end{enumerate}}
\exempleRedige{Pour $f(t)=\mathrm{e}^{-0{,}4t}$, le coefficient est
$a=-0{,}4$, donc
\[f'(t)=-0{,}4\mathrm{e}^{-0{,}4t}.\]}
\pieges{\begin{itemize}
  \item Oublier le facteur $a$.
  \item Perdre le signe négatif de $a$.
  \item Remplacer l'exposant $at$ par $a$.
\end{itemize}}
\verifier{Le signe de la dérivée doit être cohérent avec la croissance si $a>0$ et la décroissance si $a<0$.}
\sEntrainer{\refExos{M3}}
\end{fichemethode}
