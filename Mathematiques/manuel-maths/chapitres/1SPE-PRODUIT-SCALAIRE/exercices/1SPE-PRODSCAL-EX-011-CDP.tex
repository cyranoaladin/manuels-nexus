\begin{coupdepouce}{1SPE-PRODSCAL-EX-011}
\coupDePouce{1}{Pour la somme, utiliser $\|\vec{u}+\vec{v}\|^2=\|\vec{u}\|^2+2\,\vec{u}\cdot\vec{v}+\|\vec{v}\|^2$ ; pour la différence, remplacer $\vec{v}$ par $-\vec{v}$.}
\coupDePouce{2}{Développer $(\vec{u}+\vec{v})\cdot(\vec{u}-\vec{v})$ par bilinéarité : les deux termes croisés s'annulent par symétrie.}
