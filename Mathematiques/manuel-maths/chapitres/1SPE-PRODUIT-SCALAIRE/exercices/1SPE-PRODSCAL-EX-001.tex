\begin{exercice}{1SPE-PRODSCAL-EX-001}{1}{5}
Calculer le produit scalaire $\vec{u} \cdot \vec{v}$ dans chaque cas.
\begin{enumerate}
  \item $\vec{u}\begin{pmatrix} 3 \\ 4 \end{pmatrix}$ et $\vec{v}\begin{pmatrix} 2 \\ -1 \end{pmatrix}$
  \item $\vec{u}\begin{pmatrix} 1 \\ -2 \end{pmatrix}$ et $\vec{v}\begin{pmatrix} 5 \\ 3 \end{pmatrix}$
  \item $\vec{u}\begin{pmatrix} 0 \\ 7 \end{pmatrix}$ et $\vec{v}\begin{pmatrix} -3 \\ 2 \end{pmatrix}$
\end{enumerate}
\end{exercice}
