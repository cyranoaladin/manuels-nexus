\begin{exercice}{1SPE-PRODSCAL-EX-002}{1}{5}
Calculer le produit scalaire $\vec{u} \cdot \vec{v}$ a l'aide de la définition géométrique.
\begin{enumerate}
  \item $\|\vec{u}\| = 3$, $\|\vec{v}\| = 2$ et $(\vec{u}, \vec{v}) = \dfrac{\pi}{6}$.
  \item $\|\vec{u}\| = 5$, $\|\vec{v}\| = 4$ et $(\vec{u}, \vec{v}) = \dfrac{\pi}{3}$.
\end{enumerate}
\end{exercice}
