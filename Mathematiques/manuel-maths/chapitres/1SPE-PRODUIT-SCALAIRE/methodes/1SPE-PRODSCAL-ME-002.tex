\begin{fichemethode}{M2}{Développer et réduire avec le produit scalaire}
\quandUtiliser{J'utilise cette méthode lorsque l'énoncé demande de développer $\|\vec{u} + \vec{v}\|^2$, $\|\vec{u} - \vec{v}\|^2$ ou $(\vec{u} + \vec{v}) \cdot (\vec{u} - \vec{v})$, ou de calculer une norme à partir du produit scalaire.}
\pasApas{\begin{enumerate}
  \item J'identifie l'expression à développer.
  \item J'applique l'identité remarquable correspondante :
    \begin{itemize}
      \item $\|\vec{u} + \vec{v}\|^2 = \|\vec{u}\|^2 + 2\,\vec{u} \cdot \vec{v} + \|\vec{v}\|^2$ ;
      \item $\|\vec{u} - \vec{v}\|^2 = \|\vec{u}\|^2 - 2\,\vec{u} \cdot \vec{v} + \|\vec{v}\|^2$ ;
      \item $(\vec{u} + \vec{v}) \cdot (\vec{u} - \vec{v}) = \|\vec{u}\|^2 - \|\vec{v}\|^2$.
    \end{itemize}
  \item Je remplace par les valeurs connues et je simplifie.
\end{enumerate}}
\exempleRedige{Calculons $\|\vec{u} + \vec{v}\|$ sachant que $\|\vec{u}\| = 3$, $\|\vec{v}\| = 4$ et $\vec{u} \cdot \vec{v} = -6$.
\commentaireMarge{Étape 2 : identité remarquable.}
$\|\vec{u} + \vec{v}\|^2 = 9 + 2 \times (-6) + 16 = 9 - 12 + 16 = 13$.
\commentaireMarge{Étape 3 : conclusion.}
Donc $\|\vec{u} + \vec{v}\| = \sqrt{13}$.}
\pieges{\begin{itemize}
  \item Écrire $\|\vec{u} + \vec{v}\|^2 = \|\vec{u}\|^2 + \|\vec{v}\|^2$ en oubliant $2\,\vec{u} \cdot \vec{v}$.
  \item Confondre $\|\vec{u}\|^2$ et $\vec{u}^2$ (notation abusive).
  \item Oublier que $\vec{u} \cdot \vec{u} = \|\vec{u}\|^2$, pas $\|\vec{u}\|$.
\end{itemize}}
\verifier{Si $\vec{u} \perp \vec{v}$, on retrouve le théorème de Pythagore : $\|\vec{u}+\vec{v}\|^2 = \|\vec{u}\|^2 + \|\vec{v}\|^2$.}
\sEntrainer{\refExos{M2}}
\end{fichemethode}
