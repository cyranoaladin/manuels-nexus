\begin{fichemethode}{M4}{Résoudre un problème géométrique avec le produit scalaire}
\quandUtiliser{J'utilise cette méthode lorsque l'énoncé demande de trouver une médiatrice, une hauteur, une projection ou une aire à l'aide du produit scalaire.}
\pasApas{\begin{enumerate}
  \item \textbf{Médiatrice :} pour $A\neq B$, avec $I$ milieu de $[AB]$, le point $M$ est sur la médiatrice si et seulement si $\overrightarrow{IM}\cdot\overrightarrow{AB}=0$, ou de manière équivalente $MA=MB$.
  \item \textbf{Hauteur :} dans un triangle $ABC$ non aplati, le pied $H$ de la hauteur issue de $A$ est défini par $H\in(BC)$ et $\overrightarrow{AH}\cdot\overrightarrow{BC}=0$. Il peut se trouver en dehors du segment $[BC]$.
  \item \textbf{Aire :} dans un triangle $ABC$ non aplati, je pose $\vec{u}=\overrightarrow{AB}$, $\vec{v}=\overrightarrow{AC}$ et $\theta=\widehat{BAC}\in]0,\pi[$. Alors $\mathcal{A}=\frac{1}{2}\|\vec{u}\|\|\vec{v}\|\sin\theta$, avec $\sin\theta=\sqrt{1-\cos^2\theta}>0$ pour cet angle intérieur.
\end{enumerate}}
\exempleRedige{Dans un repère orthonormé, soient $A(1; 3)$ et $B(5; 1)$. Déterminer l'équation de la médiatrice de $[AB]$.
\commentaireMarge{Étape 1 : milieu $I$ et vecteur normal.}
$I = (3; 2)$ et $\overrightarrow{AB}\begin{pmatrix} 4 \\ -2 \end{pmatrix}$.
$M(x; y)$ est sur la médiatrice ssi $\overrightarrow{IM} \cdot \overrightarrow{AB} = 0$ :
$(x - 3) \times 4 + (y - 2) \times (-2) = 0$, soit $4x - 2y - 8 = 0$, c'est-à-dire $2x - y = 4$.}
\pieges{\begin{itemize}
  \item Confondre médiatrice et médiane.
  \item Oublier le facteur $\frac{1}{2}$ dans la formule de l'aire.
  \item Ne pas savoir passer de $\cos$ à $\sin$ pour l'aire.
\end{itemize}}
\verifier{Pour la médiatrice, je vérifie que le milieu $I$ satisfait l'équation et que les coefficients de $x$ et de $y$ donnent un vecteur normal non nul colinéaire à $\overrightarrow{AB}$.}
\sEntrainer{\refExos{M4}}
\end{fichemethode}
