\section{Applications géométriques}

\margeAppui{Le produit scalaire permet de traduire algébriquement des propriétés géométriques classiques : médiatrice, cercle, hauteurs d'un triangle, calcul d'aires.}

\theoreme[Médiatrice et produit scalaire]{
Soient $A\neq B$ deux points du plan, $I$ le milieu de $[AB]$ et $M$ un point du plan. Alors :
\[
M \in \text{médiatrice de } [AB]
\iff MA=MB
\iff \overrightarrow{IM} \cdot \overrightarrow{AB} = 0.
\]
Une autre écriture équivalente est :
\[
M \in \text{médiatrice de } [AB] \iff \overrightarrow{MA} \cdot \overrightarrow{AB} = -\frac{1}{2}\|\overrightarrow{AB}\|^2.
\]
}

\textbf{Démonstration.} Les égalités
$\overrightarrow{MA}=\overrightarrow{MI}-\frac12\overrightarrow{AB}$ et
$\overrightarrow{MB}=\overrightarrow{MI}+\frac12\overrightarrow{AB}$ donnent, par développement :
\[
MA^2-MB^2=2\,\overrightarrow{IM}\cdot\overrightarrow{AB}.
\]
Les distances étant positives ou nulles, $MA=MB$ équivaut à $MA^2-MB^2=0$.
Enfin,
$\overrightarrow{MA}\cdot\overrightarrow{AB}
=-\overrightarrow{IM}\cdot\overrightarrow{AB}-\frac12 AB^2$,
ce qui prouve la dernière équivalence.

\propriete[Produit scalaire et cercle de diamètre $AB$]{
Pour $A\neq B$, $I$ milieu de $[AB]$ et tout point $M$, on a l'identité :
\[
\overrightarrow{MA}\cdot\overrightarrow{MB}=MI^2 - \frac{AB^2}{4}.
\]
Ainsi, $M$ appartient au cercle de diamètre $[AB]$ si et seulement si
$\overrightarrow{MA}\cdot\overrightarrow{MB}=0$.
Les points $A$ et $B$ appartiennent aussi à ce cercle.
}

\textbf{Démonstration.} En développant
$(\overrightarrow{MI}-\frac12\overrightarrow{AB})\cdot
(\overrightarrow{MI}+\frac12\overrightarrow{AB})$,
les termes croisés s'annulent et donnent l'identité annoncée.
Le produit est donc nul si et seulement si $MI^2=AB^2/4$, c'est-à-dire
$MI=AB/2$, puisque les distances sont positives ou nulles.

L'identité de polarisation
\[
\overrightarrow{MA}\cdot\overrightarrow{MB}
=\frac{MA^2+MB^2-AB^2}{2}
\]
est également vraie pour tout point $M$ ; elle ne caractérise pas à elle seule un lieu géométrique.

\propriete[Hauteur d'un triangle]{
Dans un triangle $ABC$ non aplati, la hauteur issue de $A$ est la droite passant par $A$ et perpendiculaire à $(BC)$. Son pied est le point $H\in(BC)$ qui vérifie :
\[
\overrightarrow{AH} \cdot \overrightarrow{BC} = 0.
\]
}

\propriete[Aire d'un triangle]{
L'aire d'un triangle $ABC$ non aplati est donnée par :
\[
\mathcal{A} = \frac{1}{2} \|\overrightarrow{AB}\| \times \|\overrightarrow{AC}\| \times |\sin(\widehat{BAC})|.
\]
L'angle intérieur $\widehat{BAC}$ appartient à $]0,\pi[$, donc son sinus est strictement positif.
On le calcule à partir du cosinus obtenu par le produit scalaire :
$\sin(\widehat{BAC})=\sqrt{1-\cos^2(\widehat{BAC})}$.
}

\exemple{
Dans un repère orthonormé, soient $A(0; 0)$, $B(4; 0)$ et $C(1; 3)$. Calculer l'aire du triangle $ABC$.

$\overrightarrow{AB}\begin{pmatrix} 4 \\ 0 \end{pmatrix}$, $\overrightarrow{AC}\begin{pmatrix} 1 \\ 3 \end{pmatrix}$.

$\overrightarrow{AB} \cdot \overrightarrow{AC} = 4$, $\|\overrightarrow{AB}\| = 4$, $\|\overrightarrow{AC}\| = \sqrt{10}$.

$\cos(\widehat{BAC}) = \dfrac{4}{4\sqrt{10}} = \dfrac{1}{\sqrt{10}}$, donc $\sin(\widehat{BAC}) = \dfrac{3}{\sqrt{10}}$.

$\mathcal{A} = \dfrac{1}{2} \times 4 \times \sqrt{10} \times \dfrac{3}{\sqrt{10}} = 6$ unités d'aire.
}

\erreurFrequente{
Confondre médiatrice et médiane. La médiatrice de $[AB]$ est perpendiculaire à $(AB)$ et passe par le milieu ; la médiane relie un sommet au milieu du côté opposé.
}

\erreurFrequente{
Oublier le facteur $\dfrac{1}{2}$ dans la formule de l'aire du triangle.
}

\erreurFrequente{
Confondre les conditions $\overrightarrow{IM}\cdot\overrightarrow{AB}=0$ et
$\overrightarrow{MA}\cdot\overrightarrow{MB}=0$ : la première décrit la médiatrice, la seconde le cercle de diamètre $[AB]$, lorsque $A\neq B$ et $I$ est le milieu de $[AB]$.
}
