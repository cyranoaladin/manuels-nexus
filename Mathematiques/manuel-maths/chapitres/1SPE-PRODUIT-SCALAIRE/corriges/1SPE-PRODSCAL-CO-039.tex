\begin{corrige}{1SPE-PRODSCAL-EX-039}
\begin{enumerate}
\item $\overrightarrow{BC}\begin{pmatrix} -3 \\ 3 \end{pmatrix}$. Hauteur issue de $A$ : $\overrightarrow{AM} \cdot \overrightarrow{BC} = 0$, soit $-3x + 3y = 0$, d'ou $x = y$.
\item $\overrightarrow{AC}\begin{pmatrix} 2 \\ 3 \end{pmatrix}$. Hauteur issue de $B$ : $2(x-5) + 3y = 0$, soit $2x + 3y = 10$.
\item $x = y$ et $2x + 3x = 10$, donc $x = 2$, $y = 2$. L'orthocentre est $H(2; 2)$.
\end{enumerate}
\end{corrige}
