\begin{corrige}{1SPE-PRODSCAL-EX-017}
\begin{enumerate}
\item $\vec{u} \cdot \vec{v} = 1 \times (-2) + 2 \times 1 = -2 + 2 = 0$. Les vecteurs sont orthogonaux.
\item $\|\vec{u} + \vec{v}\|^2 = \|\vec{u}\|^2 + 2 \times 0 + \|\vec{v}\|^2 = 5 + 5 = 10$.
\item Comme $\vec{u} \cdot \vec{v} = 0$, on a bien $\|\vec{u} + \vec{v}\|^2 = \|\vec{u}\|^2 + \|\vec{v}\|^2 = 10$. C'est le théorème de Pythagore.
\end{enumerate}
\end{corrige}
