\begin{corrige}{1SPE-PRODSCAL-EX-037}
\begin{enumerate}
\item $\overrightarrow{BA}\begin{pmatrix} -4 \\ 0 \end{pmatrix}$, $\overrightarrow{BC}\begin{pmatrix} 0 \\ 3 \end{pmatrix}$.

$\overrightarrow{BA} \cdot \overrightarrow{BC} = 0$. Le triangle est rectangle en $B$.

\item $\mathcal{A} = \dfrac{1}{2} \times BA \times BC = \dfrac{1}{2} \times 4 \times 3 = 6$.

\item La hauteur issue de $C$ relative a $[AB]$ est la perpendiculaire a $(AB)$ passant par $C$. Comme $(AB)$ est l'axe des abscisses, le pied est $H(4; 0) = B$.
\end{enumerate}
\end{corrige}
