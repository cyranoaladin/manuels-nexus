\begin{corrige}{1SPE-PRODSCAL-EX-007}
\begin{enumerate}
\item $\|\vec{u}\| = \sqrt{16 + 9} = 5$ et $\|\vec{v}\| = \sqrt{4 + 1} = \sqrt{5}$.
\item $\vec{u} \cdot \vec{v} = 4 \times (-2) + (-3) \times 1 = -8 - 3 = -11$.
\item $\cos\theta = \dfrac{-11}{5\sqrt{5}} = \dfrac{-11}{5\sqrt{5}} = \dfrac{-11\sqrt{5}}{25}$.
\item Le produit scalaire est négatif, donc l'angle $\theta$ est obtus ($\theta > \frac{\pi}{2}$).
\end{enumerate}
