\begin{corrige}{1SPE-PRODSCAL-EX-045}
\begin{center}\textbf{Approfondissement — Vers la Terminale}\end{center}
\begin{enumerate}
\item $AM^2 = \dfrac{2 \times 36 + 2 \times 64 - 100}{4} = \dfrac{72 + 128 - 100}{4} = \dfrac{100}{4} = 25$. Donc $AM = 5$.
\item $AB^2 + AC^2 = 36 + 64 = 100 = BC^2$. Le triangle est rectangle en $A$. La médiane issue de $A$ est égale a la moitie de l'hypotenuse : $AM = BC/2 = 5$. \checkmark
\end{enumerate}
\end{corrige}
