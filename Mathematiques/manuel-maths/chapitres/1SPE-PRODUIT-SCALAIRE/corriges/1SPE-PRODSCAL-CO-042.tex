\begin{corrige}{1SPE-PRODSCAL-EX-042}
\begin{enumerate}
\item $\cos\widehat{BAC} = \dfrac{b^2 + c^2 - a^2}{2bc} = \dfrac{16 + 25 - 9}{40} = \dfrac{32}{40} = \dfrac{4}{5}$.
\item $\widehat{BAC} = \arccos\!\left(\dfrac{4}{5}\right) \approx 0{,}644$ rad $\approx 36{,}9°$.
\item On vérifie : $3^2 + 4^2 = 9 + 16 = 25 = 5^2$. Le triangle est rectangle en $C$ (triplet pythagoricien). $\widehat{BCA} = \dfrac{\pi}{2}$.
\end{enumerate}
