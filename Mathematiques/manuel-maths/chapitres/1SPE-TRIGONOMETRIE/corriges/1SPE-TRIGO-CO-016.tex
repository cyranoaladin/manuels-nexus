\begin{corrige}{1SPE-TRIGO-EX-016}

\textbf{1.} $\cos x + \cos(\pi-x) = \cos x + (-\cos x) = 0$.

\textbf{2.} $\sin x + \sin(\pi-x) = \sin x + \sin x = 2\sin x$.

\textbf{3.} $\sin x \sin(\pi-x) + \cos x \cos(\pi-x) = \sin x \cdot \sin x + \cos x \cdot (-\cos x) = \sin^2 x - \cos^2 x$.

\end{corrige}
