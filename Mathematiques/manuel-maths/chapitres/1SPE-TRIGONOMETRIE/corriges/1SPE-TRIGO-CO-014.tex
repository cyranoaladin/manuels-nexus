\begin{corrige}{1SPE-TRIGO-EX-014}

\textbf{1.} $\dfrac{5\pi}{6} = \pi - \dfrac{\pi}{6}$ :
$\cos\!\left(\dfrac{5\pi}{6}\right) = -\cos\!\left(\dfrac{\pi}{6}\right) = -\dfrac{\sqrt{3}}{2}$, \quad
$\sin\!\left(\dfrac{5\pi}{6}\right) = \sin\!\left(\dfrac{\pi}{6}\right) = \dfrac{1}{2}$.

\textbf{2.} $\dfrac{7\pi}{4} = 2\pi - \dfrac{\pi}{4} = -\dfrac{\pi}{4} + 2\pi$ :
$\cos\!\left(\dfrac{7\pi}{4}\right) = \cos\!\left(\dfrac{\pi}{4}\right) = \dfrac{\sqrt{2}}{2}$, \quad
$\sin\!\left(\dfrac{7\pi}{4}\right) = -\sin\!\left(\dfrac{\pi}{4}\right) = -\dfrac{\sqrt{2}}{2}$.

\textbf{3.} $\dfrac{11\pi}{6} = 2\pi - \dfrac{\pi}{6} = -\dfrac{\pi}{6} + 2\pi$ :
$\cos\!\left(\dfrac{11\pi}{6}\right) = \cos\!\left(\dfrac{\pi}{6}\right) = \dfrac{\sqrt{3}}{2}$, \quad
$\sin\!\left(\dfrac{11\pi}{6}\right) = -\sin\!\left(\dfrac{\pi}{6}\right) = -\dfrac{1}{2}$.

\end{corrige}
