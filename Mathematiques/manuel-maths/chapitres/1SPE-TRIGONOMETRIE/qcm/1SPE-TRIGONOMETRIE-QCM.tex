\section*{\textcolor{chapcolor}{\MakeUppercase{Trigonométrie — QCM d'auto-évaluation}}}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte.}

\begin{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q1]} Quelle est la mesure en radians de $60°$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{\pi}{3}$
    \item $\frac{\pi}{4}$
    \item $\frac{\pi}{6}$
    \item $\frac{\pi}{2}$
  \end{enumerate}

\item \textbf{[Q2]} Quelle est la mesure principale de $\frac{9\pi}{4}$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{9\pi}{4}$
    \item $\frac{\pi}{4}$
    \item $-\frac{7\pi}{4}$
    \item $\frac{5\pi}{4}$
  \end{enumerate}

\item \textbf{[Q3]} Combien vaut $180°$ en radians ?
  \begin{enumerate}[label=\Alph*.]
    \item $2\pi$
    \item $\frac{\pi}{2}$
    \item $\pi$
    \item $\frac{3\pi}{2}$
  \end{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q4]} Quelle est la valeur de $\cos\left(\frac{\pi}{3}\right)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{1}{2}$
    \item $\frac{\sqrt{2}}{2}$
    \item $\frac{\sqrt{3}}{2}$
    \item $0$
  \end{enumerate}

\item \textbf{[Q5]} Quelle est la valeur de $\sin\left(\frac{5\pi}{6}\right)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $-\frac{1}{2}$
    \item $\frac{1}{2}$
    \item $\frac{\sqrt{3}}{2}$
    \item $-\frac{\sqrt{3}}{2}$
  \end{enumerate}

\item \textbf{[Q6]} Que vaut $\cos^2\left(\frac{\pi}{4}\right) + \sin^2\left(\frac{\pi}{4}\right)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{1}{2}$
    \item $\sqrt{2}$
    \item $1$
    \item $2$
  \end{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q7]} Sur un cercle de rayon $3$ cm, quelle est la longueur de l'arc intercepte par un angle de $\frac{2\pi}{3}$ radians ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{2\pi}{3}$ cm
    \item $6\pi$ cm
    \item $3\pi$ cm
    \item $2\pi$ cm
  \end{enumerate}

\item \textbf{[Q8]} Après enroulement sur le cercle trigonométrique, le réel $\frac{5\pi}{2}$ a la même image que :
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{\pi}{2}$
    \item $\pi$
    \item $\frac{3\pi}{2}$
    \item $2\pi$
  \end{enumerate}

\item \textbf{[Q9]} Quels réels, avec $k \in \mathbb{Z}$, ont la même image que $\frac{\pi}{4}$ sur le cercle trigonométrique ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{\pi}{4}+k\pi$
    \item $\frac{\pi}{4}+2k\pi$
    \item $-\frac{\pi}{4}+2k\pi$
    \item $\frac{3\pi}{4}+2k\pi$
  \end{enumerate}

\item \textbf{[Q10]} Le point image du réel $-\frac{\pi}{2}$ sur le cercle trigonométrique a pour coordonnées :
  \begin{enumerate}[label=\Alph*.]
    \item $(1;0)$
    \item $(0;1)$
    \item $(0;-1)$
    \item $(-1;0)$
  \end{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q11]} Quelle est la valeur de $\cos\left(\frac{2\pi}{3}\right)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{1}{2}$
    \item $-\frac{\sqrt{3}}{2}$
    \item $\frac{\sqrt{3}}{2}$
    \item $-\frac{1}{2}$
  \end{enumerate}

\item \textbf{[Q12]} Quelle est la valeur de $\sin\left(\frac{7\pi}{6}\right)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $-\frac{1}{2}$
    \item $\frac{1}{2}$
    \item $\frac{\sqrt{3}}{2}$
    \item $-\frac{\sqrt{3}}{2}$
  \end{enumerate}

\item \textbf{[Q13]} On sait que $\cos x=-\frac45$ et que l'image de $x$ est dans le troisième quadrant. Que vaut $\sin x$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac35$
    \item $-\frac35$
    \item $\frac45$
    \item $-\frac45$
  \end{enumerate}

\item \textbf{[Q14]} Le point $M\left(-\frac{\sqrt2}{2};\frac{\sqrt2}{2}\right)$ est l'image de :
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{\pi}{4}$
    \item $\frac{5\pi}{4}$
    \item $\frac{3\pi}{4}$
    \item $\frac{7\pi}{4}$
  \end{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q15]} Sur un cercle de rayon $10$ cm, un arc mesure $5$ cm. Quelle est la mesure de l'angle correspondant en radians ?
  \begin{enumerate}[label=\Alph*.]
    \item $50$
    \item $2$
    \item $5$
    \item $0{,}5$
  \end{enumerate}

\end{enumerate}

    \item \textbf{D} — L'élève a confondu $180°$ et $270°$. \emph{Renvoi : C1, conversion.}

  \item \textbf{Q4}

    \item \textbf{C} — Le calcul $3\times\pi$ oublie le facteur $2/3$ de la mesure en radians. \emph{Renvoi : C1, longueur d'arc.}

  \item \textbf{Q8}
