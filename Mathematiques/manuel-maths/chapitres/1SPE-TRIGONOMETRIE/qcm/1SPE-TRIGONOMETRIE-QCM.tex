\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} Quelle est la mesure en radians de $60°$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{\pi}{6}$
    \item $\frac{\pi}{4}$
    \item $\frac{\pi}{3}$
    \item $\frac{\pi}{2}$
  \end{enumerate}

\item \textbf{[Q2]} Quelle est la mesure principale de $\frac{9\pi}{4}$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{\pi}{4}$
    \item $\frac{9\pi}{4}$
    \item $-\frac{7\pi}{4}$
    \item $\frac{5\pi}{4}$
  \end{enumerate}

\item \textbf{[Q3]} Combien vaut $180°$ en radians ?
  \begin{enumerate}[label=\Alph*.]
    \item $2\pi$
    \item $\pi$
    \item $\frac{\pi}{2}$
    \item $\frac{3\pi}{2}$
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q4]} Quelle est la valeur de $\cos\left(\frac{\pi}{3}\right)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{\sqrt{3}}{2}$
    \item $\frac{\sqrt{2}}{2}$
    \item $\frac{1}{2}$
    \item $0$
  \end{enumerate}

\item \textbf{[Q5]} Quelle est la valeur de $\sin\left(\frac{5\pi}{6}\right)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $-\frac{1}{2}$
    \item $\frac{1}{2}$
    \item $\frac{\sqrt{3}}{2}$
    \item $-\frac{\sqrt{3}}{2}$
  \end{enumerate}

\item \textbf{[Q6]} Que vaut $\cos^2\left(\frac{\pi}{4}\right) + \sin^2\left(\frac{\pi}{4}\right)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{1}{2}$
    \item $1$
    \item $\sqrt{2}$
    \item $2$
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q7]} Que vaut $\cos(a+b)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\cos a\cos b + \sin a\sin b$
    \item $\cos a\cos b - \sin a\sin b$
    \item $\sin a\cos b + \cos a\sin b$
    \item $\sin a\cos b - \cos a\sin b$
  \end{enumerate}

\item \textbf{[Q8]} Que vaut $\sin(2a)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $2\sin a$
    \item $\sin^2 a$
    \item $2\sin a\cos a$
    \item $\cos^2 a - \sin^2 a$
  \end{enumerate}

\item \textbf{[Q9]} Quelle est la valeur de $\cos(2a)$ si $\cos a = \frac{1}{2}$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $1$
    \item $\frac{1}{2}$
    \item $-\frac{1}{2}$
    \item $0$
  \end{enumerate}

\item[] \textbf{Capacite C4}

\item \textbf{[Q10]} Combien l'equation $\cos x = \frac{1}{2}$ a-t-elle de solutions sur $[0;2\pi]$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $0$
    \item $1$
    \item $2$
    \item une infinite
  \end{enumerate}

\item \textbf{[Q11]} Les solutions de $\sin x = 0$ dans $\mathbb{R}$ sont :
  \begin{enumerate}[label=\Alph*.]
    \item $x = 2k\pi$
    \item $x = k\pi$
    \item $x = \frac{\pi}{2} + k\pi$
    \item $x = \frac{k\pi}{2}$
  \end{enumerate}

\item \textbf{[Q12]} Sur $[0;2\pi]$, l'ensemble des solutions de $\cos x \geqslant 0$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $[0;\frac{\pi}{2}]$
    \item $[0;\frac{\pi}{2}] \cup [\frac{3\pi}{2};2\pi]$
    \item $[\frac{\pi}{2};\frac{3\pi}{2}]$
    \item $[0;\pi]$
  \end{enumerate}

\item[] \textbf{Capacite C5}

\item \textbf{[Q13]} La fonction cosinus est :
  \begin{enumerate}[label=\Alph*.]
    \item paire et periodique de periode $\pi$
    \item impaire et periodique de periode $2\pi$
    \item paire et periodique de periode $2\pi$
    \item ni paire ni impaire
  \end{enumerate}

\item \textbf{[Q14]} Sur $[0;2\pi]$, la fonction sinus atteint son maximum en :
  \begin{enumerate}[label=\Alph*.]
    \item $x = 0$
    \item $x = \frac{\pi}{2}$
    \item $x = \pi$
    \item $x = \frac{3\pi}{2}$
  \end{enumerate}

\item \textbf{[Q15]} Quel est le maximum de la fonction $f(x) = 3\cos x + 5$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $3$
    \item $5$
    \item $8$
    \item $2$
  \end{enumerate}

\end{enumerate}
