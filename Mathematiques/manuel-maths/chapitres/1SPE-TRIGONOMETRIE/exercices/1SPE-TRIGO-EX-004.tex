\begin{exercice}{1SPE-TRIGO-EX-004}{1}{10}

\begin{enumerate}
  \item $\dfrac{\pi}{6}$
  \item $\dfrac{2\pi}{3}$
  \item $-\dfrac{\pi}{4}$
  \item $\dfrac{5\pi}{4}$
