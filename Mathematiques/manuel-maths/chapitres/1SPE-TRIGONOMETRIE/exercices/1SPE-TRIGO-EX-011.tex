\begin{exercice}{1SPE-TRIGO-EX-011}{1}{8}

\begin{enumerate}
  \item $\cos\!\left(\dfrac{\pi}{6}\right)$
  \item $\sin\!\left(\dfrac{\pi}{4}\right)$
  \item $\cos\!\left(\dfrac{\pi}{3}\right) + \sin\!\left(\dfrac{\pi}{6}\right)$
  \item $\cos^2\!\left(\dfrac{\pi}{4}\right) + \sin^2\!\left(\dfrac{\pi}{3}\right)$
\end{enumerate}
