\begin{center}\textbf{Approfondissement — Vers la Terminale}\end{center}
\begin{fichemethode}{M5}{Exploiter les variations et la periodicite de cos et sin}

\quandUtiliser{%
  Utiliser cette methode lorsque l'enonce demande de comparer des cosinus ou
  des sinus sans calculatrice, de dresser un tableau de variation sur
  $[0\,;\,2\pi]$, ou de ramener l'etude d'une fonction trigonometrique a une
  periode.
}

\pasApas{%
  \begin{enumerate}
    \item Retenir les variations sur $[0\,;\,2\pi]$ : $\cos$ decroit de 1 a
      $-1$ sur $[0\,;\,\pi]$ puis croit de $-1$ a 1 sur $[\pi\,;\,2\pi]$ ;
      $\sin$ croit de 0 a 1 sur $\left[0\,;\,\frac{\pi}{2}\right]$, decroit de
      1 a $-1$ sur $\left[\frac{\pi}{2}\,;\,\frac{3\pi}{2}\right]$, puis croit
      sur $\left[\frac{3\pi}{2}\,;\,2\pi\right]$.
    \item Justifier au besoin par lecture du cercle trigonometrique (abscisse
      pour $\cos$, ordonnee pour $\sin$).
    \item Periodicite : $\cos(x + 2\pi) = \cos x$ et $\sin(x + 2\pi) = \sin x$
      pour tout reel $x$ — toute etude se ramene a un intervalle de longueur
      $2\pi$.
    \item Pour comparer deux valeurs : ramener les deux angles dans un MEME
      intervalle de monotonie, puis appliquer le sens de variation.
    \item Completer avec les symetries utiles : $\cos(-x) = \cos x$ (paire),
      $\sin(-x) = -\sin x$ (impaire).
  \end{enumerate}
}

\exempleRedige{%
  Comparer sans calculatrice $\cos\dfrac{2\pi}{5}$ et $\cos\dfrac{3\pi}{5}$,
  puis $\sin\dfrac{9\pi}{4}$ et $\sin\dfrac{\pi}{3}$.

  \medskip
  \commentaireMarge{Deux angles dans $[0\,;\,\pi]$, zone de decroissance de
  $\cos$.}%
  $\dfrac{2\pi}{5} < \dfrac{3\pi}{5}$ et tous deux sont dans $[0\,;\,\pi]$ ou
  $\cos$ est decroissante : $\cos\dfrac{2\pi}{5} > \cos\dfrac{3\pi}{5}$.

  \medskip
  \commentaireMarge{Ramener $\frac{9\pi}{4}$ dans $[0\,;\,2\pi]$ par
  periodicite.}%
  $\sin\dfrac{9\pi}{4} = \sin\!\left(\dfrac{9\pi}{4} - 2\pi\right)
  = \sin\dfrac{\pi}{4}$. Sur $\left[0\,;\,\dfrac{\pi}{2}\right]$, $\sin$ est
  croissante et $\dfrac{\pi}{4} < \dfrac{\pi}{3}$ :
  $\sin\dfrac{9\pi}{4} < \sin\dfrac{\pi}{3}$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piege 1.} Appliquer un sens de variation a deux angles qui ne
      sont PAS dans le meme intervalle de monotonie.
    \item \textbf{Piege 2.} Croire $\cos$ croissante parce que l'angle
      augmente : sur $[0\,;\,\pi]$, c'est l'inverse.
    \item \textbf{Piege 3.} Oublier la periodicite et conclure sur
      $\frac{9\pi}{4}$ comme si l'angle etait dans $[0\,;\,2\pi]$.
  \end{itemize}
}

\verifier{%
  Placer les angles sur le cercle trigonometrique et lire abscisses/ordonnees
  pour confirmer la comparaison ; controler a la calculatrice en valeurs
  approchees.
}
