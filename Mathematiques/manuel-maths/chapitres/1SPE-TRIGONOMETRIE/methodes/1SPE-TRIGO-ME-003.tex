\begin{center}\textbf{Approfondissement — Vers la Terminale}\end{center}
\begin{fichemethode}{M3}{Utiliser les formules d'addition et de duplication}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande une valeur exacte de
  $\cos$ ou $\sin$ d'un angle qui se decompose en somme ou difference d'angles
  remarquables, ou fait intervenir $\cos(2a)$, $\sin(2a)$.
}

\pasApas{%
  \begin{enumerate}
    \item Écrire la formule adaptee : $\cos(a+b) = \cos a\cos b - \sin a\sin b$,
      $\sin(a+b) = \sin a\cos b + \cos a\sin b$ (et leurs variantes en $a-b$).
    \item Duplication : $\sin(2a) = 2\sin a\cos a$ et
      $\cos(2a) = \cos^2 a - \sin^2 a = 2\cos^2 a - 1 = 1 - 2\sin^2 a$.
    \item Decomposer l'angle cible en angles remarquables (ex. :
      $\frac{\pi}{12} = \frac{\pi}{3} - \frac{\pi}{4}$).
    \item Substituer les valeurs exactes connues et simplifier (radicaux).
    \item Si une seule ligne trigonométrique est connue, retrouver l'autre par
      $\cos^2 a + \sin^2 a = 1$ en fixant le SIGNE grace a l'intervalle de $a$.
  \end{enumerate}
}

\exempleRedige{%
  1) Calculer $\cos\dfrac{\pi}{12}$. \quad 2) Sachant que
  $\cos a = \dfrac{3}{5}$ avec $a \in \left[0\,;\,\dfrac{\pi}{2}\right]$,
  calculer $\sin(2a)$ et $\cos(2a)$.

  \medskip
  \commentaireMarge{$\frac{\pi}{12} = \frac{\pi}{3} - \frac{\pi}{4}$.}%
  1) $\cos\dfrac{\pi}{12} = \cos\dfrac{\pi}{3}\cos\dfrac{\pi}{4}
  + \sin\dfrac{\pi}{3}\sin\dfrac{\pi}{4}
  = \dfrac{1}{2}\cdot\dfrac{\sqrt{2}}{2} + \dfrac{\sqrt{3}}{2}\cdot
  \dfrac{\sqrt{2}}{2} = \dfrac{\sqrt{2} + \sqrt{6}}{4}$.

  \medskip
  \commentaireMarge{Signe de $\sin a$ fixe par l'intervalle.}%
  2) $\sin^2 a = 1 - \dfrac{9}{25} = \dfrac{16}{25}$ et $a \in
  \left[0\,;\,\frac{\pi}{2}\right]$ donc $\sin a = \dfrac{4}{5}$.
  Alors $\sin(2a) = 2 \times \dfrac{4}{5} \times \dfrac{3}{5} =
  \dfrac{24}{25}$ et $\cos(2a) = 2\left(\dfrac{3}{5}\right)^2 - 1
  = -\dfrac{7}{25}$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Écrire $\cos(a+b) = \cos a + \cos b$ : les lignes
      trigonométriques ne sont PAS lineaires.
    \item \textbf{Piège 2.} Erreur de signe dans $\cos(a+b)$ (c'est un MOINS)
      ou dans $\cos(a-b)$ (c'est un PLUS).
    \item \textbf{Piège 3.} Oublier de déterminer le signe de la ligne
      manquante avec l'intervalle avant d'appliquer une formule de
      duplication.
  \end{itemize}
}

\verifier{%
  Contrôler le résultat a la calculatrice en valeur approchee
  ($\cos\frac{\pi}{12} \approx 0{,}966$), vérifier que les valeurs restent
  dans $[-1\,;\,1]$, et tester $\sin^2(2a) + \cos^2(2a) = 1$.
}
