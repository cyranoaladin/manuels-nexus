\section{Point de vue algébrique}

% ============================================================
% STRATE 1 — Rappels et calculs
% ============================================================

\definition[1]{
L'ensemble $\mathbb{C}$ des nombres complexes est l'ensemble des nombres de la forme $z=a+ib$, où $a,b\in\mathbb{R}$ et $i^2=-1$. On appelle $a=\mathrm{Re}(z)$ la \textbf{partie réelle} et $b=\mathrm{Im}(z)$ la \textbf{partie imaginaire} de $z$. Le \textbf{conjugué} de $z$ est $\bar{z}=a-ib$.
}

\propriete[Inverse d'un complexe non nul]{
Pour $z=a+ib\neq0$, l'inverse de $z$ est $\dfrac{1}{z} = \dfrac{\bar z}{a^2+b^2} = \dfrac{a-ib}{a^2+b^2}$.
}

\exemple{
Calculons $\dfrac{1}{3+4i}$ : on multiplie numérateur et dénominateur par le conjugué $3-4i$ :
\[
  \frac{1}{3+4i} = \frac{3-4i}{(3+4i)(3-4i)} = \frac{3-4i}{9+16} = \frac{3-4i}{25} = \frac{3}{25}-\frac{4}{25}i.
\]
}

\propriete[Équation $az=b$]{
Pour $a\in\mathbb{C}^*$ et $b\in\mathbb{C}$, l'équation $az=b$ admet une unique solution $z=\dfrac{b}{a}$.
}

\exemple{
Résoudre $(2+i)z = 3-i$ : $z = \dfrac{3-i}{2+i} = \dfrac{(3-i)(2-i)}{(2+i)(2-i)} = \dfrac{6-3i-2i+i^2}{4+1} = \dfrac{5-5i}{5} = 1-i$.
}

\theoreme[Propriétés du conjugué]{
Pour tous $z,w\in\mathbb{C}$ et $n\in\mathbb{N}$ : $\overline{zw}=\bar z\bar w$ ; si $z\neq0$, $\overline{1/z}=1/\bar z$ ; $\overline{z^n}=(\bar z)^n$.
}

\demonstration{
Écrivons $z=a+ib$, $w=c+id$. Alors $zw=(ac-bd)+i(ad+bc)$, donc $\overline{zw}=(ac-bd)-i(ad+bc)$. Par ailleurs $\bar z\bar w=(a-ib)(c-id)=(ac-bd)-i(ad+bc)$ (même calcul avec $-i$). On a bien $\overline{zw}=\bar z\bar w$.

Pour l'inverse, $z\times\dfrac{1}{z}=1$, donc en conjuguant : $\bar z\times\overline{1/z}=\overline{1}=1$, d'où $\overline{1/z}=1/\bar z$ (car $\bar z\neq0$).

Pour la puissance, on raisonne par récurrence à partir de $\overline{zw}=\bar z\bar w$ appliqué $n-1$ fois : $\overline{z^n}=\overline{z\times z^{n-1}}=\bar z\times\overline{z^{n-1}}=\dots=(\bar z)^n$.
}

\theoreme[Formule du binôme dans $\mathbb{C}$]{
Pour tous $z,w\in\mathbb{C}$ et $n\in\mathbb{N}$ : $(z+w)^n = \displaystyle\sum_{k=0}^n \binom{n}{k} z^k w^{n-k}$.
}

\margeAppui{
La démonstration (par récurrence sur $n$, en utilisant la relation de Pascal $\binom{n+1}{k}=\binom{n}{k-1}+\binom{n}{k}$) est identique à celle établie dans $\mathbb{R}$, car elle ne repose que sur les propriétés de la distributivité, valables aussi dans $\mathbb{C}$.
}

\erreurFrequente{
\textbf{Oublier que $i^2=-1$ dans un développement.} Lors du développement d'un produit de complexes, il faut systématiquement remplacer $i^2$ par $-1$ (et $i^3$ par $-i$, $i^4$ par $1$) : c'est la source d'erreur la plus fréquente dans les calculs algébriques sur $\mathbb{C}$.
}
