\section{Point de vue géométrique}

% ============================================================
% STRATE 1 — Module, argument, affixe
% ============================================================

\definition[1]{
Le plan est muni d'un repère orthonormé $(O,\vec{u},\vec{v})$. À tout point $M(a,b)$, on associe le nombre complexe $z=a+ib$, appelé \textbf{affixe} de $M$ (et du vecteur $\overrightarrow{OM}$).
}

\definition[2]{
Le \textbf{module} de $z=a+ib$ est $|z|=\sqrt{a^2+b^2}$ (distance $OM$). Pour $z\neq0$, un \textbf{argument} de $z$ est une mesure $\theta$ de l'angle $(\vec u, \overrightarrow{OM})$, vérifiant $a=|z|\cos\theta$ et $b=|z|\sin\theta$.
}

\exemple{
Pour $z=1+i\sqrt{3}$ : $|z|=\sqrt{1+3}=2$. On a $\cos\theta=\dfrac{1}{2}$ et $\sin\theta=\dfrac{\sqrt3}{2}$, donc $\theta=\dfrac{\pi}{3}$.
}

% ============================================================
% STRATE 2 — Demonstration : |z|^2 = z zbar, module d'un produit
% ============================================================

\theoreme[Module au carré]{
Pour tout $z\in\mathbb{C}$ : $|z|^2 = z\bar z$.
}

\demonstration{
Écrivons $z=a+ib$. Alors $z\bar z = (a+ib)(a-ib) = a^2-(ib)^2 = a^2-i^2b^2 = a^2+b^2$ (car $i^2=-1$). Or $|z|^2 = a^2+b^2$ par définition. Donc $|z|^2=z\bar z$.
}

\theoreme[Module d'un produit, d'une puissance]{
Pour tous $z,w\in\mathbb{C}$ et $n\in\mathbb{N}$ : $|zw|=|z||w|$ et $|z^n|=|z|^n$.
}

\demonstration{
En utilisant $|z|^2=z\bar z$ et $\overline{zw}=\bar z\bar w$ (démontré précédemment) :
\[
  |zw|^2 = (zw)\overline{(zw)} = zw\bar z\bar w = (z\bar z)(w\bar w) = |z|^2|w|^2.
\]
Comme $|zw|$ et $|z||w|$ sont deux réels \textbf{positifs} dont les carrés sont égaux, ils sont égaux : $|zw|=|z||w|$. Pour la puissance, on applique ce résultat $n-1$ fois par récurrence : $|z^n|=|z\times z^{n-1}|=|z|\times|z^{n-1}|=\dots=|z|^n$.
}

\erreurFrequente{
\textbf{Croire que $|z+w|=|z|+|w|$.} Contrairement au module d'un produit, le module d'une \textbf{somme} n'est en général pas la somme des modules : on a seulement l'inégalité triangulaire $|z+w|\leq|z|+|w|$, avec égalité seulement si $z$ et $w$ ont même argument (vecteurs colinéaires de même sens).
}
