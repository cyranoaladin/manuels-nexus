\begin{exercice}{TEXP-CAG-EX-008}{2}{15}
Soient $z$ et $z'$ deux nombres complexes.
\begin{enumerate}
  \item Démontrer que $\overline{zz'}=\overline{z}\,\overline{z'}$ en posant
        $z=a+\mathrm{i}b$ et $z'=a'+\mathrm{i}b'$.
  \item En déduire par récurrence que $\overline{z^n}=\overline{z}^{\,n}$ pour
        tout entier $n\geq1$.
  \item Démontrer que si $z\neq0$, $\overline{\left(\frac{1}{z}\right)}
        =\frac{1}{\overline{z}}$.
\end{enumerate}
\end{exercice}
