\begin{exercice}{TEXP-CAG-EX-005}{1}{12}
On pose $z_1=3+2\mathrm{i}$ et $z_2=1-4\mathrm{i}$.
\begin{enumerate}
  \item Calculer $z_1+z_2$, $z_1z_2$ et $\overline{z_1}$.
  \item Calculer $\dfrac{1}{z_2}$ sous forme algébrique.
  \item En déduire $\dfrac{z_1}{z_2}$ sous forme algébrique.
\end{enumerate}
\end{exercice}
