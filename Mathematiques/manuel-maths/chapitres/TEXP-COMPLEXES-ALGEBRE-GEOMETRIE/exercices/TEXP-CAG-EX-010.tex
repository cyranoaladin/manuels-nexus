\begin{exercice}{TEXP-CAG-EX-010}{2}{15}
\begin{enumerate}
  \item Démontrer que pour tout complexe $z$, $|z|^2=z\overline{z}$.
  \item En déduire que $|zz'|=|z||z'|$ pour tous complexes $z$ et $z'$.
  \item En déduire par récurrence que $|z^n|=|z|^n$, et calculer
        $\left|(1+\mathrm{i})^{12}\right|$.
\end{enumerate}
\end{exercice}
