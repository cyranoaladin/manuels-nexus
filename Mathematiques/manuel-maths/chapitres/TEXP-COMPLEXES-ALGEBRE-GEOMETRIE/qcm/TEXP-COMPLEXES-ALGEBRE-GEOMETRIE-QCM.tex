\subsection*{C1/C3 --- Calcul algébrique}

\begin{enumerate}

\item \textbf{[Q1]} Pour $z=a+ib$, l'inverse de $z$ (non nul) est :
  \begin{enumerate}[label=\Alph*.]
    \item $a-ib$.
    \item $\dfrac{a-ib}{a^2+b^2}$.
    \item $\dfrac{a+ib}{a^2+b^2}$.
    \item $\dfrac{1}{a}+\dfrac{1}{ib}$.
  \end{enumerate}

\bigskip

\item \textbf{[Q2]} $\overline{zw}$ est égal à :
  \begin{enumerate}[label=\Alph*.]
    \item $\bar z + \bar w$.
    \item $\bar z \times \bar w$.
    \item $z \times w$.
    \item $\bar z / \bar w$.
  \end{enumerate}

\bigskip

\subsection*{C2 --- Équations}

\item \textbf{[Q3]} L'équation $az=b$ (avec $a\neq0$) admet :
  \begin{enumerate}[label=\Alph*.]
    \item aucune solution.
    \item une unique solution.
    \item une infinité de solutions.
    \item deux solutions.
  \end{enumerate}

\subsection*{C4/C5 --- Module, argument}

\item \textbf{[Q4]} Le module de $z=a+ib$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $a+b$.
    \item $\sqrt{a^2+b^2}$.
    \item $a^2+b^2$.
    \item $a-b$.
  \end{enumerate}

\bigskip

\item \textbf{[Q5]} $|z|^2$ est toujours égal à :
  \begin{enumerate}[label=\Alph*.]
    \item $z^2$.
    \item $z\bar z$.
    \item $2z$.
    \item $\bar z^2$.
  \end{enumerate}

\bigskip

\item \textbf{[Q6]} $|zw|$ est égal à :
  \begin{enumerate}[label=\Alph*.]
    \item $|z|+|w|$.
    \item $|z|\times|w|$.
    \item $|z|-|w|$.
    \item $|z|/|w|$.
  \end{enumerate}

\end{enumerate}
