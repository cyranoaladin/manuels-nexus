\section*{\textcolor{chapcolor}{\MakeUppercase{Nombres complexes : algebre et geometrie}}}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q1]} La forme algébrique de $(1+i)^2$ est :
  \begin{enumerate}[label=\Alph*.]
    \item 2i
    \item -2
    \item 2
    \item -2i
  \end{enumerate}

\item \textbf{[Q2]} Le conjugue de $z = 3 - 4i$ est :
  \begin{enumerate}[label=\Alph*.]
    \item 3 + 4i
    \item -3 + 4i
    \item -3 - 4i
    \item 4 - 3i
  \end{enumerate}

\item[] \textbf{Capacité C4}

\item \textbf{[Q3]} Le module de $z = 3 + 4i$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item 7
    \item 5
    \item 25
    \item 1
  \end{enumerate}

\item \textbf{[Q4]} Un argument de $z = -1 + i$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $\pi/4$
    \item $-\pi/4$
    \item $3\pi/4$
    \item $-3\pi/4$
  \end{enumerate}

\item[] \textbf{Capacité C3}

\item \textbf{[Q5]} Pour tous nombres complexes $z$ et $w$, le conjugue du produit $zw$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $\bar z+w$
    \item $z\bar w$
    \item $\bar z-\bar w$
    \item $\bar z\,\bar w$
  \end{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q6]} La solution de l'équation $(2+i)z=5$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $2+i$
    \item $2-i$
    \item $\tfrac52$
    \item $1-2i$
  \end{enumerate}

\item[] \textbf{Capacité C5}

\item \textbf{[Q7]} Pour démontrer $|z|^2=z\bar z$, on pose $z=a+ib$ et on calcule :
  \begin{enumerate}[label=\Alph*.]
    \item $z\bar z=(a+ib)^2=a^2-b^2+2iab$
    \item $z\bar z=a^2-b^2$
    \item $z\bar z=(a+ib)(a-ib)=a^2+b^2$
    \item $z\bar z=|z|$
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des réponses erronées}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{B} — Choisir $-2$ conserve à tort une partie réelle négative ; comme $1+i^2=0$, le développement laisse uniquement le terme $2i$. \emph{Renvoi : C1.}
    \item \textbf{C} — Choisir $2$ oublie que $i^2=-1$ annule les deux termes réels $1+i^2$ ; le développement donne $(1+i)^2=2i$. \emph{Renvoi : C1.}
    \item \textbf{D} — Erreur de signe du produit. Le développement (1+i)\textasciicircum{}2=1+2i+i\textasciicircum{}2=2i. \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{B} — Changement de signe de la partie réelle. La conjugaison change le signe de la partie imaginaire : le conjugué de 3-4i est 3+4i. \emph{Renvoi : C1.}
    \item \textbf{C} — Changement des deux signes. La conjugaison change le signe de la partie imaginaire : le conjugué de 3-4i est 3+4i. \emph{Renvoi : C1.}
    \item \textbf{D} — Inversion des parties réelle et imaginaire. La conjugaison change le signe de la partie imaginaire : le conjugué de 3-4i est 3+4i. \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — Addition des parties au lieu des carres. Le module de 3+4i vaut racine(3\textasciicircum{}2+4\textasciicircum{}2)=5. \emph{Renvoi : C4.}
    \item \textbf{C} — Oubli de la racine carrée. Le module de 3+4i vaut racine(3\textasciicircum{}2+4\textasciicircum{}2)=5. \emph{Renvoi : C4.}
    \item \textbf{D} — Soustraction des carres. Le module de 3+4i vaut racine(3\textasciicircum{}2+4\textasciicircum{}2)=5. \emph{Renvoi : C4.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — Premier quadrant au lieu du second. Le point (-1,1) est dans le deuxième quadrant et forme un angle 3pi/4 avec l'axe réel positif. \emph{Renvoi : C4.}
    \item \textbf{B} — Quatrieme quadrant. Le point (-1,1) est dans le deuxième quadrant et forme un angle 3pi/4 avec l'axe réel positif. \emph{Renvoi : C4.}
    \item \textbf{D} — Troisième quadrant. Le point (-1,1) est dans le deuxième quadrant et forme un angle 3pi/4 avec l'axe réel positif. \emph{Renvoi : C4.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — Cette expression remplace le produit par une somme et ne conjugue pas le second facteur. La conjugaison est multiplicative : le conjugué de zw est le produit des conjugués de z et de w. \emph{Renvoi : C3.}
    \item \textbf{B} — Conjuguer un produit exige de conjuguer chacun des deux facteurs, pas seulement $w$. La conjugaison est multiplicative : le conjugué de zw est le produit des conjugués de z et de w. \emph{Renvoi : C3.}
    \item \textbf{C} — La conjugaison conserve le produit ; elle ne le transforme pas en une différence. La conjugaison est multiplicative : le conjugué de zw est le produit des conjugués de z et de w. \emph{Renvoi : C3.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{A} — En multipliant numérateur et dénominateur par le conjugué : $z=\dfrac{5(2-i)}{(2+i)(2-i)}=\dfrac{5(2-i)}{5}=2-i$. Garder $+i$ revient à ne pas conjuguer. \emph{Renvoi : C2.}
    \item \textbf{C} — Diviser seulement les parties réelles ignore la partie imaginaire du dénominateur : $(2+i)\times\tfrac52=5+\tfrac52i\neq5$. \emph{Renvoi : C2.}
    \item \textbf{D} — Les parties réelle et imaginaire ont été échangées : $(2+i)(1-2i)=4-3i\neq5$, alors que $(2+i)(2-i)=5$. \emph{Renvoi : C2.}
  \end{itemize}
  \item \textbf{Q7}
  \begin{itemize}
    \item \textbf{A} — C'est $z^2$ et non $z\bar z$ : le conjugué change le signe de la partie imaginaire, ce qui fait disparaître le terme en $i$. \emph{Renvoi : C5.}
    \item \textbf{B} — $a^2-b^2$ est la partie réelle de $z^2$. L'identité $(a+ib)(a-ib)=a^2-(ib)^2$ donne $a^2+b^2$, puisque $(ib)^2=-b^2$. \emph{Renvoi : C5.}
    \item \textbf{D} — Le produit $z\bar z$ vaut le module au \textbf{carré} : $|z|=\sqrt{a^2+b^2}$, tandis que $z\bar z=a^2+b^2$. Les deux ne coïncident que si $|z|\in\{0;1\}$. \emph{Renvoi : C5.}
  \end{itemize}
\end{itemize}
