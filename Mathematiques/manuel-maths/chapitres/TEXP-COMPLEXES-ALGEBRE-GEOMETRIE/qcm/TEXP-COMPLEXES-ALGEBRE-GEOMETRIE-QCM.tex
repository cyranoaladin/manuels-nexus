\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} La forme algebrique de $(1+i)^2$ est :
  \begin{enumerate}[label=\Alph*.]
    \item 2
    \item -2
    \item 2i
    \item -2i
  \end{enumerate}

\item \textbf{[Q2]} Le conjugue de $z = 3 - 4i$ est :
  \begin{enumerate}[label=\Alph*.]
    \item 3 + 4i
    \item -3 + 4i
    \item -3 - 4i
    \item 4 - 3i
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q3]} Le module de $z = 3 + 4i$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item 7
    \item 5
    \item 25
    \item 1
  \end{enumerate}

\item \textbf{[Q4]} Un argument de $z = -1 + i$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $\pi/4$
    \item $-\pi/4$
    \item $-3\pi/4$
    \item $3\pi/4$
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q5]} L'equation $z^2 + 1 = 0$ a pour solutions dans $\mathbb{C}$ :
  \begin{enumerate}[label=\Alph*.]
    \item 1 et -1
    \item i et -i
    \item 1+i et 1-i
    \item Pas de solution
  \end{enumerate}

\end{enumerate}
