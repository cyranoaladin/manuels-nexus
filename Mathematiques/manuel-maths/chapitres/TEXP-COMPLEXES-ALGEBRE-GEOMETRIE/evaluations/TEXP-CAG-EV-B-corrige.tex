\begin{corrige}{TEXP-CAG-EV-B-EX2}
\textbf{1.} Écrivons $z=a+ib$. Alors $z\bar z = (a+ib)(a-ib) = a^2-i^2b^2 = a^2+b^2$ (car $i^2=-1$). Or $|z|^2=a^2+b^2$ par définition, donc $|z|^2=z\bar z$.

\textbf{2.} En utilisant le résultat précédent et $\overline{zw}=\bar z\bar w$ :
\[
  |zw|^2 = (zw)\overline{(zw)} = zw\bar z\bar w = (z\bar z)(w\bar w) = |z|^2|w|^2.
\]
$|zw|$ et $|z||w|$ sont deux réels positifs de carrés égaux, donc égaux : $|zw|=|z||w|$.
\end{corrige}
