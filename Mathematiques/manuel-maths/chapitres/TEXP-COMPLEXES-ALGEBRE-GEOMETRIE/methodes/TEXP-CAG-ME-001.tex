\begin{fichemethode}{M1}{Calculer avec les nombres complexes : somme, produit, conjugue, inverse}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande de mettre un calcul complexe
  sous FORME ALGEBRIQUE $a + ib$ : sommes, produits, conjugues, inverses et
  quotients.
}

\pasApas{%
  \begin{enumerate}
    \item Somme/difference : regrouper parties réelles entre elles et parties
      imaginaires entre elles.
    \item Produit : developper comme un produit de binomes en remplacant
      $i^2$ par $-1$, puis regrouper.
    \item Conjugue : $\overline{a + ib} = a - ib$ (changer le signe de la
      partie imaginaire uniquement).
    \item Inverse/quotient : multiplier numérateur et dénominateur par le
      CONJUGUE du dénominateur ; le dénominateur devient
      $|z|^2 = a^2 + b^2$, réel.
    \item Presenter le résultat sous la forme $a + ib$ avec $a, b$ réels
      identifies.
  \end{enumerate}
}

\exempleRedige{%
  Calculer $(3 + 2i)(1 - 4i)$ puis $\dfrac{1}{3 + 2i}$.

  \medskip
  \commentaireMarge{$i^2 = -1$ transforme $-8i^2$ en $+8$.}%
  $(3 + 2i)(1 - 4i) = 3 - 12i + 2i - 8i^2 = 3 - 10i + 8 = 11 - 10i$.

  \medskip
  \commentaireMarge{Multiplier haut et bas par le conjugue $3 - 2i$.}%
  $\dfrac{1}{3 + 2i} = \dfrac{3 - 2i}{(3 + 2i)(3 - 2i)} = \dfrac{3 - 2i}{9 + 4}
  = \dfrac{3}{13} - \dfrac{2}{13}\,i$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Oublier $i^2 = -1$ dans un developpement (le terme
      $-8i^2$ vaut $+8$, pas $-8$).
    \item \textbf{Piège 2.} Conjuguer en changeant TOUS les signes :
      $\overline{3 + 2i} = 3 - 2i$, le 3 ne change pas.
    \item \textbf{Piège 3.} Laisser un $i$ au dénominateur : la forme
      algébrique exige un dénominateur réel.
  \end{itemize}
}

\verifier{%
  Contrôler un produit en substituant une vérification numérique (module du
  produit = produit des modules), et un inverse en multipliant le résultat par
  le nombre initial (on doit trouver 1).
}
