\begin{fichemethode}{M4}{Déterminer module, argument et affixe}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande le module et un argument d'un
  complexe donne sous forme algébrique, ou de lire/placer une affixe dans le
  plan complexe.
}

\pasApas{%
  \begin{enumerate}
    \item Affixe : au point $M(a\,;\,b)$ du plan correspond
      $z = a + ib$ (et reciproquement) ; au vecteur $\vec{u}(a\,;\,b)$
      egalement.
    \item Module : $|z| = \sqrt{a^2 + b^2}$ (distance $OM$).
    \item Argument (pour $z \neq 0$) : chercher $\theta$ tel que
      $\cos\theta = \dfrac{a}{|z|}$ ET $\sin\theta = \dfrac{b}{|z|}$ — les
      deux conditions ensemble determinent $\theta$ modulo $2\pi$.
    \item Reconnaitre les angles usuels
      ($\frac{\pi}{6}, \frac{\pi}{4}, \frac{\pi}{3}$...) ou donner une valeur
      approchee ; écrire alors $z = |z|(\cos\theta + i\sin\theta)$.
    \item Contrôler sur la figure : le quadrant du point doit correspondre au
      signe de $\cos\theta$ et $\sin\theta$.
  \end{enumerate}
}

\exempleRedige{%
  Déterminer le module et un argument de $z = 1 + i\sqrt{3}$.

  \medskip
  \commentaireMarge{$|z| = \sqrt{a^2 + b^2}$.}%
  $|z| = \sqrt{1 + 3} = 2$.

  \medskip
  \commentaireMarge{Les DEUX lignes trigonométriques sont necessaires.}%
  On cherche $\theta$ avec $\cos\theta = \dfrac{1}{2}$ et
  $\sin\theta = \dfrac{\sqrt{3}}{2}$ : $\theta = \dfrac{\pi}{3}$ convient.

  \medskip
  Donc $|z| = 2$ et $\arg(z) = \dfrac{\pi}{3}\ [2\pi]$ :
  $z = 2\left(\cos\dfrac{\pi}{3} + i\sin\dfrac{\pi}{3}\right)$, point du
  premier quadrant, coherent avec $a > 0$ et $b > 0$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Déterminer $\theta$ avec la seule tangente
      $\frac{b}{a}$ : elle ne distingue pas $\theta$ de $\theta + \pi$ —
      contrôler les signes de $\cos$ et $\sin$.
    \item \textbf{Piège 2.} Écrire $|z| = a^2 + b^2$ (oubli de la racine).
    \item \textbf{Piège 3.} Confondre l'affixe du point $M(a\,;\,b)$ avec le
      couple : l'affixe est le NOMBRE $a + ib$.
  \end{itemize}
}

\verifier{%
  Reconstituer $|z|(\cos\theta + i\sin\theta)$ et vérifier qu'on retrouve la
  forme algébrique de depart ; placer le point pour contrôler le quadrant.
}

\sEntrainer{\refExos{M4}}

\end{fichemethode}
