\begin{fichemethode}{M5}{Demontrer l'identite du module carre et les proprietes du module}

\quandUtiliser{%
  Utiliser cette methode lorsque l'enonce demande de DEMONTRER l'identite
  $|z|^2 = z\bar{z}$, puis d'en deduire $|z_1 z_2| = |z_1|\,|z_2|$ et
  $|z^n| = |z|^n$.
}

\pasApas{%
  \begin{enumerate}
    \item Identite fondamentale : poser $z = a + ib$ ($a, b$ reels) et
      calculer $z\bar{z} = (a + ib)(a - ib) = a^2 + b^2 = |z|^2$.
    \item Produit : ecrire $|z_1 z_2|^2 = (z_1 z_2)\overline{(z_1 z_2)}$ et
      utiliser $\overline{z_1 z_2} = \overline{z_1}\,\overline{z_2}$
      (fiche M3).
    \item Regrouper : $|z_1 z_2|^2 = (z_1\overline{z_1})(z_2\overline{z_2})
      = |z_1|^2\,|z_2|^2$.
    \item Conclure par passage a la racine (modules positifs) :
      $|z_1 z_2| = |z_1|\,|z_2|$.
    \item Puissance : recurrence sur $n$ avec la propriete du produit dans
      l'heredite : $|z^{n+1}| = |z^n \cdot z| = |z^n|\,|z| = |z|^{n+1}$.
  \end{enumerate}
}

\exempleRedige{%
  Demontrer que pour tous $z_1, z_2 \in \mathbb{C}$,
  $|z_1 z_2| = |z_1|\,|z_2|$.

  \medskip
  \commentaireMarge{Tout repose sur $|z|^2 = z\bar{z}$.}%
  $|z_1 z_2|^2 = (z_1 z_2)\,\overline{z_1 z_2}
  = z_1 z_2\,\overline{z_1}\,\overline{z_2}
  = \left(z_1\overline{z_1}\right)\left(z_2\overline{z_2}\right)
  = |z_1|^2\,|z_2|^2$.

  \medskip
  \commentaireMarge{La racine est legitime : les deux membres sont positifs.}%
  Les deux membres etant des reels positifs, on prend la racine carree :
  $|z_1 z_2| = |z_1|\,|z_2|$. \hfill$\square$

  \medskip
  Controle numerique : $|(3+2i)^4| = |3+2i|^4 = 13^2 = 169$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piege 1.} Ecrire $|z|^2 = z^2$ : c'est $z\bar{z}$, un REEL
      positif, pas le carre complexe.
    \item \textbf{Piege 2.} Passer a la racine sans mentionner que les deux
      membres sont positifs.
    \item \textbf{Piege 3.} Croire que le module est additif :
      $|z_1 + z_2| \neq |z_1| + |z_2|$ en general (seulement une inegalite
      triangulaire).
  \end{itemize}
}

\verifier{%
  Tester l'identite et les proprietes sur des valeurs concretes
  ($z = 3 + 2i$ : $z\bar{z} = 13 = |z|^2$), et relire la chaine en verifiant
  l'usage de la propriete du conjugue du produit.
}
