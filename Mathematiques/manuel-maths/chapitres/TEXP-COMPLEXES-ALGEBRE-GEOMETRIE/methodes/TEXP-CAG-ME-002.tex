\begin{fichemethode}{M2}{Résoudre az = b et les équations melant z et son conjugue}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande de résoudre dans $\mathbb{C}$
  une équation linéaire $az = b$, ou une équation contenant a la fois $z$ et
  $\bar{z}$ (qu'on traite en posant $z = x + iy$).
}

\pasApas{%
  \begin{enumerate}
    \item Équation $az = b$ ($a \neq 0$) : diviser, $z = \dfrac{b}{a}$, puis
      mettre le quotient sous forme algébrique (conjugue du dénominateur,
      fiche M1).
    \item Équation avec $\bar{z}$ : poser $z = x + iy$ ($x, y$ réels), donc
      $\bar{z} = x - iy$.
    \item Substituer et developper pour obtenir une égalité de la forme
      $A + iB = A' + iB'$ avec $A, B, A', B'$ réels.
    \item Identifier parties réelles et imaginaires : deux équations REELLES
      en $x$ et $y$.
    \item Résoudre le système et conclure sous la forme $z = x + iy$, puis
      vérifier dans l'équation initiale.
  \end{enumerate}
}

\exempleRedige{%
  1) Résoudre $(2 + i)z = 3 - i$. \quad 2) Résoudre
  $z + 2\bar{z} = 6 - i$.

  \medskip
  \commentaireMarge{Quotient mis sous forme algébrique.}%
  1) $z = \dfrac{3 - i}{2 + i} = \dfrac{(3 - i)(2 - i)}{5}
  = \dfrac{6 - 3i - 2i - 1}{5} = \dfrac{5 - 5i}{5} = 1 - i$.

  \medskip
  \commentaireMarge{Identification des parties réelle et imaginaire.}%
  2) Posons $z = x + iy$ : $z + 2\bar{z} = 3x - iy$. L'équation devient
  $3x - iy = 6 - i$, d'ou $3x = 6$ et $-y = -1$ : $x = 2$, $y = 1$.
  Donc $z = 2 + i$. Contrôle : $2 + i + 2(2 - i) = 6 - i$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Identifier les parties réelle et imaginaire sans
      avoir vérifie que $x$ et $y$ sont REELS (l'identification n'est valable
      que sous cette hypothèse).
    \item \textbf{Piège 2.} Écrire $\overline{z} = -z$ : le conjugue ne change
      que le signe de la partie IMAGINAIRE.
    \item \textbf{Piège 3.} Diviser par $a$ sans le mettre ensuite sous forme
      algébrique : le résultat doit être au format $a + ib$.
  \end{itemize}
}

\verifier{%
  Substituer la solution dans l'équation initiale (calcul complet avec le
  conjugue), et contrôler la forme algébrique du résultat.
}

\sEntrainer{\refExos{M2}}

\end{fichemethode}
