\begin{corrige}{TEXP-CAG-EX-006}
\textbf{1.} $\mathrm{i}^2=-1$, $\mathrm{i}^3=-\mathrm{i}$, $\mathrm{i}^4=1$.
Les puissances sont périodiques de période $4$ ; comme
$2026=4\times506+2$, on a $\mathrm{i}^{2026}=\mathrm{i}^2=-1$.

\textbf{2.} $(2+\mathrm{i})^3=(2+\mathrm{i})(3+4\mathrm{i})
=6+8\mathrm{i}+3\mathrm{i}-4=2+11\mathrm{i}$ : partie réelle $2$, partie
imaginaire $11$.

\textbf{3.} $(1+\mathrm{i})^2=1+2\mathrm{i}-1=2\mathrm{i}$. Donc
$(1+\mathrm{i})^{10}=\big((1+\mathrm{i})^2\big)^5=(2\mathrm{i})^5
=32\,\mathrm{i}^5=32\mathrm{i}$.
\end{corrige}
