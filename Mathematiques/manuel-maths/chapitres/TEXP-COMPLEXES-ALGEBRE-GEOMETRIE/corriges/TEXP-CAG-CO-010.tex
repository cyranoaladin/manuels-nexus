\begin{corrige}{TEXP-CAG-EX-010}
\textbf{1.} Avec $z=a+\mathrm{i}b$ :
$z\overline{z}=(a+\mathrm{i}b)(a-\mathrm{i}b)=a^2+b^2=|z|^2$.

\textbf{2.} $|zz'|^2=zz'\overline{zz'}=z\overline{z}\,z'\overline{z'}
=|z|^2|z'|^2$. Les modules étant positifs, on obtient $|zz'|=|z||z'|$.

\textbf{3.} La récurrence est immédiate à partir de 2 :
$|z^{n+1}|=|z^n||z|=|z|^n|z|=|z|^{n+1}$. Comme $|1+\mathrm{i}|=\sqrt2$,
$\left|(1+\mathrm{i})^{12}\right|=(\sqrt2)^{12}=2^6=64$.
\end{corrige}
