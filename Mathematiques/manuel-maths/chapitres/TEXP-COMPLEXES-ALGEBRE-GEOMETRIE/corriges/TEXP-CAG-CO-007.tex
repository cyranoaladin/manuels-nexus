\textbf{1.} $z=\dfrac{5-\mathrm{i}}{2+\mathrm{i}}
=\dfrac{(5-\mathrm{i})(2-\mathrm{i})}{5}=\dfrac{9-7\mathrm{i}}{5}$.

\textbf{2.} Posons $z=x+\mathrm{i}y$. Alors
$z+2\overline{z}=3x-\mathrm{i}y$. L'identification donne $3x=6$ et $-y=-3$,
soit $x=2$ et $y=3$ : $z=2+3\mathrm{i}$.

\textbf{3.} Avec $z=x+\mathrm{i}y$ : $z\overline{z}=x^2+y^2=4$ et
$z+\overline{z}=2x=4$, donc $x=2$ puis $y^2=0$ : l'unique solution est $z=2$.
\end{corrige}
