\begin{exercice}{1SPE-SUITES-RE-C3-EX1}{1}{10}
On considère la suite $(u_n)$ définie pour tout $n \in \mathbb{N}$ par $u_n = 3 \times 2^n$.
\begin{enumerate}
  \item Calculer $u_0$, $u_1$, $u_2$, $u_3$.
  \item Calculer $\dfrac{u_{n+1}}{u_n}$ pour tout $n \in \mathbb{N}$.
  \item En déduire que $(u_n)$ est géométrique. Préciser la raison $q$ et le premier terme $u_0$.
  \item Exprimer $u_{n+1}$ en fonction de $u_n$.
\end{enumerate}
\end{exercice}

% n = symbols('n', integer=True, nonneg=True)
% u = 3 * 2**n
% assert u.subs(n, 0) == 3
% assert u.subs(n, 1) == 6
% assert u.subs(n, 2) == 12
% assert u.subs(n, 3) == 24
% quotient = simplify(u.subs(n, n+1) / u)
% assert quotient == 2
% END-VERIFY

\begin{corrige}{1SPE-SUITES-RE-C3-EX1}

\textbf{Question 1.}
\[
u_0 = 3 \times 2^0 = 3. \quad u_1 = 3 \times 2^1 = 6. \quad u_2 = 3 \times 2^2 = 12. \quad u_3 = 3 \times 2^3 = 24.
\]

\commentaireMarge{Dans cet exercice, les termes sont non nuls : le quotient $u_{n+1}/u_n$ peut servir à reconnaître la suite géométrique.}

\textbf{Question 2.} Pour tout $n \in \mathbb{N}$, $u_n = 3 \times 2^n
\neq 0$. Le quotient est :
\[
\frac{u_{n+1}}{u_n} = \frac{3 \times 2^{n+1}}{3 \times 2^n} = 2^{n+1-n} = 2.
\]

\commentaireMarge{Le quotient est constant égal à $2$ : c'est une caractérisation par quotient valable ici, car les termes sont non nuls.}

\textbf{Question 3.} Le quotient $\dfrac{u_{n+1}}{u_n} = 2$ est constant. La suite $(u_n)$ est donc géométrique de raison $q = 2$ et de premier terme $u_0 = 3$.

\textbf{Question 4.} $u_{n+1} = q \times u_n = 2 u_n$.

\end{corrige}

\begin{exercice}{1SPE-SUITES-RE-C3-EX2}{1}{10}
On donne les premiers termes : $v_0 = 2$, $v_1 = 6$, $v_2 = 18$, $v_3 = 54$.
\begin{enumerate}
  \item Calculer $\dfrac{v_1}{v_0}$, $\dfrac{v_2}{v_1}$, $\dfrac{v_3}{v_2}$.
  \item Ces trois quotients suffisent-ils à prouver que toute la suite est
  géométrique ? Justifier.
  \item On ajoute maintenant l'information : $v_{n+1}=3v_n$ pour tout
  $n\in\mathbb{N}$. Démontrer que la suite est géométrique et exprimer $v_n$
  en fonction de $n$.
  \item Calculer $v_6$.
\end{enumerate}
\end{exercice}

\coupDePouce{1}{Les quotients observés permettent seulement de proposer une
raison candidate. Pour une preuve, il faut une relation valable pour tout
$n$ ; utiliser ensuite l'information $v_{n+1}=3v_n$.}

% assert v(2) == 18
% assert v(3) == 54
% assert v(1)/v(0) == 3
% assert v(2)/v(1) == 3

% prefix_with_counterexample = [2, 6, 18, 54, 55]
% assert prefix_with_counterexample[4] != 3 * prefix_with_counterexample[3]
% assert v(6) == 2 * 3**6
% END-VERIFY

\begin{corrige}{1SPE-SUITES-RE-C3-EX2}

\textbf{Question 1.}
\[
\frac{v_1}{v_0} = \frac{6}{2} = 3. \quad \frac{v_2}{v_1} = \frac{18}{6} = 3. \quad \frac{v_3}{v_2} = \frac{54}{18} = 3.
\]

\commentaireMarge{Les trois quotients observés sont égaux à $3$ : cela suggère
la raison candidate $3$, sans constituer une preuve pour tous les rangs.}

\textbf{Question 2.} Ces trois quotients ne suffisent pas à déterminer les
termes suivants. Par exemple, on pourrait avoir $v_4=55$ tout en conservant les
quatre valeurs données ; ce prolongement ne vérifierait pas $v_4=3v_3$.

\commentaireMarge{La relation multiplicative est désormais donnée pour tout
$n$ : c'est elle qui permet de conclure.}

\textbf{Question 3.} Pour tout $n\in\mathbb{N}$, $v_{n+1}=3v_n$. Par
définition, $(v_n)$ est donc géométrique de raison $q=3$ et de premier terme
$v_0=2$. Ainsi, $v_n=2\times3^n$.

\textbf{Question 4.}
\[
v_6 = 2 \times 3^6 = 2 \times 729 = 1\,458.
\]

\end{corrige}

\begin{exercice}{1SPE-SUITES-RE-C3-EX3}{1}{12}
Une suite géométrique $(w_n)$ vérifie $w_0 = 5$ et $w_3 = 40$.
\begin{enumerate}
  \item Montrer que $w_3 = w_0 \times q^3$, puis calculer $q$.
  \item Exprimer $w_n$ en fonction de $n$.
  \item Calculer $w_5$.
  \item Montrer que $(w_n)$ est une suite géométrique en vérifiant que le quotient $\dfrac{w_{n+1}}{w_n}$ est bien constant.
\end{enumerate}
\end{exercice}

\begin{corrige}{1SPE-SUITES-RE-C3-EX3}

\commentaireMarge{$w_3 = w_0 \times q^3$ : trois multiplications par $q$.}

\textbf{Question 1.} Pour une suite géométrique de raison $q$ : $w_n = w_0 \times q^n$. Donc $w_3 = 5q^3$.
\[
5q^3 = 40 \implies q^3 = 8 \implies q = 2.
\]

\textbf{Question 2.} $w_n = 5 \times 2^n$.

\textbf{Question 3.}
\[
w_5 = 5 \times 2^5 = 5 \times 32 = 160.
\]

\textbf{Question 4.} Pour tout $n \in \mathbb{N}$, $w_n = 5 \times 2^n
\neq 0$. Le quotient :
\[
\frac{w_{n+1}}{w_n} = \frac{5 \times 2^{n+1}}{5 \times 2^n} = 2.
\]
Le quotient est constant égal à $2$ : la suite $(w_n)$ est bien géométrique de raison $2$.

\end{corrige}
