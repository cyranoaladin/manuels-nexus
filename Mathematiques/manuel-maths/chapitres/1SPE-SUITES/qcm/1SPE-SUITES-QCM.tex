\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} Soit $(u_n)$ la suite definie pour tout $n \in \mathbb{N}$ par $u_n = 2n^2 - 3n + 1$. Quelle est la valeur de $u_3$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $10$
    \item $19$
    \item $8$
    \item $7$
  \end{enumerate}

\item \textbf{[Q2]} Soit $(v_n)$ la suite definie par $v_0 = 4$ et $v_{n+1} = 3v_n - 2$ pour tout $n \in \mathbb{N}$. Quelle est la valeur de $v_2$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $10$
    \item $28$
    \item $34$
    \item $22$
  \end{enumerate}

\item \textbf{[Q3]} Parmi les assertions suivantes, laquelle est exacte concernant une suite definie par recurrence ?
  \begin{enumerate}[label=\Alph*.]
    \item On peut calculer $u_{10}$ directement en substituant $n = 10$ dans la relation $u_{n+1} = f(u_n)$.
    \item Pour calculer $u_1$, on substitue $n = 0$ dans la relation $u_{n+1} = f(u_n)$.
    \item Une suite definie par formule explicite necessite de connaitre le terme precedent.
    \item La notation $u(n)$ avec des parentheses est la notation mathematique standard pour une suite.
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q4]} Soit $(u_n)$ la suite definie pour tout $n \in \mathbb{N}$ par $u_n = 4n - 7$. Quelle est la raison de cette suite arithmetique ?
  \begin{enumerate}[label=\Alph*.]
    \item $-7$
    \item $4n$
    \item $4$
    \item $-3$
  \end{enumerate}

\item \textbf{[Q5]} Soit $(u_n)$ une suite arithmetique de raison $r = -3$ et de premier terme $u_1 = 8$. Quelle est la valeur de $u_6$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $-7$
    \item $-22$
    \item $-10$
    \item $u_6 = 8 + 6 \times (-3) = -10$
  \end{enumerate}

\item \textbf{[Q6]} Parmi les suites suivantes, laquelle est arithmetique ?
  \begin{enumerate}[label=\Alph*.]
    \item $u_n = 3^n$
    \item $u_n = n^2 + 2$
    \item $u_n = 5n - 1$
    \item $u_n = \dfrac{1}{n+1}$
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q7]} Soit $(u_n)$ la suite geometrique de raison $q = 2$ et de premier terme $u_0 = 3$. Quelle est la formule explicite de $u_n$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $u_n = 3 + 2n$
    \item $u_n = 3 \times 2^{n-1}$
    \item $u_n = 2 \times 3^n$
    \item $u_n = 3 \times 2^n$
  \end{enumerate}

\item \textbf{[Q8]} La suite $(v_n)$ est definie par $v_0 = 6$ et $v_{n+1} = \dfrac{v_n}{3}$ pour tout $n \in \mathbb{N}$. Quelle est la raison de cette suite ?
  \begin{enumerate}[label=\Alph*.]
    \item $3$
    \item $-3$
    \item $\dfrac{1}{3}$
    \item $2$
  \end{enumerate}

\item \textbf{[Q9]} Parmi les assertions suivantes, laquelle est exacte pour identifier une suite geometrique ?
  \begin{enumerate}[label=\Alph*.]
    \item Il suffit de verifier que $u_1/u_0 = u_2/u_1$ pour conclure que la suite est geometrique.
    \item Une suite geometrique peut comporter un terme nul.
    \item Pour montrer qu'une suite est geometrique, il faut calculer $u_{n+1}/u_n$ et verifier que ce quotient est constant pour tout $n$.
    \item La raison d'une suite geometrique est toujours un entier positif.
  \end{enumerate}

\item \textbf{[Q11]} Soit la suite geometrique $(u_n)$ de premier terme $u_0 = 1$ et de raison $q = 3$. Quelle est la valeur de $S = u_0 + u_1 + u_2 + u_3 + u_4$ ?

\item[] \textbf{Capacite C5}

\item \textbf{[Q13]} Soit $(u_n)$ la suite arithmetique de raison $r = -4$ et de premier terme $u_0 = 10$. Quel est le sens de variation de cette suite ?
  \begin{enumerate}[label=\Alph*.]
    \item Croissante
    \item Constante
    \item Strictement decroissante
    \item D'abord croissante, puis decroissante
  \end{enumerate}

\item \textbf{[Q14]} Pour etudier le sens de variation de la suite $(u_n)$ definie par $u_n = 5 \times (0{,}8)^n$ (avec $u_n > 0$ pour tout $n$), quelle est la methode la plus efficace ?
  \begin{enumerate}[label=\Alph*.]
    \item Methode A : calculer le signe de $u_{n+1} - u_n$.
    \item Methode B : comparer le quotient $u_{n+1}/u_n$ a $1$.
    \item Methode C : etudier la derivee de la fonction $f(x) = 5 \times (0{,}8)^x$.
    \item Calculer les 5 premiers termes et observer la tendance.
  \end{enumerate}

\item \textbf{[Q15]} Soit $(u_n)$ la suite definie par $u_n = n^2 - 6n + 5$. A partir de quel rang cette suite est-elle strictement croissante ?
  \begin{enumerate}[label=\Alph*.]
    \item Pour tout $n \in \mathbb{N}$
    \item A partir de $n = 3$
    \item A partir de $n = 2$
    \item A partir de $n = 6$
  \end{enumerate}

\item[] \textbf{Capacite C6}

\item \textbf{[Q16]} Un capital de $2\,000$ euros est place a un taux annuel de $4\,\%$. Quel est le coefficient multiplicateur applicable chaque annee ?
  \begin{enumerate}[label=\Alph*.]
    \item $0{,}04$
    \item $1{,}4$
    \item $1{,}04$
    \item $4$
  \end{enumerate}

\item \textbf{[Q17]} Une ville compte $50\,000$ habitants et perd $300$ habitants par an. Quel modele est le plus adapte pour representer l'evolution de sa population ?
  \begin{enumerate}[label=\Alph*.]
    \item Une suite geometrique, car la population diminue.
    \item Une suite arithmetique, car la variation absolue annuelle est constante ($-300$).
    \item Une suite geometrique, car on multiplie chaque annee par un coefficient.
    \item Ni une suite arithmetique ni une suite geometrique.
  \end{enumerate}

\item \textbf{[Q18]} Un placement diminue de $5\,\%$ par an. Si le capital initial est $u_0 = 10\,000$ euros, quelle est l'expression de $u_n$ (capital apres $n$ annees) ?
  \begin{enumerate}[label=\Alph*.]
    \item $u_n = 10\,000 - 500n$
    \item $u_n = 10\,000 \times 1{,}05^n$
    \item $u_n = 10\,000 \times 0{,}95^n$
    \item $u_n = 10\,000 \times 0{,}05^n$
  \end{enumerate}

\item[] \textbf{Capacite C7}

\item \textbf{[Q19]} On considere le programme Python suivant :

def terme(n):
    u = 2
    for k in range(n):
        u = 3 * u + 1

Quelle valeur renvoie l'appel terme(3) ?
  \begin{enumerate}[label=\Alph*.]

  \end{enumerate}

\item \textbf{[Q20]} On cherche le plus petit entier $n$ tel que $u_n > 100$, ou $u_0 = 1$ et $u_{n+1} = 2u_n$. Parmi les codes suivants, lequel est correct ?
  \begin{enumerate}[label=\Alph*.]
    \item u = 1

    u = 2 * u
    n = n + 1
    \item u = 1
n = 0

    u = 2 * u
    n = n + 1
    \item u = 1

    u = 2 * u
    n = n + 1
    \item u = 0
n = 0

    u = 2 * u
    n = n + 1
  \end{enumerate}

\item \textbf{[Q21]} On veut calculer $S = u_0 + u_1 + \cdots + u_5$ ou $u_k = 2^k$. Un eleve ecrit :

S = 0
for k in range(5):
    S = S + 2**k

Quelle erreur contient ce code ?
  \begin{enumerate}[label=\Alph*.]
    \item L'accumulateur devrait etre initialise a $1$.
    \item range(5) produit les indices $0, 1, 2, 3, 4$ : le terme $u_5 = 2^5$ est omis. Il faudrait range(6).
    \item Il faut ecrire $2^k$ et non $2**k$ en Python.
    \item Il faut une boucle while et non for pour calculer une somme.
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des reponses erronees}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{A} — L'eleve a calcule $2 \times 3^2 - 3 \times 3 + 1 = 18 - 9 + 1$ mais a oublie le coefficient 2 dans $2 \times 9 = 18$ et a pose $2 \times 3 + 1 = 10$. Erreur de calcul partiel. \emph{Renvoi : M1, etape 2.}
    \item \textbf{B} — L'eleve a remplace $n$ par $4$ au lieu de $3$ (decalage d'indice : confusion entre 'le 3e terme' et $u_3$). \emph{Renvoi : M1, etape 1.}
    \item \textbf{D} — L'eleve a calcule $2 \times 3 - 3 + 1 = 4$ ou a confondu $n^2$ avec $n$ dans la formule, obtenant $2 \times 3 - 3 + 1 = 4$ ou une variante. Erreur sur l'evaluation de $n^2$. \emph{Renvoi : M1, etape 2.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}

    \item \textbf{D} — L'eleve a calcule $v_1 = 3 \times 4 - 2 = 10$ puis $v_2 = 3 \times 10 - 2 = 28$ mais a commis une erreur arithmetique ($3 \times 8 - 2 = 22$) en utilisant $v_1 = 8$ (probable confusion avec $2 \times 4 = 8$). \emph{Renvoi : M1, etape 3.}
