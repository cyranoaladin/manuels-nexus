\begin{exercice}{1SPE-SUITES-EX-042}{3}{35}
Pour tout entier naturel $n$, on pose
\[
u_n=2\left(\frac34\right)^n
\qquad\text{et}\qquad
S_n=\sum_{k=0}^{n}u_k.
\]

\begin{enumerate}
  \item Calculer $u_0$, $u_1$, $u_2$, puis $S_0$, $S_1$ et $S_2$.
  Émettre une conjecture sur le comportement de $(S_n)$.

  \item À l'aide de la formule de la somme des termes d'une suite géométrique,
  montrer que, pour tout entier naturel $n$,
  \[
  S_n=8\left(1-\left(\frac34\right)^{n+1}\right).
  \]

  \item Montrer que $S_{n+1}-S_n=u_{n+1}>0$ et que $S_n<8$.
  Interpréter ces deux résultats.

  \item Calculer $S_{10}$, $S_{30}$ et $S_{100}$, arrondis au millième.
  À partir de ces seules observations numériques, conjecturer le comportement
  de $S_n$ lorsque $n$ devient grand.

  \item On définit maintenant
  \[
  v_n=3\left(\frac12\right)^n
  \qquad\text{et}\qquad
  T_n=\sum_{k=0}^{n}v_k.
  \]
  Établir une expression de $T_n$, montrer que $T_n<6$, puis calculer $T_{10}$.
  À ce rang, laquelle des deux sommes est la plus proche de la valeur qu'elle
  semble approcher ? Justifier numériquement.
\end{enumerate}
\end{exercice}
