\section*{\textcolor{chapcolor}{\MakeUppercase{Limites des fonctions — QCM d'auto-evaluation}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} $\lim_{x \to +\infty} \dfrac{3x^2 - x}{x^2 + 5}$
  \begin{enumerate}[label=\Alph*.]
    \item $0$
    \item $3$
    \item $-1/5$
    \item $+\infty$
  \end{enumerate}

\item \textbf{[Q2]} $\lim_{x \to +\infty} (x^2 - 3x + 1)$
  \begin{enumerate}[label=\Alph*.]
    \item $1$
    \item $-\infty$
    \item $+\infty$
    \item On ne peut pas conclure
  \end{enumerate}

\item \textbf{[Q3]} La forme $\dfrac{+\infty}{+\infty}$ est :
  \begin{enumerate}[label=\Alph*.]
    \item egale a 1
    \item egale a $+\infty$
    \item une forme indeterminee
    \item egale a 0
  \end{enumerate}

\item \textbf{[Q4]} $\lim_{x \to 0^+} \dfrac{1}{x}$
  \begin{enumerate}[label=\Alph*.]
    \item $0$
    \item $1$
    \item $+\infty$
    \item $-\infty$
  \end{enumerate}

\item \textbf{[Q5]} $\lim_{x \to +\infty} x\,\mathrm{e}^{-x}$
  \begin{enumerate}[label=\Alph*.]
    \item $+\infty$
    \item $0$
    \item $1$
    \item $-\infty$
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q6]} $y = 2$ est AH a $\mathcal{C}_f$ si :
  \begin{enumerate}[label=\Alph*.]
    \item $f(2) = 0$
    \item $\lim_{x \to 2} f(x) = +\infty$
    \item $\lim_{x \to \pm\infty} f(x) = 2$
    \item La courbe coupe $y = 2$
  \end{enumerate}

\item \textbf{[Q7]} AV de $f(x) = \dfrac{x+1}{x-3}$ :
  \begin{enumerate}[label=\Alph*.]
    \item $x = -1$
    \item $x = 3$
    \item $y = 1$
    \item $x = 0$
  \end{enumerate}

\item \textbf{[Q8]} AH de $f(x) = \dfrac{4x}{x+1}$ en $+\infty$ :
  \begin{enumerate}[label=\Alph*.]
    \item $y = 0$
    \item $y = 4$
    \item $y = 1$
    \item Pas d'AH
  \end{enumerate}

\item \textbf{[Q9]} $f(x) = \mathrm{e}^{-x} + 3$ admet :
  \begin{enumerate}[label=\Alph*.]
    \item AH $y=0$ et AV $x=3$
    \item AH $y=3$ en $+\infty$ et pas d'AV
    \item AH $y=3$ en $-\infty$ et pas d'AV
    \item Pas d'asymptote
  \end{enumerate}

\item \textbf{[Q10]} Si $\lim_{x \to 5^+} f(x) = -\infty$, alors :
  \begin{enumerate}[label=\Alph*.]
    \item $y=5$ est AH
    \item $x=5$ est AV
    \item $f(5) = -\infty$
    \item Courbe coupe l'axe en 5
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q11]} $\lim_{x \to +\infty} \dfrac{\mathrm{e}^x}{x^3}$
  \begin{enumerate}[label=\Alph*.]
    \item $0$
    \item $1$
    \item $+\infty$
    \item $1/3$
  \end{enumerate}

\item \textbf{[Q12]} $\lim_{x \to -\infty} x^2\,\mathrm{e}^x$
  \begin{enumerate}[label=\Alph*.]
    \item $+\infty$
    \item $0$
    \item $-\infty$
    \item $1$
  \end{enumerate}

\item \textbf{[Q13]} Pour $\lim_{+\infty}(\mathrm{e}^x - x^2)$ :
  \begin{enumerate}[label=\Alph*.]
    \item $+\infty - \infty = 0$
    \item Factoriser par $\mathrm{e}^x$, utiliser $x^2/\mathrm{e}^x \to 0$
    \item Factoriser par $x^2$
    \item FI sans conclusion
  \end{enumerate}

\item \textbf{[Q14]} $\mathrm{e}^x \geq 1 + x$ pour $x \geq 0$ s'obtient par :
  \begin{enumerate}[label=\Alph*.]
    \item Recurrence
    \item Etude de $g(x) = \mathrm{e}^x - 1 - x$
    \item Division euclidienne
    \item Formule de Taylor
  \end{enumerate}

\item \textbf{[Q15]} $\lim_{x \to +\infty} \dfrac{x^{100}}{\mathrm{e}^x}$
  \begin{enumerate}[label=\Alph*.]
    \item $+\infty$
    \item $100$
    \item $0$
    \item $1$
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des reponses erronees}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{A} — L'eleve confond avec une fraction dont le degre du numerateur est inferieur. \emph{Renvoi : M1, etape 3.}
    \item \textbf{C} — L'eleve a divise les termes constants (-1 et 5) au lieu des coefficients dominants. \emph{Renvoi : M1, etape 3.}
    \item \textbf{D} — L'eleve pense que tout quotient de polynomes de meme degre diverge. \emph{Renvoi : M1, etape 2.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — L'eleve a garde seulement le terme constant. \emph{Renvoi : C1, terme preponderant.}
    \item \textbf{B} — L'eleve a regarde le coefficient de $-3x$ au lieu de $x^2$. \emph{Renvoi : C1, terme preponderant.}
    \item \textbf{D} — L'eleve croit que $+\infty - \infty$ est toujours indetermine, meme apres factorisation. \emph{Renvoi : M1, piege 1.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — L'eleve simplifie mecaniquement infini/infini = 1. \emph{Renvoi : C1, formes indeterminees.}
    \item \textbf{B} — L'eleve pense que +inf/+inf = +inf. \emph{Renvoi : C1, formes indeterminees.}
    \item \textbf{D} — Confusion avec une autre forme. \emph{Renvoi : C1, formes indeterminees.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — L'eleve confond avec $\lim_{x \to +\infty} 1/x$. \emph{Renvoi : C1, limite en un point.}
    \item \textbf{B} — L'eleve evalue l'inverse en une valeur non nulle au lieu de faire tendre x vers 0 : 1/x explose, il ne se stabilise pas a 1. \emph{Renvoi : M1, limites en un point.}
    \item \textbf{D} — L'eleve confond $0^+$ et $0^-$. \emph{Renvoi : C1, limites a gauche/droite.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — L'eleve croit que $x \to +\infty$ l'emporte. \emph{Renvoi : M3, croissances comparees.}
    \item \textbf{C} — L'eleve confond avec un taux de variation. \emph{Renvoi : C3, resultat fondamental.}
    \item \textbf{D} — Signe errone : x et exp(-x) sont tous deux positifs pour x > 0, le produit ne peut pas tendre vers moins l'infini. \emph{Renvoi : M3, croissances comparees.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{A} — Confusion entre AH et zero de la fonction. \emph{Renvoi : C2, definition AH.}
    \item \textbf{B} — C'est la definition d'une AV, pas d'une AH. \emph{Renvoi : C2, definition AV.}
    \item \textbf{D} — L'intersection n'est pas le critere. \emph{Renvoi : C2, erreur frequente.}
  \end{itemize}
  \item \textbf{Q7}
  \begin{itemize}
    \item \textbf{A} — L'eleve a pris le zero du numerateur au lieu du denominateur. \emph{Renvoi : M2, etape 2.}
    \item \textbf{C} — C'est l'AH, pas l'AV. \emph{Renvoi : C2, definition AH vs AV.}
    \item \textbf{D} — x = 0 annule le numerateur, pas le denominateur : c'est un zero de la fonction, pas une asymptote verticale. \emph{Renvoi : M2, etape 1.}
  \end{itemize}
  \item \textbf{Q8}
  \begin{itemize}
    \item \textbf{A} — L'eleve confond avec le cas deg(num) < deg(den). \emph{Renvoi : M1, etape 3.}
    \item \textbf{C} — L'eleve a calcule le rapport des termes constants. \emph{Renvoi : M1, etape 3.}
    \item \textbf{D} — L'eleve conclut a l'absence d'asymptote horizontale des qu'il y a une asymptote verticale ; les deux sont independantes. \emph{Renvoi : M2, etape 3.}
  \end{itemize}
  \item \textbf{Q9}
  \begin{itemize}
    \item \textbf{A} — L'exponentielle n'a pas de valeur interdite. \emph{Renvoi : C2, AV seulement aux valeurs interdites.}
    \item \textbf{C} — $\mathrm{e}^{-x} \to 0$ en $+\infty$, pas en $-\infty$. \emph{Renvoi : C1, limites de l'exponentielle.}
    \item \textbf{D} — exp(-x) tend vers 0 en plus l'infini, donc f tend vers 3 : la droite y = 3 est bien asymptote horizontale. \emph{Renvoi : M2, etape 3.}
  \end{itemize}
  \item \textbf{Q10}
  \begin{itemize}
    \item \textbf{A} — Confusion AH/AV. \emph{Renvoi : C2, definitions.}
    \item \textbf{C} — $f(5)$ n'est pas defini en general, la limite n'est pas une valeur. \emph{Renvoi : C2, definition AV.}
    \item \textbf{D} — Une limite infinie en 5 signifie que la courbe s'echappe vers le bas au voisinage de 5, elle ne coupe pas l'axe des abscisses en ce point. \emph{Renvoi : M2, etape 2.}
  \end{itemize}
  \item \textbf{Q11}
  \begin{itemize}
    \item \textbf{A} — L'eleve a inverse numerateur et denominateur dans la croissance comparee. \emph{Renvoi : C3, theoreme.}
    \item \textbf{B} — L'eleve traite le quotient comme une forme indeterminee resolue par simplification ; c'est la croissance comparee qui tranche, l'exponentielle l'emportant sur toute puissance. \emph{Renvoi : M3, croissances comparees.}
    \item \textbf{D} — L'eleve a divise l'exposant par le degre. \emph{Renvoi : C3, theoreme.}
  \end{itemize}
  \item \textbf{Q12}
  \begin{itemize}
    \item \textbf{A} — L'eleve ne considere que $x^2 \to +\infty$ et oublie $\mathrm{e}^x \to 0$. \emph{Renvoi : C3, consequence en $-\infty$.}
    \item \textbf{C} — L'eleve confond $x$ et $x^2$ (le signe). \emph{Renvoi : C3, consequence en $-\infty$.}
    \item \textbf{D} — Resultat obtenu en evaluant en x = 0 au lieu de faire tendre x vers moins l'infini, ou l'exponentielle ecrase le facteur polynomial. \emph{Renvoi : M3, croissances comparees.}
  \end{itemize}
  \item \textbf{Q13}
  \begin{itemize}
    \item \textbf{A} — $+\infty - \infty$ est une FI, pas un resultat. \emph{Renvoi : C1, formes indeterminees.}
    \item \textbf{C} — Factoriser par x\textasciicircum{}2 laisse le quotient exp(x)/x\textasciicircum{}2 en facteur, dont la limite reste a determiner : on n'a pas leve l'indetermination, on l'a deplacee. \emph{Renvoi : M3, methode de factorisation.}
    \item \textbf{D} — On peut conclure grace aux croissances comparees. \emph{Renvoi : M3, pas a pas.}
  \end{itemize}
  \item \textbf{Q14}
  \begin{itemize}
    \item \textbf{A} — La recurrence s'applique aux entiers, pas aux reels. \emph{Renvoi : C3, demonstration.}
    \item \textbf{C} — La division euclidienne concerne les polynomes. \emph{Renvoi : C3, demonstration.}
    \item \textbf{D} — La formule de Taylor n'est pas au programme de terminale ; l'inegalite s'obtient par etude du signe de la fonction x -> exp(x) - 1 - x. \emph{Renvoi : cours C3, inegalite de convexite.}
  \end{itemize}
  \item \textbf{Q15}
  \begin{itemize}
    \item \textbf{A} — L'eleve pense que $x^{100}$ croit plus vite que $\mathrm{e}^x$. \emph{Renvoi : C3, theoreme : pour TOUT $n$.}
    \item \textbf{B} — L'exposant n'intervient pas dans la valeur de la limite. \emph{Renvoi : C3, theoreme.}
    \item \textbf{D} — Meme confusion qu'en Q12 : l'exposant 100, si grand soit-il, ne change pas la hierarchie des croissances comparees. \emph{Renvoi : M3, croissances comparees.}
  \end{itemize}
\end{itemize}
