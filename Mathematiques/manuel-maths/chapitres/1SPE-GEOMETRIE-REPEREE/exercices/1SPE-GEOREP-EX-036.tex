\begin{exercice}{1SPE-GEOREP-EX-036}{2}{15}
Soit le cercle $\mathcal{C}$ de centre $\Omega(3 ; 1)$ et de rayon $2$, et la droite $\mathcal{D} : 3x + 4y - 5 = 0$.
\begin{enumerate}
  \item Calculer la distance de $\Omega$ à $\mathcal{D}$.
  \item En déduire la position relative de $\mathcal{D}$ et $\mathcal{C}$.
\end{enumerate}
\end{exercice}
