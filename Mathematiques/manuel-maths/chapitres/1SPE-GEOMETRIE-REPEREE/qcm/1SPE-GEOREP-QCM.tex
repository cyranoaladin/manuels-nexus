\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} La droite passant par $A(1 ; 3)$ et $B(3 ; 7)$ a pour equation :
  \begin{enumerate}[label=\Alph*.]
    \item $2x - y + 1 = 0$
    \item $x - 2y + 5 = 0$
    \item $2x + y - 5 = 0$
    \item $x + 2y - 7 = 0$
  \end{enumerate}

\item \textbf{[Q2]} Le point $P(2 ; -1)$ appartient a la droite $3x + 2y - 4 = 0$.
  \begin{enumerate}[label=\Alph*.]
    \item Vrai
    \item Faux
    \item On ne peut pas savoir
    \item Seulement si le repere est orthonorme
  \end{enumerate}

\item \textbf{[Q3]} La droite d'equation $x = 5$ est :
  \begin{enumerate}[label=\Alph*.]
    \item horizontale
    \item verticale
    \item de coefficient directeur $5$
    \item de coefficient directeur $1/5$
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q4]} Un vecteur normal a la droite $4x - 3y + 2 = 0$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $\vec{n}(3 ; 4)$
    \item $\vec{n}(4 ; -3)$
    \item $\vec{n}(-3 ; 4)$
    \item $\vec{n}(4 ; 3)$
  \end{enumerate}

\item \textbf{[Q5]} Un vecteur directeur de la droite $2x + 5y - 1 = 0$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $\vec{u}(2 ; 5)$
    \item $\vec{u}(-5 ; 2)$
    \item $\vec{u}(5 ; 2)$
    \item $\vec{u}(5 ; -2)$
  \end{enumerate}

\item \textbf{[Q6]} Les droites $D_1 : 2x - y + 3 = 0$ et $D_2 : x + 2y - 1 = 0$ sont :
  \begin{enumerate}[label=\Alph*.]
    \item paralleles
    \item confondues
    \item perpendiculaires
    \item secantes non perpendiculaires
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q7]} L'equation $(x - 3)^2 + (y + 1)^2 = 16$ represente un cercle de :
  \begin{enumerate}[label=\Alph*.]
    \item centre $(3 ; -1)$ et rayon $16$
    \item centre $(3 ; 1)$ et rayon $4$
    \item centre $(3 ; -1)$ et rayon $4$
    \item centre $(-3 ; 1)$ et rayon $4$
  \end{enumerate}

\item \textbf{[Q8]} Le cercle de centre $(0 ; 0)$ et de rayon $3$ a pour equation :
  \begin{enumerate}[label=\Alph*.]
    \item $x^2 + y^2 = 3$
    \item $x^2 + y^2 = 9$
    \item $(x + 3)^2 + (y + 3)^2 = 0$
    \item $x + y = 3$
  \end{enumerate}

\item \textbf{[Q9]} L'equation $x^2 + y^2 - 2x + 4y - 4 = 0$ represente :
  \begin{enumerate}[label=\Alph*.]
    \item un cercle de centre $(1 ; -2)$ et de rayon $3$
    \item un cercle de centre $(-1 ; 2)$ et de rayon $9$
    \item un cercle de centre $(1 ; -2)$ et de rayon $9$
    \item aucun cercle
  \end{enumerate}

\item[] \textbf{Capacite C4}

\item \textbf{[Q10]} La distance du point $P(1 ; 2)$ a la droite $3x + 4y - 7 = 0$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $4/5$
    \item $4$
    \item $7/5$
    \item $0$
  \end{enumerate}

\item \textbf{[Q11]} Si la distance du centre d'un cercle de rayon $5$ a une droite est $5$, alors :
  \begin{enumerate}[label=\Alph*.]
    \item la droite est secante au cercle
    \item la droite est exterieure au cercle
    \item la droite est tangente au cercle
    \item on ne peut pas conclure
  \end{enumerate}

\item \textbf{[Q12]} Les droites $D_1 : 3x - y + 2 = 0$ et $D_2 : 6x - 2y + 4 = 0$ sont :
  \begin{enumerate}[label=\Alph*.]
    \item strictement paralleles
    \item secantes
    \item perpendiculaires
    \item confondues
  \end{enumerate}

\item[] \textbf{Capacite C5}

\item \textbf{[Q13]} L'aire du triangle de sommets $A(0 ; 0)$, $B(4 ; 0)$ et $C(0 ; 6)$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $24$
    \item $12$
    \item $10$
    \item $48$
  \end{enumerate}

\item \textbf{[Q14]} Le projete orthogonal de $P$ sur une droite $D$ est :
  \begin{enumerate}[label=\Alph*.]
    \item le milieu de $P$ et du pied de la perpendiculaire
    \item le point de $D$ le plus proche de $P$
    \item le symetrique de $P$ par rapport a $D$
    \item l'intersection de $D$ avec l'axe des abscisses
  \end{enumerate}

\item \textbf{[Q15]} Si $ABCD$ est un parallelogramme et $AC = BD$, alors $ABCD$ est :
  \begin{enumerate}[label=\Alph*.]
    \item un losange
    \item un rectangle
    \item un carre
    \item un trapeze
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des reponses erronees}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{B} — Inversion du coefficient directeur (1/2 au lieu de 2). \emph{Renvoi : M1, etape 1.}
    \item \textbf{C} — Erreur de signe sur le coefficient de y. \emph{Renvoi : M1, etape 2.}
    \item \textbf{D} — Confusion entre les coefficients a et b. \emph{Renvoi : M1, etape 2.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{B} — Erreur de calcul : $3(2) + 2(-1) - 4 = 6 - 2 - 4 = 0$. \emph{Renvoi : M1, etape 3.}
    \item \textbf{C} — On peut toujours verifier en substituant les coordonnees. \emph{Renvoi : M1, etape 3.}
    \item \textbf{D} — L'appartenance a une droite ne depend pas du type de repere. \emph{Renvoi : Cours C1.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — Confusion : $y = 5$ est horizontale, $x = 5$ est verticale. \emph{Renvoi : Cours C1.}
    \item \textbf{C} — La droite $x = 5$ n'a pas de coefficient directeur (verticale). \emph{Renvoi : Cours C1.}
    \item \textbf{D} — La droite $x = 5$ n'a pas de coefficient directeur (verticale). \emph{Renvoi : Cours C1.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — C'est le vecteur directeur, pas le vecteur normal. \emph{Renvoi : M2, etape 1.}
    \item \textbf{C} — Inversion des signes : le vecteur normal est $(a ; b)$ et non $(-b ; a)$. \emph{Renvoi : M2, etape 1.}
    \item \textbf{D} — Erreur de signe sur la deuxieme coordonnee. \emph{Renvoi : M2, etape 1.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — C'est le vecteur normal, pas le vecteur directeur. \emph{Renvoi : M2, etape 1.}
    \item \textbf{C} — Oubli du signe negatif sur la premiere coordonnee. \emph{Renvoi : M2, etape 1.}
    \item \textbf{D} — Le signe negatif est sur la premiere coordonnee, pas la deuxieme. \emph{Renvoi : M2, etape 1.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{A} — $a_1 b_2 - a_2 b_1 = 2(2) - 1(-1) = 5 
eq 0$, pas paralleles. \emph{Renvoi : Cours C4.}
    \item \textbf{B} — Les coefficients ne sont pas proportionnels. \emph{Renvoi : Cours C4.}
    \item \textbf{D} — $a_1 a_2 + b_1 b_2 = 2(1) + (-1)(2) = 0$, donc perpendiculaires. \emph{Renvoi : M2, propriete 3.}
  \end{itemize}
  \item \textbf{Q7}
  \begin{itemize}
    \item \textbf{A} — Le rayon est $\sqrt{16} = 4$, pas $16$. \emph{Renvoi : M3, piege 2.}
    \item \textbf{B} — Erreur de signe : $(y + 1)^2 = (y - (-1))^2$, donc le centre a pour ordonnee $-1$. \emph{Renvoi : M3, piege 1.}
    \item \textbf{D} — Erreur de signe sur les deux coordonnees. \emph{Renvoi : M3, piege 1.}
  \end{itemize}
  \item \textbf{Q8}
  \begin{itemize}
    \item \textbf{A} — Le membre de droite est $r^2 = 9$, pas $r = 3$. \emph{Renvoi : M3, piege 2.}
    \item \textbf{C} — Le centre est $(0;0)$, pas $(-3;-3)$, et le rayon ne peut pas etre $0$. \emph{Renvoi : Cours C3.}
    \item \textbf{D} — C'est l'equation d'une droite, pas d'un cercle. \emph{Renvoi : Cours C3.}
  \end{itemize}
  \item \textbf{Q9}
  \begin{itemize}
    \item \textbf{B} — Erreur de signe sur le centre : c'est $(-A/2, -B/2) = (1, -2)$. \emph{Renvoi : M3, piege 1.}
    \item \textbf{C} — Le rayon est $\sqrt{1 + 4 + 4} = 3$, pas $9$. \emph{Renvoi : M3, piege 2.}
    \item \textbf{D} — $r^2 = 1 + 4 + 4 = 9 > 0$, c'est bien un cercle. \emph{Renvoi : Cours C3.}
  \end{itemize}
  \item \textbf{Q10}
  \begin{itemize}
    \item \textbf{B} — Oubli de diviser par $\sqrt{a^2+b^2}$ : le numerateur est $|3+8-7|=4$, mais il faut diviser par $\sqrt{9+16}=5$. \emph{Renvoi : M4, etape 1.}
    \item \textbf{C} — Erreur de calcul dans le numerateur : $|3+8-7|=4$ et non $7$. \emph{Renvoi : M4, etape 1.}
    \item \textbf{D} — $3(1)+4(2)-7 = 4 
eq 0$, donc $P$ n'est pas sur la droite et la distance n'est pas nulle. \emph{Renvoi : M4, etape 1.}
  \end{itemize}
  \item \textbf{Q11}
  \begin{itemize}
    \item \textbf{A} — La droite est secante si $d < r$, or ici $d = r$. \emph{Renvoi : Cours C4, propriete 3.}
    \item \textbf{B} — La droite est exterieure si $d > r$, or ici $d = r$. \emph{Renvoi : Cours C4, propriete 3.}
    \item \textbf{D} — La comparaison $d$ vs $r$ suffit toujours a conclure. \emph{Renvoi : Cours C4, propriete 3.}
  \end{itemize}
  \item \textbf{Q12}
  \begin{itemize}
    \item \textbf{A} — $D_2 = 2 D_1$ exactement (tous les coefficients proportionnels, meme $c$), donc confondues. \emph{Renvoi : Cours C4, propriete 1.}
    \item \textbf{B} — Les vecteurs normaux sont colineaires, donc pas secantes. \emph{Renvoi : Cours C4.}
    \item \textbf{C} — $a_1 a_2 + b_1 b_2 = 18 + 2 = 20 
eq 0$. \emph{Renvoi : Cours C2, propriete 3.}
  \end{itemize}
  \item \textbf{Q13}
  \begin{itemize}
    \item \textbf{A} — Oubli du facteur $1/2$ dans la formule de l'aire. \emph{Renvoi : Cours C5, propriete 4.}
    \item \textbf{C} — Erreur de calcul : le determinant vaut $24$, pas $20$. \emph{Renvoi : Cours C5, propriete 4.}
    \item \textbf{D} — Erreur de calcul : l'aire est $|4 \times 6|/2 = 12$, pas $48$. \emph{Renvoi : Cours C5.}
  \end{itemize}
  \item \textbf{Q14}
  \begin{itemize}
    \item \textbf{A} — Le projete orthogonal EST le pied de la perpendiculaire, pas le milieu. \emph{Renvoi : Cours C5, propriete 2.}
    \item \textbf{C} — Le symetrique est le point tel que le projete est le milieu de $[PP']$. \emph{Renvoi : Cours C5, propriete 3.}
    \item \textbf{D} — Le projete n'a rien a voir avec l'axe des abscisses en general. \emph{Renvoi : Cours C5.}
  \end{itemize}
  \item \textbf{Q15}
  \begin{itemize}
    \item \textbf{A} — Un losange a ses diagonales perpendiculaires, pas necessairement egales. \emph{Renvoi : Cours C5, propriete 1.}
    \item \textbf{C} — Un carre est un cas particulier (rectangle ET losange), on n'a pas assez d'information. \emph{Renvoi : Cours C5.}
    \item \textbf{D} — Un parallelogramme est un cas particulier de trapeze, mais la propriete le precise en rectangle. \emph{Renvoi : Cours C5.}
  \end{itemize}
\end{itemize}
