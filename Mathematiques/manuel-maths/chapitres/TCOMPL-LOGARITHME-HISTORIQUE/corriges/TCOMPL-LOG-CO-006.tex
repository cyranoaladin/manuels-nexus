\begin{corrige}{TCOMPL-LOG-EX-006}
\textbf{1.} $\ln8-\ln2+\ln5=\ln\dfrac{8\times5}{2}=\ln20$.

\textbf{2.} Le domaine impose $x>3$. L'équation devient
$\ln\big(x(x-3)\big)=\ln10$, soit $x^2-3x-10=0$, de racines $-2$ et $5$.
Seule $x=5$ appartient au domaine.

\textbf{3.} Le domaine impose $x>\tfrac12$. La fonction $\ln$ étant
strictement croissante, l'inéquation équivaut à $2x-1\leq7$, soit $x\leq4$.
L'ensemble solution est $\left]\tfrac12\,;\,4\right]$.
\end{corrige}
