\begin{corrige}{TCOMPL-LOG-EX-016}

\textbf{Question 1.} $f'(x) = \dfrac{\mathrm{e}^x(x+1) - \mathrm{e}^x}{(x+1)^2} = \dfrac{x\,\mathrm{e}^x}{(x+1)^2}$.

\commentaireMarge{Le signe de $f'$ est celui de $x$ car $\mathrm{e}^x > 0$ et $(x+1)^2 > 0$.}

$f'(x) = 0 \iff x = 0$. $f' < 0$ sur $\left]-\infty, -1\right[ \cup \left]-1, 0\right[$ et $f' > 0$ sur $\left]0, +\infty\right[$.

\textbf{Question 2.} $\lim_{x \to +\infty} f = +\infty$, $\lim_{x \to -\infty} f = 0$ (croissances compareees), $\lim_{x \to -1^+} f = +\infty$.

\textbf{Question 3.} Minimum local $f(0) = 1$. La droite $x = -1$ est asymptote verticale.

\end{corrige}
