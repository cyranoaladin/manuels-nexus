\begin{corrige}{TCOMPL-LOG-EX-005}
\textbf{1.} $\mathrm{e}^x=5\iff x=\ln5$. $\ln x=-2\iff x=\mathrm{e}^{-2}$.

\textbf{2.} $\displaystyle\lim_{x\to0^+}\ln x=-\infty$ et
$\displaystyle\lim_{x\to+\infty}\ln x=+\infty$ : $\ln$ est la réciproque de
l'exponentielle, dont les limites en $-\infty$ et $+\infty$ sont $0^+$ et
$+\infty$.

\textbf{3.} La dérivée de $\ln u$ est $\dfrac{u'}{u}$, donc celle de
$\ln(3x+1)$ est $\dfrac{3}{3x+1}$.
\end{corrige}
