\begin{exercice}{TCOMPL-LOG-EV-A-EX1}{2}{25}
\textbf{Exercice 1} (10 points) --- \textit{Capacités C2, C4}

\medskip

\begin{enumerate}
  \item (3 pts) Démontrer que pour $a,b>0$, $\ln(ab) = \ln(a)+\ln(b)$ (on admettra $\mathrm{e}^{\ln x}=x$).
  \item (7 pts) Résoudre l'équation $\ln(x) + \ln(x+3) = \ln(4)$ (préciser le domaine).
\end{enumerate}
\end{exercice}

\begin{exercice}{TCOMPL-LOG-EV-A-EX2}{2}{25}
\textbf{Exercice 2} (10 points) --- \textit{Capacité C5}

\medskip

Soit $f(x) = \ln(5x+2)$.
\begin{enumerate}
  \item (2 pts) Donner le domaine de définition de $f$.
  \item (5 pts) Calculer $f'(x)$.
  \item (3 pts) Étudier le signe de $f'(x)$ et en déduire le sens de variation de $f$.
\end{enumerate}
\end{exercice}
