\begin{corrige}{TCOMPL-LOG-EV-B-EX1}
\textbf{1.} Pour $x>0$, $\ln x$ est l'unique réel $y$ tel que $\mathrm{e}^y=x$. On a $\ln 1 = 0$ et $\ln \mathrm{e} = 1$.

\textbf{2.} Avec $b=\dfrac{1}{a}$ : $\ln\!\left(a\times\dfrac{1}{a}\right) = \ln(a) + \ln\!\left(\dfrac{1}{a}\right)$. Or $a\times\dfrac{1}{a}=1$ et $\ln(1)=0$, donc $0 = \ln(a)+\ln\!\left(\dfrac{1}{a}\right)$, soit $\ln\!\left(\dfrac{1}{a}\right) = -\ln(a)$.
\end{corrige}
