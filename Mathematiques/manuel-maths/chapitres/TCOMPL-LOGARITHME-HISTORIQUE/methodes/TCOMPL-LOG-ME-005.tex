\begin{fichemethode}{M5}{Démontrer que la dérivée de ln est x -> 1/x}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande de DEMONTRER que
  $\ln'(x) = \dfrac{1}{x}$ sur $]0\,;\,+\infty[$, en ADMETTANT que $\ln$ est
  dérivable.
}

\pasApas{%
  \begin{enumerate}
    \item Enoncer le point de depart : pour tout $x > 0$, $e^{\ln x} = x$, et
      $\ln$ est dérivable sur $]0\,;\,+\infty[$ (admis).
    \item Dériver les deux membres de l'égalité $e^{\ln x} = x$ : a droite, la
      dérivée de $x \mapsto x$ vaut 1.
    \item A gauche, appliquer la dérivation d'une composee
      $(e^{u})' = u'\,e^{u}$ avec $u = \ln$ :
      $\bigl(e^{\ln x}\bigr)' = \ln'(x)\,e^{\ln x}$.
    \item Egaler : $\ln'(x)\,e^{\ln x} = 1$, et remplacer $e^{\ln x}$ par $x$ :
      $\ln'(x) \times x = 1$.
    \item Conclure en divisant par $x \neq 0$ : $\ln'(x) = \dfrac{1}{x}$ pour
      tout $x > 0$.
  \end{enumerate}
}

\exempleRedige{%
  Démontrer que pour tout $x > 0$, $\ln'(x) = \dfrac{1}{x}$ (derivabilite de
  $\ln$ admise).

  \medskip
  \commentaireMarge{Dériver l'identite $e^{\ln x} = x$.}%
  Pour tout $x > 0$, $e^{\ln x} = x$. Les deux membres sont derivables sur
  $]0\,;\,+\infty[$ ; en derivant, avec $(e^u)' = u'e^u$ :
  $\ln'(x)\,e^{\ln x} = 1$.

  \medskip
  \commentaireMarge{Remplacer $e^{\ln x}$ par $x$.}%
  Or $e^{\ln x} = x$, donc $x\,\ln'(x) = 1$. Comme $x > 0$, on peut diviser :
  $\ln'(x) = \dfrac{1}{x}$. \hfill$\square$
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Oublier de mentionner l'hypothèse ADMISE (la
      derivabilite de $\ln$) : sans elle, on ne peut pas dériver la composee.
    \item \textbf{Piège 2.} Dériver $e^{\ln x}$ en $e^{\ln x}$ tout court, en
      oubliant le facteur $\ln'(x)$ de la composee.
    \item \textbf{Piège 3.} Diviser par $x$ sans signaler $x \neq 0$ (garanti
      ici par $x > 0$).
  \end{itemize}
}

\verifier{%
  Contrôler la cohérence du résultat : $\dfrac{1}{x} > 0$ sur
  $]0\,;\,+\infty[$, ce qui confirme la stricte croissance de $\ln$ ; et la
  tangente en 1 a pour pente $\ln'(1) = 1$ (fiche M1).
}

\end{fichemethode}
