\begin{fichemethode}{M4}{Demontrer ln(ab) = ln(a) + ln(b) et ln(1/a) = -ln(a)}

\quandUtiliser{%
  Utiliser cette methode lorsque l'enonce demande de DEMONTRER les proprietes
  algebriques du logarithme a partir de la relation fonctionnelle de
  l'exponentielle ($e^{u+v} = e^u e^v$) et de la definition de $\ln$ comme
  reciproque.
}

\pasApas{%
  \begin{enumerate}
    \item Poser le cadre : $a > 0$, $b > 0$, et rappeler les deux outils :
      $e^{u+v} = e^u\,e^v$ et $e^{\ln x} = x$ pour $x > 0$.
    \item Calculer $e^{\ln a + \ln b}$ avec la relation fonctionnelle :
      $e^{\ln a + \ln b} = e^{\ln a}\,e^{\ln b} = ab$.
    \item Comparer : on a aussi $e^{\ln(ab)} = ab$. Deux reels ayant la meme
      image par $\exp$ sont egaux ($\exp$ est strictement croissante donc
      injective) : $\ln(ab) = \ln a + \ln b$.
    \item Pour $\ln\!\dfrac{1}{a}$ : appliquer la propriete a
      $a \times \dfrac{1}{a} = 1$ :
      $\ln(a) + \ln\!\dfrac{1}{a} = \ln(1) = 0$, d'ou
      $\ln\!\dfrac{1}{a} = -\ln(a)$.
    \item Citer a chaque etape la propriete utilisee (relation fonctionnelle,
      reciproque, injectivite) : c'est une demonstration, pas un calcul.
  \end{enumerate}
}

\exempleRedige{%
  Demontrer que pour tous $a, b > 0$, $\ln(ab) = \ln(a) + \ln(b)$.

  \medskip
  \commentaireMarge{Tout passe par l'exponentielle.}%
  D'une part $e^{\ln(ab)} = ab$ (definition de la reciproque, $ab > 0$).
  D'autre part
  $e^{\ln a + \ln b} = e^{\ln a} \times e^{\ln b} = a \times b$
  (relation fonctionnelle de $\exp$, puis definition de la reciproque).

  \medskip
  \commentaireMarge{Injectivite de $\exp$.}%
  Ainsi $e^{\ln(ab)} = e^{\ln a + \ln b}$. La fonction $\exp$ etant strictement
  croissante sur $\mathbb{R}$, elle ne prend jamais deux fois la meme valeur :
  $\ln(ab) = \ln(a) + \ln(b)$. \hfill$\square$
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piege 1.} Utiliser la propriete a demontrer dans sa propre
      demonstration (raisonnement circulaire).
    \item \textbf{Piege 2.} Oublier de justifier le passage
      $e^X = e^Y \Rightarrow X = Y$ (injectivite / stricte croissance de
      $\exp$).
    \item \textbf{Piege 3.} Negliger les hypotheses $a > 0$, $b > 0$ : sans
      elles, $\ln(a)$ ou $\ln(b)$ n'existe pas.
  \end{itemize}
}

\verifier{%
  Tester la propriete demontree sur des valeurs concretes ($a = e$, $b = e^2$ :
  $\ln(e^3) = 3 = 1 + 2$), et relire la chaine deductive pour s'assurer
  qu'aucune etape n'utilise le resultat final.
}

\end{fichemethode}
