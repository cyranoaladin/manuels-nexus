\begin{exercice}{TCOMPL-LOG-EX-008}{2}{15}
On admet l'équation fonctionnelle : pour tous réels $a>0$ et $b>0$,
$\ln(ab)=\ln a+\ln b$.
\begin{enumerate}
  \item Démontrer que $\ln\!\left(\dfrac{1}{a}\right)=-\ln a$ pour tout
        $a>0$.
  \item En déduire que $\ln\!\left(\dfrac{a}{b}\right)=\ln a-\ln b$.
  \item Démontrer par récurrence que $\ln(a^{\,n})=n\ln a$ pour tout entier
        $n\geq1$.
\end{enumerate}
\end{exercice}
