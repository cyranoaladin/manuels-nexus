\begin{exercice}{TCOMPL-LOG-EX-022}{1}{15}
Soit $f = \mathrm{e}^x$. On rappelle que $f$ est convexe sur $\mathbb{R}$.
\begin{enumerate}
  \item Enoncer l'inegalite de Jensen pour $f$ aux points $x$ et $y$ avec $t = \dfrac{1}{2}$.
  \item Verifier cette inegalite pour $x = 0$ et $y = 2$.
\end{enumerate}

\end{exercice}
