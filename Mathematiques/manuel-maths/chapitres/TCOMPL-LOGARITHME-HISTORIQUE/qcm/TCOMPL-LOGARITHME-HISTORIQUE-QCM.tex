\section*{\textcolor{chapcolor}{\MakeUppercase{Logarithme neperien et applications}}}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q1]} La fonction logarithme neperien $\ln(x)$ est définie sur :
  \begin{enumerate}[label=\Alph*.]
    \item $]0, +\infty[$
    \item $\mathbb{R}$
    \item $[0, +\infty[$
    \item $\mathbb{R}^*$
  \end{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q2]} Pour tous $a,b > 0$, $\ln(a \times b) =$
  \begin{enumerate}[label=\Alph*.]
    \item $\ln(a) + \ln(b)$
    \item $\ln(a) \times \ln(b)$
    \item $\ln(a+b)$
    \item $\ln(a) - \ln(b)$
  \end{enumerate}

\item[] \textbf{Capacité C5}

\item \textbf{[Q3]} La dérivée de $f(x) = \ln(x)$ sur $]0, +\infty[$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $e^x$
    \item $\frac{1}{x}$
    \item $\ln(x)$
    \item $-\frac{1}{x^2}$
  \end{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q4]} La limite de $\ln(x)$ quand $x \to 0^+$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $0$
    \item $1$
    \item $-\infty$
    \item $+\infty$
  \end{enumerate}

\item[] \textbf{Capacité C3}

\item \textbf{[Q5]} Pour $a>0$ et tout entier $n$, $\ln(a^n) =$
  \begin{enumerate}[label=\Alph*.]
    \item $\ln(a)^n$
    \item $a \ln(n)$
    \item $\frac{1}{n} \ln(a)$
    \item $n \ln(a)$
  \end{enumerate}

\item[] \textbf{Capacité C4}

\item \textbf{[Q6]} On admet l'équation fonctionnelle $\ln(ab)=\ln a+\ln b$ pour tous $a,b>0$. Pour en déduire $\ln\!\left(\tfrac1a\right)=-\ln a$, il suffit de l'appliquer à :
  \begin{enumerate}[label=\Alph*.]
    \item $b=a$
    \item $b=\tfrac1a$, puis d'utiliser $\ln 1=0$
    \item $b=0$
    \item $a=1$ et $b=1$
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des réponses erronées}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{B} — Les nombres négatifs n'ont pas de logarithme réel. Le logarithme népérien, réciproque de l'exponentielle, est défini exactement sur ]0,+infini[. \emph{Renvoi : C1.}
    \item \textbf{C} — ln(0) n'est pas défini. Le logarithme népérien, réciproque de l'exponentielle, est défini exactement sur ]0,+infini[. \emph{Renvoi : C1.}
    \item \textbf{D} — Exclusion des nombres strictement négatifs. Le logarithme népérien, réciproque de l'exponentielle, est défini exactement sur ]0,+infini[. \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{B} — Transformation du produit en somme, pas en produit. L'équation fonctionnelle donne ln(ab)=ln(a)+ln(b) pour a et b strictement positifs. \emph{Renvoi : C2.}
    \item \textbf{C} — ln ne conserve pas l'addition. L'équation fonctionnelle donne ln(ab)=ln(a)+ln(b) pour a et b strictement positifs. \emph{Renvoi : C2.}
    \item \textbf{D} — Formule du quotient. L'équation fonctionnelle donne ln(ab)=ln(a)+ln(b) pour a et b strictement positifs. \emph{Renvoi : C2.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — Dérivée de l'exponentielle. Sur ]0,+infini[, la dérivée du logarithme népérien est la fonction x vers 1/x. \emph{Renvoi : C5.}
    \item \textbf{C} — Pas de changement par dérivation. Sur ]0,+infini[, la dérivée du logarithme népérien est la fonction x vers 1/x. \emph{Renvoi : C5.}
    \item \textbf{D} — Dérivée de 1/x. Sur ]0,+infini[, la dérivée du logarithme népérien est la fonction x vers 1/x. \emph{Renvoi : C5.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — ln(1) = 0, pas la limite en 0. Lorsque x tend vers 0 par valeurs positives, ln(x) décroît sans borne et tend vers -infini. \emph{Renvoi : C1.}
    \item \textbf{B} — Valeur erronee. Lorsque x tend vers 0 par valeurs positives, ln(x) décroît sans borne et tend vers -infini. \emph{Renvoi : C1.}
    \item \textbf{D} — Erreur de signe de l'infini. Lorsque x tend vers 0 par valeurs positives, ln(x) décroît sans borne et tend vers -infini. \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — Puissance a l'extérieur du logarithme. Pour a>0 et n entier, l'équation fonctionnelle itérée donne ln(a\textasciicircum{}n)=n ln(a). \emph{Renvoi : C3.}
    \item \textbf{B} — Inversion de a et n. Pour a>0 et n entier, l'équation fonctionnelle itérée donne ln(a\textasciicircum{}n)=n ln(a). \emph{Renvoi : C3.}
    \item \textbf{C} — Inverse du facteur. Pour a>0 et n entier, l'équation fonctionnelle itérée donne ln(a\textasciicircum{}n)=n ln(a). \emph{Renvoi : C3.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{A} — Prendre $b=a$ donne $\ln(a^2)=2\ln a$ : une identité vraie, mais où $\tfrac1a$ n'apparaît jamais. Avec $b=\tfrac1a$ on obtient $\ln 1=\ln a+\ln\tfrac1a$, d'où $\ln\tfrac1a=-\ln a$. \emph{Renvoi : C4.}
    \item \textbf{C} — La valeur $0$ n'appartient pas au domaine de $\ln$, défini sur $]0;+\infty[$ : l'équation fonctionnelle ne s'y applique pas. Le bon choix est $b=\tfrac1a$. \emph{Renvoi : C4.}
    \item \textbf{D} — Avec $a=b=1$ l'équation donne $\ln 1=2\ln 1$, donc $\ln 1=0$ : c'est un préliminaire utile, mais il ne fait pas apparaître $\tfrac1a$. Il faut ensuite poser $b=\tfrac1a$. \emph{Renvoi : C4.}
  \end{itemize}
\end{itemize}
