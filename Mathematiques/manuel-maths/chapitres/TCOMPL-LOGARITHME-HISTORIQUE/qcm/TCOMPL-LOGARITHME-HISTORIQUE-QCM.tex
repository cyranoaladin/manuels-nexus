\subsection*{C1 --- Définition}

\begin{enumerate}

\item \textbf{[Q1]} Le domaine de définition de $\ln$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $\mathbb{R}$.
    \item $]0,+\infty[$.
    \item $[0,+\infty[$.
    \item $\mathbb{R}^*$.
  \end{enumerate}

\bigskip

\subsection*{C4 --- Équation fonctionnelle}

\item \textbf{[Q2]} $\ln(ab)$ (pour $a,b>0$) est égal à :
  \begin{enumerate}[label=\Alph*.]
    \item $\ln(a) \times \ln(b)$.
    \item $\ln(a) + \ln(b)$.
    \item $\ln(a+b)$.
    \item $\ln(a) - \ln(b)$.
  \end{enumerate}

\bigskip

\subsection*{C2 --- Résolution d'équations}

\item \textbf{[Q3]} Pour résoudre une équation contenant $\ln$, il faut d'abord :
  \begin{enumerate}[label=\Alph*.]
    \item ignorer le domaine de définition.
    \item déterminer le domaine de définition (arguments strictement positifs).
    \item remplacer $\ln$ par $\exp$.
    \item supposer que toutes les solutions trouvées sont valables.
  \end{enumerate}

\bigskip

\subsection*{C3 --- Seuils}

\item \textbf{[Q4]} Pour une suite géométrique décroissante de raison $0<q<1$, résoudre $q^n < k$ en passant au logarithme nécessite de :
  \begin{enumerate}[label=\Alph*.]
    \item ne rien changer au sens de l'inégalité.
    \item inverser le sens de l'inégalité, car $\ln q < 0$.
    \item multiplier par $-1$ les deux membres avant de continuer.
    \item cette méthode ne fonctionne pas pour $q<1$.
  \end{enumerate}

\bigskip

\subsection*{C5 --- Dérivée}

\item \textbf{[Q5]} La dérivée de $\ln$ sur $]0,+\infty[$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $\ln(x)$.
    \item $\dfrac{1}{x}$.
    \item $x$.
    \item $\mathrm{e}^x$.
  \end{enumerate}

\bigskip

\item \textbf{[Q6]} La dérivée de $\ln(u(x))$, pour $u$ dérivable et strictement positive, est :
  \begin{enumerate}[label=\Alph*.]
    \item $u'(x)$.
    \item $\dfrac{u'(x)}{u(x)}$.
    \item $\dfrac{1}{u'(x)}$.
    \item $u(x) \times u'(x)$.
  \end{enumerate}

\end{enumerate}
