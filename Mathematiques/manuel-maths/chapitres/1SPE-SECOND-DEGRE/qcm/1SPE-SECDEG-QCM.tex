\section*{\textcolor{chapcolor}{\MakeUppercase{Fonctions polynomes du second degre — QCM d'auto-evaluation}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} Parmi les expressions suivantes, laquelle est un polynome du second degre ?
  \begin{enumerate}[label=\Alph*.]
    \item $f(x) = 3x + 7$
    \item $f(x) = 2x^2 - 5x + 1$
    \item $f(x) = x^3 - 2x + 4$
    \item $f(x) = \dfrac{1}{x^2} + 3$
  \end{enumerate}

\item \textbf{[Q2]} Soit $f(x) = 2x^2 - 8x + 6$. Quelle est la forme canonique de $f$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $f(x) = 2(x - 2)^2 - 2$
    \item $f(x) = 2(x - 4)^2 - 10$
    \item $f(x) = 2(x - 2)^2 + 2$
    \item $f(x) = (x - 2)^2 - 2$
  \end{enumerate}

\item \textbf{[Q3]} On developpe $3(x-1)^2 + 5$. Quel resultat obtient-on ?
  \begin{enumerate}[label=\Alph*.]
    \item $3x^2 - 6x + 8$
    \item $3x^2 - 6x + 3$
    \item $3x^2 - 3x + 8$
    \item $3x^2 + 8$
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q4]} Soit $f(x) = -x^2 + 4x - 3$. Quelle est l'abscisse du sommet de la parabole representant $f$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $x = -2$
    \item $x = 2$
    \item $x = -4$
    \item $x = 4$
  \end{enumerate}

\item \textbf{[Q5]} Soit $f(x) = x^2 - 6x + 5$. Sur quel intervalle $f$ est-elle decroissante ?
  \begin{enumerate}[label=\Alph*.]
    \item $[3\,;\,+\infty[$
    \item $]-\infty\,;\,3]$
    \item $]-\infty\,;\,5]$
    \item $[5\,;\,+\infty[$
  \end{enumerate}

\item \textbf{[Q6]} Parmi les assertions suivantes concernant $f(x) = 2x^2 - 4x + 7$, laquelle est vraie ?
  \begin{enumerate}[label=\Alph*.]
    \item Le minimum de $f$ vaut $7$.
    \item La parabole est tournee vers le bas (coefficient directeur negatif).
    \item Le minimum de $f$ vaut $5$.
    \item La fonction $f$ n'admet pas de minimum.
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q7]} On considere $f(x) = x^2 - 5x + 6$. Quelle est la valeur du discriminant $\Delta$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\Delta = 1$
    \item $\Delta = 49$
    \item $\Delta = -11$
    \item $\Delta = 25$
  \end{enumerate}

\item \textbf{[Q8]} L'equation $2x^2 - 4x + 2 = 0$ admet :
  \begin{enumerate}[label=\Alph*.]
    \item Deux racines reelles distinctes.
    \item Une racine double, egale a $1$.
    \item Aucune racine reelle.
    \item Une racine double, egale a $2$.
  \end{enumerate}

\item \textbf{[Q9]} Resoudre $x^2 - 3x - 4 = 0$. Les solutions sont :
  \begin{enumerate}[label=\Alph*.]
    \item $x_1 = -1$ et $x_2 = 4$
    \item $x_1 = 1$ et $x_2 = -4$
    \item $x_1 = -1$ et $x_2 = -4$
    \item $x_1 = 1$ et $x_2 = 4$
  \end{enumerate}

\item[] \textbf{Capacite C4}

\item \textbf{[Q10]} Soit $f(x) = 2x^2 - 8x + 6$. Les racines de $f$ sont $x_1 = 1$ et $x_2 = 3$. Quelle est la forme factorisee de $f(x)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $f(x) = (x - 1)(x - 3)$
    \item $f(x) = 2(x - 1)(x - 3)$
    \item $f(x) = 2(x + 1)(x + 3)$
    \item $f(x) = (2x - 1)(2x - 3)$
  \end{enumerate}

\item \textbf{[Q11]} On sait que $f(x) = (x-2)(x-5)$. Sur quel intervalle $f(x)$ est-il negatif ou nul ?
  \begin{enumerate}[label=\Alph*.]
    \item $]-\infty\,;\,2] \cup [5\,;\,+\infty[$
    \item $[2\,;\,5]$
    \item $]-\infty\,;\,2]$
    \item $[5\,;\,+\infty[$
  \end{enumerate}

\item \textbf{[Q12]} Le trinome $f(x) = x^2 + 2x + 5$ de discriminant $\Delta = -16 < 0$ admet quel signe ?
  \begin{enumerate}[label=\Alph*.]
    \item Il est positif pour $x > 0$ et negatif pour $x < 0$.
    \item Il est negatif pour tout $x \in \mathbb{R}$.
    \item Il est positif pour tout $x \in \mathbb{R}$.
    \item Il change de signe en $x = -1$.
  \end{enumerate}

\item[] \textbf{Capacite C5}

\item \textbf{[Q13]} Resoudre $x^2 - x - 6 \leq 0$ (les racines de $x^2 - x - 6$ sont $-2$ et $3$).
  \begin{enumerate}[label=\Alph*.]
    \item $x \in [-2\,;\,3]$
    \item $x \in ]-\infty\,;\,-2] \cup [3\,;\,+\infty[$
    \item $x \in [-2\,;\,0]$
    \item $x \in [0\,;\,3]$
  \end{enumerate}

\item \textbf{[Q14]} L'inequation $-x^2 + 4x - 3 \geq 0$ a pour ensemble de solutions :
  \begin{enumerate}[label=\Alph*.]
    \item $]-\infty\,;\,1] \cup [3\,;\,+\infty[$
    \item $[1\,;\,3]$
    \item $]-\infty\,;\,-1] \cup [3\,;\,+\infty[$
    \item $[-1\,;\,3]$
  \end{enumerate}

\item \textbf{[Q15]} Pour quelle valeur de $k$ l'equation $x^2 - 2x + k = 0$ admet-elle exactement une solution reelle ?
  \begin{enumerate}[label=\Alph*.]
    \item $k = 0$
    \item $k = 2$
    \item $k = 1$
    \item $k = -1$
  \end{enumerate}

\item[] \textbf{Capacite C6}

\item \textbf{[Q16]} On decoupe des carres de cote $x$ aux coins d'une feuille $30 \times 20$ pour former une boite. Le volume est $V(x) = x(30-2x)(20-2x)$. Quelle est la valeur de $x$ pour $x = 4$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $V(4) = 1344$
    \item $V(4) = 1280$
    \item $V(4) = 2640$
    \item $V(4) = 1056$
  \end{enumerate}

\item \textbf{[Q17]} On cherche a maximiser $f(x) = -2x^2 + 8x + 3$ sur $\mathbb{R}$. Quelle est la valeur du maximum ?
  \begin{enumerate}[label=\Alph*.]
    \item $f_{\max} = 3$
    \item $f_{\max} = 11$
    \item $f_{\max} = 8$
    \item $f_{\max} = 2$
  \end{enumerate}

\item \textbf{[Q18]} Un balle est lancee verticalement. Sa hauteur est $h(t) = -5t^2 + 20t$ (en metres, $t$ en secondes). A quel instant atteint-elle sa hauteur maximale ?
  \begin{enumerate}[label=\Alph*.]
    \item $t = 4$ s
    \item $t = 1$ s
    \item $t = 2$ s
    \item $t = 20$ s
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des reponses erronees}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{A} — L'eleve a confondu polynome du premier degre et du second degre. $3x+7$ est de degre 1 car le coefficient de $x^2$ est nul ($a = 0$). \emph{Renvoi : C1, definition.}
    \item \textbf{C} — L'eleve a retenu $x^3 - 2x + 4$ comme polynome du second degre alors que le terme dominant est $x^3$ : c'est un polynome du troisieme degre. \emph{Renvoi : C1, definition.}
    \item \textbf{D} — L'eleve a confondu $\frac{1}{x^2} = x^{-2}$ (puissance negative) avec $x^2$ (second degre). Un polynome n'admet que des puissances entieres positives. \emph{Renvoi : C1, definition.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{B} — L'eleve a calcule $\alpha = -b/(2a) = 8/2 = 4$ au lieu de $\alpha = 8/4 = 2$ (oubli du facteur $2a$ au denominateur, pas seulement $2$). \emph{Renvoi : C1, forme canonique.}
    \item \textbf{C} — L'eleve a trouve $\alpha = 2$ correctement mais a calcule $\beta = f(2) = 2(4) - 8(2) + 6 = 8 - 16 + 6 = -2$ et a ecrit $+2$ au lieu de $-2$. Erreur de signe sur $\beta$. \emph{Renvoi : C1, forme canonique.}
    \item \textbf{D} — L'eleve a oublie le facteur $a = 2$ devant la forme canonique : $f(x) = a(x-\alpha)^2 + \beta$ avec $a = 2$. \emph{Renvoi : C1, forme canonique.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{B} — L'eleve a developpe $3(x-1)^2 = 3x^2 - 6x + 3$ mais a oublie d'ajouter le $+5$ : le terme constant est $3 + 5 = 8$, pas $3$. \emph{Renvoi : R1, identites remarquables.}
    \item \textbf{C} — L'eleve a calcule $3(x-1)^2 = 3(x^2 - x + 1)$ au lieu de $3(x^2 - 2x + 1)$ : erreur sur le terme en $x$ de l'identite $(x-1)^2 = x^2 - 2x + 1$ (oubli du coefficient 2). \emph{Renvoi : R1, identites remarquables.}
    \item \textbf{D} — L'eleve a confondu $(x-1)^2$ avec $x^2 - 1$ (difference de carres au lieu du carre d'un binome) : $3(x^2 - 1) + 5 = 3x^2 + 2$. Double erreur de formule. \emph{Renvoi : R1, identites remarquables.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — L'eleve a applique $\alpha = -b/(2a)$ mais a oublie de tenir compte du signe de $a = -1$ : $\alpha = -4/(2 \times (-1)) = -4/(-2) = 2$, et non $-2$. Erreur de signe sur $a$. \emph{Renvoi : C2, sommet.}
    \item \textbf{C} — L'eleve a calcule $\alpha = b/(2a) = 4/(2 \times (-1)) = -2$, confondant la formule $\alpha = -b/(2a)$ avec $b/(2a)$ (oubli du signe moins devant $b$). \emph{Renvoi : C2, sommet.}
    \item \textbf{D} — L'eleve a utilise directement $b = 4$ comme abscisse du sommet, sans appliquer la formule $\alpha = -b/(2a)$. \emph{Renvoi : C2, sommet.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — L'eleve a confondu l'intervalle de decroissance avec l'intervalle de croissance. Comme $a = 1 > 0$, la parabole est en forme de U : $f$ est decroissante sur $]-\infty\,;\,\alpha]$ et croissante sur $[\alpha\,;\,+\infty[$. \emph{Renvoi : C2, tableau de variations.}
    \item \textbf{C} — L'eleve a confondu l'ordonnee $f(\alpha) = -4$ ou la valeur $c = 5$ avec l'abscisse du sommet $\alpha = 3$. \emph{Renvoi : C2, sommet.}
    \item \textbf{D} — L'eleve a utilise $c = 5$ comme abscisse de changement de monotonie au lieu de $\alpha = -b/(2a) = 6/2 = 3$. \emph{Renvoi : C2, sommet.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{A} — L'eleve a confondu le minimum $f(\alpha)$ avec le terme constant $c = 7$. Le minimum est $f(1) = 2 - 4 + 7 = 5$, pas $7$. \emph{Renvoi : C2, sommet.}
    \item \textbf{B} — L'eleve a dit que $a = 2 < 0$ alors que $a = 2 > 0$. La parabole est tournee vers le haut (U) et $f$ admet un minimum. \emph{Renvoi : C2, orientation de la parabole.}
    \item \textbf{D} — L'eleve a conclu que $f$ n'admet pas de minimum, peut-etre en confondant avec le cas $a < 0$ (maximum) ou en ne reconnaissant pas que $a > 0$ implique un minimum. \emph{Renvoi : C2, tableau de variations.}
  \end{itemize}
  \item \textbf{Q7}
  \begin{itemize}
    \item \textbf{B} — L'eleve a calcule $\Delta = b^2 + 4ac = 25 + 24 = 49$ au lieu de $\Delta = b^2 - 4ac = 25 - 24 = 1$ : erreur de signe dans la formule (addition au lieu de soustraction). \emph{Renvoi : C3, discriminant.}
    \item \textbf{C} — L'eleve a utilise $\Delta = -b^2 + 4ac = -25 + 24 = -1$... ou a calcule $\Delta = b^2 - 4ac$ avec $b = -5$ et $c = 1$, $a = 6$ (permutation des coefficients). Erreur d'identification de $a$, $b$, $c$. \emph{Renvoi : C3, discriminant.}
    \item \textbf{D} — L'eleve a calcule $\Delta = b^2 = (-5)^2 = 25$, oubliant le terme $-4ac = -4 \times 1 \times 6 = -24$ dans la formule. \emph{Renvoi : C3, discriminant.}
  \end{itemize}
  \item \textbf{Q8}
  \begin{itemize}
    \item \textbf{A} — L'eleve a conclu $\Delta > 0$ sans calculer : $\Delta = (-4)^2 - 4 \times 2 \times 2 = 16 - 16 = 0$. Un discriminant nul donne une racine double, pas deux racines distinctes. \emph{Renvoi : C3, cas du discriminant.}
    \item \textbf{C} — L'eleve a conclu $\Delta < 0$ par erreur. Le calcul donne $\Delta = 16 - 16 = 0 \geq 0$, donc il existe au moins une solution reelle. \emph{Renvoi : C3, cas du discriminant.}
    \item \textbf{D} — L'eleve a calcule la racine double $x_0 = -b/(2a) = 4/4 = 1$, mais a ecrit $2$ au lieu de $1$. Erreur arithmetique dans le calcul de $x_0$. \emph{Renvoi : C3, racine double.}
  \end{itemize}
  \item \textbf{Q9}
  \begin{itemize}
    \item \textbf{B} — L'eleve a applique les formules avec un signe erronne : $x = (-b \pm \sqrt{\Delta})/(2a)$ avec $b = 3$ mais a oublie que $b = -3$. Les formules donnent $x = (3 \pm 5)/2$, soit $x_1 = -1$ et $x_2 = 4$. \emph{Renvoi : C3, formule des racines.}
    \item \textbf{C} — L'eleve a trouve les bonnes valeurs en module mais a attribue les mauvais signes aux deux racines. $\Delta = 9 + 16 = 25$, $x_1 = (3-5)/2 = -1$ et $x_2 = (3+5)/2 = 4$. \emph{Renvoi : C3, formule des racines.}
    \item \textbf{D} — L'eleve a peut-etre factorise par inspection en cherchant deux entiers de somme $-3$ et produit $-4$ : il faut $(-1) \times 4 = -4$ et $(-1) + 4 = 3 
eq -3$. Confusion des signes. \emph{Renvoi : C3, formule des racines.}
  \end{itemize}
  \item \textbf{Q10}
  \begin{itemize}
    \item \textbf{A} — L'eleve a oublie le facteur $a = 2$ dans la forme factorisee $f(x) = a(x - x_1)(x - x_2)$. La formule impose de conserver le coefficient dominant. \emph{Renvoi : C4, factorisation.}
    \item \textbf{C} — L'eleve a utilise $+x_1$ et $+x_2$ au lieu de $-x_1$ et $-x_2$ : la forme factorisee est $a(x - x_1)(x - x_2)$, pas $a(x + x_1)(x + x_2)$. \emph{Renvoi : C4, factorisation.}
    \item \textbf{D} — L'eleve a reparti le coefficient $a = 2$ sur chacun des deux facteurs : $(2x - x_1)(2x - x_2)$ n'est pas la forme correcte, car $2 \times 2 = 4 
eq a = 2$. \emph{Renvoi : C4, factorisation.}
  \end{itemize}
  \item \textbf{Q11}
  \begin{itemize}
    \item \textbf{A} — L'eleve a confondu l'intervalle ou $f$ est negatif avec l'intervalle ou $f$ est positif. Le produit $(x-2)(x-5)$ est negatif entre les deux racines ($a > 0$ : parabole en U). \emph{Renvoi : C4, tableau de signes.}
    \item \textbf{C} — L'eleve n'a pris en compte que l'une des deux racines : $f(x) \leq 0 \Leftrightarrow 2 \leq x \leq 5$, pas seulement $x \leq 2$. \emph{Renvoi : C4, tableau de signes.}
    \item \textbf{D} — L'eleve a identifie le mauvais intervalle : $f(x) \geq 0$ pour $x \geq 5$, mais on cherche $f(x) \leq 0$. \emph{Renvoi : C4, tableau de signes.}
  \end{itemize}
  \item \textbf{Q12}
  \begin{itemize}
    \item \textbf{A} — L'eleve a applique le signe de $c = 5 > 0$ uniquement sur $]0\,;\,+\infty[$. Quand $\Delta < 0$ et $a > 0$, le trinome est positif sur $\mathbb{R}$ tout entier. \emph{Renvoi : C4, signe quand Delta < 0.}
    \item \textbf{B} — L'eleve a confondu $\Delta < 0$ avec «le trinome est negatif». La regle est : si $\Delta < 0$ et $a > 0$, le trinome est strictement positif partout. \emph{Renvoi : C4, signe quand Delta < 0.}
    \item \textbf{D} — L'eleve a calcule le sommet $\alpha = -b/(2a) = -1$ et a cru que c'etait une racine (changement de signe). Quand $\Delta < 0$, il n'y a pas de racine reelle et pas de changement de signe. \emph{Renvoi : C4, signe quand Delta < 0.}

  \begin{itemize}
    \item \textbf{B} — L'eleve a ecrit l'ensemble de solutions de l'inequation $\geq 0$ au lieu de $\leq 0$. Le trinome $x^2 - x - 6$ est negatif entre ses racines ($a = 1 > 0$, parabole en U). \emph{Renvoi : C5, resolution inequation.}
    \item \textbf{C} — L'eleve n'a pris qu'une moitie de l'intervalle solution, en arrondissant a $[x_1\,;\,0]$ au lieu de $[x_1\,;\,x_2]$. \emph{Renvoi : C5, resolution inequation.}
    \item \textbf{D} — L'eleve a pris $[0\,;\,x_2]$ au lieu de $[x_1\,;\,x_2] = [-2\,;\,3]$. Il a omis la racine $x_1 = -2$. \emph{Renvoi : C5, resolution inequation.}
  \end{itemize}

  \begin{itemize}
    \item \textbf{A} — L'eleve a confondu les intervalles de signe positif et negatif pour $a < 0$. Quand $a = -1 < 0$, la parabole est en $\cap$ : le trinome est positif entre ses racines. \emph{Renvoi : C5, cas a < 0.}
    \item \textbf{C} — L'eleve a utilise $-x_1 = -1$ au lieu de $x_1 = 1$ (oubli du signe des racines). Les racines de $-x^2+4x-3 = 0$ sont $x = 1$ et $x = 3$. \emph{Renvoi : C5, calcul des racines.}

    \item \textbf{D} — L'eleve a calcule $V(4) = 4 \times (30 - 4) \times (20 - 4) = 4 \times 26 \times ... $  ou $4 \times 22 \times 12 = 1056$ (confusion $30 - x$ et $30 - 2x$). \emph{Renvoi : C6, mise en equation.}

  \begin{itemize}

  \end{itemize}
