\begin{exercice}{1SPE-SECDEG-EX-035}{3}{35}
On considère la fonction $f$ définie sur $\mathbb{R}$ par :
\[
  f(x) = x^2 - 4x + 3.
\]

\begin{enumerate}
  \item \textbf{Formes du trinôme.}
  \begin{enumerate}
    \item Identifier les coefficients $a$, $b$, $c$ de $f$.
    \item Mettre $f$ sous forme canonique $f(x) = a(x - \alpha)^2 + \beta$. Préciser les coordonnées du sommet $S$ de la parabole.
    \item Calculer le discriminant $\Delta$ de $f$, puis déterminer les racines $x_1$ et $x_2$ de $f$.
    \item Écrire la forme factorisée de $f$.
  \end{enumerate}

  \item \textbf{Représentation graphique.}
  \begin{enumerate}
    \item La parabole est-elle tournée vers le haut ou vers le bas ? Justifier à partir de la valeur de $a$.
    \item Tracer l'allure de la parabole représentant $f$ en plaçant le sommet, les racines et l'ordonnée à l'origine.
    \item Déterminer l'axe de symétrie de la parabole et vérifier que $x_1$ et $x_2$ sont symétriques par rapport à cet axe.
  \end{enumerate}

  \item \textbf{Tableau de signes et position relative.}
  \begin{enumerate}
    \item Dresser le tableau de signes de $f(x)$ sur $\mathbb{R}$.
    \item Résoudre l'inéquation $f(x) < 0$.
    \item Déterminer les valeurs de $x$ pour lesquelles la courbe représentative de $f$ est au-dessous de l'axe des abscisses.
  \end{enumerate}

  \item \textbf{Question ouverte.} On considère la droite $d$ d'équation $y = x - 1$.
  \begin{enumerate}
    \item Montrer que les points d'intersection de la parabole et de la droite $d$ peuvent être obtenus en résolvant $x^2 - 5x + 4 = 0$.
    \item Résoudre cette équation et en déduire les coordonnées des points d'intersection.
    \item La droite $d$ est-elle sécante, tangente ou sans intersection avec la parabole ? Justifier.
  \end{enumerate}
\end{enumerate}
\end{exercice}
