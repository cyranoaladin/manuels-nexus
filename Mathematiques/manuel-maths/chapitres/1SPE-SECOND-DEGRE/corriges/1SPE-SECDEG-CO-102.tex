\corrige{1SPE-SECDEG-EX-102}

\begin{enumerate}
  \item[\textbf{a.}] $7 - 10 + 3 = 0$ : \textbf{racine évidente $1$}. L'autre
        racine vaut $\dfrac{c}{a} = \dfrac{3}{7}$. Donc
        \[
          f(x) = 7(x - 1)\left(x - \tfrac{3}{7}\right) = (x - 1)(7x - 3).
        \]
        Aucun discriminant n'a été nécessaire.

  \item[\textbf{b.}] $4 - 9 + 5 = 0$ : \textbf{racine évidente $-1$}. L'autre
        racine vaut $-\dfrac{c}{a} = -\dfrac{5}{4}$. Donc
        \[
          g(x) = 4(x + 1)\left(x + \tfrac{5}{4}\right) = (x + 1)(4x + 5).
        \]

  \item[\textbf{c.}] Ici $a = 1$ : la somme des racines vaut $9$ et leur produit
        $20$. Deux entiers de somme $9$ et de produit $20$ : $4$ et $5$.
        \textbf{Stratégie somme–produit}, sans calcul de discriminant :
        \[
          h(x) = (x - 4)(x - 5).
        \]

  \item[\textbf{d.}] Les coefficients $9$, $30$ et $25$ font reconnaître une
        \textbf{identité remarquable} : $9x^2 - 30x + 25 = (3x - 5)^2$. La
        factorisation est immédiate.

  \item[\textbf{e.}] C'est $k$. Son discriminant vaut
        $(-30)^2 - 4 \times 9 \times 25 = 900 - 900 = 0$ : elle possède une
        \textbf{racine double} $\dfrac{5}{3}$, et non deux racines distinctes.

        La propriété reste vraie à condition de compter la racine double deux
        fois : $\dfrac{5}{3} + \dfrac{5}{3} = \dfrac{10}{3} = -\dfrac{b}{a}$ et
        $\dfrac{5}{3} \times \dfrac{5}{3} = \dfrac{25}{9} = \dfrac{c}{a}$. En
        revanche, la recherche de \emph{deux nombres distincts} de somme et de
        produit donnés n'a plus de solution : c'est exactement le cas
        $s^2 - 4p = 0$.
\end{enumerate}
