\corrige{1SPE-SECDEG-EX-101}

\begin{enumerate}
  \item Ici $a = 5$, $b = -20$, $c = 15$. Donc
        \[
          x_1 + x_2 = -\dfrac{b}{a} = \dfrac{20}{5} = 4,
          \qquad
          x_1 x_2 = \dfrac{c}{a} = \dfrac{15}{5} = 3 .
        \]
        On cherche deux nombres de somme $4$ et de produit $3$ : ce sont $1$ et
        $3$. Vérification : $5(x-1)(x-3) = 5x^2 - 20x + 15 = f(x)$.

        \emph{Contrôle rapide} : $5 - 20 + 15 = 0$, donc $1$ était bien racine
        évidente.

  \item Deux racines distinctes imposent la forme factorisée, pas le
        coefficient dominant. Les fonctions cherchées sont donc toutes les
        \[
          f : x \longmapsto a(x + 4)(x - 1), \qquad a \in \mathbb{R}^{*}.
        \]

  \item On impose $f(2) = 12$ :
        \[
          a(2 + 4)(2 - 1) = 6a = 12 \quad\Longrightarrow\quad a = 2 .
        \]
        D'où $f(x) = 2(x+4)(x-1)$, soit sous forme développée réduite
        $f(x) = 2x^2 + 6x - 8$. Vérification : $f(2) = 8 + 12 - 8 = 12$.

  \item Deux réels de somme $11$ et de produit $30$ sont les racines de
        $X^2 - 11X + 30$. On calcule $s^2 - 4p = 121 - 120 = 1 > 0$ : ils
        existent et sont distincts. Ce sont $5$ et $6$, puisque $5 + 6 = 11$ et
        $5 \times 6 = 30$.
\end{enumerate}
