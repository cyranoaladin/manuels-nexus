\begin{corrige}{1SPE-SECDEG-EX-044}

\textbf{1. Discriminant et racines.}

\commentaireMarge{$a = -2$, $b = 8$, $c = -6$.}

\[
\Delta = b^2 - 4ac = 8^2 - 4 \times (-2) \times (-6) = 64 - 48 = 16.
\]
$\Delta > 0$, donc $g$ admet deux racines réelles :
\[
x_1 = \frac{-8 - \sqrt{16}}{2 \times (-2)} = \frac{-12}{-4} = 3, \qquad x_2 = \frac{-8 + \sqrt{16}}{2 \times (-2)} = \frac{-4}{-4} = 1.
\]

\medskip
\textbf{2. Forme factorisée.}

\[
g(x) = -2(x - 1)(x - 3).
\]

\medskip
\textbf{3. Signe de $g(x)$.}

\commentaireMarge{$a = -2 < 0$ : le trinôme est négatif à l'extérieur des racines, positif entre elles.}

Comme $a = -2 < 0$, $g(x)$ est positif entre les racines et négatif à l'extérieur :
\[
g(0) = -6 < 0, \qquad g(2) = 2 > 0, \qquad g(4) = -6 < 0.
\]
Donc $g(x) \geqslant 0$ pour $x \in [1\,;\,3]$ et $g(x) \leqslant 0$ pour $x \in \,]-\infty\,;\,1] \cup [3\,;\,+\infty[$.

\end{corrige}
