\begin{fichemethode}{M7}{Retrouver un trinôme à partir de ses deux racines}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande de « déterminer les fonctions
  polynômes du second degré qui s'annulent en $x_1$ et $x_2$ », de « trouver un
  trinôme dont la somme des racines vaut $S$ et le produit $P$ », ou de
  « reconnaître deux nombres connaissant leur somme et leur produit ».
}

\pasApas{%
  \begin{enumerate}
    \item \textbf{Écrire la forme factorisée.} Un polynôme du second degré qui
      s'annule exactement en $x_1$ et $x_2$ s'écrit
      $f(x) = a(x - x_1)(x - x_2)$, avec $a \neq 0$. Il y en a donc une
      \textbf{infinité}, une par valeur de $a$ : l'énoncé demande la famille, pas
      un polynôme unique.
    \item \textbf{En déduire la somme et le produit.} En développant,
      $f(x) = a\left(x^2 - (x_1 + x_2)x + x_1x_2\right)$, donc
      \[
        S = x_1 + x_2 = -\dfrac{b}{a}
        \qquad\text{et}\qquad
        P = x_1 x_2 = \dfrac{c}{a}.
      \]
      Ces deux formules ne dépendent pas de $a$ : deux polynômes de la même
      famille ont les mêmes racines, donc la même somme et le même produit.
    \item \textbf{Dans l'autre sens.} Si l'énoncé donne $S$ et $P$, les deux
      nombres cherchés sont les racines de $x^2 - Sx + P = 0$. Je calcule
      $\Delta = S^2 - 4P$ : s'il est négatif, aucun couple de réels ne convient.
    \item \textbf{Contrôler.} Je vérifie sur les racines trouvées que leur somme
      vaut bien $S$ et leur produit bien $P$ — deux additions et une
      multiplication qui coûtent moins cher qu'une erreur de signe.
  \end{enumerate}
}

\exempleRedige{%
  Déterminer les fonctions polynômes du second degré qui s'annulent exactement en
  $-1$ et $4$, puis donner la somme et le produit de leurs racines.

  \medskip
  \commentaireMarge{Forme factorisée.}%
  Ce sont les fonctions $f(x) = a(x + 1)(x - 4)$ avec $a \neq 0$.

  \medskip
  \commentaireMarge{Somme et produit.}%
  $S = -1 + 4 = 3$ et $P = (-1) \times 4 = -4$.

  \medskip
  \commentaireMarge{Contrôle par le développement.}%
  $f(x) = a(x^2 - 3x - 4)$, donc $-\dfrac{b}{a} = 3$ et $\dfrac{c}{a} = -4$ :
  les deux lectures concordent, et aucune ne dépend de $a$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Oublier le paramètre $a$.} Écrire « le polynôme est
      $(x+1)(x-4)$ » revient à en choisir un au hasard dans une infinité. Tant
      que l'énoncé ne fixe pas une valeur — par un point de la courbe, par
      exemple — la réponse est une famille.
    \item \textbf{Se tromper de signe sur la somme.} La forme factorisée est
      $(x - x_1)$ : pour la racine $-1$, le facteur est $(x + 1)$. Lire les
      racines directement sur les facteurs, sans changer le signe, donne
      $S = -3$ au lieu de $3$.
    \item \textbf{Confondre $-\dfrac{b}{a}$ et $\dfrac{b}{a}$.} Le signe moins
      vient du développement ; il ne s'invente pas et ne s'oublie pas.
  \end{itemize}
}
\end{fichemethode}
