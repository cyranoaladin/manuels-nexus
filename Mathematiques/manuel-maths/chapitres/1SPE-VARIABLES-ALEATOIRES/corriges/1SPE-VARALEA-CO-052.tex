\textbf{1.} Les cumuls sont $0{,}5$, puis $0{,}5+0{,}3 = 0{,}8$, puis
$0{,}8+0{,}2 = 1$. Le dernier vaut bien $1$ : la loi est complète.

\textbf{2.} On parcourt les valeurs dans l'ordre et on renvoie la première dont
le cumul dépasse $u$ :

\begin{python}
VALEURS = [0, 3, 7]
CUMULS = [0.5, 0.8, 1.0]

def simuler_valeur(u):
    for i in range(len(VALEURS)):
        if u < CUMULS[i]:
            return VALEURS[i]
    return VALEURS[-1]
\end{python}

\textbf{4.} On répète $n$ tirages et on divise la somme par $n$ :

\begin{python}
import random

def moyenne_echantillon(n, graine):
    random.seed(graine)
    somme = 0
    for i in range(n):
        somme = somme + simuler_valeur(random.random())
    return somme / n
\end{python}
