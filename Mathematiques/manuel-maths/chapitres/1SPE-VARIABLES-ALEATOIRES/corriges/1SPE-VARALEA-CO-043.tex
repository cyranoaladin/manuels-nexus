\begin{corrige}{1SPE-VARALEA-EX-043}

\textbf{2.} $E(G) = (18-m) \times \frac{1}{6} + (-m) \times \frac{5}{6} = \frac{18-m-5m}{6} = \frac{18-6m}{6} = 3-m$.
