\begin{corrige}{1SPE-VARALEA-EX-033}

\commentaireMarge{$1-p = V(X)/E(X)$.}

$1-p = \frac{V(X)}{E(X)} = \frac{4{,}2}{6} = 0{,}7$, donc $p = 0{,}3$.

$n = \frac{E(X)}{p} = \frac{6}{0{,}3} = 20$.

\textbf{Vérification :} $E(X) = 20 \times 0{,}3 = 6$ \checkmark \quad $V(X) = 20 \times 0{,}3 \times 0{,}7 = 4{,}2$ \checkmark.

\end{corrige}
