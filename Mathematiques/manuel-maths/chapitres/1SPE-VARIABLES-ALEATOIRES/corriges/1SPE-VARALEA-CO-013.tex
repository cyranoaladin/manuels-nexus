\begin{corrige}{1SPE-VARALEA-EX-013}

\textbf{1.} $E(X) = (-2) \times \frac{1}{4} + 0 \times \frac{1}{2} + 3 \times \frac{1}{4} = -\frac{1}{2} + 0 + \frac{3}{4} = \frac{1}{4}$.

\medskip
\textbf{2.} $E(X^2) = 4 \times \frac{1}{4} + 0 \times \frac{1}{2} + 9 \times \frac{1}{4} = 1 + 0 + \frac{9}{4} = \frac{13}{4}$.

$V(X) = \frac{13}{4} - \left(\frac{1}{4}\right)^2 = \frac{13}{4} - \frac{1}{16} = \frac{52-1}{16} = \frac{51}{16}$.

\medskip
\textbf{3.} $\sigma(X) = \sqrt{\frac{51}{16}} = \frac{\sqrt{51}}{4} \approx 1{,}79$.

\end{corrige}
