\begin{corrige}{1SPE-VARALEA-EX-039}

\textbf{1.} $E(X)=0\times\frac35+1\times\frac25=\frac25$.

\textbf{2.} $E(X^2)=0^2\times\frac35+1^2\times\frac25=\frac25$, donc
$V(X)=E(X^2)-E(X)^2=\frac25-\frac4{25}=\frac6{25}$.

\end{corrige}
