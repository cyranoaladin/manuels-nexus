\begin{corrige}{1SPE-VARALEA-EX-036}

\textbf{1.} $X \sim \mathcal{B}(100, \frac{1}{2})$.

\textbf{2.} $E(X) = 100 \times \frac{1}{2} = 50$.

\textbf{3.} $V(X) = 100 \times \frac{1}{2} \times \frac{1}{2} = 25$, donc $\sigma(X) = 5$.
