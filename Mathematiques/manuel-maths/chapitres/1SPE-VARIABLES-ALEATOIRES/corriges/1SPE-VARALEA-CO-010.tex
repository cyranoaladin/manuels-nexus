\begin{corrige}{1SPE-VARALEA-EX-010}

\textbf{1.} Si le numéro sort : $X = 36 - 1 = 35$ avec $P = \frac{1}{37}$.

Si le numéro ne sort pas : $X = -1$ avec $P = \frac{36}{37}$.

\medskip
\textbf{2.}
\[
\begin{array}{|c|c|c|}
\hline
x_i & -1 & 35 \\
\hline
P(X=x_i) & \frac{36}{37} & \frac{1}{37} \\
\hline
\end{array}
\]

\end{corrige}
