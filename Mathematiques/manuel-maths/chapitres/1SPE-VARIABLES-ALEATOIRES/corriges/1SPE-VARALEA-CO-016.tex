\begin{corrige}{1SPE-VARALEA-EX-016}

\textbf{1.} $E(X) = 35 \times \frac{1}{37} + (-1) \times \frac{36}{37} = \frac{35-36}{37} = -\frac{1}{37} \approx -0{,}027$.

\medskip
\textbf{2.} $E(X) < 0$ : le jeu est défavorable au joueur, favorable à la banque.

\medskip
\textbf{3.} Sur $370$ parties : perte moyenne $= 370 \times \frac{1}{37} = 10$~euros.
