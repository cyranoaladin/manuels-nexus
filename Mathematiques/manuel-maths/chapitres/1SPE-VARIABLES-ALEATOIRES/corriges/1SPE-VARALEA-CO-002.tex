\begin{corrige}{1SPE-VARALEA-EX-002}

\textbf{1.} $\Omega = \{PP, PF, FP, FF\}$, chaque issue a probabilité $\frac{1}{4}$.

\medskip
\textbf{2.} $X(PP)=2$, $X(PF)=X(FP)=1$, $X(FF)=0$.
\[
\begin{array}{|c|c|c|c|}
\hline
x_i & 0 & 1 & 2 \\
\hline
P(X=x_i) & \frac{1}{4} & \frac{1}{2} & \frac{1}{4} \\
\hline
\end{array}
\]

\medskip
\textbf{3.} $P(X \geq 1) = 1 - P(X=0) = 1 - \frac{1}{4} = \frac{3}{4}$.

\end{corrige}
