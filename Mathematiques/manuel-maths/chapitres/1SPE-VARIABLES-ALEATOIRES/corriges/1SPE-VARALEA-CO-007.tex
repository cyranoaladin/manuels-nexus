\begin{corrige}{1SPE-VARALEA-EX-007}

\textbf{1.} $X \in \{0, 1, 2\}$.

\medskip
\textbf{2.} L'arbre donne
$P(X=0)=\frac25\times\frac14=\frac1{10}$,
$P(X=2)=\frac35\times\frac24=\frac3{10}$ et
\[
P(X=1)=\frac35\times\frac24+\frac25\times\frac34
=\frac3{10}+\frac3{10}=\frac35.
\]

\medskip
\textbf{3.}
\[
\begin{array}{|c|c|c|c|}
\hline
x_i & 0 & 1 & 2 \\
\hline
P(X=x_i) & \frac{1}{10} & \frac{3}{5} & \frac{3}{10} \\
\hline
\end{array}
\]
$\frac{1}{10}+\frac{6}{10}+\frac{3}{10}=1$. \checkmark

\end{corrige}
