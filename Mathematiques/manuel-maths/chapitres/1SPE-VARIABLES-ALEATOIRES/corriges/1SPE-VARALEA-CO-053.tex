\begin{corrige}{1SPE-VARALEA-EX-053}

\textbf{1.} $\sqrt{25} = 5$, donc
\[
\frac{2\sigma}{\sqrt{25}} = \frac{2 \times 2{,}1}{5} = \frac{4{,}2}{5} = 0{,}84.
\]

\medskip
\textbf{2.} $|m - \mu| = |2{,}5 - 1{,}3| = 1{,}2$.

Or $1{,}2 > 0{,}84$ : l'écart dépasse le seuil. Cet échantillon est atypique,
sans être impossible : environ un échantillon sur vingt sort de cet intervalle.

\medskip
\textbf{3.} $\sqrt{100} = 10$ et $\sqrt{400} = 20$, donc
\[
\frac{2\sigma}{\sqrt{100}} = \frac{4{,}2}{10} = 0{,}42,
\qquad
\frac{2\sigma}{\sqrt{400}} = \frac{4{,}2}{20} = 0{,}21.
\]

\medskip
\textbf{4.} De $n = 25$ à $n = 100$, la quantité passe de $0{,}84$ à $0{,}42$ ;
de $n = 100$ à $n = 400$, de $0{,}42$ à $0{,}21$. Multiplier $n$ par $4$
multiplie $\sqrt{n}$ par $2$, donc \textbf{divise la quantité par $2$}.

\end{corrige}
