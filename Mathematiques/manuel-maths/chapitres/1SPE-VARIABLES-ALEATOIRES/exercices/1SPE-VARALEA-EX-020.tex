\begin{exercice}{1SPE-VARALEA-EX-020}{3}{20}
Soit $X$ une variable aléatoire telle que $E(X) = 3$. On pose $Y = 2X + 1$.

\begin{enumerate}
  \item Calculer $E(Y)$.
  \item Si $Z = -X + 5$, calculer $E(Z)$.
  \item Si $T=5X-4$, calculer $E(T)$ et interpréter le rôle du terme constant.
\end{enumerate}
\end{exercice}
