\begin{exercice}{1SPE-VARALEA-EX-027}{2}{15}

\begin{enumerate}
  \item Calculer $P(X = 0)$, $P(X = 1)$, $P(X = 2)$, $P(X = 3)$.
  \item Calculer $P(X \leq 2)$.
  \item Calculer $P(X \geq 3)$.
\end{enumerate}
