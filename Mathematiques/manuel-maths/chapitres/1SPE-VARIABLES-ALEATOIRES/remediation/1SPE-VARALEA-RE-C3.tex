\begin{exercice}{1SPE-VARALEA-RE-C3-EX1}{1}{10}
On joue $3$ fois à pile ou face avec une pièce équilibrée, de façon indépendante. Soit $X$ le nombre de Pile.
\begin{enumerate}
  \item Construire l'arbre des trois lancers.
  \item En regroupant les chemins selon le nombre de Pile, calculer $P(X=0)$, $P(X=1)$, $P(X=2)$ et $P(X=3)$.
  \item Vérifier que $\sum P(X=k) = 1$.
\end{enumerate}
\end{exercice}

\begin{corrige}{1SPE-VARALEA-RE-C3-EX1}
\textbf{1.} L'arbre a trois niveaux et huit chemins, chacun de probabilité $\left(\frac12\right)^3=\frac18$.
\textbf{2.} Un chemin comporte zéro Pile, trois en comportent un, trois en comportent deux et un en comporte trois. Ainsi $P(X=0)=\frac18$, $P(X=1)=\frac38$, $P(X=2)=\frac38$ et $P(X=3)=\frac18$.
\textbf{3.} $\frac{1+3+3+1}{8} = 1$. \checkmark
\end{corrige}
