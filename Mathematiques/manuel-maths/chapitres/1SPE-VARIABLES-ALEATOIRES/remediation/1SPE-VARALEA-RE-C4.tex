\begin{exercice}{1SPE-VARALEA-RE-C4-EX1}{1}{10}
Une variable aléatoire $X$ vérifie $E(X)=3$. On pose $Y=2X-5$.
\begin{enumerate}
  \item Calculer $E(Y)$.
  \item On pose $Z=-X+4$. Calculer $E(Z)$.
  \item On pose $T=5-3X$. Calculer $E(T)$.
\end{enumerate}
\end{exercice}

% BEGIN-VERIFY
% E = 3
% assert 2*E-5 == 1
% assert -E+4 == 1
% assert 5-3*E == -4
% END-VERIFY

\begin{corrige}{1SPE-VARALEA-RE-C4-EX1}
\textbf{1.} $E(Y)=2E(X)-5=2\times3-5=1$.
\textbf{2.} $E(Z)=-E(X)+4=-3+4=1$.
\textbf{3.} $E(T)=5-3E(X)=5-3\times3=-4$.
\end{corrige}
