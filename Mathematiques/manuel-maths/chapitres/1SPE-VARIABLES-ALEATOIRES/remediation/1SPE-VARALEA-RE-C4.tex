\begin{exercice}{1SPE-VARALEA-RE-C4-EX1}{1}{10}
Une variable aléatoire $X$ vérifie $E(X)=3$ et $V(X)=\frac94$. On pose $Y=2X-5$.
\begin{enumerate}
  \item Calculer $E(Y)$.
  \item Calculer $V(Y)$ et $\sigma(Y)$.
  \item On pose $Z=-X+4$. Calculer $E(Z)$ et $V(Z)$.
\end{enumerate}
\end{exercice}

\begin{corrige}{1SPE-VARALEA-RE-C4-EX1}
\textbf{1.} $E(Y)=2E(X)-5=2\times3-5=1$.
\textbf{2.} $V(Y)=2^2V(X)=4\times\frac94=9$, donc $\sigma(Y)=3$.
\textbf{3.} $E(Z)=-E(X)+4=1$ et $V(Z)=(-1)^2V(X)=\frac94$.
\end{corrige}
