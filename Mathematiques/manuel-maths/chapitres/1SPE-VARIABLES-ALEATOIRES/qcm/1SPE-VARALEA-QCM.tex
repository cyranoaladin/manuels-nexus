\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} On lance un de équilibré. Soit $X$ le résultat. Quelle est $P(X = 3)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{1}{6}$
    \item $\frac{1}{3}$
    \item $\frac{3}{6}$
    \item $\frac{1}{5}$
  \end{enumerate}

\item \textbf{[Q2]} La somme des probabilités d'un tableau de loi vaut toujours :
  \begin{enumerate}[label=\Alph*.]
    \item $0$
    \item $1$
    \item $\frac{1}{2}$
    \item Le nombre de valeurs prises par $X$
  \end{enumerate}

\item \textbf{[Q3]} On lance deux pieces. Soit $X$ le nombre de Pile. Quelle est $P(X = 1)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{1}{4}$
    \item $\frac{1}{3}$
    \item $\frac{1}{2}$
    \item $\frac{3}{4}$
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q4]} On lance un de équilibré. Quelle est l'espérance $E(X)$ du résultat ?
  \begin{enumerate}[label=\Alph*.]
    \item $3$
    \item $6$
    \item $4$
    \item $3{,}5$
  \end{enumerate}

\item \textbf{[Q5]} Si $V(X) = 9$, alors $\sigma(X)$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $3$
    \item $9$
    \item $81$
    \item $4{,}5$
  \end{enumerate}

\item \textbf{[Q6]} Si $E(X) = 4$ et $V(X) = 5$, quelle est $E(X^2)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $9$
    \item $21$
    \item $16$
    \item $20$
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q7]} On répète trois fois, de facon indépendante, une épreuve dont le succès a probabilité $\frac{1}{2}$. Quelle est la probabilité d'obtenir trois succès ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{1}{6}$
    \item $\frac{3}{8}$
    \item $\frac{1}{8}$
    \item $\frac{1}{2}$
  \end{enumerate}

\item \textbf{[Q8]} On répète deux fois, de facon indépendante, une épreuve dont le succès a probabilité $\frac{1}{3}$. Quelle est la probabilité d'obtenir exactement un succès ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{2}{9}$
    \item $\frac{2}{3}$
    \item $\frac{1}{9}$
    \item $\frac{4}{9}$
  \end{enumerate}

\item \textbf{[Q9]} Une épreuve a deux issues. On la répète trois fois. Combien l'arbre complet possede-t-il de chemins terminaux ?
  \begin{enumerate}[label=\Alph*.]
    \item $8$
    \item $3$
    \item $6$
    \item $9$
  \end{enumerate}

\item[] \textbf{Capacite C4}

\item \textbf{[Q10]} On sait que $E(X)=4$ et on pose $Y=3X-2$. Quelle est $E(Y)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $12$
    \item $10$
    \item $6$
    \item $2$
  \end{enumerate}

\item \textbf{[Q11]} On sait que $E(X)=5$ et on pose $Y=-2X+7$. Quelle est $E(Y)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $-10$
    \item $17$
    \item $-3$
    \item $12$
  \end{enumerate}

\item \textbf{[Q12]} On sait que $E(X)=-3$ et on pose $Y=-4X+1$. Quelle est $E(Y)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $-13$
    \item $-2$
    \item $-11$
    \item $13$
  \end{enumerate}

\item[] \textbf{Capacite C5}

\item \textbf{[Q13]} Un joueur paie 3 euros et recoit 15 euros s'il obtient un 6 au de. Gain algebrique si 6 ?
  \begin{enumerate}[label=\Alph*.]
    \item $12$ euros
    \item $15$ euros
    \item $3$ euros
    \item $18$ euros
  \end{enumerate}

\item \textbf{[Q14]} Un jeu a un gain moyen $E(G) = -0{,}5$ euro. Le jeu est :
  \begin{enumerate}[label=\Alph*.]
    \item Équitable
    \item Défavorable au joueur
    \item Favorable au joueur
    \item Impossible a déterminer
  \end{enumerate}

\item \textbf{[Q15]} Prime $c=60$ euros, sinistre $1000$ euros, probabilité $1/25$. Benefice moyen ?
  \begin{enumerate}[label=\Alph*.]
    \item $60$ euros
    \item $-40$ euros
    \item $20$ euros
    \item $40$ euros
  \end{enumerate}

\item[] \textbf{Capacite C6}

\item \textbf{[Q16]} Dans la fonction \code{simuler\_variable(n, graine)} du chapitre, que designe $n$ ?
  \begin{enumerate}[label=\Alph*.]
    \item Le nombre de valeurs prises par $X$
    \item La graine du generateur
    \item Le nombre d'échantillons simules
    \item La taille de l'échantillon simule
  \end{enumerate}

\item \textbf{[Q17]} Une fonction renvoie la moyenne d'un échantillon de taille $n$ stocke dans \code{somme}. Quelle ligne faut-il ecrire ?
  \begin{enumerate}[label=\Alph*.]
    \item \code{return somme / n}
    \item \code{return somme}
    \item \code{return somme / len(VALEURS)}
    \item \code{return n / somme}
  \end{enumerate}

\item \textbf{[Q18]} La loi vaut $P(X=-2)=0{,}2$, $P(X=1)=0{,}5$, $P(X=4)=0{,}3$ et on partage $[0\,;1[$ dans cet ordre. Quelle valeur la simulation renvoie-t-elle pour $u=0{,}55$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $-2$
    \item $1$
    \item $4$
    \item $0{,}55$
  \end{enumerate}

\item[] \textbf{Capacite C7}

\item \textbf{[Q19]} Pour la variable du chapitre, $\mu=1{,}3$ et $\sigma=2{,}1$. Pour un échantillon de taille $n=100$, que vaut $\dfrac{2\sigma}{\sqrt{n}}$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $0{,}21$
    \item $4{,}2$
    \item $0{,}42$
    \item $2{,}1$
  \end{enumerate}

\item \textbf{[Q20]} On simule des échantillons de taille $n$ de plus en plus grande. L'écart typique entre la moyenne $m$ d'un échantillon et l'espérance $\mu$ :
  \begin{enumerate}[label=\Alph*.]
    \item augmente
    \item devient nul des que $n>30$
    \item reste constant
    \item diminue
  \end{enumerate}

\item \textbf{[Q21]} On simule $N$ échantillons de taille $n$ et on compte ceux pour lesquels $|m-\mu|\leq \dfrac{2\sigma}{\sqrt{n}}$. La proportion observee est proche de :
  \begin{enumerate}[label=\Alph*.]
    \item $95\,\%$
    \item $50\,\%$
    \item $100\,\%$
    \item $5\,\%$
  \end{enumerate}

\end{enumerate}

    \item \textbf{B} — L'élève divise 1 par la valeur de X au lieu du nombre de faces. \emph{Renvoi : C1, definition.}
    \item \textbf{C} — L'élève confond P(X=3) et P(X<=3). \emph{Renvoi : C1, loi de probabilité.}
    \item \textbf{D} — L'élève divise par le nombre de faces restantes (5) au lieu du nombre total de faces (6). \emph{Renvoi : C1, denombrement des issues possibles.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — L'élève retient seulement la borne inferieure 0 de chaque probabilité au lieu de sommer toutes les probabilités de la loi. \emph{Renvoi : C1, propriete.}
    \item \textbf{C} — L'élève suppose a tort que la somme vaut 1/2 parce qu'il imagine deux issues equiprobables, alors que la somme d'une loi complete vaut 1. \emph{Renvoi : C1, propriete.}
    \item \textbf{D} — Confusion entre somme des probabilités et cardinal de X(Omega). \emph{Renvoi : C1, propriete.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — L'élève ne compte qu'une seule issue (PF ou FP) au lieu des deux. \emph{Renvoi : C1, construction loi.}
    \item \textbf{B} — L'élève divise par 3 valeurs au lieu de 4 issues. \emph{Renvoi : C1, construction loi.}
    \item \textbf{D} — L'élève calcule P(X>=1) au lieu de P(X=1). \emph{Renvoi : C1, loi de probabilité.}
  \end{itemize}
  \item \textbf{Q4}
