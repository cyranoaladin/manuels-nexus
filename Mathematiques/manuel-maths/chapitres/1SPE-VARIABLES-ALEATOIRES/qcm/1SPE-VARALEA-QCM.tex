\section*{\textcolor{chapcolor}{\MakeUppercase{Variables aleatoires — QCM d'auto-evaluation}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} On lance un de equilibre. Soit $X$ le resultat. Quelle est $P(X = 3)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{1}{3}$
    \item $\frac{1}{6}$
    \item $\frac{3}{6}$
    \item $\frac{1}{2}$
  \end{enumerate}

\item \textbf{[Q2]} La somme des probabilites d'un tableau de loi vaut toujours :
  \begin{enumerate}[label=\Alph*.]
    \item $0$
    \item $\frac{1}{2}$
    \item $1$
    \item Le nombre de valeurs prises par $X$
  \end{enumerate}

\item \textbf{[Q3]} On lance deux pieces. Soit $X$ le nombre de Pile. Quelle est $P(X = 1)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{1}{4}$
    \item $\frac{1}{2}$
    \item $\frac{1}{3}$
    \item $\frac{3}{4}$
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q4]} On lance un de equilibre. Quelle est l'esperance $E(X)$ du resultat ?
  \begin{enumerate}[label=\Alph*.]
    \item $3$
    \item $3{,}5$
    \item $4$
    \item $6$
  \end{enumerate}

\item \textbf{[Q5]} Si $V(X) = 9$, alors $\sigma(X)$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $81$
    \item $9$
    \item $3$
    \item $4{,}5$
  \end{enumerate}

\item \textbf{[Q6]} Si $E(X) = 4$ et $V(X) = 5$, quelle est $E(X^2)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $21$
    \item $9$
    \item $16$
    \item $20$
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q7]} On repete trois fois, de facon independante, une epreuve dont le succes a probabilite $\frac{1}{2}$. Quelle est la probabilite d'obtenir trois succes ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{1}{6}$
    \item $\frac{1}{8}$
    \item $\frac{3}{8}$
    \item $\frac{1}{2}$
  \end{enumerate}

\item \textbf{[Q8]} On repete deux fois, de facon independante, une epreuve dont le succes a probabilite $\frac{1}{3}$. Quelle est la probabilite d'obtenir exactement un succes ?
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{2}{9}$
    \item $\frac{4}{9}$
    \item $\frac{1}{9}$
    \item $\frac{2}{3}$
  \end{enumerate}

\item \textbf{[Q9]} Une epreuve a deux issues. On la repete trois fois. Combien l'arbre complet possede-t-il de chemins terminaux ?
  \begin{enumerate}[label=\Alph*.]
    \item $3$
    \item $8$
    \item $6$
    \item $9$
  \end{enumerate}

\item[] \textbf{Capacite C4}

\item \textbf{[Q10]} On sait que $E(X)=4$ et on pose $Y=3X-2$. Quelle est $E(Y)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $12$
    \item $10$
    \item $6$
    \item $2$
  \end{enumerate}

\item \textbf{[Q11]} On sait que $V(X)=5$ et on pose $Y=-2X+7$. Quelle est $V(Y)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $-10$
    \item $20$
    \item $3$
    \item $5$
  \end{enumerate}

\item \textbf{[Q12]} On sait que $\sigma(X)=3$ et on pose $Y=-4X+1$. Quelle est $\sigma(Y)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $-12$
    \item $12$
    \item $48$
    \item $3$
  \end{enumerate}

\item[] \textbf{Capacite C5}

\item \textbf{[Q13]} Un joueur paie 3 euros et recoit 15 euros s'il obtient un 6 au de. Gain algebrique si 6 ?
  \begin{enumerate}[label=\Alph*.]
    \item $15$ euros
    \item $12$ euros
    \item $3$ euros
    \item $18$ euros
  \end{enumerate}

\item \textbf{[Q14]} Un jeu a un gain moyen $E(G) = -0{,}5$ euro. Le jeu est :
  \begin{enumerate}[label=\Alph*.]
    \item Equitable
    \item Favorable au joueur
    \item Defavorable au joueur
    \item Impossible a determiner
  \end{enumerate}

\item \textbf{[Q15]} Prime $c=60$ euros, sinistre $1000$ euros, probabilite $1/25$. Benefice moyen ?
  \begin{enumerate}[label=\Alph*.]
    \item $60$ euros
    \item $20$ euros
    \item $-40$ euros
    \item $40$ euros
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des reponses erronees}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{A} — L'eleve divise 1 par la valeur de X au lieu du nombre de faces. \emph{Renvoi : C1, definition.}
    \item \textbf{C} — L'eleve confond P(X=3) et P(X<=3). \emph{Renvoi : C1, loi de probabilite.}
    \item \textbf{D} — L'eleve confond de et piece. \emph{Renvoi : C1, definition.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — L'eleve retient seulement la borne inferieure 0 de chaque probabilite au lieu de sommer toutes les probabilites de la loi. \emph{Renvoi : C1, propriete.}
    \item \textbf{B} — L'eleve suppose a tort que la somme vaut 1/2 parce qu'il imagine deux issues equiprobables, alors que la somme d'une loi complete vaut 1. \emph{Renvoi : C1, propriete.}
    \item \textbf{D} — Confusion entre somme des probabilites et cardinal de X(Omega). \emph{Renvoi : C1, propriete.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — L'eleve ne compte qu'une seule issue (PF ou FP) au lieu des deux. \emph{Renvoi : C1, construction loi.}
    \item \textbf{C} — L'eleve divise par 3 valeurs au lieu de 4 issues. \emph{Renvoi : C1, construction loi.}
    \item \textbf{D} — L'eleve calcule P(X>=1) au lieu de P(X=1). \emph{Renvoi : C1, loi de probabilite.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — L'eleve moyenne seulement les faces 1 a 5 : (1+2+3+4+5)/5=3, en oubliant la face 6. \emph{Renvoi : C2, esperance.}
    \item \textbf{C} — L'eleve moyenne seulement les faces 2 a 6 : (2+3+4+5+6)/5=4, en oubliant la face 1. \emph{Renvoi : C2, esperance.}
    \item \textbf{D} — Confusion esperance et valeur maximale. \emph{Renvoi : C2, esperance.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — L'eleve met V au carre au lieu de prendre la racine. \emph{Renvoi : C2, ecart type.}
    \item \textbf{B} — L'eleve confond variance et ecart type. \emph{Renvoi : C2, ecart type.}
    \item \textbf{D} — L'eleve divise V par 2. \emph{Renvoi : C2, ecart type.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{B} — L'eleve calcule E(X)+V(X) = 9. \emph{Renvoi : C2, Koenig-Huygens.}
    \item \textbf{C} — L'eleve donne [E(X)]\textasciicircum{}2 = 16. \emph{Renvoi : C2, Koenig-Huygens.}
    \item \textbf{D} — L'eleve calcule E(X)*V(X). \emph{Renvoi : C2, Koenig-Huygens.}
  \end{itemize}
  \item \textbf{Q7}
  \begin{itemize}
    \item \textbf{A} — L'eleve divise 1 par deux fois le nombre d'epreuves, au lieu de multiplier les trois probabilites du chemin SSS. \emph{Renvoi : C3, arbre pondere.}
    \item \textbf{C} — L'eleve compte trois positions possibles pour un echec ; il calcule ainsi la probabilite d'exactement deux succes. \emph{Renvoi : C3, nombre de succes.}
    \item \textbf{D} — L'eleve conserve la probabilite d'un seul succes et ne multiplie pas le long des trois branches du chemin SSS. \emph{Renvoi : C3, arbre pondere.}
  \end{itemize}
  \item \textbf{Q8}
  \begin{itemize}
    \item \textbf{A} — L'eleve calcule la probabilite d'un seul chemin, succes puis echec, et oublie le chemin echec puis succes. \emph{Renvoi : C3, arbre pondere.}
    \item \textbf{C} — L'eleve calcule la probabilite de deux succes : (1/3)\textasciicircum{}2=1/9. \emph{Renvoi : C3, nombre de succes.}
    \item \textbf{D} — L'eleve additionne 1/3 et 1/3 au lieu de multiplier sur chaque chemin puis d'additionner les deux chemins favorables. \emph{Renvoi : C3, arbre pondere.}
  \end{itemize}
  \item \textbf{Q9}
  \begin{itemize}
    \item \textbf{A} — L'eleve donne le nombre de niveaux de l'arbre au lieu du nombre de chemins terminaux. \emph{Renvoi : C3, arbre pondere.}
    \item \textbf{C} — L'eleve calcule 2 fois 3, alors que le nombre de chemins double a chaque niveau. \emph{Renvoi : C3, arbre pondere.}
    \item \textbf{D} — L'eleve calcule 3 au carre, comme s'il y avait trois issues par epreuve au lieu de deux. \emph{Renvoi : C3, arbre pondere.}
  \end{itemize}
  \item \textbf{Q10}
  \begin{itemize}
    \item \textbf{A} — L'eleve calcule 3E(X) mais oublie le terme constant -2. \emph{Renvoi : C4, linearite.}
    \item \textbf{C} — L'eleve calcule 2E(X)-2 en utilisant 2 comme coefficient de X. \emph{Renvoi : C4, linearite.}
    \item \textbf{D} — L'eleve soustrait seulement 2 a E(X) et oublie la multiplication par 3. \emph{Renvoi : C4, linearite.}
  \end{itemize}
  \item \textbf{Q11}
  \begin{itemize}
    \item \textbf{A} — L'eleve multiplie V(X) par -2 sans mettre le coefficient au carre, ce qui produit une variance negative impossible. \emph{Renvoi : C4, variance affine.}
    \item \textbf{C} — L'eleve calcule V(X)-2 et traite a tort le coefficient comme un terme soustrait. \emph{Renvoi : C4, variance affine.}
    \item \textbf{D} — L'eleve pense que ni la multiplication par -2 ni l'ajout de 7 ne modifient la variance. \emph{Renvoi : C4, variance affine.}
  \end{itemize}
  \item \textbf{Q12}
  \begin{itemize}
    \item \textbf{A} — L'eleve multiplie l'ecart type par -4 sans prendre la valeur absolue ; un ecart type ne peut pas etre negatif. \emph{Renvoi : C4, ecart type affine.}
    \item \textbf{C} — L'eleve utilise le carre du coefficient, regle de la variance, au lieu de sa valeur absolue pour l'ecart type : 4\textasciicircum{}2 fois 3=48. \emph{Renvoi : C4, ecart type affine.}
    \item \textbf{D} — L'eleve pense que la multiplication par -4 ne modifie pas l'ecart type et conserve sigma(X)=3. \emph{Renvoi : C4, ecart type affine.}
  \end{itemize}
  \item \textbf{Q13}
  \begin{itemize}
    \item \textbf{A} — Oubli de la mise dans le gain algebrique. \emph{Renvoi : C5, gain algebrique.}
    \item \textbf{C} — Confusion mise et gain. \emph{Renvoi : C5, gain algebrique.}
    \item \textbf{D} — Addition mise + gain brut. \emph{Renvoi : C5, gain algebrique.}
  \end{itemize}
  \item \textbf{Q14}
  \begin{itemize}
    \item \textbf{A} — Confusion E(G)=0 (equitable) et E(G)<0. \emph{Renvoi : C5, jeu equitable.}
    \item \textbf{B} — Signe de E(G) inverse. \emph{Renvoi : C5, interpretation esperance.}
    \item \textbf{D} — L'eleve pense que l'esperance ne suffit pas. \emph{Renvoi : C5, interpretation esperance.}
  \end{itemize}
  \item \textbf{Q15}
  \begin{itemize}
    \item \textbf{A} — L'eleve donne la prime sans soustraire l'esperance du sinistre. \emph{Renvoi : C5, assurance.}
    \item \textbf{C} — L'eleve donne l'oppose du cout moyen du sinistre, -pI=-40, sans ajouter la prime de 60 euros. \emph{Renvoi : C5, assurance.}
    \item \textbf{D} — L'eleve calcule correctement le cout moyen du sinistre pI=40 mais le confond avec le benefice moyen c-pI. \emph{Renvoi : C5, assurance.}
  \end{itemize}
\end{itemize}
