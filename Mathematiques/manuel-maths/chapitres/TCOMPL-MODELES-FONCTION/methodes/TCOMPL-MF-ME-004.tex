\begin{fichemethode}{M4}{Reconnaitre graphiquement convexité, concavité et point d'inflexion}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé fournit une COURBE (ou un trace de
  tangentes) et demande de reperer les zones de convexité, de concavité, et les
  eventuels points d'inflexion.
}

\pasApas{%
  \begin{enumerate}
    \item Position par rapport aux tangentes : la courbe est AU-DESSUS de
      chacune de ses tangentes $\to$ $f$ convexe ; AU-DESSOUS $\to$ $f$
      concave.
    \item Position par rapport aux cordes (secantes) : sous ses cordes $\to$
      convexe ; au-dessus $\to$ concave.
    \item Reperer visuellement l'orientation : « creux vers le haut » (cuvette)
      $\to$ convexe ; « creux vers le bas » (dome) $\to$ concave.
    \item Point d'inflexion : point ou la courbe TRAVERSE sa tangente — la
      convexité y change de nature.
    \item Conclure en donnant les intervalles (abscisses) de convexité, de
      concavité, et les coordonnées des points d'inflexion.
  \end{enumerate}
}

\exempleRedige{%
  La courbe de $f(x) = x^3 - 3x + 1$ est donnee. Decrire sa convexité.

  \medskip
  \commentaireMarge{Un dome a gauche, une cuvette a droite.}%
  Pour $x < 0$, la courbe est au-dessous de ses tangentes (allure en dome) :
  $f$ est concave sur $]-\infty\,;\,0]$. Pour $x > 0$, elle est au-dessus de ses
  tangentes (allure en cuvette) : $f$ est convexe sur $[0\,;\,+\infty[$.

  \medskip
  \commentaireMarge{La courbe traverse sa tangente au point d'inflexion.}%
  Au point d'abscisse 0 (de coordonnées $(0\,;\,1)$), la courbe traverse sa
  tangente : c'est un point d'inflexion.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Confondre variations et convexité : une fonction
      peut être décroissante ET convexe — les deux notions sont indépendantes.
    \item \textbf{Piège 2.} Prendre un extremum local pour un point
      d'inflexion : a l'inflexion, c'est la COURBURE qui change, pas
      necessairement le sens de variation.
    \item \textbf{Piège 3.} Conclure d'une seule tangente : la position doit
      être verifiee pour TOUTES les tangentes de la zone consideree.
  \end{itemize}
}

\verifier{%
  Tracer deux ou trois tangentes (ou cordes) dans chaque zone pour confirmer la
  position de la courbe, et contrôler la cohérence avec le signe de $f''$
  lorsqu'il est calculable (fiche M5).
}

\end{fichemethode}
