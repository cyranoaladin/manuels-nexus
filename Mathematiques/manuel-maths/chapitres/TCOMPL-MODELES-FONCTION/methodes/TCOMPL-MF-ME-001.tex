\begin{fichemethode}{M1}{Étudier une fonction : dérivée, limites, tableau de variation}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande l'étude complete d'une
  fonction : calcul de la dérivée, signe de la dérivée, limites aux bornes et
  tableau de variation.
}

\pasApas{%
  \begin{enumerate}
    \item Preciser l'ensemble de définition et les bornes de l'étude.
    \item Calculer $f'(x)$ avec les formules de dérivation (somme, produit,
      quotient, composee) et FACTORISER $f'$ autant que possible.
    \item Étudier le signe de $f'(x)$ (tableau de signes des facteurs).
    \item Calculer les limites de $f$ aux bornes de l'ensemble d'étude et les
      valeurs de $f$ aux points ou $f'$ s'annule (extremums locaux).
    \item Dresser le tableau de variation complet : signe de $f'$, fleches de
      variation, limites et extremums.
  \end{enumerate}
}

\exempleRedige{%
  Étudier $f(x) = x^3 - 3x + 1$ sur $\mathbb{R}$.

  \medskip
  \commentaireMarge{Factoriser $f'$ pour lire son signe.}%
  $f'(x) = 3x^2 - 3 = 3(x-1)(x+1)$ : $f'(x) > 0$ pour $x < -1$ ou $x > 1$,
  $f'(x) < 0$ pour $-1 < x < 1$.

  \medskip
  \commentaireMarge{Extremums locaux aux annulations de $f'$.}%
  $f(-1) = 3$ (maximum local) et $f(1) = -1$ (minimum local) ;
  $\lim\limits_{x \to -\infty} f(x) = -\infty$ et
  $\lim\limits_{x \to +\infty} f(x) = +\infty$.

  \medskip
  Tableau : $f$ croit de $-\infty$ a 3 sur $]-\infty\,;\,-1]$, décroît de 3 a
  $-1$ sur $[-1\,;\,1]$, puis croit de $-1$ a $+\infty$ sur $[1\,;\,+\infty[$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Conclure sur les variations a partir du signe de $f$
      au lieu du signe de $f'$.
    \item \textbf{Piège 2.} Ne pas factoriser $f'$ et se tromper dans le tableau
      de signes d'un trinôme.
    \item \textbf{Piège 3.} Oublier les limites ou les valeurs des extremums
      dans le tableau : il doit être complet pour être exploitable (fiche M2).
  \end{itemize}
}

\verifier{%
  Contrôler le signe de $f'$ par une valeur test dans chaque intervalle,
  vérifier les extremums en les recalculant, et confronter le tableau a
  l'allure de la courbe a la calculatrice.
}

\end{fichemethode}
