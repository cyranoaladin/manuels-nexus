\begin{fichemethode}{M7}{Déterminer une droite de regression (calcul ou logiciel)}

\quandUtiliser{%
  Utiliser cette méthode lorsque le nuage est sensiblement rectiligne
  (fiche M6) et que l'énoncé demande la droite de regression de $y$ en $x$ par
  la méthode des moindres carres, a la main ou au logiciel.
}

\pasApas{%
  \begin{enumerate}
    \item Calculer $\bar{x}$ et $\bar{y}$ (point moyen, fiche M6).
    \item Calculer la pente
      $a = \dfrac{\sum_i (x_i - \bar{x})(y_i - \bar{y})}
      {\sum_i (x_i - \bar{x})^2}$
      (organiser les calculs en tableau de colonnes).
    \item Calculer l'ordonnée a l'origine $b = \bar{y} - a\,\bar{x}$ : la
      droite passe TOUJOURS par le point moyen $G$.
    \item Au logiciel/calculatrice : mode statistiques a deux variables,
      regression linéaire $y = ax + b$ — relever $a$ et $b$ (et vérifier la
      cohérence avec le nuage).
    \item Écrire l'équation $y = ax + b$ et l'exploiter (tendance,
      interpolation), en restant prudent hors de la plage des donnees.
  \end{enumerate}
}

\exempleRedige{%
  Serie : $(0\,;\,1)$, $(1\,;\,2)$, $(2\,;\,2)$, $(3\,;\,5)$, $(4\,;\,5)$,
  de point moyen $G(2\,;\,3)$. Déterminer la droite de regression de $y$ en $x$.

  \medskip
  \commentaireMarge{Tableau des ecarts au point moyen.}%
  $\sum (x_i - \bar{x})(y_i - \bar{y})
  = (-2)(-2) + (-1)(-1) + 0 + (1)(2) + (2)(2) = 11$ et
  $\sum (x_i - \bar{x})^2 = 4 + 1 + 0 + 1 + 4 = 10$.

  \medskip
  \commentaireMarge{$b$ se deduit du passage par $G$.}%
  $a = \dfrac{11}{10} = 1{,}1$ et $b = 3 - 1{,}1 \times 2 = 0{,}8$ :
  la droite de regression est $y = 1{,}1\,x + 0{,}8$.

  \medskip
  Contrôle : $1{,}1 \times 2 + 0{,}8 = 3 = \bar{y}$ — la droite passe par $G$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Inverser les roles dans la pente : le dénominateur
      est $\sum (x_i - \bar{x})^2$, jamais un terme en $y$.
    \item \textbf{Piège 2.} Calculer $b$ avant $a$, ou l'estimer graphiquement :
      $b$ se DEDUIT de $a$ par $b = \bar{y} - a\bar{x}$.
    \item \textbf{Piège 3.} Extrapoler sans precaution très loin de la plage des
      donnees : le modèle affine n'y est plus valide a priori.
  \end{itemize}
}

\verifier{%
  Vérifier que la droite passe par le point moyen $G$, comparer $a$ et $b$ avec
  la sortie du logiciel, et contrôler visuellement que la droite traverse le
  nuage.
}

\end{fichemethode}
