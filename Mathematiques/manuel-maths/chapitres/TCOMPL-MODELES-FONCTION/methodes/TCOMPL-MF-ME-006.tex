\begin{fichemethode}{M6}{Représenter un nuage de points et calculer son point moyen}

\quandUtiliser{%
  Utiliser cette méthode lorsqu'une serie statistique a deux variables
  $(x_i\,;\,y_i)$ est donnee et que l'énoncé demande le nuage de points et les
  coordonnées du point moyen $G$.
}

\pasApas{%
  \begin{enumerate}
    \item Choisir un repere adapte : unités et graduations couvrant toutes les
      valeurs de la serie (axes eventuellement coupes par un « zigzag »).
    \item Placer chaque point $M_i(x_i\,;\,y_i)$ SANS les relier : un nuage
      n'est pas une courbe.
    \item Calculer les moyennes
      $\bar{x} = \dfrac{x_1 + \dots + x_n}{n}$ et
      $\bar{y} = \dfrac{y_1 + \dots + y_n}{n}$.
    \item Placer le point moyen $G(\bar{x}\,;\,\bar{y})$ sur le graphique (il
      n'est generalement PAS un point du nuage).
    \item Decrire l'allure du nuage : forme allongee ou non, tendance
      croissante ou décroissante — en vue d'un ajustement (fiche M7).
  \end{enumerate}
}

\exempleRedige{%
  Serie : $(0\,;\,1)$, $(1\,;\,2)$, $(2\,;\,2)$, $(3\,;\,5)$, $(4\,;\,5)$.
  Représenter le nuage et déterminer le point moyen.

  \medskip
  \commentaireMarge{Moyenne de chaque coordonnée separement.}%
  $\bar{x} = \dfrac{0+1+2+3+4}{5} = 2$ et
  $\bar{y} = \dfrac{1+2+2+5+5}{5} = 3$ : $G(2\,;\,3)$.

  \medskip
  Le nuage, allonge le long d'une direction croissante, suggere un ajustement
  affine ; $G$ se place au « centre de gravite » du nuage sans en être un point.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Relier les points du nuage entre eux : un nuage
      statistique se représente par des points isoles (croix).
    \item \textbf{Piège 2.} Croire que $G$ appartient au nuage : c'est faux en
      general (ici $(2\,;\,3)$ n'est pas dans la serie).
    \item \textbf{Piège 3.} Melanger les series dans les moyennes : $\bar{x}$
      se calcule avec les $x_i$ seulement, $\bar{y}$ avec les $y_i$ seulement.
  \end{itemize}
}

\verifier{%
  Contrôler que $\bar{x}$ et $\bar{y}$ sont compris entre le minimum et le
  maximum de leur serie, et que $G$ tombe visuellement au centre du nuage.
}

\end{fichemethode}
