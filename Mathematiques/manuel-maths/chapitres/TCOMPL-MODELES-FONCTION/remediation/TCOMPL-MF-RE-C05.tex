\subsection*{Rappel -- Fonction logarithme neperien : propriétés et dérivée}

\textbf{Propriétés :} $\ln(ab) = \ln a + \ln b$, $\ln\left(\dfrac{a}{b}\right) = \ln a - \ln b$, $\ln(\mathrm{e}^x) = x$.

\textbf{Dérivée :} $(\ln x)' = \dfrac{1}{x}$ sur $]0, +\infty[$.

\begin{exercice}{TCOMPL-MF-FR-R5-EX1}{1}{8}
\begin{enumerate}
  \item Résoudre $\ln(x + 1) = 0$.
  \item Étudier les variations de $g(x) = \ln x - x$ sur $]0, +\infty[$.
  \item Déterminer $\lim_{x \to +\infty} \dfrac{\ln x}{x^2}$.
\end{enumerate}
\end{exercice}
