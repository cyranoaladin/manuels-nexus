\textbf{1.} $k'(x) = 3x^2 - 6x$, donc $k''(x) = 6x - 6$.

\textbf{2.} $k''(x) = 6(x-1)$. Pour $x<1$, $k''(x)<0$ : $k$ est \textbf{concave} sur $]-\infty,1]$. Pour $x>1$, $k''(x)>0$ : $k$ est \textbf{convexe} sur $[1,+\infty[$.

\textbf{3.} $k''$ s'annule en $x=1$ en changeant de signe : $k$ admet un \textbf{point d'inflexion} en $x=1$, d'ordonnée $k(1) = 1-3=-2$, soit le point $(1,-2)$.
\end{corrige}
