\section*{\textcolor{chapcolor}{\MakeUppercase{Fonctions et modélisation}}}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q1]} La dérivée de $f(x) = e^{-x^2}$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $-2x e^{-x^2}$
    \item $e^{-x^2}$
    \item $2x e^{-x^2}$
    \item $-x e^{-x^2}$
  \end{enumerate}

\item[] \textbf{Capacité C5}

\item \textbf{[Q2]} Une fonction $f$ deux fois dérivable est convexe sur un intervalle $I$ si et seulement si sa dérivée seconde $f''$ est :
  \begin{enumerate}[label=\Alph*.]
    \item Positive ou nulle
    \item Nulle
    \item Négative
    \item Strictement décroissante
  \end{enumerate}

\item[] \textbf{Capacité C4}

\item \textbf{[Q3]} Si $f''$ s'annule en $a$ et change de signe en $a$, alors le point $(a,f(a))$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $f'$ s'annule
    \item $f''$ s'annule en changeant de signe
    \item $f$ s'annule
    \item $f''$ est maximale
  \end{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q4]} La limite de $\frac{e^x}{x}$ quand $x \to +\infty$ est par croissance comparee :
  \begin{enumerate}[label=\Alph*.]
    \item 0
    \item 1
    \item $+\infty$
    \item e
  \end{enumerate}

\item \textbf{[Q5]} Si $f(x) = \ln(2x+1)$, la dérivée $f'(x)$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{1}{2x+1}$
    \item $2 \ln(2x+1)$
    \item $\frac{1}{x}$
    \item $\frac{2}{2x+1}$
  \end{enumerate}

\end{enumerate}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{B} — Oubli de dériver l'interieur. Par la règle de chaîne, la dérivée de exp(-x\textasciicircum{}2) est -2x exp(-x\textasciicircum{}2). \emph{Renvoi : C1.}
    \item \textbf{C} — Erreur de signe de la dérivée de -x\textasciicircum{}2. Par la règle de chaîne, la dérivée de exp(-x\textasciicircum{}2) est -2x exp(-x\textasciicircum{}2). \emph{Renvoi : C1.}
    \item \textbf{D} — Oubli du facteur 2. Par la règle de chaîne, la dérivée de exp(-x\textasciicircum{}2) est -2x exp(-x\textasciicircum{}2). \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{B} — Fonction affine. Pour une fonction deux fois dérivable, la convexité équivaut à f''>=0 sur l'intervalle. \emph{Renvoi : C5.}
    \item \textbf{C} — Condition de concavité. Pour une fonction deux fois dérivable, la convexité équivaut à f''>=0 sur l'intervalle. \emph{Renvoi : C5.}
    \item \textbf{D} — Condition incorrecte. Pour une fonction deux fois dérivable, la convexité équivaut à f''>=0 sur l'intervalle. \emph{Renvoi : C5.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — Extremum local de f, pas inflexion. Si f'' s'annule en a et change de signe, la convexité change et le point (a,f(a)) est un point d'inflexion. \emph{Renvoi : C4.}
    \item \textbf{C} — Racine de f. Si f'' s'annule en a et change de signe, la convexité change et le point (a,f(a)) est un point d'inflexion. \emph{Renvoi : C4.}
    \item \textbf{D} — Pas de changement de concavité requis par le max de f''. Si f'' s'annule en a et change de signe, la convexité change et le point (a,f(a)) est un point d'inflexion. \emph{Renvoi : C4.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — Inverse x/e\textasciicircum{}x tend vers 0. L'exponentielle domine x à l'infini, donc exp(x)/x tend vers +infini. \emph{Renvoi : C1.}
    \item \textbf{B} — Forme indeterminee non levee. L'exponentielle domine x à l'infini, donc exp(x)/x tend vers +infini. \emph{Renvoi : C1.}
    \item \textbf{D} — Valeur en x=1. L'exponentielle domine x à l'infini, donc exp(x)/x tend vers +infini. \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — Oubli de la dérivée de u(x) = 2x+1. Pour u=2x+1, la règle (ln u)'=u'/u donne 2/(2x+1) sur le domaine de définition. \emph{Renvoi : C1.}
    \item \textbf{B} — Multiplication au lieu de dérivation. Pour u=2x+1, la règle (ln u)'=u'/u donne 2/(2x+1) sur le domaine de définition. \emph{Renvoi : C1.}
    \item \textbf{C} — Simplification abusive. Pour u=2x+1, la règle (ln u)'=u'/u donne 2/(2x+1) sur le domaine de définition. \emph{Renvoi : C1.}
  \end{itemize}
\end{itemize}
