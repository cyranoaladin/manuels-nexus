\subsection*{C1 --- Dérivée et tableau de variation}

\begin{enumerate}

\item \textbf{[Q1]} Si $f'(x) < 0$ sur un intervalle $I$, alors sur $I$, $f$ est :
  \begin{enumerate}[label=\Alph*.]
    \item croissante.
    \item décroissante.
    \item constante.
    \item convexe.
  \end{enumerate}

\subsection*{C2 --- Nombre de solutions}

\item \textbf{[Q2]} Une fonction $f$ décroît de $+\infty$ à $5$ sur $]0,a]$ puis croît de $5$ à $+\infty$ sur $[a,+\infty[$. L'équation $f(x) = 3$ admet :
  \begin{enumerate}[label=\Alph*.]
    \item aucune solution.
    \item une solution.
    \item deux solutions.
    \item une infinité de solutions.
  \end{enumerate}

\bigskip

\subsection*{C4/C5 --- Convexité}

\item \textbf{[Q3]} Une fonction dont la courbe est toujours au-dessus de ses tangentes est dite :
  \begin{enumerate}[label=\Alph*.]
    \item concave.
    \item convexe.
    \item croissante.
    \item impaire.
  \end{enumerate}

\bigskip

\item \textbf{[Q4]} Une fonction deux fois dérivable est convexe sur un intervalle $I$ si et seulement si :
  \begin{enumerate}[label=\Alph*.]
    \item $f'(x) \geq 0$ sur $I$.
    \item $f''(x) \geq 0$ sur $I$.
    \item $f(x) \geq 0$ sur $I$.
    \item $f''(x) \leq 0$ sur $I$.
  \end{enumerate}

\bigskip

\subsection*{C6/C7 --- Statistique à deux variables}

\item \textbf{[Q5]} Le point moyen $G(\bar{x}, \bar{y})$ d'un nuage de points :
  \begin{enumerate}[label=\Alph*.]
    \item n'a aucun lien avec la droite de régression.
    \item appartient toujours à la droite de régression (moindres carrés).
    \item est toujours à l'origine du repère.
    \item n'existe que si le nuage est aligné.
  \end{enumerate}

\bigskip

\item \textbf{[Q6]} Utiliser une droite de régression pour estimer une valeur très en dehors de la plage des données observées, c'est :
  \begin{enumerate}[label=\Alph*.]
    \item interpoler, toujours fiable.
    \item extrapoler, ce qui demande un regard critique.
    \item impossible mathématiquement.
    \item équivalent à calculer le point moyen.
  \end{enumerate}

\end{enumerate}
