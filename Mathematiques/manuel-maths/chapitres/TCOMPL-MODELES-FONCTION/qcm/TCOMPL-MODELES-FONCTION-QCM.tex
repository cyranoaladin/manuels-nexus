\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} La derivee de $f(x) = e^{-x^2}$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $e^{-x^2}$
    \item $-2x e^{-x^2}$
    \item $2x e^{-x^2}$
    \item $-x e^{-x^2}$
  \end{enumerate}

\item \textbf{[Q2]} Une fonction f est convexe sur un intervalle I si et seulement si sa derivee seconde $f''$ est :
  \begin{enumerate}[label=\Alph*.]
    \item Négative
    \item Nulle
    \item Positive ou nulle
    \item Strictement decroissante
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q3]} Un point d'inflexion de la courbe de f est un point ou :
  \begin{enumerate}[label=\Alph*.]
    \item $f''$ s'annule en changeant de signe
    \item $f'$ s'annule
    \item $f$ s'annule
    \item $f''$ est maximale
  \end{enumerate}

\item \textbf{[Q4]} La limite de $\frac{e^x}{x}$ quand $x \to +\infty$ est par croissance comparee :
  \begin{enumerate}[label=\Alph*.]
    \item 0
    \item 1
    \item e
    \item $+\infty$
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q5]} Si $f(x) = \ln(2x+1)$, la derivee $f'(x)$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{1}{2x+1}$
    \item $\frac{2}{2x+1}$
    \item $\frac{1}{x}$
    \item $2 \ln(2x+1)$
  \end{enumerate}

\end{enumerate}
