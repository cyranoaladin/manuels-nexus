\begin{exercice}{TCOMPL-MF-EX-006}{2}{15}
On donne le tableau de variations de $f$ sur $\mathbb{R}$ : $f$ décroît de
$+\infty$ à $-4$ sur $\left]-\infty\,;\,1\right]$, puis croît de $-4$ à
$+\infty$ sur $\left[1\,;\,+\infty\right[$.
\begin{enumerate}
  \item Combien l'équation $f(x)=0$ admet-elle de solutions ?
  \item Et l'équation $f(x)=-4$ ? $f(x)=-5$ ?
  \item Résoudre l'inéquation $f(x)\leq-4$.
\end{enumerate}
\end{exercice}
