\begin{exercice}{TCOMPL-MF-EX-007}{2}{15}
Soit $f(x)=x^3+x-3$ sur $\mathbb{R}$.
\begin{enumerate}
  \item Montrer que $f$ est strictement croissante.
  \item Justifier que $f(x)=0$ a une unique solution $\alpha$, et l'encadrer
        entre deux entiers consécutifs.
  \item Donner un encadrement de $\alpha$ d'amplitude $10^{-2}$ par
        dichotomie ou balayage.
\end{enumerate}
\end{exercice}
