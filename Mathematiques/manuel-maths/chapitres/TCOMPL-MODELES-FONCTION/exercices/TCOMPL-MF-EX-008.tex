\begin{exercice}{TCOMPL-MF-EX-008}{1}{12}
On donne la courbe de $f(x)=x^3-3x$ sur $[-2\,;\,2]$.
\begin{enumerate}
  \item Reconnaître graphiquement sur quel intervalle la courbe est concave,
        puis convexe.
  \item Déterminer l'abscisse du point d'inflexion.
  \item Confirmer par le calcul de $f''$.
\end{enumerate}
\end{exercice}
