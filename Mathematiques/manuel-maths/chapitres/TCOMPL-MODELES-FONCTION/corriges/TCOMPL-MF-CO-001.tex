\begin{corrige}{TCOMPL-MF-EX-001}
\textbf{1.} Pour $x>0$ :
\[
  f'(x) = 1 - \frac{16}{x^2}.
\]

\textbf{2.} $f'(x) = 0 \iff x^2 = 16 \iff x = 4$ (car $x>0$). Pour $0<x<4$, $f'(x)<0$ ; pour $x>4$, $f'(x)>0$ : $f$ décroît sur $]0,4]$ puis croît sur $[4,+\infty[$.

\textbf{3.} Le minimum est atteint en $x=4$, avec $f(4) = 4 + \dfrac{16}{4} = 4+4=8$. Le coût moyen minimal est de $8$ euros par objet, pour une production de $4$ objets.
\end{corrige}
