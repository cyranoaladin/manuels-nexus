\begin{corrige}{TCOMPL-MF-EX-005}

\textbf{Question 1.} On pose $u(x) = \dfrac{1}{x}$. Alors $u'(x) = -\dfrac{1}{x^2}$.

\[ f'(x) = u'(x)\,\mathrm{e}^{u(x)} = -\frac{1}{x^2}\,\mathrm{e}^{1/x}. \]

\commentaireMarge{Sur $\mathbb{R}^*$, $x^2 > 0$ et $\mathrm{e}^{1/x} > 0$, donc $f' < 0$.}

\textbf{Question 2.} Pour tout $x \in \mathbb{R}^*$, on a $x^2 > 0$ et $\mathrm{e}^{1/x} > 0$, donc $f'(x) < 0$.

La fonction $f$ est \textbf{strictement decroissante} sur $\left]-\infty, 0\right[$ et sur $\left]0, +\infty\right[$.

\end{corrige}
