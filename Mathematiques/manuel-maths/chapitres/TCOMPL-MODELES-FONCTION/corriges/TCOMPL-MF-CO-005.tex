\begin{corrige}{TCOMPL-MF-EX-005}
\textbf{1.} $f'(x)=\dfrac{2(x+3)-(2x+1)}{(x+3)^2}=\dfrac{5}{(x+3)^2}>0$ :
$f$ est strictement croissante sur $\left]-3\,;\,+\infty\right[$.

\textbf{2.} En $-3^+$, le numérateur tend vers $-5$ et le dénominateur vers
$0^+$ : $f(x)\to-\infty$. En $+\infty$, $f(x)=\dfrac{2+1/x}{1+3/x}\to2$.

\textbf{3.} $f$ croît de $-\infty$ à $2$ (limite non atteinte) sur
$\left]-3\,;\,+\infty\right[$.
