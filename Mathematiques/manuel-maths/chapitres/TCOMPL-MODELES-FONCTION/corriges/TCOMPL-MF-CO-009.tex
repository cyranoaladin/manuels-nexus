\begin{corrige}{TCOMPL-MF-EX-009}
\textbf{1.} $f'(x)=(1-x)\mathrm{e}^{-x}$ et
$f''(x)=(x-2)\mathrm{e}^{-x}$.

\textbf{2.} $\mathrm{e}^{-x}>0$, donc $f''$ a le signe de $x-2$ : $f$ est
concave sur $\left]-\infty\,;\,2\right]$ et convexe sur
$\left[2\,;\,+\infty\right[$. Le point d'inflexion est en $x=2$, d'ordonnée
$2\mathrm{e}^{-2}$.

\textbf{3.} Au voisinage de $x=2$ la courbe traverse sa tangente : elle est
au-dessous avant (partie concave) et au-dessus après (partie convexe).
\end{corrige}
