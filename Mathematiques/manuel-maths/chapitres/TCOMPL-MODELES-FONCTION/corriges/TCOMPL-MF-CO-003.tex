\begin{corrige}{TCOMPL-MF-EX-003}
\textbf{1.} $h'(x) = 3x^2 - 12x + 9$, donc $h''(x) = 6x - 12$.

\textbf{2.} $h''(x) = 6x-12 = 6(x-2)$. Pour $x<2$, $h''(x)<0$ : $h$ est \textbf{concave} sur $]-\infty,2]$. Pour $x>2$, $h''(x)>0$ : $h$ est \textbf{convexe} sur $[2,+\infty[$.

\textbf{3.} $h''$ s'annule en $x=2$ en changeant de signe (négatif avant, positif après) : $h$ admet un \textbf{point d'inflexion} d'abscisse $x=2$.
\end{corrige}
