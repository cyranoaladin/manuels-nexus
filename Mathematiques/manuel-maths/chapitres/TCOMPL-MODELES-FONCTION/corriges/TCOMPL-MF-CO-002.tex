\begin{corrige}{TCOMPL-MF-EX-002}
\textbf{1.} $10 > 8$ (le minimum) : la valeur $10$ est atteinte une fois sur chaque intervalle de monotonie ($]0,4]$ et $[4,+\infty[$), soit \textbf{deux solutions}.

\textbf{2.} $8$ est exactement le minimum de $f$, atteint en un seul point ($x=4$), à la jonction des deux intervalles de monotonie : \textbf{une seule solution}.

\textbf{3.} $7 < 8$ (le minimum) : $f$ ne prend jamais la valeur $7$, qui est strictement inférieure à son minimum : \textbf{aucune solution}.
\end{corrige}
