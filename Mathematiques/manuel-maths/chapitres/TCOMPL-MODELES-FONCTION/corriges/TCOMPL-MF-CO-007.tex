\begin{corrige}{TCOMPL-MF-EX-007}
\textbf{1.} $f'(x)=3x^2+1>0$ pour tout $x$ : $f$ est strictement croissante
sur $\mathbb{R}$.

\textbf{2.} $f$ est continue, strictement croissante, de limites $-\infty$ et
$+\infty$ : l'équation $f(x)=0$ a une unique solution $\alpha$. Comme
$f(1)=-1<0$ et $f(2)=7>0$, on a $1<\alpha<2$.

\textbf{3.} Par balayage au pas $0{,}01$ : $f(1{,}21)\approx-0{,}018<0$ et
$f(1{,}22)\approx0{,}036>0$, donc $1{,}21<\alpha<1{,}22$.
\end{corrige}
