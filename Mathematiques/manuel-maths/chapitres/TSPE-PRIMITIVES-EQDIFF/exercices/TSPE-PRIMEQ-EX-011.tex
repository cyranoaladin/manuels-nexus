\begin{exercice}{TSPE-PRIMEQ-EX-003}{1}{12}

\begin{enumerate}
  \item Determiner la solution particuliere constante $y_0$.
  \item Donner la forme generale des solutions.
  \item Determiner la solution $y$ telle que $y(0) = 1$.
\end{enumerate}
