\begin{exercice}{TSPE-PRIMEQ-EX-006}{1}{10}
On donne $F(x) = \mathrm{e}^{2x}$ et $G(x) = \mathrm{e}^{2x} - 7$.
\begin{enumerate}
  \item Verifier que $F$ et $G$ sont deux primitives d'une meme fonction $f$ que l'on determinera.
  \item Sans calcul supplementaire, donner (en justifiant par le cours) la valeur de la constante $k$ telle que $G = F + k$.
\end{enumerate}
\end{exercice}
