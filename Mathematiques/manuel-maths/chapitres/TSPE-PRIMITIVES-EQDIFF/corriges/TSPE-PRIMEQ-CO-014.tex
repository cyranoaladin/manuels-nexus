\begin{corrige}{TSPE-PRIMEQ-EX-006}

\textbf{Q1.} $F'(x) = 2\mathrm{e}^{2x}$ et $G'(x) = 2\mathrm{e}^{2x}$ : $F$ et $G$ sont bien deux primitives de $f(x) = 2\mathrm{e}^{2x}$ sur $\mathbb{R}$.

\textbf{Q2.} D'apres le theoreme du cours, deux primitives d'une meme fonction continue sur un intervalle different d'une constante. Ici, $G(x)-F(x) = -7$ pour tout $x$ (calcul direct), donc $k=-7$ : $G = F - 7$.

\end{corrige}
