\begin{corrige}{TSPE-PRIMEQ-EX-005}

\textbf{Q1.} $y_p'(x) = 2$. Et $3y_p(x) + 5 - 6x = 3(2x-1)+5-6x = 6x-3+5-6x = 2$. On a bien $y_p' = 3y_p+5-6x$. \checkmark

\textbf{Q2.} Solutions generales : $y(x) = C\mathrm{e}^{3x} + (2x-1)$, $C \in \mathbb{R}$.

\textbf{Q3.} $y(0) = C - 1 = 5$, donc $C=6$. La solution cherchee est $y(x) = 6\mathrm{e}^{3x}+2x-1$.

\end{corrige}
