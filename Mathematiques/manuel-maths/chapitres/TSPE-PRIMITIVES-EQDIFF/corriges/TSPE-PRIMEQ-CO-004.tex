\begin{corrige}{TSPE-PRIMEQ-EX-004}

\textbf{Q1.} $T' = -0{,}1(T-20) = -0{,}1T + 2$ : on a $a=-0{,}1$ et $b=2$.

\textbf{Q2.} Solution particuliere : $T_0 = -\dfrac{b}{a} = -\dfrac{2}{-0{,}1} = 20$ (coherent avec la temperature ambiante). Solutions generales : $T(t) = C\mathrm{e}^{-0{,}1t}+20$.

\textbf{Q3.} $T(0) = C+20 = 90$, donc $C=70$. Ainsi $T(t) = 70\,\mathrm{e}^{-0{,}1t}+20$.

\textbf{Q4.} $T(10) = 70\,\mathrm{e}^{-1}+20 \approx 25{,}75+20 = 45{,}75\degres$C.
