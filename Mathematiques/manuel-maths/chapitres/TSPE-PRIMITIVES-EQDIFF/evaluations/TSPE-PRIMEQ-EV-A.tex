\begin{evaluation}{TSPE-PRIMEQ-EV-A}{55}

\begin{exercice}{TSPE-PRIMEQ-EV-A-EX1}{2}{20}
\textbf{Exercice 1} (7 points) — \textit{Capacite C1}

\medskip

Determiner une primitive sur $\mathbb{R}$ de $f(x) = x^2(x^3+2)^4$.
\end{exercice}

\begin{exercice}{TSPE-PRIMEQ-EV-A-EX2}{2}{20}
\textbf{Exercice 2} (6 points) — \textit{Capacite C4}

\medskip

On donne $F(x) = \ln(x^2+1)$ et $G(x) = \ln(x^2+1) + \dfrac{1}{2}$.
\begin{enumerate}
  \item (3 pts) Verifier que $F$ et $G$ sont deux primitives d'une meme fonction $f$ sur $\mathbb{R}$, que l'on determinera.
  \item (3 pts) En utilisant le theoreme du cours, justifier sans calcul supplementaire que $G-F$ est constante, et donner sa valeur.
\end{enumerate}
\end{exercice}

\begin{exercice}{TSPE-PRIMEQ-EV-A-EX3}{2}{15}
\textbf{Exercice 3} (7 points) — \textit{Capacite C5}

\medskip

Resoudre l'equation differentielle $y' = 2y$, puis determiner la solution telle que $y(1) = 3\mathrm{e}^2$.
\end{exercice}
