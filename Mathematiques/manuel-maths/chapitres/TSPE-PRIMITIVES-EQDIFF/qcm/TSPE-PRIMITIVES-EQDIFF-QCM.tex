\section*{\textcolor{chapcolor}{\MakeUppercase{QCM d'auto-évaluation — Primitives et équations différentielles}}}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q1]} Une primitive de $f(x) = \mathrm{e}^{3x}$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $\dfrac{1}{3}\mathrm{e}^{3x}$.
    \item $\mathrm{e}^{3x}$.
    \item $3\mathrm{e}^{3x}$.
    \item $\mathrm{e}^{3x+1}$.
  \end{enumerate}

\item[] \textbf{Capacité C4}

\item \textbf{[Q2]} Si $F$ et $G$ sont deux primitives de la même fonction $f$ continue sur un intervalle $I$, alors :
  \begin{enumerate}[label=\Alph*.]
    \item $F-G$ est constante.
    \item $F-G$ est une fonction affine de pente non nulle.
    \item $F=G$.
    \item $F \times G$ est constante.
  \end{enumerate}

\item[] \textbf{Capacité C5}

\item \textbf{[Q3]} Les solutions de $y'=5y$ sur $\mathbb{R}$ sont :
  \begin{enumerate}[label=\Alph*.]
    \item $y(x) = 5x + C$.
    \item $y(x) = C\mathrm{e}^{5x}$.
    \item $y(x) = C\mathrm{e}^{-5x}$.
    \item $y(x) = 5\mathrm{e}^{Cx}$.
  \end{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q4]} Pour $y' = 3y - 12$, la solution particulière constante est :
  \begin{enumerate}[label=\Alph*.]
    \item $y_0 = 12$.
    \item $y_0 = -4$.
    \item $y_0 = 4$.
    \item $y_0 = -36$.
  \end{enumerate}

\item[] \textbf{Capacité C3}

\item \textbf{[Q5]} Pour utiliser une solution particulière $y_p$ donnee de $y'=ay+f$, il faut d'abord :
  \begin{enumerate}[label=\Alph*.]
    \item calculer $y_p(0)$.
    \item résoudre $y_p=0$.
    \item dériver $f$.
    \item vérifier que $y_p$ satisfait bien l'équation.
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des réponses erronées}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{B} — La dérivée de $\mathrm e^{3x}$ vaut $3\mathrm e^{3x}$ ; il faut donc le facteur $1/3$ pour retrouver exactement $\mathrm e^{3x}$. \emph{Renvoi : M1, primitive de $\mathrm{e}^{ax}$.}
    \item \textbf{C} — La dérivée de $3\mathrm e^{3x}$ vaut $9\mathrm e^{3x}$ ; ce choix multiplie par $3$ au lieu de compenser la dérivée de $3x$. \emph{Renvoi : M1, primitive de $\mathrm{e}^{ax}$.}
    \item \textbf{D} — Comme $\mathrm e^{3x+1}=\mathrm e\,\mathrm e^{3x}$, sa dérivée vaut $3\mathrm e\,\mathrm e^{3x}$ et non $\mathrm e^{3x}$. \emph{Renvoi : M1, primitive de $\mathrm{e}^{ax}$.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{B} — Puisque $(F-G)'=f-f=0$, la difference est constante : elle ne peut donc pas être affine de pente non nulle. \emph{Renvoi : C1, ensemble des primitives.}
    \item \textbf{C} — Deux primitives d'une même fonction ne sont pas necessairement égales ; elles peuvent differer d'une constante. \emph{Renvoi : C1, ensemble des primitives.}
    \item \textbf{D} — Aucune propriété generale ne rend le produit de deux primitives constant. \emph{Renvoi : C1, ensemble des primitives.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — Pour $y=5x+C$, on a $y'=5$ alors que $5y=25x+5C$ ; l'égalité différentielle n'est donc pas satisfaite en general. \emph{Renvoi : C5, solutions de $y'=ay$.}
    \item \textbf{C} — La dérivée de $C\mathrm e^{-5x}$ vaut $-5y$ ; cette famille resout $y'=-5y$, et non l'équation $y'=5y$. \emph{Renvoi : C5, solutions de $y'=ay$.}
    \item \textbf{D} — Pour $y=5\mathrm e^{Cx}$, on obtient $y'=Cy$ ; cette expression ne satisfait $y'=5y$ que pour $C=5$ et ne donne pas toute la famille. \emph{Renvoi : C5, solutions de $y'=ay$.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — Pour $y_0=12$, on a $3y_0-12=24$ au lieu de $0$ ; cette constante n'est donc pas une solution particulière. \emph{Renvoi : M2, solution particulière constante de $y'=ay+b$.}
    \item \textbf{B} — Pour la constante $y_0=-4$, le membre de gauche vaut $0$ mais $3y_0-12=-24$ ; l'équation n'est pas verifiee. \emph{Renvoi : M2, solution particulière constante de $y'=ay+b$.}
    \item \textbf{D} — Pour $y_0=-36$, le second membre vaut $3(-36)-12=-120$, tandis que la dérivée de la constante vaut $0$. \emph{Renvoi : M2, solution particulière constante de $y'=ay+b$.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — La valeur $y_p(0)$ peut servir ensuite avec une condition initiale, mais elle ne prouve pas que $y_p'=ay_p+f$ sur l'intervalle. \emph{Renvoi : M2, vérifier qu'une fonction est solution particulière.}
    \item \textbf{B} — Les zeros de $y_p$ ne disent pas si la fonction satisfait l'équation différentielle ; il faut substituer $y_p$ et $y_p'$ dans celle-ci. \emph{Renvoi : M2, vérifier qu'une fonction est solution particulière.}
    \item \textbf{C} — Dériver $f$ n'est pas la vérification demandee ; il faut calculer $y_p'$ puis comparer cette dérivée a $ay_p+f$. \emph{Renvoi : M2, vérifier qu'une fonction est solution particulière.}
  \end{itemize}
\end{itemize}
