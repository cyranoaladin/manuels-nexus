\section*{\textcolor{chapcolor}{\MakeUppercase{QCM d'auto-evaluation — Primitives et équations différentielles}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} Une primitive de $f(x) = \mathrm{e}^{3x}$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $3\mathrm{e}^{3x}$.
    \item $\mathrm{e}^{3x}$.
    \item $\dfrac{1}{3}\mathrm{e}^{3x}$.
    \item $\mathrm{e}^{3x+1}$.
  \end{enumerate}

\item[] \textbf{Capacite C4}

\item \textbf{[Q2]} Si $F$ et $G$ sont deux primitives de la meme fonction $f$ continue sur un intervalle $I$, alors :
  \begin{enumerate}[label=\Alph*.]
    \item $F=G$.
    \item $F-G$ est une fonction affine de pente non nulle.
    \item $F-G$ est constante.
    \item $F \times G$ est constante.
  \end{enumerate}

\item[] \textbf{Capacite C5}

\item \textbf{[Q3]} Les solutions de $y'=5y$ sur $\mathbb{R}$ sont :
  \begin{enumerate}[label=\Alph*.]
    \item $y(x) = 5x + C$.
    \item $y(x) = C\mathrm{e}^{5x}$.
    \item $y(x) = C\mathrm{e}^{-5x}$.
    \item $y(x) = 5\mathrm{e}^{Cx}$.
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q4]} Pour $y' = 3y - 12$, la solution particuliere constante est :
  \begin{enumerate}[label=\Alph*.]
    \item $y_0 = 4$.
    \item $y_0 = -4$.
    \item $y_0 = 12$.
    \item $y_0 = -36$.
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q5]} Pour utiliser une solution particuliere $y_p$ donnee de $y'=ay+f$, il faut d'abord :
  \begin{enumerate}[label=\Alph*.]
    \item calculer $y_p(0)$.
    \item verifier que $y_p$ satisfait bien l'equation.
    \item deriver $f$.
    \item resoudre $y_p=0$.
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des reponses erronees}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{A} — La derivee de $3\mathrm e^{3x}$ vaut $9\mathrm e^{3x}$ ; ce choix multiplie par $3$ au lieu de compenser la derivee de $3x$.
    \item \textbf{B} — La derivee de $\mathrm e^{3x}$ vaut $3\mathrm e^{3x}$ ; il faut donc le facteur $1/3$ pour retrouver exactement $\mathrm e^{3x}$.
    \item \textbf{D} — Comme $\mathrm e^{3x+1}=\mathrm e\,\mathrm e^{3x}$, sa derivee vaut $3\mathrm e\,\mathrm e^{3x}$ et non $\mathrm e^{3x}$.
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — Deux primitives d'une meme fonction ne sont pas necessairement egales ; elles peuvent differer d'une constante. \emph{Renvoi : C1, ensemble des primitives.}
    \item \textbf{B} — Puisque $(F-G)'=f-f=0$, la difference est constante : elle ne peut donc pas etre affine de pente non nulle. \emph{Renvoi : C1, ensemble des primitives.}
    \item \textbf{D} — Aucune propriete generale ne rend le produit de deux primitives constant. \emph{Renvoi : C1, ensemble des primitives.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — Pour $y=5x+C$, on a $y'=5$ alors que $5y=25x+5C$ ; l'egalite differentielle n'est donc pas satisfaite en general.
    \item \textbf{C} — La derivee de $C\mathrm e^{-5x}$ vaut $-5y$ ; cette famille resout $y'=-5y$, et non l'equation $y'=5y$.
    \item \textbf{D} — Pour $y=5\mathrm e^{Cx}$, on obtient $y'=Cy$ ; cette expression ne satisfait $y'=5y$ que pour $C=5$ et ne donne pas toute la famille.
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{B} — Pour la constante $y_0=-4$, le membre de gauche vaut $0$ mais $3y_0-12=-24$ ; l'equation n'est pas verifiee.
    \item \textbf{C} — Pour $y_0=12$, on a $3y_0-12=24$ au lieu de $0$ ; cette constante n'est donc pas une solution particuliere.
    \item \textbf{D} — Pour $y_0=-36$, le second membre vaut $3(-36)-12=-120$, tandis que la derivee de la constante vaut $0$.
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — La valeur $y_p(0)$ peut servir ensuite avec une condition initiale, mais elle ne prouve pas que $y_p'=ay_p+f$ sur l'intervalle.
    \item \textbf{C} — Deriver $f$ n'est pas la verification demandee ; il faut calculer $y_p'$ puis comparer cette derivee a $ay_p+f$.
    \item \textbf{D} — Les zeros de $y_p$ ne disent pas si la fonction satisfait l'equation differentielle ; il faut substituer $y_p$ et $y_p'$ dans celle-ci.
  \end{itemize}
\end{itemize}
