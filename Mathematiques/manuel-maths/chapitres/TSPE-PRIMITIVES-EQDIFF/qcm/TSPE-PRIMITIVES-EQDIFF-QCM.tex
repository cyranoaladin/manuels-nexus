\section*{\textcolor{chapcolor}{\MakeUppercase{QCM d'auto-evaluation — Primitives et équations différentielles}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} Une primitive de $f(x) = \mathrm{e}^{3x}$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $3\mathrm{e}^{3x}$.
    \item $\mathrm{e}^{3x}$.
    \item $\dfrac{1}{3}\mathrm{e}^{3x}$.
    \item $\mathrm{e}^{3x+1}$.
  \end{enumerate}

\item \textbf{[Q2]} Si $F$ et $G$ sont deux primitives de la meme fonction $f$ continue sur un intervalle $I$, alors :
  \begin{enumerate}[label=\Alph*.]
    \item $F=G$.
    \item $F-G$ est une fonction affine de pente non nulle.
    \item $F-G$ est constante.
    \item $F \times G$ est constante.
  \end{enumerate}

\item \textbf{[Q3]} Les solutions de $y'=5y$ sur $\mathbb{R}$ sont :
  \begin{enumerate}[label=\Alph*.]
    \item $y(x) = 5x + C$.
    \item $y(x) = C\mathrm{e}^{5x}$.
    \item $y(x) = C\mathrm{e}^{-5x}$.
    \item $y(x) = 5\mathrm{e}^{Cx}$.
  \end{enumerate}

\item \textbf{[Q4]} Pour $y' = 3y - 12$, la solution particuliere constante est :
  \begin{enumerate}[label=\Alph*.]
    \item $y_0 = 4$.
    \item $y_0 = -4$.
    \item $y_0 = 12$.
    \item $y_0 = -36$.
  \end{enumerate}

\item \textbf{[Q5]} Pour utiliser une solution particuliere $y_p$ donnee de $y'=ay+f$, il faut d'abord :
  \begin{enumerate}[label=\Alph*.]
    \item calculer $y_p(0)$.
    \item verifier que $y_p$ satisfait bien l'equation.
    \item deriver $f$.
    \item resoudre $y_p=0$.
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des reponses erronees}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{A} — La derivee de $3\mathrm e^{3x}$ vaut $9\mathrm e^{3x}$ ; le facteur devant une primitive doit etre $1/3$. \emph{Renvoi : C1, primitives de fonctions exponentielles.}
    \item \textbf{B} — La derivee de $\mathrm e^{3x}$ vaut $3\mathrm e^{3x}$, pas $\mathrm e^{3x}$. \emph{Renvoi : C1, primitives de fonctions exponentielles.}
    \item \textbf{D} — $\mathrm e^{3x+1}=\mathrm e\,\mathrm e^{3x}$ a pour derivee $3\mathrm e^{3x+1}$ ; ce n'est pas une primitive demandee. \emph{Renvoi : C1, primitives de fonctions exponentielles.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — Deux primitives d'une meme fonction ne sont pas necessairement egales ; elles peuvent differer d'une constante. \emph{Renvoi : C1, ensemble des primitives.}
    \item \textbf{B} — Puisque $(F-G)'=f-f=0$, la difference est constante : elle ne peut donc pas etre affine de pente non nulle. \emph{Renvoi : C1, ensemble des primitives.}
    \item \textbf{D} — Aucune propriete generale ne rend le produit de deux primitives constant. \emph{Renvoi : C1, ensemble des primitives.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — La derivee de $5x+C$ vaut $5$, et non $5(5x+C)$ en general. \emph{Renvoi : C1, equation differentielle y'=ay.}
    \item \textbf{C} — $C\mathrm e^{-5x}$ satisfait $y'=-5y$, pas $y'=5y$. \emph{Renvoi : C1, equation differentielle y'=ay.}
    \item \textbf{D} — Le parametre arbitraire doit multiplier $\mathrm e^{5x}$ ; il ne remplace pas le coefficient $5$ dans l'exposant. \emph{Renvoi : C1, equation differentielle y'=ay.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{B} — En posant $y_0=-4$, le membre de droite vaut $3(-4)-12=-24$, alors que la derivee d'une constante vaut $0$. \emph{Renvoi : C1, solution constante.}
    \item \textbf{C} — La constante $12$ annule seulement le terme soustrait ; elle ne verifie pas $3y_0-12=0$. \emph{Renvoi : C1, solution constante.}
    \item \textbf{D} — Le produit $3\times(-12)$ n'est pas la condition d'equilibre ; il faut resoudre $3y_0-12=0$. \emph{Renvoi : C1, solution constante.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — La valeur initiale $y_p(0)$ n'etablit pas que $y_p$ satisfait l'equation differentielle sur tout l'intervalle. \emph{Renvoi : C1, solution particuliere.}
    \item \textbf{C} — Deriver $f$ n'est pas la verification demandee ; il faut calculer $y_p'$ puis comparer a $ay_p+f$. \emph{Renvoi : C1, solution particuliere.}
    \item \textbf{D} — Resoudre $y_p=0$ ne verifie pas l'equation ; une solution particuliere peut ne jamais s'annuler. \emph{Renvoi : C1, solution particuliere.}
  \end{itemize}
\end{itemize}
