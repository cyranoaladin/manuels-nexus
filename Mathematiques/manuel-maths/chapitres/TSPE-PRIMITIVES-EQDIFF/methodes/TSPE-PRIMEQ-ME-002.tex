\begin{fichemethode}{M2}{Résoudre une équation différentielle $y'=ay+b$ ou $y'=ay+f$}

\quandUtiliser{%
  Utiliser cette méthode pour résoudre une équation différentielle linéaire du premier ordre, puis déterminer la solution verifiant une condition initiale.
}

\pasApas{%
  \begin{enumerate}
    \item \textbf{Identifier} $a$ (coefficient de $y$) et le second membre ($b$ constant, ou $f$ fonction).
    \item \textbf{Trouver ou vérifier une solution particulière} $y_p$ : constante si second membre constant ($y_p=-b/a$), ou vérifier la solution donnee dans l'énoncé si second membre est une fonction.
    \item \textbf{Écrire la solution generale} $y(x) = C\mathrm{e}^{ax} + y_p(x)$.
    \item \textbf{Utiliser la condition initiale} (par exemple $y(x_0)=y_0$) pour déterminer $C$.
  \end{enumerate}
}

\end{fichemethode}
