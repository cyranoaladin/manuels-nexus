\begin{fichemethode}{M1}{Calculer une primitive}

\quandUtiliser{%
  Utiliser cette methode pour trouver une primitive d'une fonction donnee.
}

\pasApas{%
  \begin{enumerate}
    \item \textbf{Reconnaitre} si $f$ est une fonction de reference (table du cours).
    \item \textbf{Sinon, chercher une forme composee} : identifier $u$ et verifier si $f = u' \times v(u)$ pour un $v$ dont on connait une primitive ($\mathrm{e}^u$, $\ln|u|$, $u^{n+1}/(n+1)$...).
    \item \textbf{Ajuster les constantes multiplicatives} si necessaire (facteur devant $u'$).
    \item \textbf{Verifier} systematiquement en derivant la primitive trouvee : on doit retrouver $f$.
  \end{enumerate}
}

\end{fichemethode}
