\begin{exercice}{TSPE-LOG-RE-C1-EX1}{1}{10}
Resoudre $\ln(3x+1) = 2$.
\end{exercice}

\begin{corrige}{TSPE-LOG-RE-C1-EX1}
Domaine : $3x+1>0$, soit $x > -\dfrac{1}{3}$.

$\ln(3x+1)=2 \iff 3x+1 = \mathrm{e}^2 \iff x = \dfrac{\mathrm{e}^2-1}{3}$.

Cette valeur verifie le domaine : $S = \left\{\dfrac{\mathrm{e}^2-1}{3}\right\}$.
\end{corrige}
