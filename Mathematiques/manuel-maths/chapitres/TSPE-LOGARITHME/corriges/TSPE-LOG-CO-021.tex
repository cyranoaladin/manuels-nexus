\begin{corrige}{TSPE-LOG-EX-005}

\textbf{a.} $f = \ln(u)$ avec $u(x)=x^2+1$, toujours strictement positif : $D_f = \mathbb{R}$. $u'(x)=2x$, donc $f'(x) = \dfrac{u'(x)}{u(x)} = \dfrac{2x}{x^2+1}$.

\textbf{b.} $g$ definie pour $x>0$ : $D_g = ]0,+\infty[$. $g'(x) = 3 \times \dfrac{1}{x} - 1 = \dfrac{3}{x} - 1$.

\end{corrige}
