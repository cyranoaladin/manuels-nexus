\begin{corrige}{TSPE-LOG-EX-006}

\textbf{Q1.} $f'(x) = 1 - \dfrac{1}{x}$.

\textbf{Q2.} $f'(x) = 0 \iff \dfrac{1}{x} = 1 \iff x = 1$.

Pour $0 < x < 1$ : $\dfrac{1}{x} > 1$, donc $f'(x) = 1 - \dfrac{1}{x} < 0$.

Pour $x > 1$ : $\dfrac{1}{x} < 1$, donc $f'(x) > 0$.

\textbf{Q3.} $f$ est decroissante sur $]0,1]$ et croissante sur $[1,+\infty[$ : $f$ admet un minimum en $x=1$, qui vaut $f(1) = 1 - \ln 1 = 1 - 0 = 1$.

\end{corrige}
