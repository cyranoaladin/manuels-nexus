\begin{corrige}{TSPE-LOG-EX-001}

\textbf{Domaine.} Il faut $x>0$ et $x-2>0$, soit $x>2$.

\textbf{Resolution.} Sur ce domaine, $\ln(x)+\ln(x-2) = \ln(x(x-2))$ (equation fonctionnelle). L'equation devient :
\[
  \ln(x(x-2)) = \ln 3 \iff x(x-2) = 3 \iff x^2 - 2x - 3 = 0.
\]

Discriminant $\Delta = 4+12=16$, racines $x = \dfrac{2 \pm 4}{2}$, soit $x=3$ ou $x=-1$.

Seul $x=3$ verifie $x>2$ : $\boxed{S=\{3\}}$.

\end{corrige}
