\begin{exercice}{TSPE-LOG-EX-006}{2}{14}
Soit $f(x) = x - \ln x$ sur $]0, +\infty[$.
\begin{enumerate}
  \item Calculer $f'(x)$.
  \item Resoudre $f'(x)=0$, puis etudier le signe de $f'(x)$ sur $]0,+\infty[$.
  \item En deduire le tableau de variations de $f$, et donner le minimum de $f$ sur $]0,+\infty[$.
\end{enumerate}
\end{exercice}
