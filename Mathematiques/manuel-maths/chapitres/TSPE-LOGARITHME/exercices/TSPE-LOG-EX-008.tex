\begin{exercice}{TSPE-LOG-EX-008}{3}{20}
On considere la fonction $f(x) = x\ln x$ sur $]0, +\infty[$, prolongee par $f(0)=0$.
\begin{enumerate}
  \item Justifier que $\displaystyle\lim_{x \to 0^+} f(x) = 0$ (le prolongement en $0$ est donc continu).
  \item Determiner $\displaystyle\lim_{x \to +\infty} f(x)$.
  \item Calculer $f'(x)$ pour $x>0$, et resoudre $f'(x)=0$.
  \item Etudier le signe de $f'(x)$ et dresser le tableau de variations de $f$ sur $]0,+\infty[$.
  \item En deduire la valeur minimale de $f$ sur $]0,+\infty[$ (valeur exacte).
\end{enumerate}
\end{exercice}
