\section*{\textcolor{chapcolor}{\MakeUppercase{QCM d'auto-évaluation — Fonction logarithme népérien}}}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q1]} $\ln(2x)$ (pour $x>0$) est égal a :
  \begin{enumerate}[label=\Alph*.]
    \item $\ln 2 + \ln x$.
    \item $2\ln x$.
    \item $\ln 2 \times \ln x$.
    \item $(\ln x)^2$.
  \end{enumerate}

\item[] \textbf{Capacité C3}

\item \textbf{[Q2]} La dérivée de $\ln(5x+1)$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $\dfrac{5}{5x+1}$.
    \item $\dfrac{1}{5x+1}$.
    \item $\dfrac{5x+1}{5}$.
    \item $5\ln(5x+1)$.
  \end{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q3]} $\displaystyle\lim_{x \to +\infty} \ln x$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $0$.
    \item $+\infty$.
    \item $-\infty$.
    \item $1$.
  \end{enumerate}

\item[] \textbf{Capacité C4}

\item \textbf{[Q4]} $\displaystyle\lim_{x \to 0^+} x\ln x$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $-\infty$.
    \item $1$.
    \item $0$.
    \item Cette limite n'existe pas.
  \end{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q5]} Pour résoudre $\mathrm{e}^{2x} = 7$, on applique aux deux membres :
  \begin{enumerate}[label=\Alph*.]
    \item la fonction carre.
    \item l'inverse.
    \item la fonction exponentielle a nouveau.
    \item la fonction $\ln$.
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des réponses erronées}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{B} — Le coefficient $2$ a été traite comme un exposant de $x$ ; la règle correcte pour le produit est $\ln(2x)=\ln 2+\ln x$.
    \item \textbf{C} — Le logarithme d'un produit devient une somme de logarithmes, pas leur produit : $\ln(ab)=\ln a+\ln b$.
    \item \textbf{D} — Le carre $(\ln x)^2$ ne provient d'aucune propriété du produit $2x$ ; il confond le facteur $2$ avec une puissance de $\ln x$.
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{B} — La formule $(\ln u)'=u'/u$ exige le numérateur $u'=5$ ; choisir $1$ oublie la dérivée de la fonction affine interieure.
    \item \textbf{C} — Le quotient a été inverse : pour $u=5x+1$, on doit calculer $u'/u=5/(5x+1)$, et non $u/u'$.
    \item \textbf{D} — Multiplier le logarithme initial par $5$ ne le derive pas ; la règle de chaine donne le quotient $5/(5x+1)$.
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — La valeur $0$ est celle de $\ln 1$, pas une limite a l'infini ; comme $\ln$ est croissante et non majorée, elle tend vers $+\infty$.
    \item \textbf{C} — La limite $-\infty$ correspond a $x\to0^+$ ; a l'infini, la croissance de $\ln x$ conduit au contraire vers $+\infty$.
    \item \textbf{D} — La valeur $1$ est $\ln(\mathrm e)$, une valeur ponctuelle ; elle ne décrit pas le comportement non borne de $\ln x$ quand $x\to+\infty$.
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — Cette reponse ne retient que $\ln x\to-\infty$ et ignore le facteur $x\to0$ ; la comparaison des croissances donne $x\ln x\to0$.
    \item \textbf{B} — La valeur $1$ ne resulte pas du produit $x\ln x$ ; en posant $x=\mathrm e^{-t}$, on obtient $-t\mathrm e^{-t}\to0$.
    \item \textbf{D} — La forme $0\times(-\infty)$ est indeterminee mais pas sans limite ; une transformation ou un résultat de croissance comparee donne la limite $0$.
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — La fonction carre n'est pas la réciproque de l'exponentielle et transforme l'équation en $\mathrm e^{4x}=49$ sans isoler $x$.
    \item \textbf{B} — Prendre les inverses donne $\mathrm e^{-2x}=1/7$, une équation équivalente mais qui n'isole pas $x$ ; il faut ensuite appliquer $\ln$.
    \item \textbf{C} — Appliquer encore l'exponentielle compose une seconde fois au lieu d'inverser la première ; seul $\ln$ donne $2x=\ln 7$.
  \end{itemize}
\end{itemize}
