\subsection*{C1/C3 — Propriétés algébriques et dérivée}

\begin{enumerate}

\item \textbf{[Q1]} $\ln(2x)$ (pour $x>0$) est egal a :
  \begin{enumerate}[label=\Alph*.]
    \item $2\ln x$.
    \item $\ln 2 + \ln x$.
    \item $\ln 2 \times \ln x$.
    \item $(\ln x)^2$.
  \end{enumerate}

\bigskip

\item \textbf{[Q2]} La derivee de $\ln(5x+1)$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $\dfrac{1}{5x+1}$.
    \item $\dfrac{5}{5x+1}$.
    \item $\dfrac{5x+1}{5}$.
    \item $5\ln(5x+1)$.
  \end{enumerate}

\bigskip

\item \textbf{[Q3]} $\displaystyle\lim_{x \to +\infty} \ln x$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $0$.
    \item $-\infty$.
    \item $+\infty$.
    \item $1$.
  \end{enumerate}

\bigskip

\subsection*{C4 — Croissances comparées}

\item \textbf{[Q4]} $\displaystyle\lim_{x \to 0^+} x\ln x$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $-\infty$.
    \item $0$.
    \item $1$.
    \item Cette limite n'existe pas.
  \end{enumerate}

\bigskip

\subsection*{C2 — Résolution de problèmes}

\item \textbf{[Q5]} Pour resoudre $\mathrm{e}^{2x} = 7$, on applique aux deux membres :
  \begin{enumerate}[label=\Alph*.]
    \item la fonction carre.
    \item la fonction $\ln$.
    \item la fonction exponentielle a nouveau.
    \item l'inverse.
  \end{enumerate}

\end{enumerate}
