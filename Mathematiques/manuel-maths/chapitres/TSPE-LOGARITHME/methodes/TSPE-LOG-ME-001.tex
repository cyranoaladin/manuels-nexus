\begin{fichemethode}{M1}{Résoudre une équation ou inéquation avec $\ln$ ou $\mathrm{e}$}

\quandUtiliser{%
  Utiliser cette methode pour resoudre une equation ou inequation faisant intervenir $\ln$ ou $\mathrm{e}$.
}

\pasApas{%
  \begin{enumerate}
    \item \textbf{Determiner le domaine} : identifier les conditions de positivite sur les arguments des $\ln$.
    \item \textbf{Isoler} le terme en $\ln$ ou en $\mathrm{e}$ si necessaire (regrouper via l'equation fonctionnelle si plusieurs termes).
    \item \textbf{Appliquer} $\exp$ (si l'equation porte sur $\ln$) ou $\ln$ (si l'equation porte sur $\mathrm{e}$) aux deux membres — en verifiant leur signe pour une inequation.
    \item \textbf{Resoudre} l'equation/inequation algebrique obtenue.
    \item \textbf{Verifier} que les solutions trouvees respectent le domaine determine a l'etape 1.
  \end{enumerate}
}

\end{fichemethode}
