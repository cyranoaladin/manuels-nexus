\begin{fichemethode}{M1}{Résoudre une équation ou inéquation avec $\ln$ ou $\mathrm{e}$}

\quandUtiliser{%
  Utiliser cette méthode pour résoudre une équation ou inéquation faisant intervenir $\ln$ ou $\mathrm{e}$.
}

\pasApas{%
  \begin{enumerate}
    \item \textbf{Déterminer le domaine} : identifier les conditions de positivité sur les arguments des $\ln$.
    \item \textbf{Isoler} le terme en $\ln$ ou en $\mathrm{e}$ si nécessaire (regrouper via l'équation fonctionnelle si plusieurs termes).
    \item \textbf{Appliquer} $\exp$ (si l'équation porte sur $\ln$) ou $\ln$ (si l'équation porte sur $\mathrm{e}$) aux deux membres — en verifiant leur signe pour une inéquation.
    \item \textbf{Résoudre} l'équation/inéquation algébrique obtenue.
    \item \textbf{Vérifier} que les solutions trouvees respectent le domaine déterminé a l'étape 1.
  \end{enumerate}
}

\end{fichemethode}
