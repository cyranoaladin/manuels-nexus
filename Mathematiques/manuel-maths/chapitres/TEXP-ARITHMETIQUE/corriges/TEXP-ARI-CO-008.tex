\begin{corrige}{TEXP-ARI-EX-008}
\textbf{1.} Un diviseur commun $d$ de $n$ et $n+1$ divise leur différence
$(n+1)-n=1$, donc $d=1$ : $\mathrm{PGCD}(n\,;\,n+1)=1$.

\textbf{2.} Un diviseur commun de $n$ et $n+2$ divise $2$, donc vaut $1$ ou
$2$. Si $n$ est pair, $2$ divise les deux et
$\mathrm{PGCD}(n\,;\,n+2)=2$ ; si $n$ est impair, $\mathrm{PGCD}(n\,;\,n+2)=1$.

\textbf{3.} Soit $d$ un diviseur commun de $2n+1$ et $3n+2$. Alors $d$ divise
$3(2n+1)-2(3n+2)=-1$, donc $d=1$ : $\mathrm{PGCD}(2n+1\,;\,3n+2)=1$ pour tout
$n$.
\end{corrige}
