\begin{corrige}{TEXP-ARI-EX-022}
\textbf{1.} De $au+bv=1$ on tire $acu+bcv=c$. Or $a$ divise $acu$, et $a$
divise $bc$ donc $a$ divise $bcv$. Par somme, $a$ divise $c$.

\textbf{2.} Prenons $a=6$, $b=4$, $c=3$ : $a$ divise $bc=12$, mais $a$ ne
divise pas $c=3$. Ici $\mathrm{PGCD}(6\,;\,4)=2\neq1$ : l'hypothèse est bien
nécessaire.

\textbf{3.} Soit $p$ premier divisant $bc$. Si $p$ ne divise pas $b$, alors
$\mathrm{PGCD}(p\,;\,b)=1$ (les seuls diviseurs de $p$ sont $1$ et $p$). Le
théorème de Gauss donne alors $p\mid c$.
\end{corrige}
