\begin{corrige}{TEXP-ARI-EX-050}

\textbf{Question 1.} $f''(x) = 12(x-1)^2 \geq 0$ : $f$ est convexe (non strictement).

\commentaireMarge{$f''=12(x-1)^2 \geq 0$ s'annule en $1$ sans changer de signe : pas d'inflexion.}

$f''(1) = 0$ mais $f''$ \textbf{ne change pas de signe} : il n'y a \textbf{pas} de point d'inflexion.

\textbf{Question 2.} $f(1) = 0$ et $f'(1) = 0$ : la tangente est $y = 0$. Comme $f$ est convexe, $f(x) \geq 0$ : la courbe est au-dessus de la tangente.

\textbf{Question 3.} L'annulation de $f''$ sans changement de signe n'indique pas une inflexion mais un << aplatissement >> de la convexite. La courbe reste convexe et reste au-dessus de sa tangente, avec un contact d'ordre superieur (ici d'ordre 4).

\end{corrige}
