\begin{corrige}{TEXP-ARI-EX-023}
\textbf{1.} $p_i$ divise le produit $p_1\cdots p_r$. S'il divisait $N$, il
diviserait la différence $N-p_1\cdots p_r=1$, ce qui est impossible pour
$p_i\geq2$.

\textbf{2.} $N\geq3$ admet au moins un diviseur premier, qui ne peut être
aucun des $p_i$ : la liste $p_1,\dots,p_r$ n'était donc pas exhaustive.
L'ensemble des nombres premiers est infini.

\textbf{3.} Avec $2,3,5,7$ : $N=2\times3\times5\times7+1=211$, qui est premier
($\sqrt{211}<15$ et aucun de $2,3,5,7,11,13$ ne le divise). Attention : $N$
n'est pas toujours premier, par exemple
$2\times3\times5\times7\times11\times13+1=30031=59\times509$.
\end{corrige}
