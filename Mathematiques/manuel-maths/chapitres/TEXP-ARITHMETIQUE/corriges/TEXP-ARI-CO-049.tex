\begin{corrige}{TEXP-ARI-EX-049}

Posons $g(x) = f(x) - f(a) - f'(a)(x - a)$.

\commentaireMarge{Poser $g$, étudier $g'$ et $g''$.}

\textbf{Initialisation.} $g(a) = 0$ et $g'(x) = f'(x) - f'(a)$, donc $g'(a) = 0$. De plus $g''(x) = f''(x) \geq 0$.

\textbf{Cas $x \geq a$.} Comme $g'' \geq 0$, $g'$ est croissante. Pour $x \geq a$, $g'(x) \geq g'(a) = 0$, donc $g$ est croissante sur $[a, x]$, et $g(x) \geq g(a) = 0$.

\textbf{Cas $x \leq a$.} Pour $x \leq a$, $g'(x) \leq g'(a) = 0$, donc $g$ est decroissante sur $[x, a]$. Ainsi $g(x) \geq g(a) = 0$.

\textbf{Conclusion.} Dans les deux cas, $g(x) \geq 0$, soit $f(x) \geq f(a) + f'(a)(x - a)$. $\blacksquare$

\end{corrige}
