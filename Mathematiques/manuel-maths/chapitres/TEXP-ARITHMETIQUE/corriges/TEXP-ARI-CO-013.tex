\begin{corrige}{TEXP-ARI-EX-013}
\textbf{1.} Si $au\equiv1\ [n]$, il existe $v$ tel que $au-nv=1$. Tout
diviseur commun de $a$ et $n$ divise alors $1$, donc
$\mathrm{PGCD}(a\,;\,n)=1$.

\textbf{2.} Si $\mathrm{PGCD}(a\,;\,n)=1$, le théorème de Bézout fournit $u,v$
tels que $au+nv=1$, d'où $au\equiv1\ [n]$ : $u$ est un inverse de $a$.

\textbf{3.} Modulo $12$, les inversibles sont les entiers premiers avec $12$ :
$1,5,7,11$, soit $4$. Modulo $13$, qui est premier, tous les entiers de $1$ à
$12$ sont inversibles, soit $12$. Un modulo premier maximise le nombre
d'inversibles : c'est ce qui rend les modulos premiers commodes en
cryptographie.
\end{corrige}
