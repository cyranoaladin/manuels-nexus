\begin{corrige}{TEXP-ARI-EX-025}
\textbf{1.} Si $p$ est premier et $p\nmid a$, alors $a^{p-1}\equiv1\ [p]$.

\textbf{2.} $7$ est premier et $7\nmid3$, donc $3^6\equiv1\ [7]$. Or
$100=6\times16+4$, d'où $3^{100}\equiv3^4=81\equiv4\ [7]$.

\textbf{3.} $13$ est premier et $13\nmid2$, donc $2^{12}\equiv1\ [13]$. Comme
$2026=12\times168+10$, on a $2^{2026}\equiv2^{10}=1024\equiv10\ [13]$ (car
$1024=78\times13+10$).
\end{corrige}
