\begin{corrige}{TEXP-ARI-EX-019}
\textbf{1.} Avec $x$ billets à $23$ et $y$ billets à $17$ :
$23x+17y=1000$. Comme $\mathrm{PGCD}(23\,;\,17)=1$ divise $1\,000$, il y a des
solutions entières.

\textbf{2.} $23=1\times17+6$, $17=2\times6+5$, $6=1\times5+1$, d'où
$1=6-5=6-(17-2\times6)=3\times6-17=3(23-17)-17=3\times23-4\times17$.
En multipliant par $1\,000$ : $x_0=3000$, $y_0=-4000$. Les solutions sont
$x=3000-17k$, $y=-4000+23k$, $k\in\mathbb{Z}$.

\textbf{3.} $x\geq0$ donne $k\leq176{,}4$ et $y\geq0$ donne $k\geq173{,}9$ :
$k\in\{174,175,176\}$, soit $(x\,;\,y)\in\{(42\,;\,2),(25\,;\,25),(8\,;\,48)\}$.
Vérification : $23\times42+17\times2=1000$.
\end{corrige}
