\begin{corrige}{TEXP-ARI-EX-032}

\textbf{Question 1.} $f'(x) = (1-x)\,\mathrm{e}^{-x}$ et $f''(x) = (x-2)\,\mathrm{e}^{-x}$.

\commentaireMarge{$f''=(x-2)\mathrm{e}^{-x}$ change de signe en $x=2$ : inflexion.}

$f'$ s'annule en $x=1$ (max local $f(1)=\mathrm{e}^{-1}$). $\lim_{x\to+\infty} f = 0$ (asymptote horizontale $y=0$).

\textbf{Question 2.} $f''(x) = 0 \iff x = 2$. Point d'inflexion en $(2,\, 2\mathrm{e}^{-2})$.

Convexe sur $\left]-\infty, 2\right]$, concave sur $\left[2, +\infty\right[$.

\textbf{Question 3.} La courbe part de $0$, monte jusqu'a $\mathrm{e}^{-1}$, descend en s'incurvant vers l'asymptote $y=0$.

\end{corrige}
