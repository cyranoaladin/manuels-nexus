\begin{fichemethode}{M2}{Resoudre ax = b [n] avec un inverse modulaire}

\quandUtiliser{%
  Utiliser cette methode lorsque l'enonce demande de resoudre une congruence
  $ax \equiv b\ [n]$ et que $a$ et $n$ sont premiers entre eux : on determine
  un inverse de $a$ modulo $n$ puis on l'applique.
}

\pasApas{%
  \begin{enumerate}
    \item Verifier $\mathrm{PGCD}(a\,;\,n) = 1$ : c'est la condition d'existence d'un
      inverse de $a$ modulo $n$.
    \item Chercher l'inverse $u$ de $a$ : un entier tel que
      $au \equiv 1\ [n]$ (essais sur les restes $1, 2, \dots, n-1$, ou
      remontee de l'algorithme d'Euclide pour de grands $n$).
    \item Multiplier la congruence par $u$ :
      $ax \equiv b\ [n] \iff x \equiv ub\ [n]$.
    \item Reduire $ub$ modulo $n$ pour donner le representant dans
      $\{0, \dots, n-1\}$.
    \item Conclure : les solutions sont les entiers
      $x \equiv x_0\ [n]$, et controler en substituant.
  \end{enumerate}
}

\exempleRedige{%
  Resoudre $5x \equiv 3\ [7]$.

  \medskip
  \commentaireMarge{$5$ et $7$ premiers entre eux : inverse garanti.}%
  $\mathrm{PGCD}(5\,;\,7) = 1$. On cherche $u$ tel que $5u \equiv 1\ [7]$ :
  $5 \times 3 = 15 = 2 \times 7 + 1$, donc $u = 3$ convient.

  \medskip
  \commentaireMarge{Multiplier les deux membres par l'inverse.}%
  $5x \equiv 3\ [7] \iff x \equiv 3 \times 3 \equiv 9 \equiv 2\ [7]$.

  \medskip
  Les solutions sont les entiers $x \equiv 2\ [7]$. Controle :
  $5 \times 2 = 10 \equiv 3\ [7]$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piege 1.} « Diviser » une congruence : on ne divise jamais,
      on MULTIPLIE par un inverse (qui n'existe que si $\mathrm{PGCD}(a\,;\,n)=1$).
    \item \textbf{Piege 2.} Oublier de reduire le resultat modulo $n$ (donner 9
      au lieu de 2 modulo 7).
    \item \textbf{Piege 3.} Conclure a une solution UNIQUE : l'ensemble des
      solutions est une classe de congruence entiere.
  \end{itemize}
}

\verifier{%
  Substituer le representant trouve dans la congruence initiale, et controler
  l'inverse par le produit $au$ reduit modulo $n$ (il doit donner 1).
}

\end{fichemethode}
