\begin{fichemethode}{M8}{Utiliser le petit théorème de Fermat pour reduire de grandes puissances}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande le reste d'une GRANDE
  puissance $a^n$ modulo un nombre PREMIER $p$ : le petit théorème de Fermat
  reduit l'exposant modulo $p - 1$.
}

\pasApas{%
  \begin{enumerate}
    \item Enoncer le théorème : si $p$ est premier et $p$ ne divise pas $a$,
      alors $a^{p-1} \equiv 1\ [p]$. (Variante : $a^p \equiv a\ [p]$ pour tout
      entier $a$.)
    \item Vérifier les hypothèses : $p$ premier, et $a$ non multiple de $p$.
    \item Écrire la division euclidienne de l'exposant par $p - 1$ :
      $n = (p-1)q + r$ avec $0 \leq r < p-1$.
    \item Reduire : $a^n = \left(a^{p-1}\right)^q \times a^r
      \equiv 1^q \times a^r \equiv a^r\ [p]$.
    \item Calculer $a^r$ modulo $p$ (petit exposant restant) et conclure par le
      reste dans $\{0, \dots, p-1\}$.
  \end{enumerate}
}

\exempleRedige{%
  Déterminer le reste de $7^{222}$ dans la division par 11.

  \medskip
  \commentaireMarge{Hypothèses : 11 premier, $11 \nmid 7$.}%
  11 est premier et ne divise pas 7, donc par le petit théorème de Fermat,
  $7^{10} \equiv 1\ [11]$.

  \medskip
  \commentaireMarge{Reduire l'exposant modulo $p-1 = 10$.}%
  $222 = 22 \times 10 + 2$, donc
  $7^{222} = \left(7^{10}\right)^{22} \times 7^{2} \equiv 1^{22} \times 49
  \equiv 49 \equiv 5\ [11]$.

  \medskip
  Le reste de $7^{222}$ par 11 est 5.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Reduire l'exposant modulo $p$ au lieu de $p - 1$.
    \item \textbf{Piège 2.} Appliquer $a^{p-1} \equiv 1\ [p]$ quand $p \mid a$
      (faux : utiliser alors $a^n \equiv 0\ [p]$) ou quand $p$ n'est pas
      premier.
    \item \textbf{Piège 3.} Oublier le facteur residuel $a^r$ après la
      reduction, ou rendre un « reste » superieur a $p - 1$.
  \end{itemize}
}

\verifier{%
  Contrôler sur un petit cas calculable directement (ex. $7^2 = 49 \equiv 5\
  [11]$), et vérifier que le reste annonce est dans $\{0, \dots, p-1\}$.
}
