\begin{fichemethode}{M1}{Determiner les diviseurs d'un entier et le PGCD par l'algorithme d'Euclide}

\quandUtiliser{%
  Utiliser cette methode lorsque l'enonce demande la liste des diviseurs d'un
  entier, ou le PGCD de deux entiers par divisions euclidiennes successives.
}

\pasApas{%
  \begin{enumerate}
    \item Diviseurs de $n$ : tester les entiers $d$ de 1 a $\sqrt{n}$ ; chaque
      diviseur $d$ trouve donne le diviseur associe $\dfrac{n}{d}$ (les
      diviseurs vont par paires).
    \item PGCD de $a$ et $b$ ($a > b$) : effectuer la division euclidienne
      $a = bq + r$ avec $0 \leq r < b$.
    \item Iterer sur le couple $(b\,;\,r)$ : le PGCD est inchange car
      $\mathrm{PGCD}(a\,;\,b) = \mathrm{PGCD}(b\,;\,r)$.
    \item S'arreter quand le reste est NUL : le PGCD est le DERNIER RESTE NON
      NUL.
    \item Controler que le resultat divise bien $a$ et $b$.
  \end{enumerate}
}

\exempleRedige{%
  Calculer $\mathrm{PGCD}(252\,;\,198)$ par l'algorithme d'Euclide.

  \medskip
  \commentaireMarge{Chaque ligne : dividende $=$ quotient $\times$ diviseur $+$
  reste.}%
  $252 = 1 \times 198 + 54$ ;\quad
  $198 = 3 \times 54 + 36$ ;\quad
  $54 = 1 \times 36 + 18$ ;\quad
  $36 = 2 \times 18 + 0$.

  \medskip
  Le dernier reste non nul est 18 : $\mathrm{PGCD}(252\,;\,198) = 18$.
  Controle : $252 = 18 \times 14$ et $198 = 18 \times 11$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piege 1.} Prendre le dernier QUOTIENT ou le dernier reste
      (nul) au lieu du dernier reste NON NUL.
    \item \textbf{Piege 2.} Pour les diviseurs, s'arreter avant $\sqrt{n}$ ou
      oublier les diviseurs associes $\frac{n}{d}$.
    \item \textbf{Piege 3.} Poursuivre l'algorithme avec le couple (quotient,
      reste) au lieu de (diviseur, reste).
  \end{itemize}
}

\verifier{%
  Verifier que le PGCD trouve divise les deux entiers de depart, et que les
  quotients $\frac{a}{d}$ et $\frac{b}{d}$ obtenus sont premiers entre eux.
}

\sEntrainer{\refExos{M1}}

\end{fichemethode}
