\begin{fichemethode}{M9}{Programmer Euclide, Bézout, le crible d'Ératosthène et la décomposition}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande d'écrire (ou de compléter) en
  Python l'algorithme d'Euclide, le calcul d'un couple de Bézout, le crible
  d'Ératosthène ou la décomposition en facteurs premiers.
}

\pasApas{%
  \begin{enumerate}
    \item Euclide : pour deux entiers non tous deux nuls, prendre d'abord
      leurs valeurs absolues. Tant que \texttt{b != 0}, remplacer
      \texttt{(a, b)} par \texttt{(b, a \% b)} ; le PGCD est le dernier
      \texttt{a}, strictement positif.
    \item Bézout : pour deux entiers non tous deux nuls, si $b=0$, renvoyer
      $(|a|,\operatorname{sgn}(a),0)$, où $\operatorname{sgn}(a)$ vaut
      $1$ si $a>0$ et $-1$ si $a<0$. Sinon, calculer
      $q=\lfloor a/b\rfloor$ et $r=a-qb$, obtenir $(d,u,v)$ pour $(b,r)$,
      puis renvoyer $(d,v,u-qv)$ : en effet,
      $bu+rv=av+b(u-qv)=d$. En Python, \texttt{q = a // b} et
      \texttt{r = a \% b} ; même si $b<0$, on a $|r|<|b|$, ce qui assure
      la terminaison de la récursion. Le reste Python peut alors être négatif.
    \item Crible : pour un entier $n<2$, renvoyer la liste vide. Sinon,
      initialiser le tableau de booléens à Vrai (sauf 0 et 1) ; pour
      chaque premier $p\leq\sqrt{n}$, barrer les multiples de $p$ à partir
      de $p^2$.
    \item Décomposition : pour un entier $n\geq1$, diviser par les candidats
      croissants tant que $p^2\leq n$ ; à la fin, si $n>1$, c'est un dernier
      facteur premier. Pour $n=1$, le dictionnaire vide représente le produit
      vide, égal à 1. Les entrées $n\leq0$ sont hors de ce domaine.
    \item Tester aussi les cas limites : une entrée nulle et l'autre négative
      pour Euclide et Bézout ; $n=0,1,2$ pour le crible ; $n=1$ et un carré
      de nombre premier pour la décomposition.
  \end{enumerate}
}

\exempleRedige{%
  Écrire l'algorithme d'Euclide et l'appliquer à $(252\,;\,198)$.

  \medskip
  \commentaireMarge{L'invariant de boucle : le PGCD du couple est conservé.}%
\lstinputlisting[language=Python]{chapitres/TEXP-ARITHMETIQUE/python/euclide.py}

  \medskip
  \texttt{euclide(252, 198)} renvoie 18, conforme au calcul manuel
  (fiche M1). Le couple \texttt{(a, b)} devient successivement
  $(198\,;\,54)$, $(54\,;\,36)$, $(36\,;\,18)$, $(18\,;\,0)$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Confondre \texttt{a \% b} (reste) et
      \texttt{a // b} (quotient) — les deux interviennent dans Bézout. Pour un diviseur négatif, le reste Python est négatif ou nul ; le reste euclidien est, lui, toujours non négatif.
    \item \textbf{Piège 2.} Dans le crible, barrer à partir de $2p$ au lieu de
      $p^2$ (correct mais inutilement coûteux) ou oublier la borne
      $\sqrt{n}$.
    \item \textbf{Piège 3.} Dans la décomposition, oublier le facteur final
      $n > 1$ après la boucle (ex. le 5 de 360, ou un grand premier).
  \end{itemize}
}

\verifier{%
  Confronter chaque sortie à un calcul manuel : PGCD par divisions posées,
  égalité $au + bv = d$ effectivement vérifiée, liste des premiers jusqu'à 30,
  produit de la décomposition égal à l'entier initial.
}
