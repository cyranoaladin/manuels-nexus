\begin{corrige}{TEXP-ARI-EV-B-EX1}
\textbf{1.} Voir la démonstration du cours : par Bézout, $au+bv=1$ pour des entiers $u,v$ ; en multipliant par $c$, $acu+bcv=c$ ; $a$ divise $acu$ et $bcv$ (car $a\mid bc$), donc $a$ divise leur somme $c$.

\textbf{2.} $7$ divise $4\times28=112$ et $\mathrm{PGCD}(7,4)=1$ : par le théorème de Gauss, $7$ divise $28$ (ce qui se vérifie directement : $28=4\times7$).
\end{corrige}
