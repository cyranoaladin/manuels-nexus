\begin{corrige}{TEXP-ARI-EV-B-EX3}
\textbf{1.} On barre les multiples de $p$ à partir de $p^2$ : tout multiple $kp$ avec $k<p$ a déjà été barré comme multiple du facteur $k$, plus petit. Et la boucle s'arrête dès que $p^2>n$ : si $m\leq n$ est composé, son plus petit facteur premier vérifie $p\leq\sqrt m\leq\sqrt n$, il a donc déjà servi.

\begin{python}
def crible(n):
    est_premier = [True] * (n + 1)
    est_premier[0] = est_premier[1] = False
    p = 2
    while p * p <= n:
        if est_premier[p]:
            for multiple in range(p * p, n + 1, p):
                est_premier[multiple] = False
        p += 1
    return [k for k in range(n + 1) if est_premier[k]]
\end{python}

\textbf{2.} On divise par les entiers croissants tant que le carré ne dépasse pas $n$ ; ce qui reste au-delà est premier.

def decomposer(n):
    facteurs, d = [], 2
    while d * d <= n:
        while n % d == 0:
            facteurs.append(d)
            n //= d
        d += 1
    if n > 1:
        facteurs.append(n)
    return facteurs
\end{python}
