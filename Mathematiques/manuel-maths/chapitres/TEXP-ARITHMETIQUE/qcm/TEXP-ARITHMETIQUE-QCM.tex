\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} Soient a et b entiers. Si a divise b et b divise c alors :
  \begin{enumerate}[label=\Alph*.]
    \item a divise b+c uniquement
    \item a divise c
    \item c divise a
    \item a = c
  \end{enumerate}

\item \textbf{[Q2]} La division euclidienne de 25 par 7 donne pour reste :
  \begin{enumerate}[label=\Alph*.]

    \item 2
    \item 4
    \item 3
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q3]} Le PGCD de 48 et 18 est :
  \begin{enumerate}[label=\Alph*.]
    \item 6
    \item 12
    \item 3
    \item 2
  \end{enumerate}

\item \textbf{[Q4]} Le theoreme de Bezout affirme que PGCD(a,b)=1 si et seulement si :
  \begin{enumerate}[label=\Alph*.]
    \item $a+b=1$
    \item $a b = 1$
    \item $\exists u,v \in \mathbb{Z}, au-bv=0$
    \item $\exists u,v \in \mathbb{Z}, au+bv=1$
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q5]} Si $a \equiv b \pmod{n}$, alors :
  \begin{enumerate}[label=\Alph*.]
    \item $n$ divise $a-b$
    \item $a$ divise $b-n$
    \item $a-b = n$
    \item $a+b$ est multiple de $n$
  \end{enumerate}

\end{enumerate}
