\section*{\textcolor{chapcolor}{\MakeUppercase{Arithmétique}}}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q1]} Soient a, b et c entiers. Si a divise b et b divise c alors :
  \begin{enumerate}[label=\Alph*.]
    \item a divise b+c uniquement
    \item a divise c
    \item c divise a
    \item a = c
  \end{enumerate}

\item \textbf{[Q2]} La division euclidienne de 25 par 7 donne pour reste :
  \begin{enumerate}[label=\Alph*.]
    \item 4
    \item 2
    \item 1
    \item 3
  \end{enumerate}

\item \textbf{[Q3]} Le PGCD de 48 et 18 est :
  \begin{enumerate}[label=\Alph*.]
    \item 6
    \item 12
    \item 3
    \item 2
  \end{enumerate}

\item[] \textbf{Capacité C5}

\item \textbf{[Q4]} Le théorème de Bezout affirme que PGCD(a,b)=1 si et seulement si :
  \begin{enumerate}[label=\Alph*.]
    \item $a+b=1$
    \item $a b = 1$
    \item $\exists u,v \in \mathbb{Z}, au+bv=1$
    \item $\exists u,v \in \mathbb{Z}, au-bv=0$
  \end{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q5]} Si $a \equiv b \pmod{n}$, alors :
  \begin{enumerate}[label=\Alph*.]
    \item $a+b$ est multiple de $n$
    \item $a$ divise $b-n$
    \item $a-b = n$
    \item $n$ divise $a-b$
  \end{enumerate}

\end{enumerate}

  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{B} — Erreur de calcul. La division 25=3*7+4 donne le reste 4, compris entre 0 et 6. \emph{Renvoi : C1.}
    \item \textbf{C} — Quotient faux. La division 25=3*7+4 donne le reste 4, compris entre 0 et 6. \emph{Renvoi : C1.}
    \item \textbf{D} — Reste incorrect. La division 25=3*7+4 donne le reste 4, compris entre 0 et 6. \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{B} — 12 ne divise pas 18. L'algorithme d'Euclide donne 48=2*18+12, 18=1*12+6, puis le PGCD vaut 6. \emph{Renvoi : C1.}
    \item \textbf{C} — 3 n'est pas le PLUS GRAND diviseur. L'algorithme d'Euclide donne 48=2*18+12, 18=1*12+6, puis le PGCD vaut 6. \emph{Renvoi : C1.}
    \item \textbf{D} — 2 n'est pas le plus grand diviseur. L'algorithme d'Euclide donne 48=2*18+12, 18=1*12+6, puis le PGCD vaut 6. \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — Équation trop simple. Le théorème de Bézout caractérise la coprimalité par l'existence de u,v entiers tels que au+bv=1. \emph{Renvoi : C5.}
    \item \textbf{B} — Produit égal a 1. Le théorème de Bézout caractérise la coprimalité par l'existence de u,v entiers tels que au+bv=1. \emph{Renvoi : C5.}
    \item \textbf{D} — Équation homogene. Le théorème de Bézout caractérise la coprimalité par l'existence de u,v entiers tels que au+bv=1. \emph{Renvoi : C5.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — Somme au lieu de difference. La congruence a équivalent à b modulo n signifie exactement que n divise a-b. \emph{Renvoi : C2.}
    \item \textbf{B} — Sens du diviseur inverse. La congruence a équivalent à b modulo n signifie exactement que n divise a-b. \emph{Renvoi : C2.}
    \item \textbf{C} — Égalité non nécessaire. La congruence a équivalent à b modulo n signifie exactement que n divise a-b. \emph{Renvoi : C2.}
  \end{itemize}
