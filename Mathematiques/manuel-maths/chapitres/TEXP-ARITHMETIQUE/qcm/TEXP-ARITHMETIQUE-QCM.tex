\subsection*{C1 --- PGCD}

\begin{enumerate}

\item \textbf{[Q1]} L'algorithme d'Euclide calcule le PGCD de $a$ et $b$ en utilisant :
  \begin{enumerate}[label=\Alph*.]
    \item des divisions euclidiennes successives.
    \item la décomposition en facteurs premiers uniquement.
    \item une recherche exhaustive de tous les diviseurs.
    \item aucune opération arithmétique.
  \end{enumerate}

\bigskip

\subsection*{C5/C6 --- Bézout, Gauss}

\item \textbf{[Q2]} Le théorème de Bézout affirme que le PGCD de $a$ et $b$ s'écrit :
  \begin{enumerate}[label=\Alph*.]
    \item $a+b$.
    \item $ax+by$ pour des entiers $x,y$.
    \item $a\times b$.
    \item toujours $1$.
  \end{enumerate}

\bigskip

\item \textbf{[Q3]} Le théorème de Gauss dit que si $a$ divise $bc$ et $a$ est premier avec $b$, alors :
  \begin{enumerate}[label=\Alph*.]
    \item $a$ divise $b$.
    \item $a$ divise $c$.
    \item $a$ est premier avec $c$.
    \item $a=1$.
  \end{enumerate}

\bigskip

\subsection*{C4 --- Équations diophantiennes}

\item \textbf{[Q4]} L'équation $ax+by=c$ admet des solutions entières si et seulement si :
  \begin{enumerate}[label=\Alph*.]
    \item $c=0$.
    \item $\mathrm{PGCD}(a,b)$ divise $c$.
    \item $a$ et $b$ sont premiers.
    \item $c$ divise $a$ et $b$.
  \end{enumerate}
