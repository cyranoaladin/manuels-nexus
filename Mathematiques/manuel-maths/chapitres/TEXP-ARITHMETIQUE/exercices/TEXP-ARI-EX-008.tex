\begin{exercice}{TEXP-ARI-EX-008}{2}{15}
Soit $n$ un entier naturel.
\begin{enumerate}
  \item Montrer que $\mathrm{PGCD}(n\,;\,n+1)=1$ pour tout $n$.
  \item Déterminer, selon la parité de $n$, la valeur de
        $\mathrm{PGCD}(n\,;\,n+2)$.
  \item Que vaut $\mathrm{PGCD}(2n+1\,;\,3n+2)$ ? Justifier.
\end{enumerate}
\end{exercice}
