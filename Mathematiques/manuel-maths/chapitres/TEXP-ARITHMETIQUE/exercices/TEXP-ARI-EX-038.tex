\begin{exercice}{TEXP-ARI-EX-038}{2}{15}
Soit $f(x) = x^4 + x^2$.
\begin{enumerate}
  \item Montrer que $f$ est strictement convexe sur $\mathbb{R}$.
  \item En deduire que $f$ admet un unique minimum et le determiner.
  \item Resoudre $f(x) \geq 2$.
\end{enumerate}

\end{exercice}
