\begin{exercice}{TEXP-ARI-EX-016}{2}{18}

$c=(am+b)\bmod26$, avec ici $a=7$ et $b=3$.
\begin{enumerate}
  \item Chiffrer le mot \og MATH \fg.
  \item Justifier que ce chiffrement est déchiffrable et déterminer la fonction
        de déchiffrement.
  \item Déchiffrer le mot chiffré \og SVX \fg.
\end{enumerate}
