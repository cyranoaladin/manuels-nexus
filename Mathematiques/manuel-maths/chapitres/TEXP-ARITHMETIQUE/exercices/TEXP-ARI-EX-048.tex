\begin{exercice}{TEXP-ARI-EX-048}{2}{20}
Soit $f(x) = \dfrac{1}{1+x}$ sur $\left]-1, +\infty\right[$.
\begin{enumerate}
  \item Montrer que $f$ est convexe.
  \item En deduire que $\dfrac{1}{1+x} \geq 1 - x$ pour tout $x > -1$.
  \item En deduire une approximation de $\dfrac{1}{1{,}05}$.
\end{enumerate}

\end{exercice}
