\begin{exercice}{TEXP-ARI-EX-022}{2}{15}
\begin{enumerate}
  \item Soient $a$, $b$, $c$ des entiers tels que $a$ divise $bc$ et
        $\mathrm{PGCD}(a\,;\,b)=1$. En partant d'une identité de Bézout
        $au+bv=1$, démontrer que $a$ divise $c$.
  \item Montrer sur un exemple que l'hypothèse $\mathrm{PGCD}(a\,;\,b)=1$ est
        indispensable.
  \item En déduire que si un nombre premier $p$ divise un produit $bc$, alors
        $p$ divise $b$ ou $p$ divise $c$.
\end{enumerate}
\end{exercice}
