\textbf{Corrige de l'evaluation A -- TSPE-DERIVATION-CONVEXITE-EV-A}

\textbf{Exercice 1.}

\textbf{1.} $u(x) = 2x^2 + 1$, $u'(x) = 4x$. $f_1'(x) = 4x\,\mathrm{e}^{2x^2+1}$.

\textbf{2.} $u(x) = x^2 + 3 > 0$ sur $\mathbb{R}$. $f_2'(x) = \dfrac{2x}{x^2 + 3}$.

\bigskip

\textbf{Exercice 2.}

\textbf{1.} $g'(x) = 2x\,\mathrm{e}^{-x} - x^2\,\mathrm{e}^{-x} = x(2-x)\,\mathrm{e}^{-x}$. Comme $\mathrm{e}^{-x} > 0$, le signe est celui de $x(2-x)$ : $g' > 0$ sur $[0, 2]$, $g' < 0$ ailleurs.

\textbf{2.} $\lim_{+\infty} x^2\,\mathrm{e}^{-x} = 0$ (croissances comparees), $\lim_{-\infty} x^2\,\mathrm{e}^{-x} = +\infty$.

\textbf{3.} Maximum local en $x = 2$ : $g(2) = 4\mathrm{e}^{-2}$. Minimum local en $x = 0$ : $g(0) = 0$.

\bigskip

\textbf{Exercice 3.}

\textbf{1.} $h'(x) = \mathrm{e}^x - 1$, $h''(x) = \mathrm{e}^x > 0$ : $h$ est convexe sur $\mathbb{R}$.

\textbf{2.} $h'(0) = 0$ et $h(0) = 0$. Par convexite, le minimum est $h(0) = 0$, donc $h(x) \geq 0$, soit $\mathrm{e}^x \geq 1 + x$.

\textbf{3.} La tangente en $0$ est $y = 1 + x$. Comme $\mathrm{e}^x$ est convexe, la courbe est au-dessus de sa tangente.

\bigskip

\textbf{Exercice 4.}

\textbf{1.} $k'(x) = 3x^2 - 12x + 9 = 3(x-1)(x-3)$. $k''(x) = 6x - 12$.

\textbf{2.} $k''(x) > 0$ sur $]2, +\infty[$ (convexe), $k''(x) < 0$ sur $]-\infty, 2[$ (concave). Point d'inflexion en $x = 2$, $k(2) = 3$.

\textbf{3.} $k(2) = 3$ et $k'(2) = 3(1)(-1) = -3$. Tangente : $y = -3x + 9$. Comme $k''(2) = 0$ avec changement de signe, la courbe traverse sa tangente en $x = 2$.
