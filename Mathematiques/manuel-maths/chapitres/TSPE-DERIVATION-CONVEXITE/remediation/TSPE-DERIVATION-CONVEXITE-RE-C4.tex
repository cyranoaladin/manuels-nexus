\subsection*{Rappel -- Erreurs dans l'esquisse de courbe}

\textbf{Erreur 1 :} Confondre le signe de $f'$ (variations) et celui de $f''$ (convexite).

\textbf{Erreur 2 :} Considerer que $f'' = 0$ implique toujours un point d'inflexion (il faut aussi un changement de signe).

\textbf{Erreur 3 :} Oublier de placer les points d'inflexion sur l'esquisse.

\begin{exercice}{TSPE-DERCONV-RE-C4-EX1}{1}{12}
Soit $f(x) = x^4 - 6x^2$.
\begin{enumerate}
  \item Calculer $f'(x)$ et $f''(x)$.
  \item La courbe admet-elle des points d'inflexion ?
  \item Sur quels intervalles la courbe est-elle convexe ?
\end{enumerate}
\end{exercice}
