\subsection*{Rappel -- Erreurs sur les inégalités par convexité}

\textbf{Erreur 1 :} Confondre convexité et concavité. Si $f'' > 0$, $f$ est convexe (en forme de $\cup$).

\textbf{Erreur 2 :} Appliquer Jensen dans le mauvais sens pour une fonction concave.

\textbf{Erreur 3 :} Oublier de vérifier la convexité avant d'utiliser l'inégalité des tangentes.

\begin{exercice}{TSPE-DERCONV-RE-C3-EX1}{1}{12}
\begin{enumerate}
  \item Montrer que $f(x) = x^2 + \mathrm{e}^x$ est convexe sur $\mathbb{R}$.
  \item En déduire que $f\!\left(\dfrac{a+b}{2}\right) \leq \dfrac{f(a) + f(b)}{2}$.
  \item Vérifier pour $a = 0$ et $b = 1$.
\end{enumerate}
\end{exercice}
