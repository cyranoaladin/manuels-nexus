\begin{corrige}{TSPE-DERCONV-EX-046}

\textbf{Question 1.} $g(a) = f(a) - f(a) - 0 = 0$ et $g'(x) = f'(x) - f'(a)$, donc $g'(a) = 0$.

\commentaireMarge{$g(a)=0$, $g'(a)=0$, $g''=f'' \geq 0$ : $g$ convexe de minimum $0$.}

\textbf{Question 2.} $g''(x) = f''(x) \geq 0$ sur $I$ : $g$ est convexe. Comme $g'(a) = 0$ et $g$ convexe, $a$ est un minimum global, donc $g(x) \geq g(a) = 0$.

\textbf{Question 3.} Ainsi $f(x) \geq f(a) + f'(a)(x-a)$ : la courbe est au-dessus de sa tangente en $a$.

\end{corrige}
