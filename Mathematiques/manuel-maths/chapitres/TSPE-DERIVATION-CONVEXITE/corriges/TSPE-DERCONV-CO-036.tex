\begin{corrige}{TSPE-DERCONV-EX-036}

\textbf{Question 1.} $f(0) = 1$ et $f'(0) = 1$. La tangente est $y = x + 1$.

\commentaireMarge{$\mathrm{e}^x$ convexe : la courbe est au-dessus de toute tangente.}

\textbf{Question 2.} $g(x) = \mathrm{e}^x - (x+1)$. On a $g(0)=0$ et $g''(x) = \mathrm{e}^x > 0$ : $g$ est convexe, donc son minimum $g(0)=0$ est global.

Ainsi $g(x) \geq 0$ : la courbe est \textbf{au-dessus} de la tangente partout.

\end{corrige}
