\section*{\textcolor{chapcolor}{\MakeUppercase{QCM d'auto-evaluation -- Complements sur la derivation, convexite}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} La derivee de $f(x) = \mathrm{e}^{x^2}$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $\mathrm{e}^{x^2}$
    \item $2x\,\mathrm{e}^{x^2}$
    \item $x\,\mathrm{e}^{x^2}$
    \item $2\,\mathrm{e}^{x^2}$
  \end{enumerate}

\item \textbf{[Q2]} La derivee de $f(x) = \ln(x^2 + 1)$ sur $\mathbb{R}$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $\dfrac{1}{x^2 + 1}$
    \item $\dfrac{2x}{x^2 + 1}$
    \item $\dfrac{2}{x^2 + 1}$
    \item $\dfrac{x}{x^2 + 1}$
  \end{enumerate}

\item \textbf{[Q3]} La derivee de $f(x) = (3x - 2)^4$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $4(3x-2)^3$
    \item $12(3x-2)^3$
    \item $3(3x-2)^4$
    \item $12(3x-2)^4$
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q4]} Soit $f(x) = x\,\mathrm{e}^{-x}$. La derivee $f'(x)$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $\mathrm{e}^{-x} - x\,\mathrm{e}^{-x}$
    \item $\mathrm{e}^{-x} + x\,\mathrm{e}^{-x}$
    \item $-\mathrm{e}^{-x}$
    \item $x\,\mathrm{e}^{-x}$
  \end{enumerate}

\item \textbf{[Q5]} La limite de $f(x) = \dfrac{\ln x}{x}$ en $+\infty$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $+\infty$
    \item $1$
    \item $0$
    \item $-\infty$
  \end{enumerate}

\item \textbf{[Q6]} La fonction $f(x) = x^3 - 3x$ admet comme extrema locaux :
  \begin{enumerate}[label=\Alph*.]
    \item un maximum local en $x = 1$ et un minimum local en $x = -1$
    \item un maximum local en $x = -1$ et un minimum local en $x = 1$
    \item un maximum local en $x = 0$
    \item aucun extremum local
  \end{enumerate}

\item[] \textbf{Capacite C5}

\item \textbf{[Q7]} La fonction $f(x) = \mathrm{e}^x$ est :
  \begin{enumerate}[label=\Alph*.]
    \item convexe sur $\mathbb{R}$
    \item concave sur $\mathbb{R}$
    \item convexe sur $\mathbb{R}^+$ et concave sur $\mathbb{R}^-$
    \item ni convexe ni concave
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q8]} L'inegalite $\mathrm{e}^x \geq 1 + x$ est valable :
  \begin{enumerate}[label=\Alph*.]
    \item pour tout $x \geq 0$
    \item pour tout $x \leq 0$
    \item pour tout $x \in \mathbb{R}$
    \item seulement en $x = 0$
  \end{enumerate}

\item[] \textbf{Capacite C5}

\item \textbf{[Q9]} La fonction $f(x) = \ln x$ est :
  \begin{enumerate}[label=\Alph*.]
    \item convexe sur $]0, +\infty[$
    \item concave sur $]0, +\infty[$
    \item convexe puis concave
    \item ni convexe ni concave
  \end{enumerate}

\item[] \textbf{Capacite C4}

\item \textbf{[Q10]} Si $f''(x) > 0$ sur un intervalle $I$, alors la courbe de $f$ est :
  \begin{enumerate}[label=\Alph*.]
    \item convexe (en forme de $\cup$)
    \item concave (en forme de $\cap$)
    \item croissante
    \item decroissante
  \end{enumerate}

\item[] \textbf{Capacite C5}

\item \textbf{[Q11]} Un point d'inflexion est un point ou :
  \begin{enumerate}[label=\Alph*.]
    \item $f'(x) = 0$
    \item la convexite change au passage de ce point
    \item $f''(x) = 0$ seulement
    \item $f(x) = 0$
  \end{enumerate}

\item \textbf{[Q12]} Si $f$ est convexe, alors la courbe est :
  \begin{enumerate}[label=\Alph*.]
    \item au-dessus de ses tangentes
    \item au-dessous de ses tangentes
    \item au-dessus de ses cordes
    \item au-dessus de ses tangentes et de ses cordes
  \end{enumerate}

\item \textbf{[Q13]} La courbe de $f(x) = x^3$ traverse sa tangente en $x = 0$ car :
  \begin{enumerate}[label=\Alph*.]
    \item $f'(0) = 0$
    \item $f''(0) = 0$ et $f''$ change de signe
    \item $f(0) = 0$
    \item $f$ est decroissante
  \end{enumerate}

\item[] \textbf{Capacite C6}

\item \textbf{[Q14]} Si $f''(x) \geq 0$ sur $I$, alors pour tout $a, x \in I$ :
  \begin{enumerate}[label=\Alph*.]
    \item $f(x) \geq f(a) + f'(a)(x - a)$
    \item $f(x) \leq f(a) + f'(a)(x - a)$
    \item $f(x) = f(a) + f'(a)(x - a)$
    \item $f(x) \geq f(a)$
  \end{enumerate}

\item \textbf{[Q15]} L'approximation $\mathrm{e}^{0{,}1} \approx 1{,}1$ obtenue par tangente en $0$ est :
  \begin{enumerate}[label=\Alph*.]
    \item par defaut (la vraie valeur est plus grande)
    \item par exces (la vraie valeur est plus petite)
    \item exacte
    \item impossible a determiner
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des reponses erronees}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{A} — La derivee de l'exponentielle est conservee, mais la derivee de la fonction interieure $x^2$ fournit aussi le facteur $2x$.
    \item \textbf{C} — Le facteur $x$ correspond a une derivee incomplete de $x^2$ : on a $(x^2)'=2x$, et non simplement $x$.
    \item \textbf{D} — Le facteur $2$ oublie la variable dans $(x^2)'=2x$ ; la regle de chaine donne donc $2x\,\mathrm{e}^{x^2}$.
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — La formule $(\ln u)'=u'/u$ exige le numerateur $u'=2x$ ; choisir $1$ revient a oublier toute la derivee de $x^2+1$.
    \item \textbf{C} — Le numerateur $2$ remplace a tort $(x^2)'=2x$ par une constante et perd donc le facteur variable $x$.
    \item \textbf{D} — Le numerateur $x$ oublie le coefficient $2$ dans $(x^2+1)'=2x$ ; le denominateur seul ne corrige pas cet oubli.
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — Le facteur exterieur $4$ est present, mais la derivee de la fonction interieure $3x-2$ apporte aussi le facteur $3$.
    \item \textbf{C} — Le facteur $3$ vient de la derivee de $3x-2$, mais la puissance doit diminuer de $4$ a $3$ et etre multipliee par $4$.
    \item \textbf{D} — Le coefficient $12$ est correct, mais la derivation de la puissance quatrieme impose $(3x-2)^3$, et non $(3x-2)^4$.
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{B} — Le signe plus suppose a tort que $(\mathrm{e}^{-x})'=\mathrm{e}^{-x}$ ; la derivee de $-x$ impose le terme $-x\mathrm{e}^{-x}$.
    \item \textbf{C} — Le resultat $-\mathrm{e}^{-x}$ ne conserve que la derivee de l'exponentielle et omet le terme obtenu en derivant le facteur $x$.
    \item \textbf{D} — Le produit $x\mathrm{e}^{-x}$ est la fonction initiale : aucun des deux termes de la formule $(uv)'=u'v+uv'$ n'a ete calcule.
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — Choisir $+\infty$ ne compare que les divergences separees de $\ln x$ et de $x$ ; le denominateur croit plus vite et impose la limite nulle.
    \item \textbf{B} — La valeur $1$ supposerait des croissances equivalentes ; or $\ln x=o(x)$, donc leur quotient tend vers $0$.
    \item \textbf{D} — Pour $x>1$, $\ln x$ et $x$ sont positifs : leur quotient ne peut pas tendre vers $-\infty$, et la croissance comparee donne $0$.
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{A} — Cette reponse inverse les deux extrema : $f'=3(x^2-1)$ est positive, puis negative, puis positive, donc le maximum est en $-1$.
    \item \textbf{C} — Le point $0$ n'est pas critique puisque $f'(0)=-3$ ; il ne peut donc pas etre le maximum local annonce.
    \item \textbf{D} — La derivee $3(x^2-1)$ change deux fois de signe, en $-1$ puis en $1$ : la fonction admet donc bien deux extrema locaux.
  \end{itemize}
  \item \textbf{Q7}
  \begin{itemize}
    \item \textbf{B} — La concavite exigerait $f''(x)\leq 0$ ; ici $f''(x)=\mathrm{e}^x>0$ sur tout $\mathbb{R}$.
    \item \textbf{C} — Le signe de $x$ ne change pas celui de $f''(x)=\mathrm{e}^x$, qui reste strictement positif sur les deux demi-droites.
    \item \textbf{D} — La derivee seconde $f''(x)=\mathrm{e}^x$ est positive partout ; ce critere etablit directement la convexite sur $\mathbb{R}$.
  \end{itemize}
  \item \textbf{Q8}
  \begin{itemize}
    \item \textbf{A} — Restreindre l'inegalite a $x\geq0$ oublie que la convexite de l'exponentielle place sa courbe au-dessus de la tangente en $0$ sur tout $\mathbb{R}$.
    \item \textbf{B} — Restreindre l'inegalite a $x\leq0$ oublie la meme propriete de tangente, valable sur tout le domaine de l'exponentielle.
    \item \textbf{D} — L'egalite a lieu en $x=0$, mais l'inegalite reste stricte pour tout $x\neq0$ car l'exponentielle est strictement convexe.
  \end{itemize}
  \item \textbf{Q9}
  \begin{itemize}
    \item \textbf{A} — La convexite exigerait $f''(x)\geq0$ ; pour $f(x)=\ln x$, on a $f''(x)=-1/x^2<0$ sur $]0,+\infty[$.
    \item \textbf{C} — Aucun changement de convexite ne se produit : $f''(x)=-1/x^2$ reste strictement negatif sur tout le domaine.
    \item \textbf{D} — Le signe constant $f''(x)=-1/x^2<0$ fournit precisement une conclusion : $\ln$ est concave sur $]0,+\infty[$.
  \end{itemize}
  \item \textbf{Q10}
  \begin{itemize}
    \item \textbf{B} — La concavite correspond au signe oppose $f''\leq0$ ; une derivee seconde strictement positive caracterise ici la convexite.
    \item \textbf{C} — Le signe de $f''$ decrit la variation de $f'$ et la convexite, pas le signe de $f'$ necessaire pour conclure que $f$ est croissante.
    \item \textbf{D} — Une derivee seconde positive n'impose pas $f'\leq0$ ; elle rend $f'$ croissante et ne suffit pas a rendre $f$ decroissante.
  \end{itemize}
  \item \textbf{Q11}
  \begin{itemize}
    \item \textbf{A} — $f'(x)=0$ caracterise un point critique lorsque $f$ est derivable, pas un changement de convexite. \emph{Renvoi : cours C5, definition.}
    \item \textbf{C} — $f''(x)=0$ ne suffit pas : par exemple $f(x)=x^4$ verifie $f''(0)=0$ sans changer de convexite en 0. \emph{Renvoi : cours C5, point d'inflexion.}
    \item \textbf{D} — $f(x)=0$ indique une intersection avec l'axe des abscisses, sans information sur la convexite. \emph{Renvoi : cours C5, definition.}
  \end{itemize}
  \item \textbf{Q12}
  \begin{itemize}
    \item \textbf{B} — La position est inversee : une fonction convexe a sa courbe au-dessus de chaque tangente, et non au-dessous.
    \item \textbf{C} — Pour deux points de la courbe d'une fonction convexe, le segment de corde se situe au-dessus de la courbe, pas au-dessous.
    \item \textbf{D} — La premiere partie est vraie, mais la courbe est au-dessous de ses cordes ; l'affirmation conjointe est donc fausse.
  \end{itemize}
  \item \textbf{Q13}
  \begin{itemize}
    \item \textbf{A} — La condition $f'(0)=0$ indique seulement une tangente horizontale ; elle n'implique pas que la courbe traverse cette tangente.
    \item \textbf{C} — L'egalite $f(0)=0$ place le point sur l'axe des abscisses, mais ne donne aucune information sur un changement de convexite.
    \item \textbf{D} — La fonction $x^3$ est croissante sur $\mathbb{R}$ ; surtout, la monotonie ne caracterise pas le franchissement d'une tangente.
  \end{itemize}
  \item \textbf{Q14}
  \begin{itemize}
    \item \textbf{B} — Le sens $\leq$ correspondrait a une fonction concave ; une fonction convexe reste au-dessus de sa tangente en $a$.
    \item \textbf{C} — L'egalite avec la tangente pour tous $a,x$ caracteriserait une fonction affine ; la convexite ne donne qu'une inegalite.
    \item \textbf{D} — La convexite compare $f(x)$ a la tangente $f(a)+f'(a)(x-a)$, pas a la seule valeur $f(a)$ ; aucune croissance n'est supposee.
  \end{itemize}
  \item \textbf{Q15}
  \begin{itemize}
    \item \textbf{B} — La fonction exponentielle est strictement convexe : sa courbe est au-dessus de sa tangente en $0$, donc $\mathrm{e}^{0{,}1}>1{,}1$.
    \item \textbf{C} — La tangente ne coincide avec la courbe qu'au point de contact $x=0$ ; a $x=0{,}1$, la stricte convexite exclut l'egalite.
    \item \textbf{D} — La convexite determine le sens de l'approximation : la courbe est au-dessus de toute tangente, donc $1{,}1$ est une valeur par defaut.
  \end{itemize}
\end{itemize}
