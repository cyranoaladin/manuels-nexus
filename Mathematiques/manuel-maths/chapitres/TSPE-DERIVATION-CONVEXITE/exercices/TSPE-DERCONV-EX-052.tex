\begin{exercice}{TSPE-DERCONV-EX-052}{3}{25}
Soit $f(x) = x^4$.
\begin{enumerate}
  \item Montrer que $f$ est convexe sur $\mathbb{R}$.
  \item Enoncer et demontrer l'inegalite de Jensen pour $f$ avec $t = \dfrac{1}{3}$, $x = 1$ et $y = 2$.
  \item Generaliser : pour quels $t \in [0,1]$ l'inegalite de Jensen est-elle une egalite ?
\end{enumerate}

\end{exercice}
