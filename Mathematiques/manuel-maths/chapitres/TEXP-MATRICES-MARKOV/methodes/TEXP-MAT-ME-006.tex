\begin{fichemethode}{M6}{Démontrer que la distribution après n transitions est pi0 P puissance n}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande de DEMONTRER la formule
  $\pi_n = \pi_0 P^n$ pour une chaine de Markov de matrice de transition $P$.
}

\pasApas{%
  \begin{enumerate}
    \item Etablir la relation d'UNE étape : par la formule des probabilités
      totales sur l'etat courant,
      $P(X_{n+1} = j) = \sum_i P(X_n = i)\,p_{ij}$, ce qui s'ecrit
      matriciellement $\pi_{n+1} = \pi_n P$.
    \item Poser la propriété $\mathcal{P}(n)$ : $\pi_n = \pi_0 P^n$.
    \item Initialisation : $n = 0$ — $\pi_0 = \pi_0 P^0 = \pi_0 I$, vraie.
    \item Heredite : si $\pi_n = \pi_0 P^n$, alors
      $\pi_{n+1} = \pi_n P = \left(\pi_0 P^n\right) P = \pi_0 P^{n+1}$
      (associativite du produit matriciel).
    \item Conclure par récurrence : $\pi_n = \pi_0 P^n$ pour tout entier
      $n \geq 0$.
  \end{enumerate}
}

\exempleRedige{%
  Rediger la démonstration pour une chaine a deux etats.

  \medskip
  \commentaireMarge{Étape 1 : probabilités totales.}%
  Le système $\{X_n = 1\}$, $\{X_n = 2\}$ est un système complet d'événements.
  Pour tout etat $j$ :
  $P(X_{n+1} = j) = P(X_n = 1)\,p_{1j} + P(X_n = 2)\,p_{2j}$,
  c'est exactement le coefficient $j$ du produit $\pi_n P$ : donc
  $\pi_{n+1} = \pi_n P$.

  \medskip
  \commentaireMarge{Étape 2 : récurrence immediate.}%
  $\mathcal{P}(0)$ est vraie ($P^0 = I$). Si $\pi_n = \pi_0 P^n$, alors
  $\pi_{n+1} = \pi_n P = \pi_0 P^n P = \pi_0 P^{n+1}$ : $\mathcal{P}(n+1)$
  est vraie. Par récurrence, $\pi_n = \pi_0 P^n$ pour tout $n$.
  \hfill$\square$
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Admettre la relation $\pi_{n+1} = \pi_n P$ sans la
      démontrer : c'est le cœur probabiliste (probabilités totales).
    \item \textbf{Piège 2.} Écrire $\pi_{n+1} = P\,\pi_n$ avec des
      distributions en ligne : l'ordre du produit suit la convention choisie.
    \item \textbf{Piège 3.} Utiliser la propriété de Markov implicitement :
      dire explicitement que $p_{ij}$ ne depend pas de $n$ ni du passe.
  \end{itemize}
}

\verifier{%
  Confronter la formule demontree a l'iteration directe sur un exemple
  numérique ($\pi_1$, $\pi_2$ calcules des deux facons, fiche M5).
}

\end{fichemethode}
