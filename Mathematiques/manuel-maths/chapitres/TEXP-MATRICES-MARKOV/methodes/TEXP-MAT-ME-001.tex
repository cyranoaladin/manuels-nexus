\begin{fichemethode}{M1}{Operer sur les matrices : somme, produit, inverse, puissance}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande des calculs matriciels :
  sommes, produits, inverse d'une matrice carrée $2 \times 2$, puissances.
}

\pasApas{%
  \begin{enumerate}
    \item Somme : coefficient par coefficient, entre matrices de MEMES
      dimensions.
    \item Produit $AB$ : le coefficient $(i\,;\,j)$ est le produit scalaire de
      la LIGNE $i$ de $A$ par la COLONNE $j$ de $B$ ; les dimensions doivent
      s'enchainer (colonnes de $A$ = lignes de $B$).
    \item Retenir que le produit n'est PAS commutatif : calculer $AB$ et $BA$
      separement si les deux sont demandes.
    \item Inverse $2 \times 2$ : si $\det A = ad - bc \neq 0$,
      $A^{-1} = \dfrac{1}{ad - bc}\begin{pmatrix} d & -b\\ -c & a
      \end{pmatrix}$ ; contrôler par $A A^{-1} = I$.
    \item Puissances : produits successifs ($A^2 = AA$, $A^3 = A^2 A$...),
      calculatrice pour les grands exposants.
  \end{enumerate}
}

\exempleRedige{%
  $A = \begin{pmatrix} 1 & 2\\ 3 & 4 \end{pmatrix}$,
  $B = \begin{pmatrix} 0 & 1\\ 1 & 0 \end{pmatrix}$. Calculer $AB$, $BA$ et
  $A^{-1}$.

  \medskip
  \commentaireMarge{Ligne de $A$ contre colonne de $B$.}%
  $AB = \begin{pmatrix} 2 & 1\\ 4 & 3 \end{pmatrix}$ et
  $BA = \begin{pmatrix} 3 & 4\\ 1 & 2 \end{pmatrix}$ : $AB \neq BA$.

  \medskip
  \commentaireMarge{Formule de l'inverse $2 \times 2$.}%
  $\det A = 1 \times 4 - 2 \times 3 = -2 \neq 0$, donc
  $A^{-1} = -\dfrac{1}{2}\begin{pmatrix} 4 & -2\\ -3 & 1 \end{pmatrix}
  = \begin{pmatrix} -2 & 1\\ 3/2 & -1/2 \end{pmatrix}$.
  Contrôle : $A A^{-1} = I_2$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Multiplier coefficient par coefficient : le produit
      matriciel est ligne contre colonne.
    \item \textbf{Piège 2.} Utiliser la commutativite ($AB = BA$) ou simplifier
      $AB = AC \Rightarrow B = C$ : faux en general.
    \item \textbf{Piège 3.} Inverser une matrice de determinant nul, ou oublier
      le facteur $\frac{1}{\det A}$.
  \end{itemize}
}

\verifier{%
  Contrôler les dimensions de chaque résultat, vérifier un produit par un
  coefficient recalcule a la main, et TOUJOURS tester $A A^{-1} = I$.
}

\end{fichemethode}
