\begin{fichemethode}{M4}{Représenter une chaine de Markov : graphe pondere, distribution initiale, matrice de transition}

\quandUtiliser{%
  Utiliser cette méthode lorsqu'une situation a 2 ou 3 etats evolue par étapes
  avec des probabilités de passage NE DEPENDANT que de l'etat courant : il faut
  construire le graphe oriente pondere, la matrice de transition et la
  distribution initiale.
}

\pasApas{%
  \begin{enumerate}
    \item Identifier les ETATS (2 ou 3) et vérifier la propriété de Markov :
      les probabilités de transition ne dependent que de l'etat present.
    \item Dessiner le graphe oriente : un sommet par etat, un arc $i \to j$
      pondere par la probabilité de passer de $i$ a $j$ (boucles comprises).
    \item Contrôler que la somme des poids SORTANT de chaque sommet vaut 1.
    \item Ranger ces poids dans la matrice de transition $P$ : LIGNE $i$ =
      etat de depart, colonne $j$ = etat d'arrivee ; chaque ligne somme a 1.
    \item Représenter la distribution initiale par une matrice LIGNE
      $\pi_0 = (p_1\ \dots)$ dont les coefficients somment a 1.
  \end{enumerate}
}

\exempleRedige{%
  Chaque année, 10\,\% des abonnes d'un operateur $A$ partent chez $B$, et
  30\,\% des abonnes de $B$ viennent chez $A$. Au depart, les parts sont
  égales. Représenter la chaine.

  \medskip
  \commentaireMarge{Deux etats, quatre arcs (dont deux boucles).}%
  Etats $A$ et $B$. Arcs : $A \to B$ : $0{,}1$ ; $A \to A$ : $0{,}9$ ;
  $B \to A$ : $0{,}3$ ; $B \to B$ : $0{,}7$ — chaque sommet emet un total
  de 1.

  \medskip
  \commentaireMarge{Lignes $=$ depart ; chaque ligne somme a 1.}%
  $P = \begin{pmatrix} 0{,}9 & 0{,}1\\ 0{,}3 & 0{,}7 \end{pmatrix}$ et
  $\pi_0 = (0{,}5\ \ 0{,}5)$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Oublier les BOUCLES $i \to i$ : sans elles, les
      sorties ne somment pas a 1.
    \item \textbf{Piège 2.} Transposer la matrice (colonnes qui somment a 1) :
      la convention ici est lignes $=$ etat de depart.
    \item \textbf{Piège 3.} Confondre proportions d'une distribution
      (photographie a l'instant $n$) et probabilités de transition (passage
      d'une étape a la suivante).
  \end{itemize}
}

\verifier{%
  Vérifier que chaque ligne de $P$ et que $\pi_0$ somment a 1, et que chaque
  poids du graphe se retrouve au bon emplacement $(i\,;\,j)$ de la matrice.
}

\sEntrainer{\refExos{M4}}

\end{fichemethode}
