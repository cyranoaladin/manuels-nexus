\begin{exercice}{TEXP-MAT-EX-001}{1}{12}
On donne les matrices $A=\begin{pmatrix}3&2\\1&1\end{pmatrix}$ et $B=\begin{pmatrix}0&1\\2&1\end{pmatrix}$.
\begin{enumerate}
  \item Calculer $A+B$ et $AB$.
  \item Calculer $\det(A)$. La matrice $A$ est-elle inversible ?
  \item Déterminer $A^{-1}$ et vérifier que $AA^{-1}=I$.
  \item Calculer $A^2$.
\end{enumerate}
\end{exercice}
