\begin{exercice}{TEXP-MAT-EX-010}{2}{18}
On étudie la suite de matrices colonnes définie par
$U_{n+1}=AU_n+C$ avec $A=\begin{pmatrix}\tfrac12&0\\0&\tfrac13\end{pmatrix}$,
$C=\begin{pmatrix}1\\2\end{pmatrix}$ et $U_0=\begin{pmatrix}0\\0\end{pmatrix}$.
\begin{enumerate}
  \item Calculer $U_1$, $U_2$ et $U_3$.
  \item Déterminer la matrice colonne constante $X$ vérifiant $X=AX+C$.
  \item Montrer que $V_n=U_n-X$ vérifie $V_{n+1}=AV_n$, et en déduire la
        limite de $U_n$.
\end{enumerate}
\end{exercice}
