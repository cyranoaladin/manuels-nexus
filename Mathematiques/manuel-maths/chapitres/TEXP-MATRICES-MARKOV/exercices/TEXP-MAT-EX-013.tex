\begin{exercice}{TEXP-MAT-EX-013}{2}{15}
Soit une chaîne de Markov de matrice de transition $T$ et de distribution
initiale $\pi_0$ (matrice ligne).
\begin{enumerate}
  \item Justifier que $\pi_1=\pi_0T$ en décomposant selon l'état de départ.
  \item Démontrer par récurrence que $\pi_n=\pi_0T^n$ pour tout $n\geq0$.
  \item Vérifier la formule sur l'exemple
        $T=\begin{pmatrix}0{,}7&0{,}3\\0{,}4&0{,}6\end{pmatrix}$ et
        $\pi_0=\begin{pmatrix}0{,}5&0{,}5\end{pmatrix}$ pour $n=3$.
\end{enumerate}
\end{exercice}
