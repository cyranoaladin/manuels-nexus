\begin{exercice}{TEXP-MAT-EX-003}{2}{15}
On considère la suite de matrices colonnes $(U_n)$ définie par $U_0=\begin{pmatrix}0\\0\end{pmatrix}$ et $U_{n+1}=AU_n+C$, avec $A=\begin{pmatrix}\tfrac{1}{2}&0\\\tfrac{1}{4}&\tfrac{3}{4}\end{pmatrix}$ et $C=\begin{pmatrix}3\\2\end{pmatrix}$.
\begin{enumerate}
  \item Calculer $U_1$ puis $U_2$ directement à partir de la relation de récurrence.
  \item Vérifier que $I-A$ est inversible, et déterminer le point fixe $X$ vérifiant $X=AX+C$.
  \item En déduire l'expression de $U_n$ en fonction de $n$, sous la forme $U_n=A^n(U_0-X)+X$.
\end{enumerate}
\end{exercice}
