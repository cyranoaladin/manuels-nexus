\begin{exercice}{TEXP-MAT-EX-014}{2}{18}
On cherche les distributions invariantes.
\begin{enumerate}
  \item Déterminer $\pi=\begin{pmatrix}a&b\end{pmatrix}$ vérifiant $\pi T=\pi$
        et $a+b=1$, pour
        $T=\begin{pmatrix}0{,}7&0{,}3\\0{,}4&0{,}6\end{pmatrix}$.
  \item Vérifier que $\pi_n$ converge vers cette distribution, en calculant
        $\pi_{20}$ à partir de $\pi_0=\begin{pmatrix}1&0\end{pmatrix}$.
  \item Même question pour la chaîne à trois états de matrice
        $S=\begin{pmatrix}0{,}5&0{,}5&0\\0&0{,}5&0{,}5\\0{,}5&0&0{,}5\end{pmatrix}$.
\end{enumerate}
\end{exercice}
