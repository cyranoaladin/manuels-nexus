\begin{exercice}{TEXP-MAT-EX-008}{1}{12}
Soient $A=\begin{pmatrix}2&1\\0&3\end{pmatrix}$ et
$B=\begin{pmatrix}1&-1\\4&2\end{pmatrix}$.
\begin{enumerate}
  \item Calculer $A+B$, $AB$ et $BA$. Que constate-t-on ?
  \item Calculer $A^2$ et $A^3$.
  \item $A$ est-elle inversible ? Si oui, donner $A^{-1}$ et vérifier
        $AA^{-1}=I_2$.
\end{enumerate}
\end{exercice}
