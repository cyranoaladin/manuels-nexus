\subsection*{Rappel -- Erreurs sur la derivee d'une fonction composee}

\textbf{Erreur 1 :} Oublier le facteur $u'(x)$. On ecrit $(\mathrm{e}^{u})' = \mathrm{e}^u$ au lieu de $u'\,\mathrm{e}^u$.

\textbf{Erreur 2 :} Confondre la derivee de $u^n$ avec celle de $x^n$. On ecrit $(u^n)' = n\,u^{n-1}$ au lieu de $n\,u'\,u^{n-1}$.

\textbf{Erreur 3 :} Oublier le domaine pour $\ln(u)$ : il faut $u > 0$.

\begin{exercice}{TEXP-MAT-RE-C1-EX1}{1}{10}
Corriger les erreurs dans les calculs suivants :
\begin{enumerate}
  \item $\left(\mathrm{e}^{3x}\right)' = \mathrm{e}^{3x}$ \quad (erreur ?)
  \item $\left((2x+1)^3\right)' = 3(2x+1)^2$ \quad (erreur ?)
  \item $f(x) = \ln(-x^2)$ est-elle bien definie sur $\mathbb{R}$ ?
\end{enumerate}
\end{exercice}
