\section*{\textcolor{chapcolor}{\MakeUppercase{Matrices et chaines de Markov}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} Une matrice de transition d'une chaine de Markov a pour propriete :
  \begin{enumerate}[label=\Alph*.]
    \item La somme des colonnes vaut 1
    \item La somme de chaque ligne vaut 1
    \item Tous ses coefficients sont negatifs
    \item Elle est toujours symetrique
  \end{enumerate}

\item \textbf{[Q2]} Si $P$ est la matrice de transition et $P_0$ l'etat initial (vecteur ligne), alors $P_n =$
  \begin{enumerate}[label=\Alph*.]
    \item $P_0 P^n$
    \item $P^n P_0$
    \item $P_0 + n P$
    \item $P_0 P n$
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q3]} Un etat stable $X$ d'une chaine de Markov verifie :
  \begin{enumerate}[label=\Alph*.]
    \item $P X = X$
    \item $X P = 0$
    \item $X P = X$ et la somme des composantes vaut 1
    \item $X^2 = P$
  \end{enumerate}

\item \textbf{[Q4]} Si une matrice de transition $P$ n'a aucun coefficient nul, la chaine de Markov est :
  \begin{enumerate}[label=\Alph*.]
    \item Instable
    \item Periodique
    \item Divergente
    \item Ergodique et admet un unique etat stable
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q5]} Pour inverser une matrice 2x2 $A = \begin{pmatrix} a & b \\ c & d \end{pmatrix}$, le determinant est :
  \begin{enumerate}[label=\Alph*.]
    \item $ad - bc$
    \item $ac - bd$
    \item $ad + bc$
    \item $a+b+c+d$
  \end{enumerate}

\end{enumerate}
