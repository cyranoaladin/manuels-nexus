\begin{exercice}{TSPE-TRIGO-EX-005}{1}{10}
Soit $f(x) = x - \sin x$ sur $[0, \pi]$.
\begin{enumerate}
  \item Calculer $f'(x)$.
  \item Justifier que $f'(x) \geq 0$ pour tout $x \in [0,\pi]$ (on rappelle que $\cos x \leq 1$).
  \item En deduire le sens de variation de $f$ sur $[0,\pi]$.
\end{enumerate}
\end{exercice}
