\section*{\textcolor{chapcolor}{\MakeUppercase{QCM d'auto-évaluation -- Fonctions sinus et cosinus}}}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q1]} L'équation cos x = 1/2 admet sur [-pi,pi] :
  \begin{enumerate}[label=\Alph*.]
    \item deux solutions opposees.
    \item une seule solution.
    \item trois solutions.
    \item aucune solution.
  \end{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q2]} sin'(x) vaut :
  \begin{enumerate}[label=\Alph*.]
    \item cos x
    \item -cos x
    \item -sin x
    \item sin x
  \end{enumerate}

\item \textbf{[Q3]} cos'(x) vaut :
  \begin{enumerate}[label=\Alph*.]
    \item sin x
    \item -sin x
    \item cos x
    \item -cos x
  \end{enumerate}

\item \textbf{[Q4]} Pour dériver x -> sin(3x), on obtient :
  \begin{enumerate}[label=\Alph*.]
    \item cos(3x)
    \item 3cos(x)
    \item 3cos(3x)
    \item cos(x)
  \end{enumerate}

\item \textbf{[Q5]} Pour trouver le maximum d'une fonction f construite avec sin ou cos sur un intervalle, on cherche :
  \begin{enumerate}[label=\Alph*.]
    \item les points ou f s'annule.
    \item les bornes de l'intervalle uniquement.
    \item les points ou f' est maximale.
    \item les points critiques interieurs et les bornes, puis on compare les valeurs de f.
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des réponses erronées}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{B} — Une seule solution oublie la symetrie du cosinus : avec $\cos(\pi/3)=1/2$, on a aussi $\cos(-\pi/3)=1/2$. \emph{Renvoi : M1, symétrie des solutions sur $[-\pi,\pi]$.}
    \item \textbf{C} — Ajouter $0$ comme troisième solution est faux, car $\cos(0)=1$ ; seules $-\pi/3$ et $\pi/3$ conviennent sur l'intervalle. \emph{Renvoi : M1, lecture du cercle trigonométrique.}
    \item \textbf{D} — Le cercle trigonométrique fournit explicitement $\cos(\pi/3)=1/2$ et, par parite, la seconde solution $-\pi/3$. \emph{Renvoi : M1, valeurs remarquables du cosinus.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{B} — Le signe moins appartient a la formule $(\cos x)'=-\sin x$ ; la dérivée de $\sin x$ est $\cos x$ sans changement de signe. \emph{Renvoi : C2, dérivées de référence de sinus et cosinus.}
    \item \textbf{C} — Le résultat $-\sin x$ est la dérivée de $\cos x$, pas celle de $\sin x$ ; les deux formules de référence ont été interverties. \emph{Renvoi : C2, dérivées de référence de sinus et cosinus.}
    \item \textbf{D} — Conserver $\sin x$ revient a ne pas dériver la fonction ; le taux de variation de $\sin$ est donne par $\cos x$. \emph{Renvoi : C2, dérivées de référence de sinus et cosinus.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — Le signe moins manque dans la formule de référence $(\cos x)'=-\sin x$ ; $\sin x$ seul donne donc le signe oppose. \emph{Renvoi : C2, dérivées de référence de sinus et cosinus.}
    \item \textbf{C} — Conserver $\cos x$ revient a ne pas dériver ; la dérivée fait intervenir l'autre fonction trigonométrique, $-\sin x$. \emph{Renvoi : C2, dérivées de référence de sinus et cosinus.}
    \item \textbf{D} — Le résultat $-\cos x$ serait la dérivée de $-\sin x$ ; pour $\cos x$, il faut changer de fonction et obtenir $-\sin x$. \emph{Renvoi : C2, dérivées de référence de sinus et cosinus.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — La dérivée exterieure est correcte, mais la règle de chaine exige de multiplier par la dérivée de $3x$, qui vaut $3$. \emph{Renvoi : C2, dérivée d'une fonction composée.}
    \item \textbf{B} — Le facteur $3$ est present, mais l'argument de la fonction cosinus doit rester $3x$ après dérivation de la composee. \emph{Renvoi : C2, dérivée d'une fonction composée.}
    \item \textbf{D} — Ce choix perd simultanement le facteur interieur $3$ et remplace a tort l'argument $3x$ par $x$. \emph{Renvoi : C2, dérivée d'une fonction composée.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — Les zeros de $f$ donnent les intersections avec l'axe des abscisses, pas ses maxima : une valeur maximale peut être non nulle. \emph{Renvoi : M2, recherche d'un optimum sur un intervalle.}
    \item \textbf{B} — Comparer seulement les bornes peut manquer un maximum interieur : il faut aussi examiner les points critiques de l'intervalle. \emph{Renvoi : M2, recherche d'un optimum sur un intervalle.}
    \item \textbf{C} — Un maximum de $f'$ localise la croissance la plus rapide de $f$, et non une valeur maximale de $f$ sur l'intervalle. \emph{Renvoi : M2, recherche d'un optimum sur un intervalle.}
  \end{itemize}
\end{itemize}
