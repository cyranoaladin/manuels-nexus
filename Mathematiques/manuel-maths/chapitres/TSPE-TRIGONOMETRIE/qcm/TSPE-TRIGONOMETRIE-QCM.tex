\subsection*{C1 — Équations et inéquations $\cos x = a$}

\begin{enumerate}

\item \textbf{[Q1]} L'equation $\cos x = \dfrac{1}{2}$ admet sur $[-\pi,\pi]$ :
  \begin{enumerate}[label=\Alph*.]
    \item une seule solution.
    \item deux solutions opposees.
    \item trois solutions.
    \item aucune solution.
  \end{enumerate}

\bigskip

\item \textbf{[Q2]} $\sin'(x)$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $-\cos x$.
    \item $\cos x$.
    \item $-\sin x$.
    \item $\sin x$.
  \end{enumerate}

\bigskip

\item \textbf{[Q3]} $\cos'(x)$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $\sin x$.
    \item $-\sin x$.
    \item $\cos x$.
    \item $-\cos x$.
  \end{enumerate}

\subsection*{C2 — Optimisation avec des fonctions trigonométriques}

\item \textbf{[Q4]} Pour deriver $x \mapsto \sin(3x)$, on obtient :
  \begin{enumerate}[label=\Alph*.]
    \item $\cos(3x)$.
    \item $3\cos(3x)$.
    \item $3\cos(x)$.
    \item $\cos(x)$.
  \end{enumerate}

\bigskip

\item \textbf{[Q5]} Pour trouver le maximum d'une fonction $f$ construite avec $\sin$ ou $\cos$ sur un intervalle, on cherche :
  \begin{enumerate}[label=\Alph*.]
    \item les points ou $f$ s'annule.
    \item les points ou $f'$ s'annule en changeant de signe (de $+$ vers $-$).
    \item les points ou $f'$ est maximale.
    \item les bornes de l'intervalle uniquement.
  \end{enumerate}

\end{enumerate}
