\begin{corrige}{TSPE-TRIGO-EV-A-EX1}
\textbf{Q1.} $\cos\!\left(\dfrac{3\pi}{4}\right) = -\dfrac{\sqrt{2}}{2}$, donc les solutions sur $[-\pi,\pi]$ sont $x=\dfrac{3\pi}{4}$ et $x=-\dfrac{3\pi}{4}$.

\textbf{Q2.} La courbe de $\cos$ est au-dessus de $-\dfrac{\sqrt{2}}{2}$ entre les deux solutions precedentes : $S = \left[-\dfrac{3\pi}{4}, \dfrac{3\pi}{4}\right]$.
\end{corrige}

\begin{corrige}{TSPE-TRIGO-EV-A-EX2}
\textbf{Q1.} $f'(x) = \cos x - \sin x$.

\textbf{Q2.} $f'(x)=0 \iff \cos x = \sin x$. Sur $\left[-\dfrac{\pi}{2},\dfrac{\pi}{2}\right]$, cela se produit en $x=\dfrac{\pi}{4}$. On a $f'(0)=1>0$ et $f'\!\left(\dfrac{\pi}{2}\right)=0-1=-1<0$ : $f'$ est positive avant $\dfrac{\pi}{4}$ et negative apres.

\textbf{Q3.} $f$ est donc croissante sur $\left[-\dfrac{\pi}{2},\dfrac{\pi}{4}\right]$ et decroissante sur $\left[\dfrac{\pi}{4},\dfrac{\pi}{2}\right]$ : le maximum de $f$ vaut $f\!\left(\dfrac{\pi}{4}\right) = \dfrac{\sqrt{2}}{2}+\dfrac{\sqrt{2}}{2} = \sqrt{2}$, atteint en $x=\dfrac{\pi}{4}$.
\end{corrige}
