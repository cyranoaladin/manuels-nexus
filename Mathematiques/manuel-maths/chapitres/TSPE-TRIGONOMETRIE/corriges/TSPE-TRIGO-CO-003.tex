\begin{corrige}{TSPE-TRIGO-EX-003}

\textbf{Q1.} $h(0) = 10 + 10\cos(0) = 10+10 = 20$. A $t=0$, le wagon est au point le plus haut, a $20$ metres du sol.

\textbf{Q2.} $h(t) = 10+5\sqrt{3} \iff 10\cos t = 5\sqrt{3} \iff \cos t = \dfrac{\sqrt{3}}{2}$.

On reconnait $\cos\!\left(\dfrac{\pi}{6}\right) = \dfrac{\sqrt{3}}{2}$ : les solutions sur $[-\pi,\pi]$ sont $t = \dfrac{\pi}{6}$ et $t = -\dfrac{\pi}{6}$.

\textbf{Q3.} $h(t) \geq 10+5\sqrt{3} \iff \cos t \geq \dfrac{\sqrt{3}}{2}$. La courbe de $\cos$ est au-dessus de $\dfrac{\sqrt{3}}{2}$ entre les deux solutions de l'equation (la « bosse centrale ») : l'ensemble cherche est $\boxed{\left[-\dfrac{\pi}{6}, \dfrac{\pi}{6}\right]}$.

\end{corrige}
