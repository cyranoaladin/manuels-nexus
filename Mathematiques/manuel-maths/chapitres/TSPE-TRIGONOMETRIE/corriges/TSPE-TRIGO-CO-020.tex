\begin{corrige}{TSPE-TRIGO-EX-020}
\textbf{1.} $\cos x=\tfrac{\sqrt3}{2}$ donne $x=\pm\tfrac{\pi}{6}$ dans
$[-\pi\,;\,\pi]$.

\textbf{2.} $\cos x=-\tfrac12$ en $x=\pm\tfrac{2\pi}{3}$. La fonction
$\cos$ étant décroissante sur $[0\,;\,\pi]$, l'inéquation est vérifiée pour
$x\in\left[-\pi\,;\,-\tfrac{2\pi}{3}\right]\cup
\left[\tfrac{2\pi}{3}\,;\,\pi\right]$.

\textbf{3.} $2\cos^2x-1=\cos2x$, donc l'équation équivaut à $\cos2x=0$, soit
$2x=\tfrac{\pi}{2}+k\pi$ et $x=\tfrac{\pi}{4}+k\tfrac{\pi}{2}$. Dans
$[-\pi\,;\,\pi]$ : $x\in\left\{\pm\tfrac{\pi}{4},\pm\tfrac{3\pi}{4}\right\}$.
\end{corrige}
