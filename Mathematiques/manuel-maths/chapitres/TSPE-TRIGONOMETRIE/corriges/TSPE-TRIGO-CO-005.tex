\begin{corrige}{TSPE-TRIGO-EX-005}

\textbf{Q1.} $f'(x) = 1 - \cos x$.

\textbf{Q2.} Pour tout reel $x$, $\cos x \leq 1$, donc $-\cos x \geq -1$, donc $f'(x) = 1 - \cos x \geq 1 - 1 = 0$.

\textbf{Q3.} $f' \geq 0$ sur $[0,\pi]$ : $f$ est croissante sur $[0,\pi]$.

\end{corrige}
