\begin{exercice}{TSPE-TRIGONOMETRIE-FR-R1-016-EX1}{2}{12}
Pour tout réel $x$, simplifier les expressions suivantes.
\begin{enumerate}
  \item $\cos x + \cos(\pi - x)$
  \item $\sin x + \sin(\pi - x)$
  \item $\sin x \cdot \sin(\pi - x) + \cos x \cdot \cos(\pi - x)$
\end{enumerate}
\end{exercice}

\medskip
\noindent\textbf{Corrigé.}

\textbf{1.} $\cos x + \cos(\pi-x) = \cos x + (-\cos x) = 0$.

\textbf{2.} $\sin x + \sin(\pi-x) = \sin x + \sin x = 2\sin x$.

\textbf{3.} $\sin x \sin(\pi-x) + \cos x \cos(\pi-x) = \sin x \cdot \sin x + \cos x \cdot (-\cos x) = \sin^2 x - \cos^2 x = -\cos(2x)$.
