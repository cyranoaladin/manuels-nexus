\begin{exercice}{TSPE-TRIGO-RE-C1-EX1}{1}{10}
Resoudre dans $[-\pi,\pi]$ :
\begin{enumerate}
  \item $\cos x = \dfrac{\sqrt{3}}{2}$.
  \item $\cos x \geq 0$.
\end{enumerate}
\end{exercice}

% from sympy import cos, pi, sqrt, simplify

% assert cos(pi/2) == 0
% END-VERIFY

\begin{corrige}{TSPE-TRIGO-RE-C1-EX1}
\textbf{a.} $\cos\!\left(\dfrac{\pi}{6}\right) = \dfrac{\sqrt{3}}{2}$, donc $S = \left\{-\dfrac{\pi}{6}\,;\,\dfrac{\pi}{6}\right\}$.

\textbf{b.} $\cos x = 0$ pour $x = \pm\dfrac{\pi}{2}$. La courbe de $\cos$ est positive entre ces deux valeurs : $S = \left[-\dfrac{\pi}{2}, \dfrac{\pi}{2}\right]$.
\end{corrige}
