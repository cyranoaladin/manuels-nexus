\item[] \textbf{Capacite C4}

\item \textbf{[Q1]} Deux primitives d'une meme fonction continue sur un intervalle differents :
  \begin{enumerate}[label=\Alph*.]
    \item d'un facteur multiplicatif.
    \item d'une constante additive.
    \item toujours de 1.
    \item elles sont necessairement egales.
  \end{enumerate}

\item[] \textbf{Capacite C5}

\item \textbf{[Q2]} Une primitive de u'/u (avec u>0) est :
  \begin{enumerate}[label=\Alph*.]
    \item u\textasciicircum{}2
    \item ln(u)
    \item e\textasciicircum{}u
    \item 1/u
  \end{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q3]} Pour f continue et positive sur [a,b], l'integrale de f de a a b represente :
  \begin{enumerate}[label=\Alph*.]
    \item la pente de la tangente a la courbe.
    \item l'aire sous la courbe entre a et b.
    \item la valeur de f en b moins celle en a.
    \item toujours un nombre negatif.
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q4]} Si F est une primitive de f, alors l'integrale de a a b de f(x)dx vaut :
  \begin{enumerate}[label=\Alph*.]
    \item F(a)-F(b)
    \item F(b)-F(a)
    \item F(a)+F(b)
    \item F(b)*F(a)
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q5]} Pour une fonction croissante, la methode des rectangles a gauche donne :
  \begin{enumerate}[label=\Alph*.]
    \item un majorant de l'integrale.
    \item un minorant de l'integrale.
    \item la valeur exacte de l'integrale.
    \item une valeur negative.
  \end{enumerate}

\item[] \textbf{Capacite C6}

\item \textbf{[Q6]} Pour f continue, positive et croissante, la fonction F(x)=integrale de a a x de f(t)dt verifie :
  \begin{enumerate}[label=\Alph*.]
    \item F'=f
    \item F=f'
    \item F est constante.
    \item F n'est pas derivee.
  \end{enumerate}

\end{enumerate}
