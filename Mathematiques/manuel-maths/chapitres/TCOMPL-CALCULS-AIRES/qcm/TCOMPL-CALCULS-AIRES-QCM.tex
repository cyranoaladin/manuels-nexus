\subsection*{C4/C5 --- Primitives}

\begin{enumerate}

\item \textbf{[Q1]} Deux primitives d'une même fonction continue sur un intervalle diffèrent :
  \begin{enumerate}[label=\Alph*.]
    \item d'un facteur multiplicatif.
    \item d'une constante additive.
    \item toujours de $1$.
    \item elles sont nécessairement égales.
  \end{enumerate}

\bigskip

\item \textbf{[Q2]} Une primitive de $\dfrac{u'}{u}$ (avec $u>0$) est :
  \begin{enumerate}[label=\Alph*.]
    \item $u^2$.
    \item $\ln u$.
    \item $\mathrm{e}^u$.
    \item $\dfrac{1}{u}$.
  \end{enumerate}

\bigskip

\subsection*{C1/C3 --- Intégrale et calcul}

\item \textbf{[Q3]} Pour $f$ continue et positive sur $[a,b]$, $\displaystyle\int_a^b f(x)\,\mathrm{d}x$ représente :
  \begin{enumerate}[label=\Alph*.]
    \item la pente de la tangente à la courbe.
    \item l'aire sous la courbe entre $a$ et $b$.
    \item la valeur de $f$ en $b$ moins la valeur en $a$.
    \item toujours un nombre négatif.
  \end{enumerate}

\bigskip

\item \textbf{[Q4]} Si $F$ est une primitive de $f$, alors $\displaystyle\int_a^b f(x)\,\mathrm{d}x$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $F(a)-F(b)$.
    \item $F(b)-F(a)$.
    \item $F(a)+F(b)$.
    \item $F(b)\times F(a)$.
  \end{enumerate}

\bigskip

\subsection*{C2 --- Encadrement}

\item \textbf{[Q5]} Pour une fonction croissante, la méthode des rectangles \og à gauche \fg{} donne :
  \begin{enumerate}[label=\Alph*.]
    \item un majorant de l'intégrale.
    \item un minorant de l'intégrale.
    \item la valeur exacte de l'intégrale.
    \item une valeur négative.
  \end{enumerate}

\bigskip

\subsection*{C6 --- Fonction intégrale}

\item \textbf{[Q6]} Pour $f$ continue, positive et croissante, la fonction $F(x)=\displaystyle\int_a^x f(t)\,\mathrm{d}t$ vérifie :
  \begin{enumerate}[label=\Alph*.]
    \item $F'=f$.
    \item $F=f'$.
    \item $F$ est constante.
    \item $F$ n'est pas dérivable.
  \end{enumerate}

\end{enumerate}
