\begin{corrige}{TCOMPL-AIR-EV-B-EX1}
\textbf{1.} $f(x)=g(x) \iff -x^2+6x=2x \iff -x^2+4x=0 \iff x(4-x)=0 \iff x=0$ ou $x=4$.

\textbf{2.} $f(x)-g(x) = -x^2+4x = -x(x-4)$, polynôme du second degré de coefficient dominant négatif, positif \textbf{entre} ses racines $0$ et $4$ : $f\geq g$ sur $[0,4]$.

\textbf{3.} L'aire vaut :
\[
  \int_0^4 \big(f(x)-g(x)\big)\,\mathrm{d}x = \int_0^4 \big(-x^2+4x\big)\,\mathrm{d}x = \left[-\frac{x^3}{3}+2x^2\right]_0^4 = -\frac{64}{3}+32 = \frac{32}{3} \text{ unités d'aire.}
\]
\end{corrige}

\begin{corrige}{TCOMPL-AIR-EV-B-EX2}
\textbf{1.} Pour $f$ continue, positive et croissante sur $[a,b]$, la fonction $F(x)=\displaystyle\int_a^x f(t)\,\mathrm{d}t$ est dérivable sur $[a,b]$ et $F'=f$.

\textbf{2.} Avec $h=\dfrac{4-0}{4}=1$ :
\[
  1\times\big(\sqrt{0}+\sqrt{1}+\sqrt{2}+\sqrt{3}\big) \ \leq\ \int_0^4 \sqrt{x}\,\mathrm{d}x \ \leq\ 1\times\big(\sqrt{1}+\sqrt{2}+\sqrt{3}+\sqrt{4}\big).
\]
\end{corrige}

\begin{corrige}{TCOMPL-AIR-EV-B-EX3}
\textbf{1.} $h$ est continue et positive sur $[0\,;4]$ : $\displaystyle\int_0^4 h(x)\,\mathrm{d}x$ est l'aire, en unités d'aire, du domaine compris entre la courbe de $h$, l'axe des abscisses et les droites $x=0$ et $x=4$. Ce domaine est le triangle rectangle de sommets $(0;0)$, $(4;0)$ et $(4;4)$, d'aire $\dfrac{4\times4}{2}=8$.

\textbf{2.} Sur $[4\,;6]$, le domaine est le trapèze de hauteur $6-4=2$ et de bases $h(4)=4$ et $h(6)=6$, d'aire $\dfrac{(4+6)\times2}{2}=10$. Le domaine sur $[0\,;6]$ est le triangle de sommets $(0;0)$, $(6;0)$ et $(6;6)$, d'aire $\dfrac{6\times6}{2}=18$. On vérifie bien
\[
  \int_0^6 h = 18 = 8 + 10 = \int_0^4 h + \int_4^6 h,
\]
ce qui est la relation de Chasles : découper le domaine en $x=4$ découpe l'aire en deux parts qui se rajoutent.
\end{corrige}
