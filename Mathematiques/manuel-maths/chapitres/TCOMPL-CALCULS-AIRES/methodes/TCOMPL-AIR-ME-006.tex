\begin{fichemethode}{M6}{Démontrer que la fonction intégrale a pour dérivée f}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande de DEMONTRER que
  $F(x) = \displaystyle\int_a^x f(t)\,dt$ est dérivable de dérivée $f$,
  lorsque $f$ est continue, positive et croissante.
}

\pasApas{%
  \begin{enumerate}
    \item Fixer $x$ et prendre $h > 0$ : interpréter
      $F(x+h) - F(x) = \displaystyle\int_x^{x+h} f(t)\,dt$ comme l'aire sous
      la courbe entre $x$ et $x+h$ (relation de Chasles).
    \item Encadrer cette aire par deux rectangles de largeur $h$ : comme $f$
      est CROISSANTE, $h\,f(x) \leq F(x+h) - F(x) \leq h\,f(x+h)$.
    \item Diviser par $h > 0$ :
      $f(x) \leq \dfrac{F(x+h) - F(x)}{h} \leq f(x+h)$.
    \item Faire tendre $h$ vers 0 : par CONTINUITE de $f$,
      $f(x+h) \to f(x)$ ; le théorème des gendarmes donne
      $\dfrac{F(x+h) - F(x)}{h} \to f(x)$.
    \item Traiter de même $h < 0$ (encadrement renverse sur $[x+h\,;\,x]$),
      puis conclure : $F$ est dérivable en $x$ et $F'(x) = f(x)$.
  \end{enumerate}
}

\exempleRedige{%
  Rediger l'encadrement central pour $f$ continue, positive et croissante,
  et le contrôler sur $f(t) = t^2$ en $x = 1$, $h = 0{,}1$.

  \medskip
  \commentaireMarge{Aire entre deux rectangles de largeur $h$.}%
  Sur $[x\,;\,x+h]$, $f$ etant croissante, on a
  $f(x) \leq f(t) \leq f(x+h)$ pour tout $t$ : l'aire
  $\int_x^{x+h} f(t)\,dt$ est comprise entre celles des rectangles de
  hauteurs $f(x)$ et $f(x+h)$ :
  $h\,f(x) \leq F(x+h) - F(x) \leq h\,f(x+h)$.

  \medskip
  \commentaireMarge{Contrôle numérique de l'encadrement.}%
  Pour $f(t) = t^2$ ($F(x) = \frac{x^3}{3}$), $x = 1$, $h = 0{,}1$ :
  $0{,}1 \leq F(1{,}1) - F(1) = \dfrac{331}{3000} \approx 0{,}1103
  \leq 0{,}121$ — l'encadrement est vérifie, et le quotient
  $\approx 1{,}103$ se rapproche bien de $f(1) = 1$ quand $h$ diminue.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Oublier l'hypothèse de CROISSANCE : sans elle,
      l'encadrement par $f(x)$ et $f(x+h)$ est faux (il faudrait min/max).
    \item \textbf{Piège 2.} Oublier la CONTINUITE au moment du passage a la
      limite : c'est elle qui donne $f(x+h) \to f(x)$.
    \item \textbf{Piège 3.} Ne traiter que $h > 0$ : la derivabilite exige
      aussi la limite a gauche.
  \end{itemize}
}

\verifier{%
  Contrôler l'encadrement sur un exemple numérique (rectangles), et vérifier
  la conclusion sur une fonction dont on connait une primitive
  ($F(x) = \frac{x^3}{3}$ pour $f(t) = t^2$ : $F' = f$).
}
