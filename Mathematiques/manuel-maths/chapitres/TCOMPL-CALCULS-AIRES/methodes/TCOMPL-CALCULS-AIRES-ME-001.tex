\begin{fichemethode}{M1}{Interpréter une intégrale comme une aire et utiliser Chasles}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande d'interpréter $\int_a^b f(x)\,dx$
  pour une fonction continue \textbf{positive}, de lire une intégrale sur un graphique,
  ou de decouper un calcul d'aire avec la relation de Chasles.
}

\pasApas{%
  \begin{enumerate}
    \item Vérifier que $f$ est continue et \textbf{positive} sur $[a\,;\,b]$ :
      alors $\int_a^b f(x)\,dx$ est l'aire, en unités d'aire, du domaine compris
      entre la courbe de $f$, l'axe des abscisses et les droites $x=a$ et $x=b$.
    \item Identifier le domaine sur la figure (le hachurer si besoin).
    \item Si le domaine est une figure usuelle (rectangle, triangle, trapeze),
      calculer son aire geometriquement : c'est la valeur de l'intégrale.
    \item Pour decouper le calcul en morceaux, appliquer la relation de Chasles :
      pour tout $c$, $\displaystyle\int_a^b f = \int_a^c f + \int_c^b f$.
    \item Convertir en unités d'aire si le repere n'est pas orthonorme d'unité 1.
  \end{enumerate}
}

\exempleRedige{%
  Calculer $I = \displaystyle\int_0^3 x\,dx$ par interprétation géométrique,
  puis retrouver le résultat avec Chasles en $c = 1$.

  \medskip
  \commentaireMarge{Le domaine est un triangle rectangle.}%
  La fonction $x \mapsto x$ est continue et positive sur $[0\,;\,3]$. Le domaine
  sous la courbe est le triangle de sommets $(0;0)$, $(3;0)$, $(3;3)$ :
  $I = \dfrac{3 \times 3}{2} = \dfrac{9}{2}$.

  \medskip
  \commentaireMarge{Chasles : on decoupe en $[0;1]$ puis $[1;3]$.}%
  Avec Chasles : $\displaystyle\int_0^1 x\,dx = \dfrac{1}{2}$ (triangle) et
  $\displaystyle\int_1^3 x\,dx = \dfrac{(1+3)\times 2}{2} = 4$ (trapeze), d'ou
  $I = \dfrac{1}{2} + 4 = \dfrac{9}{2}$ : les deux calculs concordent.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Oublier l'hypothèse $f \geq 0$ : si $f$ change de signe,
      l'intégrale n'est plus l'aire du domaine (les parties sous l'axe comptent negativement).
    \item \textbf{Piège 2.} Confondre unité d'aire et cm$^2$ : l'unité d'aire depend
      des unités des axes du repere.
    \item \textbf{Piège 3.} Écrire Chasles avec un signe faux :
      $\int_a^b = \int_a^c + \int_c^b$ vaut pour TOUT $c$, mais chaque borne doit
      être recopiee dans le bon ordre.
  \end{itemize}
}

\verifier{%
  Comparer le résultat a un ordre de grandeur lu sur la figure (encadrer le domaine
  par un rectangle) et vérifier que la somme des morceaux de Chasles redonne
  exactement l'intégrale totale.
}

\end{fichemethode}
