\begin{corrige}{TCOMPL-AIR-EX-007}

\textbf{Question 1.} $u(x) = x^2-2$, $u'(x) = 2x$. Donc $f'(x) = 2x\,\mathrm{e}^{x^2-2}$.

\commentaireMarge{$f'(0)=0$ : la tangente est horizontale.}

\textbf{Question 2.} $f(0) = \mathrm{e}^{-2}$ et $f'(0) = 0$. La tangente en $0$ est :
\[ y = \mathrm{e}^{-2}. \]

\textbf{Question 3.} Etudions $g(x) = f(x) - \mathrm{e}^{-2} = \mathrm{e}^{x^2-2} - \mathrm{e}^{-2}$.

Au voisinage de $0$, $x^2-2 \geq -2$ donc $\mathrm{e}^{x^2-2} \geq \mathrm{e}^{-2}$, soit $g(x) \geq 0$.

La courbe est \textbf{au-dessus} de la tangente au voisinage de $0$ (et en fait partout).

\end{corrige}
