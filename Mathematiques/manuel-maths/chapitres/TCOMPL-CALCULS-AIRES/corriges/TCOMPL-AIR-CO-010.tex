\begin{corrige}{TCOMPL-AIR-EX-010}
\textbf{1.} $F(x+h)-F(x)=\int_x^{x+h}f(t)\,\mathrm{d}t$ est l'aire sous la
courbe entre $x$ et $x+h$. Comme $f$ est croissante, sur cet intervalle
$f(x)\leq f(t)\leq f(x+h)$, d'où
$h\,f(x)\leq F(x+h)-F(x)\leq h\,f(x+h)$.

\textbf{2.} En divisant par $h>0$ :
$f(x)\leq\dfrac{F(x+h)-F(x)}{h}\leq f(x+h)$.

\textbf{3.} $f$ étant continue, $f(x+h)\to f(x)$ quand $h\to0^+$. Le théorème
des gendarmes donne $\dfrac{F(x+h)-F(x)}{h}\to f(x)$ : $F$ est dérivable en
$x$ et $F'(x)=f(x)$.
\end{corrige}
