\begin{corrige}{TCOMPL-AIR-EX-006}
\textbf{1.} Avec quatre sous-intervalles de largeur $\tfrac14$, la somme
inférieure vaut
$\tfrac14\left(0+\tfrac1{16}+\tfrac4{16}+\tfrac9{16}\right)=\tfrac{7}{32}$ et
la somme supérieure
$\tfrac14\left(\tfrac1{16}+\tfrac4{16}+\tfrac9{16}+1\right)=\tfrac{15}{32}$.
Donc $\tfrac{7}{32}\leq\int_0^1x^2\,\mathrm{d}x\leq\tfrac{15}{32}$.

\textbf{2.} $\int_0^1x^2\,\mathrm{d}x=\left[\tfrac{x^3}{3}\right]_0^1
=\tfrac13$. On a bien $\tfrac{7}{32}\approx0{,}219\leq\tfrac13\approx0{,}333
\leq\tfrac{15}{32}\approx0{,}469$.

\textbf{3.} La valeur moyenne vaut
$\dfrac{1}{1-0}\int_0^1x^2\,\mathrm{d}x=\tfrac13$ : c'est la hauteur du
rectangle de base $[0\,;\,1]$ ayant la même aire que le domaine sous la
courbe.
\end{corrige}
