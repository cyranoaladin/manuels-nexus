\begin{corrige}{TCOMPL-AIR-EX-002}
\textbf{1.} Une primitive de $f$ est $F(x) = x^3-x^2+x$. Donc :
\[
  \int_0^2 f(x)\,\mathrm{d}x = F(2)-F(0) = (8-4+2) - 0 = 6.
\]

\textbf{2.} La valeur moyenne est $m = \dfrac{1}{2-0}\times6 = 3$.
\end{corrige}
