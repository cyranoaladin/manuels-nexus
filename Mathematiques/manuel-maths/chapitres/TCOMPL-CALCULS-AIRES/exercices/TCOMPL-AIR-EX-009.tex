\begin{exercice}{TCOMPL-AIR-EX-009}{2}{15}
Soient $F$ et $G$ deux primitives d'une même fonction $f$ continue sur un
intervalle $I$.
\begin{enumerate}
  \item Calculer la dérivée de $F-G$ sur $I$.
  \item En déduire que $F-G$ est constante sur $I$.
  \item Illustrer avec $f(x)=2x$ : donner deux primitives et vérifier.
\end{enumerate}
\end{exercice}
