\begin{exercice}{TCOMPL-AIR-EX-005}{1}{12}
Soit $f$ la fonction définie sur $[0\,;\,4]$ par $f(x)=2$ si $x\in[0\,;\,2]$
et $f(x)=3$ si $x\in\left]2\,;\,4\right]$.
\begin{enumerate}
  \item Représenter $f$ et calculer, par des aires de rectangles,
        $\displaystyle\int_0^4 f(x)\,\mathrm{d}x$.
  \item Vérifier la relation de Chasles
        $\displaystyle\int_0^4=\int_0^2+\int_2^4$.
  \item Que vaut $\displaystyle\int_2^2 f(x)\,\mathrm{d}x$ ? Justifier
        géométriquement.
\end{enumerate}
\end{exercice}
