\begin{exercice}{TCOMPL-AIR-EX-010}{3}{18}
Soit $f$ continue, positive et croissante sur $[a\,;\,b]$, et
$F(x)=\displaystyle\int_a^x f(t)\,\mathrm{d}t$.
\begin{enumerate}
  \item Pour $h>0$, encadrer $F(x+h)-F(x)$ par deux aires de rectangles.
  \item En déduire un encadrement de $\dfrac{F(x+h)-F(x)}{h}$.
  \item Conclure que $F'(x)=f(x)$.
\end{enumerate}
\end{exercice}
