\begin{exercice}{TCOMPL-AIR-EX-006}{2}{15}
Soit $f(x)=x^2$ sur $[0\,;\,1]$.
\begin{enumerate}
  \item Encadrer $\displaystyle\int_0^1 x^2\,\mathrm{d}x$ par les sommes de
        rectangles inférieurs et supérieurs à pas $\tfrac14$.
  \item Calculer la valeur exacte de l'intégrale et vérifier l'encadrement.
  \item Calculer la valeur moyenne de $f$ sur $[0\,;\,1]$ et l'interpréter.
\end{enumerate}
\end{exercice}
