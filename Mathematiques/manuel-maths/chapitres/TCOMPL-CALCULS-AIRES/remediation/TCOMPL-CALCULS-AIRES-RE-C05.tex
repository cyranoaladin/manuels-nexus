\subsection*{Rappel -- Fonction logarithme neperien : proprietes et derivee}

\textbf{Proprietes :} $\ln(ab) = \ln a + \ln b$, $\ln\left(\dfrac{a}{b}\right) = \ln a - \ln b$, $\ln(\mathrm{e}^x) = x$.

\textbf{Derivee :} $(\ln x)' = \dfrac{1}{x}$ sur $]0, +\infty[$.

\begin{exercice}{TCOMPL-AIR-FR-R5-EX1}{1}{8}
\begin{enumerate}
  \item Resoudre $\ln(x + 1) = 0$.
  \item Etudier les variations de $g(x) = \ln x - x$ sur $]0, +\infty[$.
  \item Determiner $\lim_{x \to +\infty} \dfrac{\ln x}{x^2}$.
\end{enumerate}
\end{exercice}
