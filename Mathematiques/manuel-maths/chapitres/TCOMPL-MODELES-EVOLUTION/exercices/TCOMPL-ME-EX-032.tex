\begin{exercice}{TCOMPL-ME-EX-032}{2}{20}
Pour tout entier $n \geq 1$, on pose $t_n = 2 + \dfrac{\cos n}{n}$.
\begin{enumerate}
  \item Justifier que, pour tout $n \geq 1$, $\ 2-\dfrac{1}{n} \leq t_n \leq 2+\dfrac{1}{n}$.
  \item En déduire la limite de $(t_n)$ en citant le théorème utilisé.
  \item Un élève écrit : « comme $\dfrac{1}{n+1} < \dfrac{1}{n}$ pour tout $n \geq 1$, les deux suites ont des limites différentes ». Réfuter cette affirmation.
\end{enumerate}
\end{exercice}
