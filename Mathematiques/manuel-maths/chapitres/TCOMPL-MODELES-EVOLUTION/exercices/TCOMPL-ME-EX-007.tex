\begin{exercice}{TCOMPL-ME-EX-007}{2}{15}
Soit $(v_n)$ la suite géométrique de premier terme $v_0=3$ et de raison
$q=0{,}4$.
\begin{enumerate}
  \item Exprimer $v_n$ en fonction de $n$ et déterminer $\lim v_n$.
  \item Calculer $S_n=v_0+v_1+\dots+v_n$ en fonction de $n$.
  \item Déterminer $\lim S_n$.
\end{enumerate}
\end{exercice}
