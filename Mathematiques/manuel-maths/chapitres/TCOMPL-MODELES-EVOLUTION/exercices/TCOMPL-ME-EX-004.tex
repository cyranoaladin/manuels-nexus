\begin{exercice}{TCOMPL-ME-EX-004}{1}{15}
On définit la suite $(u_n)$ par $u_0 = 10$ et $u_{n+1} = \dfrac{1}{2}\left(u_n + \dfrac{6}{u_n}\right)$ (méthode de Héron pour approcher $\sqrt{6}$).
\begin{enumerate}
  \item Identifier la fonction $f$ telle que $u_{n+1}=f(u_n)$.
  \item Chercher un point fixe de $f$ (solution de $f(x)=x$ pour $x>0$) : quelle valeur remarquable obtient-on ?
  \item Calculer $u_1$, $u_2$ à la calculatrice ou à la main, et constater la rapidité de la convergence.
\end{enumerate}
\end{exercice}
