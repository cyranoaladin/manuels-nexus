\begin{exercice}{TCOMPL-ME-EX-006}{2}{15}
Soit $f(x)=\sqrt{x+2}$ et la suite définie par $u_0=0$ et $u_{n+1}=f(u_n)$.
\begin{enumerate}
  \item Calculer $u_1$, $u_2$ et $u_3$ (valeurs approchées au millième).
  \item Représenter graphiquement les premiers termes en utilisant la courbe
        de $f$ et la droite $y=x$.
  \item Conjecturer le comportement de la suite et déterminer le point fixe
        de $f$.
\end{enumerate}
\end{exercice}
