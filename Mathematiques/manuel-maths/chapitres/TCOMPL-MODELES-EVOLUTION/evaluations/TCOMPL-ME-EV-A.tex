\begin{evaluation}{TCOMPL-ME-EV-A}{50}

\begin{exercice}{TCOMPL-ME-EV-A-EX1}{1}{20}
\textbf{Exercice 1} (8 points) --- \textit{Capacités C1, C2}

\medskip

Un système d'arrosage délivre $500$ mL d'eau chaque jour à une plante en pot ; $70\%$ de l'eau présente dans le pot s'évapore ou est absorbée chaque jour (donc $30\%$ subsiste).
\begin{enumerate}
  \item (3 pts) Identifier la raison $q$ de la suite géométrique associée.
  \item (5 pts) Déterminer la quantité limite d'eau accumulée dans le pot au bout d'un grand nombre de jours.
\end{enumerate}
\end{exercice}

\begin{exercice}{TCOMPL-ME-EV-A-EX2}{2}{30}
\textbf{Exercice 2} (12 points) --- \textit{Capacité C4}

\medskip

Une suite $(w_n)$ vérifie $w_{n+1} = 0{,}4\,w_n + 12$, avec $w_0=0$.
\begin{enumerate}
  \item (3 pts) Déterminer la suite constante $c$ solution de la relation.
  \item (4 pts) Justifier que $(w_n - c)$ est géométrique, en préciser la raison.
  \item (5 pts) En déduire l'expression explicite de $w_n$ en fonction de $n$, et calculer $w_3$.
\end{enumerate}
\end{exercice}

\end{evaluation}
