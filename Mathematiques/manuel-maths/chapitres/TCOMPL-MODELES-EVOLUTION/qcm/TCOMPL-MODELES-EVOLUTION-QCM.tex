\section*{\textcolor{chapcolor}{\MakeUppercase{Modeles d'evolution discrète et continue}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} Une suite arithmetique de raison r et premier terme $u_0$ a pour terme general :
  \begin{enumerate}[label=\Alph*.]
    \item $u_n = u_0 + n r$
    \item $u_n = u_0 \times r^n$
    \item $u_n = u_0 + r^n$
    \item $u_n = u_0 n r$
  \end{enumerate}

\item \textbf{[Q2]} Une suite geometrique de raison $q > 1$ et $u_0 > 0$ est :
  \begin{enumerate}[label=\Alph*.]
    \item Decroissante et converge vers 0
    \item Constante
    \item Strictement croissante et diverge vers $+\infty$
    \item Alternée
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q3]} L'equation differentielle $y' = a y$ a pour solutions les fonctions :
  \begin{enumerate}[label=\Alph*.]
    \item $f(x) = a x + C$
    \item $f(x) = C e^{a x}$
    \item $f(x) = C e^{-a x}$
    \item $f(x) = C x^a$
  \end{enumerate}

\item \textbf{[Q4]} Le modele de Malthus suppose un taux de croissance :
  \begin{enumerate}[label=\Alph*.]
    \item Decroissant
    \item Nul
    \item Logistique
    \item Constant et proportionnel a la population
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q5]} Le modele logistique de Verhulst comporte une capacite d'accueil K qui est :
  \begin{enumerate}[label=\Alph*.]
    \item La limite de la population quand $t \to +\infty$
    \item La population initiale $u_0$
    \item Le taux de croissance initial
    \item Le temps de demi-vie
  \end{enumerate}

\end{enumerate}
