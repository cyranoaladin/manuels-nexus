\section*{QCM d'auto-évaluation --- Modèles d'évolution}

\subsection*{C1/C2 --- Modélisation, suite géométrique}

\begin{enumerate}

\item \textbf{[Q1]} Pour une suite géométrique de raison $q$ avec $0<q<1$, la limite de $u_n$ quand $n\to+\infty$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $+\infty$.
    \item $0$.
    \item $u_0$.
    \item $q$.
  \end{enumerate}

\bigskip

\item \textbf{[Q2]} La limite de la somme $S_n$ des termes d'une suite géométrique de premier terme $u_0$ et de raison $q \in\, ]0,1[$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $u_0(1-q)$.
    \item $\dfrac{u_0}{1-q}$.
    \item $\dfrac{u_0}{1+q}$.
    \item $u_0 \times q$.
  \end{enumerate}

\bigskip

\subsection*{C3/C4 --- Suites récurrentes}

\item \textbf{[Q3]} Sur la représentation graphique d'une suite $u_{n+1}=f(u_n)$ (courbe de $f$ et droite $y=x$), un point fixe candidat pour la limite correspond à :
  \begin{enumerate}[label=\Alph*.]
    \item l'origine du repère.
    \item un point d'intersection entre la courbe de $f$ et la droite $y=x$.
    \item le point $u_0$.
    \item n'importe quel point de la courbe.
  \end{enumerate}

\bigskip

\item \textbf{[Q4]} Pour une suite arithmético-géométrique $u_{n+1}=au_n+b$ (avec $a\neq1$), la suite constante $c$ solution vérifie :
  \begin{enumerate}[label=\Alph*.]
    \item $c = a+b$.
    \item $c = ac+b$.
    \item $c = a-b$.
    \item $c = b/a$.
  \end{enumerate}

\bigskip

\subsection*{C5 --- Équation différentielle}

\item \textbf{[Q5]} La solution générale de $y'=ay$ (avec $a$ réel) est :
  \begin{enumerate}[label=\Alph*.]
    \item $y(x) = ax + C$.
    \item $y(x) = C\,\mathrm{e}^{ax}$.
    \item $y(x) = a\,\mathrm{e}^{x}$.
    \item $y(x) = C^{ax}$.
  \end{enumerate}

\bigskip

\item \textbf{[Q6]} Pour résoudre $y'=ay+b$, on cherche d'abord :
  \begin{enumerate}[label=\Alph*.]
    \item la limite de $y$ en $+\infty$.
    \item une solution particulière constante $y_p=-b/a$.
    \item la dérivée seconde de $y$.
    \item la valeur de $y$ en $x=1$.
  \end{enumerate}

\end{enumerate}
