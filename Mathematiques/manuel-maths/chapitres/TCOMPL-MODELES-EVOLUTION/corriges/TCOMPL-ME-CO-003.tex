\begin{corrige}{TCOMPL-ME-EX-003}
\textbf{1.} $y_p$ vérifie $-0{,}25\,y_p+60=0$, soit $y_p = \dfrac{60}{0{,}25} = 240$.

\textbf{2.} La solution générale est $y(t) = C\,\mathrm{e}^{-0{,}25t} + 240$. La condition $y(0)=180$ donne $C+240=180$, soit $C=-60$ :
\[
  y(t) = -60\,\mathrm{e}^{-0{,}25t} + 240.
\]

\textbf{3.} Quand $t\to+\infty$, $\mathrm{e}^{-0{,}25t}\to0$, donc $y(t) \to 240$ : la température de l'objet se stabilise à $240$ degrés (température du four).
\end{corrige}
