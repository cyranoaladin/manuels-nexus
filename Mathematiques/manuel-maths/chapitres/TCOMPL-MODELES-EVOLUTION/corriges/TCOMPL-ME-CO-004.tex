\begin{corrige}{TCOMPL-ME-EX-004}
\textbf{1.} $f(x) = \dfrac{1}{2}\left(x+\dfrac{6}{x}\right)$ pour $x>0$.

\textbf{2.} $f(x)=x \iff \dfrac{1}{2}\left(x+\dfrac{6}{x}\right)=x \iff x+\dfrac{6}{x}=2x \iff \dfrac{6}{x}=x \iff x^2=6 \iff x=\sqrt{6}$ (car $x>0$). Le point fixe est $\sqrt{6}$.

\textbf{3.} $u_1 = \dfrac{1}{2}\left(10+\dfrac{6}{10}\right) = \dfrac{1}{2}\times10{,}6 = 5{,}3$. $u_2 = \dfrac{1}{2}\left(5{,}3+\dfrac{6}{5{,}3}\right) \approx 3{,}216$. On observe une convergence très rapide vers $\sqrt{6}\approx2{,}449$ : c'est la célèbre méthode de Héron (ou de Newton).
\end{corrige}
