\section*{\textcolor{chapcolor}{\MakeUppercase{Calcul d'integrales et primitives}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} Soit $f(x) = x^2$. L'integrale $\int_{0}^{3} f(x) dx$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item 3
    \item 9
    \item 27
    \item 18
  \end{enumerate}

\item \textbf{[Q2]} La valeur moyenne de $f(x) = 2x$ sur $[0, 4]$ est :
  \begin{enumerate}[label=\Alph*.]
    \item 2
    \item 8
    \item 4
    \item 16
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q3]} Si $F$ est une primitive de $f$ sur $[a, b]$, alors $\int_a^b f(x) dx$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $F(b) - F(a)$
    \item $F(a) - F(b)$
    \item $f(b) - f(a)$
    \item $F(b) + F(a)$
  \end{enumerate}

\item \textbf{[Q4]} $\int_0^1 e^{2x} dx$ est egale a :
  \begin{enumerate}[label=\Alph*.]
    \item $e^2 - 1$
    \item $\frac{e^2 - 1}{2}$
    \item $2(e^2 - 1)$
    \item $e^2$
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q5]} Par integration par parties de $\int x e^x dx$, la primitive est :
  \begin{enumerate}[label=\Alph*.]
    \item $x e^x$
    \item $e^x$
    \item $(x+1)e^x$
    \item $(x-1)e^x$
  \end{enumerate}

\end{enumerate}
