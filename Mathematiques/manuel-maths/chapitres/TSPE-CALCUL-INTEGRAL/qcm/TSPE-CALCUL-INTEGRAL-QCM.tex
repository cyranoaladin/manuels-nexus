\subsection*{C2/C7 — Calcul et théorème fondamental}

\begin{enumerate}

\item \textbf{[Q1]} Si $F$ est une primitive de $f$ sur $[a,b]$, alors $\displaystyle\int_a^b f(x)\,\mathrm{d}x$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $F(a)-F(b)$.
    \item $F(b)-F(a)$.
    \item $F(b)+F(a)$.
    \item $F'(b)-F'(a)$.
  \end{enumerate}

\bigskip

\item \textbf{[Q2]} Pour une fonction $f$ continue, positive et croissante, la fonction $x \mapsto \int_a^x f(t)\,\mathrm{d}t$ est :
  \begin{enumerate}[label=\Alph*.]
    \item une primitive de $f$.
    \item une primitive de $f'$.
    \item constante.
    \item toujours nulle.
  \end{enumerate}

\subsection*{C8 — Intégration par parties}

\item \textbf{[Q3]} La formule d'integration par parties s'ecrit :
  \begin{enumerate}[label=\Alph*.]
    \item $\int_a^b uv' = \int_a^b u'v$.
    \item $\int_a^b uv' = [uv]_a^b + \int_a^b u'v$.
    \item $\int_a^b uv' = [uv]_a^b - \int_a^b u'v$.
    \item $\int_a^b uv' = [u'v']_a^b$.
  \end{enumerate}

\bigskip

\subsection*{C3/C4 — Comparaison et aires}

\item \textbf{[Q4]} Si $f \leq g$ sur $[a,b]$, alors :
  \begin{enumerate}[label=\Alph*.]
    \item $\int_a^b f \geq \int_a^b g$.
    \item $\int_a^b f \leq \int_a^b g$.
    \item $\int_a^b f = \int_a^b g$.
    \item on ne peut rien dire.
  \end{enumerate}

\bigskip

\item \textbf{[Q5]} L'aire entre les courbes de $f$ et $g$ sur $[a,b]$, avec $g \leq f$, se calcule par :
  \begin{enumerate}[label=\Alph*.]
    \item $\int_a^b (f+g)$.
    \item $\int_a^b (f-g)$.
    \item $\int_a^b (g-f)$.
    \item $\int_a^b f \times \int_a^b g$.
  \end{enumerate}

\end{enumerate}
