\section*{\textcolor{chapcolor}{\MakeUppercase{Calcul d'integrales et primitives}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q1]} Soit $f(x) = x^2$. L'integrale $\int_{0}^{3} f(x) dx$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item 3
    \item 9
    \item 27
    \item 18
  \end{enumerate}

\item \textbf{[Q2]} La valeur moyenne de $f(x) = 2x$ sur $[0, 4]$ est :
  \begin{enumerate}[label=\Alph*.]
    \item 2
    \item 8
    \item 4
    \item 16
  \end{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q3]} On sait que $2 \leqslant f(x) \leqslant 5$ sur $[0,3]$. En notant $I=\int_0^3 f(x)\,dx$, quel encadrement est garanti ?
  \begin{enumerate}[label=\Alph*.]
    \item $2 \leqslant I \leqslant 5$
    \item $6 \leqslant I \leqslant 15$
    \item $0 \leqslant I \leqslant 3$
    \item $10 \leqslant I \leqslant 25$
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q4]} Si $f(x) \leqslant g(x)$ pour tout $x\in[0,1]$, avec $f$ et $g$ continues, alors :
  \begin{enumerate}[label=\Alph*.]
    \item $\int_0^1 f(x)\,dx \leqslant \int_0^1 g(x)\,dx$
    \item $\int_0^1 f(x)\,dx \geqslant \int_0^1 g(x)\,dx$
    \item $\int_0^1 f(x)\,dx = \int_0^1 g(x)\,dx$
    \item $\int_0^1 f(x)\,dx \leqslant 0$
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q5]} Par integration par parties de $\int x e^x dx$, une primitive est :
  \begin{enumerate}[label=\Alph*.]
    \item $x e^x$
    \item $e^x$
    \item $(x+1)e^x$
    \item $(x-1)e^x$
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des reponses erronees}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{A} — Erreur de primitive $x$ au lieu de $x^3/3$. \emph{Renvoi : C2.}
    \item \textbf{C} — Le calcul $3^3=27$ oublie le facteur $1/3$ de la primitive $x^3/3$ ; la valeur correcte est donc $27/3=9$. \emph{Renvoi : C2.}
    \item \textbf{D} — Le resultat $18$ double a tort la valeur correcte $9$ ; aucune symetrie ni aucun coefficient ne justifie ce facteur $2$ sur $[0,3]$. \emph{Renvoi : C2.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — Le choix $2$ remplace la valeur moyenne sur tout l'intervalle par la valeur ponctuelle $f(1)=2$, qui ne tient pas compte de $[0,4]$. \emph{Renvoi : C2.}
    \item \textbf{B} — Le resultat $8$ est $f(4)=8$ : il evalue la fonction a la borne droite au lieu de calculer puis moyenner l'integrale. \emph{Renvoi : C2.}
    \item \textbf{D} — L'integrale vaut $16$ ; choisir $16$ oublie ensuite le facteur $1/(4-0)$ indispensable dans la definition de la valeur moyenne. \emph{Renvoi : C2.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — Les bornes $2$ et $5$ concernent les valeurs de $f$ ; leur integration sur un intervalle de longueur $3$ multiplie chacune par $3$. \emph{Renvoi : C1.}
    \item \textbf{C} — La positivite donne seulement $I\geq0$ ; la majoration $I\leq3$ oublie que $f$ peut atteindre $5$ sur un intervalle de longueur $3$. \emph{Renvoi : C1.}
    \item \textbf{D} — Multiplier les bornes de $f$ par $5$ utilise la borne superieure comme longueur ; la longueur de $[0,3]$ vaut $3$. \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{B} — Le sens de l'inegalite est inverse : l'integration conserve l'ordre entre deux fonctions continues comparees sur le meme intervalle. \emph{Renvoi : C3.}
    \item \textbf{C} — L'hypothese $f\leqslant g$ n'impose pas leur egalite ; par exemple $f=0$ et $g=1$ donnent des integrales distinctes. \emph{Renvoi : C3.}
    \item \textbf{D} — La comparaison entre $f$ et $g$ ne donne aucun signe absolu pour $f$ ; il faudrait en plus supposer $f\leqslant0$. \emph{Renvoi : C3.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — Oubli du terme du a la derivee de $x$. \emph{Renvoi : C2.}
    \item \textbf{B} — Choisir $e^x$ revient a supprimer le facteur $x$ avant d'integrer ; sa derivee vaut seulement $e^x$, et non $xe^x$. \emph{Renvoi : C2.}
    \item \textbf{C} — Erreur de signe dans le terme $-e^x$. \emph{Renvoi : C2.}
  \end{itemize}
\end{itemize}
