\section*{\textcolor{chapcolor}{\MakeUppercase{Calcul d'intégrales et primitives}}}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q1]} Soit $f(x) = x^2$. L'intégrale $\int_{0}^{3} f(x) dx$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item 9
    \item 3
    \item 27
    \item 18
  \end{enumerate}

\item \textbf{[Q2]} La valeur moyenne de $f(x) = 2x$ sur $[0, 4]$ est :
  \begin{enumerate}[label=\Alph*.]
    \item 4
    \item 8
    \item 2
    \item 16
  \end{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q3]} On sait que $2 \leqslant f(x) \leqslant 5$ sur $[0,3]$. En notant $I=\int_0^3 f(x)\,dx$, quel encadrement est garanti ?
  \begin{enumerate}[label=\Alph*.]
    \item $2 \leqslant I \leqslant 5$
    \item $6 \leqslant I \leqslant 15$
    \item $0 \leqslant I \leqslant 3$
    \item $10 \leqslant I \leqslant 25$
  \end{enumerate}

\item[] \textbf{Capacité C3}

\item \textbf{[Q4]} Si $f(x) \leqslant g(x)$ pour tout $x\in[0,1]$, avec $f$ et $g$ continues, alors :
  \begin{enumerate}[label=\Alph*.]
    \item $\int_0^1 f(x)\,dx = \int_0^1 g(x)\,dx$
    \item $\int_0^1 f(x)\,dx \geqslant \int_0^1 g(x)\,dx$
    \item $\int_0^1 f(x)\,dx \leqslant \int_0^1 g(x)\,dx$
    \item $\int_0^1 f(x)\,dx \leqslant 0$
  \end{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q5]} Par integration par parties de $\int x e^x dx$, une primitive est :
  \begin{enumerate}[label=\Alph*.]
    \item $x e^x$
    \item $e^x$
    \item $(x+1)e^x$
    \item $(x-1)e^x$
  \end{enumerate}

\item[] \textbf{Capacité C4}

\item \textbf{[Q6]} Sur $[0;1]$, l'aire du domaine compris entre les courbes de $f(x)=x$ et $g(x)=x^2$ vaut :
  \begin{enumerate}[label=\Alph*.]
    \item $\tfrac12$
    \item $\tfrac16$
    \item $\tfrac13$
    \item $\tfrac56$
  \end{enumerate}

\item[] \textbf{Capacité C5}

\item \textbf{[Q7]} Soit $I_n=\displaystyle\int_0^1 x^n e^x\,dx$ pour $n\geq1$. Une intégration par parties donne la relation de récurrence :
  \begin{enumerate}[label=\Alph*.]
    \item $I_n=e+nI_{n-1}$
    \item $I_n=nI_{n-1}-e$
    \item $I_n=e-nI_{n-1}$
    \item $I_n=I_{n-1}$
  \end{enumerate}

\item[] \textbf{Capacité C6}

\item \textbf{[Q8]} Un véhicule roule à la vitesse $v(t)$, en m$\cdot$s$^{-1}$, entre $t=0$ et $t=60$ s. La quantité $\dfrac1{60}\displaystyle\int_0^{60}v(t)\,dt$ représente :
  \begin{enumerate}[label=\Alph*.]
    \item la distance parcourue, en mètres
    \item l'accélération moyenne
    \item la vitesse maximale atteinte
    \item la vitesse moyenne sur la minute, en m$\cdot$s$^{-1}$
  \end{enumerate}

\item[] \textbf{Capacité C7}

\item \textbf{[Q9]} Soit $f$ continue, positive et croissante sur $[a;b]$, et $F(x)=\displaystyle\int_a^x f(t)\,dt$. Pour démontrer $F'(x)=f(x)$, l'encadrement utilisé pour $h>0$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $h\,f(x)\leq F(x+h)-F(x)\leq h\,f(x+h)$
    \item $h\,f(x+h)\leq F(x+h)-F(x)\leq h\,f(x)$
    \item $F(x+h)-F(x)=h\,f(x)$
    \item $F(x+h)-F(x)\leq0$
  \end{enumerate}

\item[] \textbf{Capacité C8}

\item \textbf{[Q10]} La formule $\displaystyle\int_a^b u'v=[uv]_a^b-\int_a^b uv'$ se démontre en intégrant entre $a$ et $b$ :
  \begin{enumerate}[label=\Alph*.]
    \item la dérivée du quotient $\tfrac uv$
    \item la relation $(uv)'=u'v+uv'$
    \item la relation $(uv)'=u'v'$
    \item la formule de dérivation d'une fonction composée
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des réponses erronées}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{B} — Choisir $3$ revient a prendre $x$ pour primitive de $x^2$ puis a calculer $[x]_0^3$ ; or une primitive correcte est $x^3/3$. \emph{Renvoi : C2.}
    \item \textbf{C} — Le calcul $3^3=27$ oublie le facteur $1/3$ de la primitive $x^3/3$ ; la valeur correcte est donc $27/3=9$. \emph{Renvoi : C2.}
    \item \textbf{D} — Le résultat $18$ double a tort la valeur correcte $9$ ; aucune symetrie ni aucun coefficient ne justifie ce facteur $2$ sur $[0,3]$. \emph{Renvoi : C2.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{B} — Le résultat $8$ est $f(4)=8$ : il evalue la fonction a la borne droite au lieu de calculer puis moyenner l'intégrale. \emph{Renvoi : C2.}
    \item \textbf{C} — Le choix $2$ remplace la valeur moyenne sur tout l'intervalle par la valeur ponctuelle $f(1)=2$, qui ne tient pas compte de $[0,4]$. \emph{Renvoi : C2.}
    \item \textbf{D} — L'intégrale vaut $16$ ; choisir $16$ oublie ensuite le facteur $1/(4-0)$ indispensable dans la définition de la valeur moyenne. \emph{Renvoi : C2.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — Les bornes $2$ et $5$ concernent les valeurs de $f$ ; leur integration sur un intervalle de longueur $3$ multiplie chacune par $3$. \emph{Renvoi : C1.}
    \item \textbf{C} — La positivité donne seulement $I\geq0$ ; la majoration $I\leq3$ oublie que $f$ peut atteindre $5$ sur un intervalle de longueur $3$. \emph{Renvoi : C1.}
    \item \textbf{D} — Multiplier les bornes de $f$ par $5$ utilise la borne superieure comme longueur ; la longueur de $[0,3]$ vaut $3$. \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — L'hypothèse $f\leqslant g$ n'impose pas leur égalité ; par exemple $f=0$ et $g=1$ donnent des intégrales distinctes. \emph{Renvoi : C3.}
    \item \textbf{B} — Le sens de l'inégalité est inverse : l'integration conserve l'ordre entre deux fonctions continues comparees sur le même intervalle. \emph{Renvoi : C3.}
    \item \textbf{D} — La comparaison entre $f$ et $g$ ne donne aucun signe absolu pour $f$ ; il faudrait en plus supposer $f\leqslant0$. \emph{Renvoi : C3.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — La dérivée de $xe^x$ vaut $e^x+xe^x$ : le terme $e^x$ issu de la dérivée de $x$ empeche cette expression d'être une primitive de $xe^x$. \emph{Renvoi : C2.}
    \item \textbf{B} — Choisir $e^x$ revient a supprimer le facteur $x$ avant d'integrer ; sa dérivée vaut seulement $e^x$, et non $xe^x$. \emph{Renvoi : C2.}
    \item \textbf{C} — La dérivée de $(x+1)e^x$ vaut $(x+2)e^x$ ; l'integration par parties soustrait $e^x$ et conduit a $(x-1)e^x$. \emph{Renvoi : C2.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{A} — $\tfrac12$ est l'aire sous la seule courbe de $f$. L'aire entre deux courbes intègre leur \textbf{différence} : $\int_0^1(x-x^2)\,dx=\tfrac12-\tfrac13=\tfrac16$. \emph{Renvoi : C4.}
    \item \textbf{C} — $\tfrac13$ est l'aire sous la seule courbe de $g$. Il faut retrancher cette aire de celle sous $f$, soit $\tfrac12-\tfrac13=\tfrac16$. \emph{Renvoi : C4.}
    \item \textbf{D} — $\tfrac12+\tfrac13=\tfrac56$ additionne les deux aires au lieu de les soustraire ; sur $[0;1]$ on a $x^2\leq x$, et le domaine entre les courbes est leur différence. \emph{Renvoi : C4.}
  \end{itemize}
  \item \textbf{Q7}
  \begin{itemize}
    \item \textbf{A} — La formule d'intégration par parties retranche l'intégrale restante : $I_n=[x^ne^x]_0^1-\int_0^1 nx^{n-1}e^x\,dx=e-nI_{n-1}$. \emph{Renvoi : C5.}
    \item \textbf{B} — Les deux signes sont inversés. Le crochet $[x^ne^x]_0^1$ vaut $e$ et arrive en tête avec le signe $+$ ; l'intégrale restante est retranchée. \emph{Renvoi : C5.}
    \item \textbf{D} — La suite n'est pas constante : $I_0=e-1\approx1{,}718$ et $I_1=1$. La relation $I_1=e-1\times I_0=e-(e-1)=1$ le confirme. \emph{Renvoi : C5.}
  \end{itemize}
  \item \textbf{Q8}
  \begin{itemize}
    \item \textbf{A} — La distance est l'intégrale seule, $\int_0^{60}v(t)\,dt$. La division par la durée transforme cette distance en une moyenne, et change l'unité de mètres en m$\cdot$s$^{-1}$. \emph{Renvoi : C6.}
    \item \textbf{B} — L'accélération moyenne est $\dfrac{v(60)-v(0)}{60}$ : une variation de vitesse rapportée au temps, pas une moyenne de la vitesse elle-même. \emph{Renvoi : C6.}
    \item \textbf{C} — Une moyenne ne dépasse jamais le maximum et ne l'atteint que si $v$ est constante. Sur un trajet réel, la vitesse moyenne est strictement inférieure à la vitesse maximale. \emph{Renvoi : C6.}
  \end{itemize}
  \item \textbf{Q9}
  \begin{itemize}
    \item \textbf{B} — L'encadrement est inversé : $f$ croissante donne $f(x)\leq f(t)\leq f(x+h)$ pour $t\in[x;x+h]$, donc l'aire est comprise entre les deux rectangles, le petit d'abord. \emph{Renvoi : C7.}
    \item \textbf{C} — L'égalité ne vaut que si $f$ est constante sur $[x;x+h]$ : l'aire sous la courbe n'est pas exactement celle d'un rectangle. C'est le passage à la limite qui rétablit l'égalité. \emph{Renvoi : C7.}
    \item \textbf{D} — $f$ est positive, donc $F$ est croissante : la différence $F(x+h)-F(x)$ est positive. Elle est même minorée par $h\,f(x)$. \emph{Renvoi : C7.}
  \end{itemize}
  \item \textbf{Q10}
  \begin{itemize}
    \item \textbf{A} — Le quotient conduit à une autre formule et suppose $v$ ne s'annulant pas. La formule cherchée porte sur un produit, et vient de la dérivée d'un produit. \emph{Renvoi : C8.}
    \item \textbf{C} — La dérivée d'un produit n'est pas le produit des dérivées : avec $u=v=x$, $(x^2)'=2x$ alors que $u'v'=1$. \emph{Renvoi : C8.}
    \item \textbf{D} — La dérivation d'une composée fournit le changement de variable, une autre technique d'intégration. L'intégration par parties vient de la dérivée du produit. \emph{Renvoi : C8.}
  \end{itemize}
\end{itemize}
