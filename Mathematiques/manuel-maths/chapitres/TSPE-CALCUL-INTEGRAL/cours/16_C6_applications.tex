\section{Interprétation de l'intégrale dans un contexte concret}

% ============================================================
% STRATE 1 — Exemple d'interprétation physique
% ============================================================

\exemple{
La vitesse d'un mobile (en m/s) est donnee par $v(t) = 3t^2+2t$ pour $t \in [0,4]$ (temps en secondes).

\textbf{1. Interprétation de l'intégrale.} La distance parcourue entre les instants $t=1$ et $t=3$ est $\displaystyle\int_1^3 v(t)\,\mathrm{d}t$ (l'intégrale de la vitesse donne la distance, primitive de la vitesse = position).

\[
  \int_1^3 (3t^2+2t)\,\mathrm{d}t = \big[t^3+t^2\big]_1^3 = (27+9)-(1+1) = 34 \text{ metres}.
\]

\textbf{2. Vitesse moyenne.} La vitesse moyenne sur $[1,3]$ est $\mu = \dfrac{1}{3-1}\displaystyle\int_1^3 v(t)\,\mathrm{d}t = \dfrac{34}{2} = 17$ m/s.

\margeAppui{
Selon le contexte, l'intégrale d'une grandeur "par unité de temps (ou d'espace)" donne la grandeur cumulee : intégrale d'un debit = volume ; intégrale d'une puissance = énergie ; intégrale d'une densite = masse, etc.
}

% ============================================================
% STRATE 2 — Erreurs frequentes
% ============================================================

\erreurFrequente{
\textbf{Confondre valeur moyenne et intégrale.} La valeur moyenne est l'intégrale \emph{divisée par la longueur de l'intervalle} $(b-a)$. Oublier cette division est une erreur frequente.
}
