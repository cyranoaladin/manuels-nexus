\section{Algorithmique : rectangles, milieux, trapèzes}

% ============================================================
% STRATE 1 --- Probleme, modele, algorithmes
% ============================================================

\margeAppui{
« Methodes des rectangles, des milieux, des trapezes » figure parmi les
\textbf{exemples d'algorithme} du programme. Il les cite, il ne les impose pas.
}

\textbf{Le probleme.} On veut encadrer
$\displaystyle\int_1^2 \frac{\mathrm{d}x}{x}$, c'est-a-dire $\ln 2$, sans
utiliser la fonction $\ln$ -- comme on devrait le faire pour une fonction dont
on ne connait aucune primitive. Trois facons de remplacer la courbe par
quelque chose de calculable, et une comparaison de leurs erreurs.

\propriete[Les trois sommes]{
On decoupe $[a\,;\,b]$ en $n$ intervalles de largeur $h = \dfrac{b-a}{n}$, et
l'on note $x_k = a + kh$. Alors
\[
  R_{\text{g}} = h\sum_{k=0}^{n-1} f(x_k),
  \qquad
  M = h\sum_{k=0}^{n-1} f\!\left(\frac{x_k + x_{k+1}}{2}\right),
  \qquad
  T = \frac{h}{2}\sum_{k=0}^{n-1}\bigl(f(x_k) + f(x_{k+1})\bigr).
\]
$M$ remplace $f$ par sa valeur au \emph{milieu} de chaque intervalle, $T$ par
le segment joignant les deux extremites.

Entree : la fonction, les bornes, $n$. Sortie : les trois sommes.
}

\propriete[Ce que la convexite garantit]{
Si $f$ est \textbf{convexe} sur $[a\,;\,b]$, alors sa courbe est au-dessous de
chacune de ses cordes et au-dessus de chacune de ses tangentes. On en deduit
\[
  M \;\leq\; \int_a^b f(x)\,\mathrm{d}x \;\leq\; T .
\]
La methode des milieux minore, celle des trapezes majore : l'integrale est
encadree sans aucun calcul de primitive.
}

\margeAppui{
Ici $f(x) = \dfrac{1}{x}$, de derivee seconde $\dfrac{2}{x^3} > 0$ sur
$[1\,;\,2]$ : $f$ y est bien convexe.
}

\begin{python}
def quadratures(f, a, b, n):
    """Sommes de gauche, des milieux et des trapezes sur n intervalles."""
    if n < 1 or b <= a:
        raise ValueError("il faut n >= 1 et a < b")
    h = (b - a) / n
    gauche = h * sum(f(a + k * h) for k in range(n))
    milieux = h * sum(f(a + (k + 0.5) * h) for k in range(n))
    trapezes = h * sum((f(a + k * h) + f(a + (k + 1) * h)) / 2 for k in range(n))
    return gauche, milieux, trapezes
\end{python}

\exemple{
Pour $f(x) = \dfrac{1}{x}$ sur $[1\,;\,2]$, avec $\ln 2 = 0{,}693\,147\ldots$ :

\begin{center}
\begin{tabular}{|c|c|c|c|}
\hline
$n$ & milieux $M$ & trapezes $T$ & $\dfrac{2M + T}{3}$ \\
\hline
$10$ & $0{,}692\,835$ & $0{,}693\,771$ & $0{,}693\,147$ \\
\hline
$100$ & $0{,}693\,144$ & $0{,}693\,153$ & $0{,}693\,147$ \\
\hline
$1000$ & $0{,}693\,147$ & $0{,}693\,147$ & $0{,}693\,147$ \\
\hline
\end{tabular}
\end{center}

L'encadrement $M \leq \ln 2 \leq T$ tient a chaque ligne. Mieux : l'erreur des
trapezes est le \textbf{double} de celle des milieux, et de signe oppose. La
combinaison $\dfrac{2M+T}{3}$ annule donc les deux erreurs a l'ordre principal
-- avec dix intervalles seulement, elle donne six decimales de $\ln 2$.
}

\propriete[Travail demande]{%
\begin{enumerate}
  \item Verifier l'encadrement $M \leq \ln 2 \leq T$ sur les trois lignes.
  \item Calculer le rapport $\dfrac{\ln 2 - M}{T - \ln 2}$ pour chaque $n$ :
        que constate-t-on ?
  \item Multiplier $n$ par $10$ : par combien l'erreur est-elle divisee ?
  \item Reprendre avec $f(x) = -\dfrac{1}{x}$ : dans quel sens l'encadrement
        se fait-il ? Pourquoi ?
\end{enumerate}
}

\exemple{
\textbf{Reponses.} 2. Le rapport vaut $\frac{1}{2}$ : l'erreur des trapezes
est le double de celle des milieux. 3. Par $100$ : les deux methodes sont
d'ordre $2$. 4. La fonction devient \emph{concave} : les roles s'echangent,
$T \leq I \leq M$.
}

% STRATE 2 --- Erreurs frequentes
% ============================================================

\erreurFrequente{
\textbf{Croire que la methode des milieux majore toujours l'erreur des
trapezes.} C'est l'inverse : a nombre d'intervalles egal, les milieux sont
\emph{deux fois} plus precis. L'intuition « le trapeze epouse mieux la
courbe » est fausse pour une fonction convexe.
}

\erreurFrequente{
\textbf{Utiliser l'encadrement sans verifier la convexite.} $M \leq I \leq T$
suppose $f$ convexe sur tout l'intervalle. Sans cette hypothese, les deux
sommes restent des valeurs approchees, mais rien ne garantit plus qu'elles
encadrent l'integrale : sur une fonction qui change de convexite, elles
peuvent se trouver du meme cote.
}
