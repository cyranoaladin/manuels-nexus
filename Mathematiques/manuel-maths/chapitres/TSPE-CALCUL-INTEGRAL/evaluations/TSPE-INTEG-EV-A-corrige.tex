\begin{corrige}{TSPE-INTEG-EV-A-EX1}
$f$ est décroissante sur $[1,2]$ (fonction inverse). Minimum : $f(2)=\dfrac{1}{2}$. Maximum : $f(1)=1$.

Encadrement : $\dfrac{1}{2}\times(2-1) \leq \displaystyle\int_1^2 f(x)\,\mathrm{d}x \leq 1\times(2-1)$, soit $\dfrac{1}{2} \leq \displaystyle\int_1^2 \dfrac{1}{x}\,\mathrm{d}x \leq 1$.
\end{corrige}

\begin{corrige}{TSPE-INTEG-EV-A-EX2}
$u(x)=x$, $u'(x)=1$ ; $v'(x)=\mathrm{e}^{-x}$, $v(x)=-\mathrm{e}^{-x}$.
\[
  \int_0^2 x\,\mathrm{e}^{-x}\,\mathrm{d}x = \big[-x\,\mathrm{e}^{-x}\big]_0^2 - \int_0^2 (-\mathrm{e}^{-x})\,\mathrm{d}x = -2\mathrm{e}^{-2} + \big[-\mathrm{e}^{-x}\big]_0^2 = -2\mathrm{e}^{-2} + (-\mathrm{e}^{-2}+1) = 1-3\mathrm{e}^{-2}.
\]
\end{corrige}

\begin{corrige}{TSPE-INTEG-EV-A-EX3}
\textbf{Q1.} $f(x)=g(x) \iff 3-x^2=x^2-1 \iff 2x^2=4 \iff x=\pm\sqrt{2}$.

\textbf{Q2.} $f(x)-g(x) = 4-2x^2$. En $x=0$ : $4>0$, et $4-2x^2 \geq 0 \iff x^2 \leq 2$, donc $f \geq g$ sur $[-\sqrt{2},\sqrt{2}]$.

\textbf{Q3.} $\mathcal{A} = \displaystyle\int_{-\sqrt{2}}^{\sqrt{2}} (4-2x^2)\,\mathrm{d}x = \left[4x-\dfrac{2x^3}{3}\right]_{-\sqrt{2}}^{\sqrt{2}} = \dfrac{20\sqrt{2}}{3}$ unités d'aire.
\end{corrige}

\begin{corrige}{TSPE-INTEG-EV-A-EX4}
\textbf{1.} Pour $x\in[0\,;1]$, $0\leq x^2\leq1$ donc $1\leq1+x^2\leq2$, et en passant aux inverses (fonction décroissante sur les réels strictement positifs) :
$\dfrac12\leq\dfrac{1}{1+x^2}\leq1$. En intégrant ces inégalités sur $[0\,;1]$, de longueur $1$ :
\[
  \frac12 \leq \int_0^1\frac{\mathrm{d}x}{1+x^2} \leq 1 .
\]
La valeur exacte est $\dfrac{\pi}{4}\approx0{,}785$, qui appartient bien à $\left[\dfrac12\,;1\right]$.

\textbf{2.} $\displaystyle\int_0^4 p(t)\,\mathrm{d}t=\left[3t-\frac{t^2}{4}\right]_0^4=12-4=8$ : c'est l'\textbf{énergie} consommée en quatre heures, en kW$\cdot$h. La quantité $\dfrac14\times8=2$ est la \textbf{puissance moyenne} sur ces quatre heures, en kW — celle d'un appareil qui consommerait autant en fonctionnant de façon constante.
\end{corrige}
