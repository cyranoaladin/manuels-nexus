\begin{corrige}{TSPE-INTEG-EV-B-EX1}
Une primitive de $\cos$ est $\sin$.
\[
  \int_0^{\pi/2} \cos(x)\,\mathrm{d}x = \big[\sin x\big]_0^{\pi/2} = \sin\!\left(\frac{\pi}{2}\right) - \sin(0) = 1-0 = \boxed{1}.
\]
\end{corrige}

\begin{corrige}{TSPE-INTEG-EV-B-EX2}
$u(x)=\ln x$, $u'(x)=\dfrac{1}{x}$ ; $v'(x)=1$, $v(x)=x$.
\[
  \int_1^2 \ln(x)\,\mathrm{d}x = \big[x\ln x\big]_1^2 - \int_1^2 \frac{x}{x}\,\mathrm{d}x = (2\ln 2 - 0) - \int_1^2 1\,\mathrm{d}x = 2\ln 2 - 1.
\]
\end{corrige}

\begin{corrige}{TSPE-INTEG-EV-B-EX3}
\textbf{Q1.} $K_0 = \displaystyle\int_0^1 \mathrm{e}^x\,\mathrm{d}x = \big[\mathrm{e}^x\big]_0^1 = \mathrm{e}-1$.

\textbf{Q2.} $x^n\mathrm{e}^x \geq 0$ sur $[0,1]$ (produit de deux termes positifs), donc $K_n$ est l'intégrale d'une fonction positive : $K_n \geq 0$.

\textbf{Q3.} Avec $u(x)=x^n$ ($u'(x)=nx^{n-1}$) et $v'(x)=\mathrm{e}^x$ ($v(x)=\mathrm{e}^x$) :
\[
  K_n = \big[x^n\mathrm{e}^x\big]_0^1 - \int_0^1 nx^{n-1}\mathrm{e}^x\,\mathrm{d}x = \mathrm{e} - n\,K_{n-1}.
\]
\end{corrige}

\begin{corrige}{TSPE-INTEG-EV-B-EX4}
\textbf{1.} Par la relation de Chasles, $F(x+h)-F(x)=\displaystyle\int_x^{x+h}f(t)\,\mathrm{d}t$. Comme $f$ est croissante, $f(x)\leq f(t)\leq f(x+h)$ pour tout $t\in[x\,;x+h]$. En intégrant ces inégalités sur un intervalle de longueur $h$ :
\[
  h\,f(x)\ \leq\ F(x+h)-F(x)\ \leq\ h\,f(x+h),
\]
c'est-à-dire que l'aire est comprise entre celle du rectangle « par défaut » et celle du rectangle « par excès ». En divisant par $h>0$ :
$f(x)\leq\dfrac{F(x+h)-F(x)}{h}\leq f(x+h)$.

\textbf{2.} $f$ est \textbf{continue} en $x$ : $f(x+h)\to f(x)$ quand $h\to0^+$. Les deux bornes de l'encadrement tendent donc vers $f(x)$, et le théorème des gendarmes donne
$\lim\limits_{h\to0^+}\dfrac{F(x+h)-F(x)}{h}=f(x)$. Le même raisonnement avec $h<0$ donne la limite à gauche. Donc $F$ est dérivable en $x$ et $F'(x)=f(x)$ : $F$ est une primitive de $f$.

\textbf{3.} Si $G$ est une autre primitive de $f$, alors $(G-F)'=0$ sur l'intervalle $[a\,;b]$, donc $G=F+k$ pour une constante $k$. Ainsi
\[
  G(b)-G(a)=\big(F(b)+k\big)-\big(F(a)+k\big)=F(b)-F(a)=\int_a^b f-\int_a^a f=\int_a^b f .
\]
La constante disparaît : toute primitive convient.
\end{corrige}
