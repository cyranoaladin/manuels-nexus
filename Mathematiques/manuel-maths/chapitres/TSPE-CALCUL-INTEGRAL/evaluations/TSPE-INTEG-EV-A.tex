\begin{evaluation}{TSPE-INTEG-EV-A}{55}

\begin{exercice}{TSPE-INTEG-EV-A-EX1}{2}{15}
\textbf{Exercice 1} (5 points) — \textit{Capacité C1}

\medskip

Soit $f(x)=\dfrac{1}{x}$ sur $[1,2]$. Déterminer le minimum et le maximum de $f$ sur $[1,2]$, puis encadrer $\displaystyle\int_1^2 f(x)\,\mathrm{d}x$.
\end{exercice}

\begin{exercice}{TSPE-INTEG-EV-A-EX2}{2}{20}
\textbf{Exercice 2} (6 points) — \textit{Capacité C8}

\medskip

Calculer $\displaystyle\int_0^2 x\,\mathrm{e}^{-x}\,\mathrm{d}x$ par integration par parties (poser $u(x)=x$, $v'(x)=\mathrm{e}^{-x}$).
\end{exercice}

\begin{exercice}{TSPE-INTEG-EV-A-EX3}{2}{20}
\textbf{Exercice 3} (5 points) — \textit{Capacité C4}

\medskip

On considere $f(x)=3-x^2$ et $g(x)=x^2-1$.
\begin{enumerate}
  \item (2 pts) Déterminer les points d'intersection des deux courbes.
  \item (2 pts) Justifier que $f \geq g$ entre ces deux points.
  \item (1 pt) Calculer l'aire entre les deux courbes sur cet intervalle.
\end{enumerate}
\end{exercice}

\begin{exercice}{TSPE-INTEG-EV-A-EX4}{2}{20}
\textbf{Exercice 4} (4 points) — \textit{Capacités C3, C6}

\medskip

\begin{enumerate}
  \item (2 pts) Montrer que, pour tout $x\in[0\,;1]$, $\dfrac12\leq\dfrac{1}{1+x^2}\leq1$. En déduire un encadrement de $\displaystyle\int_0^1\frac{\mathrm{d}x}{1+x^2}$, puis vérifier que la valeur exacte $\dfrac{\pi}{4}$ le respecte.
  \item (2 pts) La puissance consommée par un appareil, en kW, est $p(t)=3-0{,}5t$ pour $t$ en heures, $t\in[0\,;4]$. Calculer $\displaystyle\int_0^4 p(t)\,\mathrm{d}t$ et $\dfrac14\displaystyle\int_0^4 p(t)\,\mathrm{d}t$, puis dire ce que chacune de ces deux quantités représente, avec son unité.
\end{enumerate}
\end{exercice}
