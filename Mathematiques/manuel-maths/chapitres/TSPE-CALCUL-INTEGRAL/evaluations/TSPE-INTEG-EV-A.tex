\begin{evaluation}{TSPE-INTEG-EV-A}{55}

\begin{exercice}{TSPE-INTEG-EV-A-EX1}{2}{15}
\textbf{Exercice 1} (5 points) — \textit{Capacite C1}

\medskip

Soit $f(x)=\dfrac{1}{x}$ sur $[1,2]$. Determiner le minimum et le maximum de $f$ sur $[1,2]$, puis encadrer $\displaystyle\int_1^2 f(x)\,\mathrm{d}x$.
\end{exercice}

\begin{exercice}{TSPE-INTEG-EV-A-EX2}{2}{20}
\textbf{Exercice 2} (8 points) — \textit{Capacite C8}

\medskip

Calculer $\displaystyle\int_0^2 x\,\mathrm{e}^{-x}\,\mathrm{d}x$ par integration par parties (poser $u(x)=x$, $v'(x)=\mathrm{e}^{-x}$).
\end{exercice}

\begin{exercice}{TSPE-INTEG-EV-A-EX3}{2}{20}
\textbf{Exercice 3} (7 points) — \textit{Capacite C4}

\medskip

On considere $f(x)=3-x^2$ et $g(x)=x^2-1$.
\begin{enumerate}
  \item (2 pts) Determiner les points d'intersection des deux courbes.
  \item (2 pts) Justifier que $f \geq g$ entre ces deux points.
  \item (3 pts) Calculer l'aire entre les deux courbes sur cet intervalle.
\end{enumerate}
\end{exercice}

\end{evaluation}
