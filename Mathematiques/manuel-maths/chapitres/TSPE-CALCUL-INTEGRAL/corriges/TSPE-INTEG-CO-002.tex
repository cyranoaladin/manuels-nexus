\begin{corrige}{TSPE-INTEG-EX-002}
Une primitive de $\dfrac{1}{x}$ sur $]0,+\infty[$ est $\ln x$.
\[
  \int_1^{\mathrm{e}} \frac{1}{x}\,\mathrm{d}x = \ln(\mathrm{e}) - \ln(1) = 1 - 0 = \boxed{1}.
\]
\end{corrige}
