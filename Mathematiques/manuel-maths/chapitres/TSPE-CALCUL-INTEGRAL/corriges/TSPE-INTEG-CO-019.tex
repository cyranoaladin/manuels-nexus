\begin{corrige}{TSPE-INTEG-EX-003}

\textbf{Q1.} Pour $x \in [0,1]$, $0 \leq x \leq 1$, donc en multipliant par $x^2 \geq 0$ : $x^3 = x \times x^2 \leq 1 \times x^2 = x^2$.

\textbf{Q2.} D'apres la propriete de comparaison des integrales (fonctions continues, $x^3 \leq x^2$ sur $[0,1]$) : $\displaystyle\int_0^1 x^3\,\mathrm{d}x \leq \int_0^1 x^2\,\mathrm{d}x$.

\textbf{Q3.} $\displaystyle\int_0^1 x^2\,\mathrm{d}x = \left[\dfrac{x^3}{3}\right]_0^1 = \dfrac{1}{3}$. $\displaystyle\int_0^1 x^3\,\mathrm{d}x = \left[\dfrac{x^4}{4}\right]_0^1 = \dfrac{1}{4}$.

On verifie bien $\dfrac{1}{4} \leq \dfrac{1}{3}$.

\end{corrige}
