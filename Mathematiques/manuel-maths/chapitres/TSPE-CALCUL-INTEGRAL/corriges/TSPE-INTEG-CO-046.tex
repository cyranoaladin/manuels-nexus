\begin{corrige}{TSPE-INTEG-EX-006}

\textbf{Q1.} $\displaystyle\int_0^3 (2t+1)\,\mathrm{d}t = \left[t^2+t\right]_0^3 = (9+3)-0 = 12$.

\textbf{Q2.} Temperature moyenne : $\mu = \dfrac{1}{3-0}\displaystyle\int_0^3 T(t)\,\mathrm{d}t = \dfrac{12}{3} = 4\degres$C.
