\begin{corrige}{TSPE-INTEG-EX-001}

\textbf{Q1.} $f$ est strictement croissante sur $[0,2]$ (somme de $x^2$ croissante sur $[0,+\infty[$ et d'une constante). Minimum : $f(0)=1$. Maximum : $f(2)=5$.

\textbf{Q2.} Par la propriete d'encadrement, avec $m=1$, $M=5$, $b-a=2$ :
\[
  2 \leq \int_0^2 f(x)\,\mathrm{d}x \leq 10.
\]
