\begin{corrige}{TSPE-INTEG-EX-004}

\textbf{Q1.} $f(x)=g(x) \iff x=x^2 \iff x(1-x)=0 \iff x=0$ ou $x=1$. Points d'intersection : $(0,0)$ et $(1,1)$.

\textbf{Q2.} Pour $x \in [0,1]$, $f(x)-g(x) = x-x^2 = x(1-x)$. Comme $x \geq 0$ et $1-x \geq 0$ sur $[0,1]$, on a $f(x)-g(x) \geq 0$, soit $f \geq g$.

\textbf{Q3.} $\mathcal{A} = \displaystyle\int_0^1 (x-x^2)\,\mathrm{d}x = \left[\dfrac{x^2}{2}-\dfrac{x^3}{3}\right]_0^1 = \dfrac{1}{2}-\dfrac{1}{3} = \boxed{\dfrac{1}{6}}$ unite d'aire.

\end{corrige}
