\begin{exercice}{TSPE-INTEG-EX-007}{3}{18}
On considere $f(t) = 2t+1$ sur $[0,5]$, une fonction continue, positive et croissante. On pose $A(x) = \displaystyle\int_0^x f(t)\,\mathrm{d}t$ pour $x \in [0,5]$.
\begin{enumerate}
  \item En reprenant la demonstration du cours, justifier que pour $h>0$ et $x_0 \in [0,5[$ :
  \[
    h\,f(x_0) \leq A(x_0+h) - A(x_0) \leq h\,f(x_0+h).
  \]
  \item En deduire (sans calculer $A(x)$ explicitement) que $A'(x_0) = f(x_0)$.
  \item Verifier ce resultat en calculant explicitement $A(x)$ (aire d'un trapeze), puis $A'(x)$.
\end{enumerate}
\end{exercice}
