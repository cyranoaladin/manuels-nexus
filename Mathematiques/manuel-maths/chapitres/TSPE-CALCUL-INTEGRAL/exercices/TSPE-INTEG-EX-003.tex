\begin{exercice}{TSPE-INTEG-EX-003}{2}{12}
\begin{enumerate}
  \item Justifier que $x^3 \leq x^2$ pour tout $x \in [0,1]$.
  \item En deduire que $\displaystyle\int_0^1 x^3\,\mathrm{d}x \leq \int_0^1 x^2\,\mathrm{d}x$.
  \item Calculer les deux integrales et verifier l'inegalite obtenue.
\end{enumerate}
\end{exercice}
