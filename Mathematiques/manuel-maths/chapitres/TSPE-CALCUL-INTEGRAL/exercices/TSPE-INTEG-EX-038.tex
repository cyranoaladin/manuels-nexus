\begin{exercice}{TSPE-INTEG-EX-006}{2}{12}
La temperature d'une piece (en \degres C) est modelisee par $T(t) = 2t+1$ pour $t \in [0,3]$ (temps en heures).
\begin{enumerate}
  \item Calculer $\displaystyle\int_0^3 T(t)\,\mathrm{d}t$.
  \item En deduire la temperature moyenne de la piece entre $t=0$ et $t=3$.
\end{enumerate}
\end{exercice}
