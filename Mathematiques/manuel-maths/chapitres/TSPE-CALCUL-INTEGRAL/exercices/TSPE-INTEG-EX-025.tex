\begin{exercice}{TSPE-INTEG-EX-001}{1}{8}
Soit $f(x) = x^2+1$ sur $[0,2]$.
\begin{enumerate}
  \item Determiner le minimum et le maximum de $f$ sur $[0,2]$.
  \item En deduire un encadrement de $\displaystyle\int_0^2 f(x)\,\mathrm{d}x$.
\end{enumerate}
\end{exercice}
