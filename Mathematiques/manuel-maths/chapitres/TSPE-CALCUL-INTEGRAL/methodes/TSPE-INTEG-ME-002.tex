\begin{fichemethode}{M2}{Calculer l'aire entre deux courbes}

\quandUtiliser{%
  Utiliser cette méthode pour calculer l'aire du domaine compris entre les courbes de deux fonctions $f$ et $g$.
}

\pasApas{%
  \begin{enumerate}
    \item \textbf{Déterminer les points d'intersection} en resolvant $f(x)=g(x)$.
    \item \textbf{Déterminer quelle courbe est au-dessus} sur chaque intervalle (étudier le signe de $f-g$, par exemple en un point test).
    \item \textbf{Integrer} $|f-g|$ (c'est-a-dire la plus grande fonction moins la plus petite) sur l'intervalle considere.
    \item \textbf{Exprimer le résultat en unités d'aire}, en precisant l'unité si le repere n'est pas orthonorme.
  \end{enumerate}
}

\end{fichemethode}
