\begin{fichemethode}{M4}{Compter les chemins de longueur donnee avec les puissances de la matrice}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande le NOMBRE de chemins (ou
  chaines) de longueur $n$ entre deux sommets : il se lit dans la puissance
  $n$-ieme de la matrice d'adjacence.
}

\pasApas{%
  \begin{enumerate}
    \item Écrire la matrice d'adjacence $M$ dans un ordre de sommets fixe
      (fiche M3).
    \item Calculer $M^n$ (produits successifs, ou calculatrice pour $n$
      grand) — attention, produit MATRICIEL, pas coefficient par coefficient.
    \item Citer le théorème (démonstration : fiche M5) : le coefficient
      $(i\,;\,j)$ de $M^n$ est le nombre de chemins de longueur $n$ du sommet
      $i$ au sommet $j$.
    \item Lire le coefficient demande, a l'intersection de la bonne ligne
      (depart) et de la bonne colonne (arrivee).
    \item Si possible, ENUMERER les chemins pour une petite longueur afin de
      contrôler la lecture.
  \end{enumerate}
}

\exempleRedige{%
  Graphe d'arêtes $AB$, $AC$, $BC$, $CD$. Combien de chaines de longueur 2 de
  $A$ a $D$ ? de longueur 3 de $A$ a $B$ ?

  \medskip
  \commentaireMarge{Le coefficient (1;4) de $M^2$ repond a la question.}%
  $M^2 = \begin{pmatrix}
  2 & 1 & 1 & 1\\
  1 & 2 & 1 & 1\\
  1 & 1 & 3 & 0\\
  1 & 1 & 0 & 1
  \end{pmatrix}$ : le coefficient ligne $A$, colonne $D$ vaut 1 — l'unique
  chaine est $A - C - D$.

  \medskip
  \commentaireMarge{Enumeration de contrôle.}%
  Le coefficient ligne $A$, colonne $B$ de $M^3$ vaut 3 : les trois chaines de
  longueur 3 de $A$ a $B$ sont $A\,B\,A\,B$, $A\,C\,A\,B$ et $A\,B\,C\,B$
  (une chaine peut repasser par un sommet deja visite).
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Elever les coefficients au carre un par un :
      $M^2$ est un produit MATRICIEL.
    \item \textbf{Piège 2.} Confondre ligne et colonne : la ligne est le
      DEPART, la colonne l'ARRIVEE (indifferent ici par symetrie, crucial en
      oriente).
    \item \textbf{Piège 3.} Oublier que les chemins comptes peuvent repasser
      par un même sommet ou une même arête : ce ne sont pas des chemins
      « sans repetition ».
  \end{itemize}
}

\verifier{%
  Enumerer les chemins a la main pour une petite longueur et comparer au
  coefficient, et contrôler $M^{n} = M^{n-1} M$ sur un produit.
}

\sEntrainer{\refExos{M4}}

\end{fichemethode}
