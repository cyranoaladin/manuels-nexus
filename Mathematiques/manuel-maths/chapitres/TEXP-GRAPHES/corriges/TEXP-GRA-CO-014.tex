\begin{corrige}{TEXP-GRA-EX-014}
\textbf{1.} $M^2$ donne le coefficient $(1\,;\,4)=1$ : un seul chemin de
longueur $2$, à savoir $1-3-4$.

\textbf{2.} Le coefficient $(1\,;\,5)$ de $M^2$ vaut $1$ : le chemin $1-3-5$.
Celui de $M^3$ vaut $2$ : les deux chemins de longueur $3$ sont $1-2-3-5$ et
$1-3-4-5$.

\textbf{3.} Le coefficient $(3\,;\,3)$ de $M^3$ vaut $4$ : il compte les
chemins fermés de longueur $3$ partant de $3$. Chacun correspond à un triangle
contenant $3$ parcouru dans un sens ; il y a deux triangles ($1,2,3$ et
$3,4,5$), chacun parcouru dans deux sens.
\end{corrige}
