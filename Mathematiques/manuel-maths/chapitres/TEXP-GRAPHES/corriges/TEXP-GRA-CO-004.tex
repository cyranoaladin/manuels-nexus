\begin{corrige}{TEXP-GRA-EX-004}
\textbf{1.} En numérotant $A,B,C,D$ dans cet ordre :
$$M=\begin{pmatrix}0&1&1&0\\1&0&1&0\\1&1&0&1\\0&0&1&0\end{pmatrix}$$

\textbf{2.}
$$M^2=\begin{pmatrix}2&1&1&1\\1&2&1&1\\1&1&3&0\\1&1&0&1\end{pmatrix}$$
Le coefficient $(M^2)_{1,4}=1$ (ligne $A$, colonne $D$) : il existe $1$ chaîne de longueur $2$ entre $A$ et $D$, à savoir $A\text{-}C\text{-}D$.

\textbf{3.}
$$M^3=\begin{pmatrix}2&3&4&1\\3&2&4&1\\4&4&2&3\\1&1&3&0\end{pmatrix}$$
Le coefficient $(M^3)_{1,4}=1$ : il existe $1$ chaîne de longueur $3$ entre $A$ et $D$, à savoir $A\text{-}B\text{-}C\text{-}D$.
\end{corrige}
