\begin{exercice}{TEXP-GRA-EX-006}{1}{12}
On considère le graphe $G$ d'ordre $5$, de sommets $A,B,C,D,E$ et d'arêtes
$\{A,B\}$, $\{A,C\}$, $\{B,C\}$, $\{C,D\}$, $\{D,E\}$.
\begin{enumerate}
  \item Donner l'ordre de $G$ et le degré de chaque sommet.
  \item Vérifier la relation entre la somme des degrés et le nombre d'arêtes.
  \item $G$ est-il connexe ? complet ? Justifier.
\end{enumerate}
\end{exercice}
