\begin{exercice}{TEXP-GRA-EX-010}{3}{18}
Soit $M$ la matrice d'adjacence d'un graphe à $p$ sommets. On note $m^{(n)}_{ij}$
le coefficient $(i\,;\,j)$ de $M^n$.
\begin{enumerate}
  \item Vérifier la propriété pour $n=1$ : que compte $m^{(1)}_{ij}$ ?
  \item En écrivant $M^{n+1}=M^n\times M$, montrer que si $m^{(n)}_{ij}$ compte
        les chemins de longueur $n$ de $i$ à $j$, alors $m^{(n+1)}_{ij}$ compte
        ceux de longueur $n+1$.
  \item Conclure par récurrence.
\end{enumerate}
\end{exercice}
