\begin{exercice}{TEXP-GRA-EX-004}{2}{15}
On considère le graphe $H$ à $4$ sommets $A,B,C,D$ (numérotés dans cet ordre), constitué d'un triangle $A,B,C$ (arêtes $\{A,B\}$, $\{A,C\}$, $\{B,C\}$) auquel on ajoute un sommet $D$ relié uniquement à $C$ (arête $\{C,D\}$).
\begin{enumerate}
  \item Construire la matrice d'adjacence $M$ de $H$.
  \item Calculer $M^2$. En déduire le nombre de chaînes de longueur $2$ entre $A$ et $D$, et écrire explicitement cette (ces) chaîne(s).
  \item Calculer $M^3$. En déduire le nombre de chaînes de longueur $3$ entre $A$ et $D$, et écrire explicitement cette (ces) chaîne(s).
\end{enumerate}
\end{exercice}
