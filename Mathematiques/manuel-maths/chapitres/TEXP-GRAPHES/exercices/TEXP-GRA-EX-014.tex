\begin{exercice}{TEXP-GRA-EX-014}{3}{18}
On considère le graphe « papillon » : sommets $1,\dots,5$, arêtes $\{1,2\}$,
$\{1,3\}$, $\{2,3\}$, $\{3,4\}$, $\{3,5\}$, $\{4,5\}$.
\begin{enumerate}
  \item Calculer le nombre de chemins de longueur $2$ entre $1$ et $4$.
  \item Existe-t-il un chemin de longueur $2$ entre $1$ et $5$ ? Et de
        longueur $3$ ?
  \item Que vaut le coefficient $(3\,;\,3)$ de $M^3$ ? Interpréter.
\end{enumerate}
\end{exercice}
