\begin{exercice}{TEXP-GRA-EX-003}{2}{12}
On reprend le graphe $G$ de l'exercice précédent : sommets $S_1,S_2,S_3,S_4,S_5$, arêtes $\{S_1,S_2\}$, $\{S_1,S_3\}$, $\{S_2,S_3\}$, $\{S_3,S_4\}$, $\{S_4,S_5\}$.
\begin{enumerate}
  \item Construire la matrice d'adjacence $M$ de $G$, en numérotant les sommets dans l'ordre $S_1,S_2,S_3,S_4,S_5$.
  \item Vérifier que $M$ est symétrique. Pourquoi cela est-il attendu pour ce graphe ?
  \item Combien la matrice $M$ compte-t-elle de coefficients égaux à $1$ ? Relier ce nombre au nombre d'arêtes de $G$.
\end{enumerate}
\end{exercice}
