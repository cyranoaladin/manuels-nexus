\begin{exercice}{TEXP-GRA-EX-001}{1}{12}
On considère le graphe $G$ à $5$ sommets $S_1,S_2,S_3,S_4,S_5$ et aux arêtes $\{S_1,S_2\}$, $\{S_1,S_3\}$, $\{S_2,S_3\}$, $\{S_3,S_4\}$, $\{S_4,S_5\}$.
\begin{enumerate}
  \item Donner l'ordre de $G$.
  \item Déterminer le degré de chaque sommet.
  \item Vérifier que la somme des degrés est égale au double du nombre d'arêtes.
  \item Le graphe $G$ est-il complet ? Justifier en donnant le nombre d'arêtes d'un graphe complet à $5$ sommets.
\end{enumerate}
\end{exercice}
