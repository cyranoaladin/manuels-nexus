\section*{\textcolor{chapcolor}{\MakeUppercase{Graphes et matrices d'adjacence}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} Dans un graphe simple non oriente d'ordre n, le degre maximal d'un sommet est :
  \begin{enumerate}[label=\Alph*.]
    \item n
    \item n-1
    \item 2n
    \item n\textasciicircum{}2
  \end{enumerate}

\item \textbf{[Q2]} La somme des degres de tous les sommets d'un graphe est egale a :
  \begin{enumerate}[label=\Alph*.]
    \item Le nombre d'aretes
    \item Le double du nombre de sommets
    \item Le double du nombre d'aretes
    \item Le carre du nombre de sommets
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q3]} Un graphe admet une chaine eulerienne si et seulement si le nombre de sommets de degre impair est :
  \begin{enumerate}[label=\Alph*.]
    \item 0 ou 2
    \item Exactement 1
    \item Un nombre pair quelconque
    \item Toujours 0
  \end{enumerate}

\item \textbf{[Q4]} Si M est la matrice d'adjacence, le nombre de chemins de longueur k de i a j est donne par le coefficient $(i,j)$ de :
  \begin{enumerate}[label=\Alph*.]
    \item $M + k$
    \item $k M$
    \item $M^{k-1}$
    \item $M^k$
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q5]} L'algorithme de Dijkstra permet de trouver :
  \begin{enumerate}[label=\Alph*.]
    \item Un cycle eulerien
    \item La matrice d'adjacence
    \item Le plus court chemin dans un graphe a poids positifs
    \item La coloration minimale
  \end{enumerate}

\end{enumerate}
