\section*{\textcolor{chapcolor}{\MakeUppercase{Graphes et matrices d'adjacence}}}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q1]} Dans un graphe simple non oriente d'ordre n, le degré maximal d'un sommet est :
  \begin{enumerate}[label=\Alph*.]
    \item n-1
    \item n
    \item 2n
    \item n\textasciicircum{}2
  \end{enumerate}

\item \textbf{[Q2]} La somme des degrés de tous les sommets d'un graphe est égale a :
  \begin{enumerate}[label=\Alph*.]
    \item Le double du nombre d'arêtes
    \item Le double du nombre de sommets
    \item Le nombre d'arêtes
    \item Le carre du nombre de sommets
  \end{enumerate}

\item[] \textbf{Capacité C3}

\item \textbf{[Q3]} La matrice d'adjacence d'un graphe simple non oriente est :
  \begin{enumerate}[label=\Alph*.]
    \item antisymétrique avec diagonale égale à 1
    \item symétrique avec diagonale nulle
    \item toujours triangulaire
    \item une matrice dont chaque ligne somme à 1
  \end{enumerate}

\item[] \textbf{Capacité C4}

\item \textbf{[Q4]} Si M est la matrice d'adjacence, le nombre de chemins de longueur k de i a j est donne par le coefficient $(i,j)$ de :
  \begin{enumerate}[label=\Alph*.]
    \item $M + k$
    \item $k M$
    \item $M^k$
    \item $M^{k-1}$
  \end{enumerate}

\item[] \textbf{Capacité C5}

\item \textbf{[Q5]} Pourquoi le coefficient $(i,j)$ de $M^{n+1}$ compte-t-il les chemins de longueur $n+1$ de $i$ a $j$ ?
  \begin{enumerate}[label=\Alph*.]
    \item parce que tous les coefficients de $M$ valent 1
    \item parce que $M^{n+1}=M+n+1$
    \item car un graphe possède toujours un unique chemin entre deux sommets
    \item car $\sum_k(M^n)_{ik}M_{kj}$ classe les chemins selon l'avant-dernier sommet $k$
  \end{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q6]} On veut modéliser par un graphe un réseau de six villes reliées par des routes à double sens. Le modèle adapté est :
  \begin{enumerate}[label=\Alph*.]
    \item un graphe orienté, avec une arête par sens de circulation
    \item un graphe non orienté, un sommet par ville et une arête par route
    \item un graphe dont les sommets sont les routes
    \item un arbre, puisqu'il y a six villes
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des réponses erronées}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{B} — Compte incluant le sommet lui-même sans boucle. Dans un graphe simple d'ordre n, un sommet peut être adjacent aux n-1 autres sommets au maximum. \emph{Renvoi : C1.}
    \item \textbf{C} — Valeur trop grande. Dans un graphe simple d'ordre n, un sommet peut être adjacent aux n-1 autres sommets au maximum. \emph{Renvoi : C1.}
    \item \textbf{D} — Valeur quadratique fausse. Dans un graphe simple d'ordre n, un sommet peut être adjacent aux n-1 autres sommets au maximum. \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{B} — Confusions entre sommets et arêtes. Chaque arête contribue 1 au degré de chacune de ses deux extrémités, donc la somme des degrés vaut deux fois le nombre d'arêtes. \emph{Renvoi : C1.}
    \item \textbf{C} — Chaque arête contribue pour 2 degrés. Chaque arête contribue 1 au degré de chacune de ses deux extrémités, donc la somme des degrés vaut deux fois le nombre d'arêtes. \emph{Renvoi : C1.}
    \item \textbf{D} — Formule quadratique incorrecte. Chaque arête contribue 1 au degré de chacune de ses deux extrémités, donc la somme des degrés vaut deux fois le nombre d'arêtes. \emph{Renvoi : C1.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — L'absence de boucle impose une diagonale nulle, et une arête non orientée produit deux coefficients égaux. La matrice d'adjacence d'un graphe simple non orienté est symétrique et sa diagonale est nulle. \emph{Renvoi : C3.}
    \item \textbf{C} — Aucun ordre général des sommets ne rend la matrice triangulaire lorsqu'il existe des arêtes dans les deux directions matricielles. La matrice d'adjacence d'un graphe simple non orienté est symétrique et sa diagonale est nulle. \emph{Renvoi : C3.}
    \item \textbf{D} — La somme d'une ligne est le degré du sommet correspondant, qui n'est généralement pas égal à 1. La matrice d'adjacence d'un graphe simple non orienté est symétrique et sa diagonale est nulle. \emph{Renvoi : C3.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — Addition incorrecte. Le coefficient (i,j) de M\textasciicircum{}k compte les chemins de longueur k allant du sommet i au sommet j. \emph{Renvoi : C4.}
    \item \textbf{B} — Multiplication par un scalaire. Le coefficient (i,j) de M\textasciicircum{}k compte les chemins de longueur k allant du sommet i au sommet j. \emph{Renvoi : C4.}
    \item \textbf{D} — Puissance decalee de 1. Le coefficient (i,j) de M\textasciicircum{}k compte les chemins de longueur k allant du sommet i au sommet j. \emph{Renvoi : C4.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — Une matrice d'adjacence contient aussi des zéros ; le comptage vient du produit matriciel, pas de coefficients tous égaux à 1. La formule (M\textasciicircum{}(n+1))\_(i,j)=somme\_k (M\textasciicircum{}n)\_(i,k)M\_(k,j) classe les chemins selon leur avant-dernier sommet k. \emph{Renvoi : C5.}
    \item \textbf{B} — Une puissance matricielle est obtenue par produits successifs et ne se réduit pas à une addition scalaire. La formule (M\textasciicircum{}(n+1))\_(i,j)=somme\_k (M\textasciicircum{}n)\_(i,k)M\_(k,j) classe les chemins selon leur avant-dernier sommet k. \emph{Renvoi : C5.}
    \item \textbf{C} — Un graphe peut avoir zéro, un ou plusieurs chemins entre deux sommets ; l'unicité n'est pas une propriété générale. La formule (M\textasciicircum{}(n+1))\_(i,j)=somme\_k (M\textasciicircum{}n)\_(i,k)M\_(k,j) classe les chemins selon leur avant-dernier sommet k. \emph{Renvoi : C5.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{A} — Une route à double sens n'impose aucune orientation. Doubler les arcs n'ajoute rien et fausse la lecture des degrés, donc les raisonnements de parcours. \emph{Renvoi : C2.}
    \item \textbf{C} — Prendre les routes comme sommets construit un autre graphe, qui répond aux questions de parcours d'arêtes. La question porte sur les liaisons entre villes : ce sont les villes qui sont les sommets. \emph{Renvoi : C2.}
    \item \textbf{D} — Un arbre suppose l'absence de cycle et exactement cinq arêtes pour six sommets. Rien ne l'assure : un réseau routier comporte presque toujours des cycles. \emph{Renvoi : C2.}
  \end{itemize}
\end{itemize}
