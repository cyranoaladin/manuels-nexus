\section*{\textcolor{chapcolor}{\MakeUppercase{Suites : limites et convergence — QCM d'auto-évaluation}}}

\begin{center}
\textit{Pour chaque question, une seule réponse est exacte.}
\end{center}

\begin{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q1]} Soit $(u_n)$ définie par $u_n = \dfrac{5n - 1}{2n + 3}$. Quelle est $\lim_{n \to +\infty} u_n$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $5/2$
    \item $5/3$
    \item $+\infty$
    \item $0$
  \end{enumerate}

\item \textbf{[Q2]} La suite $u_n = (-1)^n + 1/n$ est :
  \begin{enumerate}[label=\Alph*.]
    \item convergente vers 0
    \item divergente sans limite
    \item divergente vers $+\infty$
    \item convergente vers 1
  \end{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q3]} Dans une preuve par récurrence, l'étape d'heredite consiste a :
  \begin{enumerate}[label=\Alph*.]
    \item Vérifier P(0)
    \item Montrer P(n) pour tout n
    \item Supposer P(n) et montrer P(n+1)
    \item Remplacer n par n+1
  \end{enumerate}

\item \textbf{[Q4]} Dans l'heredite de la propriété $2^n > n$ pour $n \geqslant 1$, quelle redaction est complete ?
  \begin{enumerate}[label=\Alph*.]
    \item $2^{n+1} = 2\times 2^n > 2n \geqslant n+1$, car $n \geqslant 1$
    \item $2^{n+1} > n+1$ directement par l'hypothèse de récurrence
    \item $2^{n+1} = 2\times 2^n > 2n$, donc $2^{n+1}>n+1$ sans autre justification
    \item Montrer de nouveau $2^n>n$ sans supposer l'hypothèse de récurrence
  \end{enumerate}

\item[] \textbf{Capacité C3}

\item \textbf{[Q5]} $u_0=100$, $u_{n+1}=0{,}9u_n+20$. Limite ?
  \begin{enumerate}[label=\Alph*.]
    \item 180
    \item 200
    \item 220
    \item 0
  \end{enumerate}

\item \textbf{[Q6]} $u_{n+1}=0{,}7u_n+9$ et $v_n=u_n-30$. Quelle est la raison de $(v_n)$ ?
  \begin{enumerate}[label=\Alph*.]
    \item 9
    \item 30
    \item 0,7
    \item 0,3
  \end{enumerate}

\item[] \textbf{Capacité C4}

\item \textbf{[Q7]} Laquelle est vraie ?
  \begin{enumerate}[label=\Alph*.]
    \item Toute suite croissante diverge
    \item Convergente => croissante
    \item Majorée => converge
    \item Croissante majorée => converge
  \end{enumerate}

\item \textbf{[Q8]} $(u_n)$ croissante, $u_n \geq n$. Conclusion ?
  \begin{enumerate}[label=\Alph*.]
    \item diverge vers $+\infty$
    \item bornee
    \item converge vers 1
    \item décroissante
  \end{enumerate}

\item[] \textbf{Capacité C5}

\item \textbf{[Q9]} Pour $a \geqslant -1$ et $n \in \mathbb{N}$, l'inégalité de Bernoulli affirme :
  \begin{enumerate}[label=\Alph*.]
    \item $(1+a)^n \leq 1+na$
    \item $(1+a)^n \geq 1+na$
    \item $(1+a)^n = 1+na$
    \item $(1+a)^n \geqslant 1+n^2a$
  \end{enumerate}

\item \textbf{[Q10]} $\lim 0{,}95^n$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $+\infty$
    \item 0,95
    \item 0
    \item pas de limite
  \end{enumerate}

\item \textbf{[Q11]} $\lim 1{,}02^n$ ?
  \begin{enumerate}[label=\Alph*.]
    \item 0
    \item 1
    \item 1,02
    \item $+\infty$
  \end{enumerate}

\item[] \textbf{Capacité C6}

\item \textbf{[Q12]} $u_n \leq v_n$ et $\lim u_n = +\infty$. Conclusion sur $v_n$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $\lim v_n = +\infty$
    \item $\lim v_n = 0$
    \item $(v_n)$ converge
    \item On ne peut rien conclure
  \end{enumerate}

\item \textbf{[Q13]} Le théorème des gendarmes s'applique quand :
  \begin{enumerate}[label=\Alph*.]
    \item limites différentes
    \item même limite
    \item une seule limite
    \item divergence
  \end{enumerate}

\item[] \textbf{Capacité C7}

\item \textbf{[Q14]} $\lim_{x \to -\infty} e^x$ ?
  \begin{enumerate}[label=\Alph*.]
    \item $-\infty$
    \item 1
    \item 0
    \item $+\infty$
  \end{enumerate}

\item \textbf{[Q15]} $\lim_{x \to +\infty} e^x / x^2$ ?
  \begin{enumerate}[label=\Alph*.]
    \item 0
    \item 1
    \item indeterminee
    \item $+\infty$
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des réponses erronées}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{B} — L'élève a divise les termes constants (-1 et 3) au lieu des coefficients dominants. \emph{Renvoi : M1, étape 3.}
    \item \textbf{C} — L'élève pense que tout quotient diverge parce que le numérateur tend vers $+\infty$ ; or le dénominateur croît à la même vitesse, et le quotient des termes dominants donne $5n/2n = 5/2$. \emph{Renvoi : M1, étape 2.}
    \item \textbf{D} — Conclure $0$ applique a tort la limite de $1/n$ au quotient entier ; les termes dominants $5n$ et $2n$ donnent au contraire $5/2$. \emph{Renvoi : M1, étape 1.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — Ne regarder que $1/n\to0$ oublie le terme oscillant $(-1)^n$ ; les sous-suites paires et impaires ont des limites différentes. \emph{Renvoi : C1, définition divergence.}
    \item \textbf{C} — Le facteur $(-1)^n$ reste borne entre $-1$ et $1$ et $1/n\to0$ ; aucun terme ne croit sans borne vers $+\infty$. \emph{Renvoi : C1, exemples.}
    \item \textbf{D} — Les termes pairs tendent bien vers $1$, mais les termes impairs tendent vers $-1$ ; la suite complete ne peut donc pas converger vers $1$. \emph{Renvoi : C1, unicite limite.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — Vérifier $P(0)$ ou le premier rang etablit seulement l'initialisation ; l'heredite doit partir de $P(n)$ pour obtenir $P(n+1)$. \emph{Renvoi : M2, étape 1.}
    \item \textbf{B} — Affirmer directement $P(n)$ pour tout $n$ est la conclusion de la récurrence ; il manque l'implication de $P(n)$ vers $P(n+1)$. \emph{Renvoi : M2, étape 4.}
    \item \textbf{D} — On ne remplace pas mecaniquement; on doit utiliser l'hypothèse. \emph{Renvoi : M2, piège 3.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{B} — L'hypothèse $2^n>n$ ne donne pas directement $2^{n+1}>n+1$ : il faut multiplier par 2 puis comparer $2n$ et $n+1$. \emph{Renvoi : M2, redaction de l'heredite.}
    \item \textbf{C} — La chaine omet la justification finale $2n \geqslant n+1$, qui repose sur l'hypothèse $n \geqslant 1$. \emph{Renvoi : M2, redaction de l'heredite.}
    \item \textbf{D} — L'heredite suppose la propriété au rang $n$ et doit démontrer celle au rang $n+1$ ; redemontrer le rang $n$ ne fait pas avancer la preuve. \emph{Renvoi : M2, piège 2.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — Le résultat $180$ applique encore le facteur $0{,}9$ au point fixe correct $200$ ; une limite doit vérifier directement $\ell=0{,}9\ell+20$. \emph{Renvoi : M3, étape 1.}
    \item \textbf{C} — Le résultat $220$ ajoute $20$ au point fixe $200$ une seconde fois ; dans l'équation, ce terme est deja pris en compte avant la division par $0{,}1$. \emph{Renvoi : M3, étape 1.}
    \item \textbf{D} — La limite n'est pas celle de $0{,}9^n$ seul : le terme additif $20$ deplace le point fixe, qui vérifie $\ell=0{,}9\ell+20$. \emph{Renvoi : M3, étape 6.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{A} — Confusion entre la constante additive et la raison : le calcul $v_{n+1}=u_{n+1}-30=0{,}7u_n+9-30=0{,}7(u_n-30)=0{,}7v_n$ montre que $9$ disparaît et que la raison est $0{,}7$. \emph{Renvoi : M3, étape 3.}
    \item \textbf{B} — La valeur $30=9/(1-0{,}7)$ est le point fixe soustrait pour definir $v_n$ ; la relation obtenue est $v_{n+1}=0{,}7v_n$. \emph{Renvoi : M3, étape 1.}
    \item \textbf{D} — Le nombre $0{,}3=1-0{,}7$ intervient dans le calcul du point fixe $9/0{,}3$ ; le coefficient multiplicatif de $v_n$ reste $0{,}7$. \emph{Renvoi : M3, étape 3.}
  \end{itemize}
  \item \textbf{Q7}
  \begin{itemize}
    \item \textbf{A} — Contre-exemple : $u_n = 1-1/n$ est croissante et converge vers $1$ ; une suite croissante ne diverge que si elle n'est pas majorée. \emph{Renvoi : C4, théorème.}
    \item \textbf{B} — Contre-exemple : $u_n = (-1)^n/n$ converge vers 0 sans être croissante. \emph{Renvoi : C1, définition.}
    \item \textbf{C} — Être majorée ne suffit pas : la suite $(-1)^n$ est majorée mais diverge ; le théorème exige aussi que la suite soit croissante. \emph{Renvoi : C4, théorème.}
  \end{itemize}
  \item \textbf{Q8}
  \begin{itemize}
    \item \textbf{B} — $u_n \geq n$ implique que $(u_n)$ n'est pas majorée. \emph{Renvoi : C4, théorème.}
    \item \textbf{C} — Si $u_n \geq n$, la suite ne peut converger vers un réel fini. \emph{Renvoi : C4, théorème.}
    \item \textbf{D} — Contradictoire avec l'hypothèse : une suite croissante ne peut pas être décroissante. \emph{Renvoi : RE-C4.}
  \end{itemize}
  \item \textbf{Q9}
  \begin{itemize}
    \item \textbf{A} — Le sens $\leqslant$ est inverse : pour $a=1$ et $n=2$, on obtient $4$ a gauche et $3$ a droite, donc seule l'inégalité $\geqslant$ est vraie. \emph{Renvoi : C5, théorème.}
    \item \textbf{C} — C'est une inégalité, pas une égalité (sauf si $a=0$ ou $n \leq 1$). \emph{Renvoi : C5, théorème.}
    \item \textbf{D} — Le coefficient de $a$ est $n$, pas $n^2$ ; par exemple $a=1$ et $n=2$ donnent $4 \not\geqslant 5$. \emph{Renvoi : cours C5.}
  \end{itemize}
  \item \textbf{Q10}
  \begin{itemize}
    \item \textbf{A} — La divergence vers $+\infty$ concerne une raison $q>1$ ; ici $0<0{,}95<1$, donc les puissances decroissent vers $0$. \emph{Renvoi : C5, tableau.}
    \item \textbf{B} — La valeur $0{,}95$ est la raison et le premier facteur, pas la limite ; multiplier indefiniment par un nombre de $]0,1[$ conduit a $0$. \emph{Renvoi : C5, tableau.}
    \item \textbf{D} — Une suite géométrique de raison strictement comprise entre 0 et 1 converge vers 0 ; l'absence de limite concerne les raisons inferieures ou égales a -1. \emph{Renvoi : M4, suites géométriques.}
  \end{itemize}
  \item \textbf{Q11}
  \begin{itemize}
    \item \textbf{A} — La limite $0$ vaut seulement lorsque $|q|<1$ ; ici $q=1{,}02>1$, donc les puissances croissent sans borne. \emph{Renvoi : C5, tableau.}
    \item \textbf{B} — Une limite égale a $1$ correspondrait a la suite constante de raison $q=1$ ; avec $q=1{,}02$, chaque multiplication augmente le terme. \emph{Renvoi : C5, tableau.}
    \item \textbf{C} — Confusion entre la raison de la suite et sa limite : la raison 1,02 est superieure a 1, donc la suite diverge vers plus l'infini. \emph{Renvoi : M4, suites géométriques.}
  \end{itemize}
  \item \textbf{Q12}
  \begin{itemize}
    \item \textbf{B} — La conclusion $v_n\to0$ contredit $v_n\geq u_n$ : pour tout seuil $A$, les termes $u_n$ puis $v_n$ finissent par depasser $A$. \emph{Renvoi : C6, théorème comparaison.}
    \item \textbf{C} — La minoration par une suite tendant vers plus l'infini interdit la convergence : v\_n est repousse vers plus l'infini. \emph{Renvoi : M5, théorème de comparaison.}
    \item \textbf{D} — Le théorème de comparaison conclut précisément : $v_n$ majore une suite qui dépasse tout réel, donc $v_n$ dépasse tout réel aussi et $\lim v_n = +\infty$. \emph{Renvoi : C6, théorème comparaison.}
  \end{itemize}
  \item \textbf{Q13}
  \begin{itemize}
    \item \textbf{A} — Si les deux suites encadrantes ont des limites différentes, l'encadrement laisse un intervalle entier de valeurs possibles : le théorème exige la même limite des deux côtés. \emph{Renvoi : C6, théorème gendarmes.}
    \item \textbf{C} — Connaitre la limite d'une seule borne ne suffit pas a coincer la suite ; les deux suites encadrantes doivent tendre vers la même limite $\ell$. \emph{Renvoi : C6, théorème gendarmes.}
    \item \textbf{D} — Le théorème des gendarmes conclut a une convergence : il encadre par deux suites de même limite finie, il ne s'applique pas a une divergence. \emph{Renvoi : M5, théorème des gendarmes.}
  \end{itemize}
  \item \textbf{Q14}
  \begin{itemize}
    \item \textbf{A} — $e^x > 0$ pour tout $x$, donc la limite ne peut être $-\infty$. \emph{Renvoi : C7, démonstration.}
    \item \textbf{B} — La valeur $1$ est seulement $e^0$ ; lorsque $x$ décroît sans borne, $e^x=1/e^{-x}$ et le dénominateur tend vers $+\infty$, donc la limite vaut $0$. \emph{Renvoi : C7, démonstration.}
    \item \textbf{D} — Sens de variation inverse : l'exponentielle tend vers 0 en moins l'infini et vers plus l'infini en plus l'infini. \emph{Renvoi : cours C7.}
  \end{itemize}
  \item \textbf{Q15}
  \begin{itemize}
    \item \textbf{A} — L'exponentielle croit plus vite que tout polynôme. \emph{Renvoi : C7, croissance comparee.}
    \item \textbf{B} — L'élève simplifie comme s'il s'agissait d'un quotient de termes de même nature ; la croissance comparee donne plus l'infini. \emph{Renvoi : M3, croissances comparees.}
    \item \textbf{C} — C'est une forme indeterminee qui se resout par les croissances comparees. \emph{Renvoi : C7, croissance comparee.}
  \end{itemize}
\end{itemize}
