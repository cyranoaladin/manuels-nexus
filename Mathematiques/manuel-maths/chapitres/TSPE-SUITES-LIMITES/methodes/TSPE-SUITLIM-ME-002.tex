\begin{fichemethode}{M2}{Rediger une démonstration par récurrence}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande de « démontrer par récurrence que\ldots », « prouver que pour tout $n$\ldots » ou lorsqu'une propriété porte sur tous les entiers a partir d'un rang.
}

\pasApas{%
  \begin{enumerate}
    \item \textbf{Écrire la propriété.} « Soit $\mathcal{P}(n)$ la propriété : \guillemotleft\ldots\guillemotright. »
    \item \textbf{Initialisation.} Vérifier que $\mathcal{P}(n_0)$ est vraie en remplacant $n$ par $n_0$ dans la propriété.
    \item \textbf{Heredite.} Écrire « Soit $n \geq n_0$ fixe. Supposons $\mathcal{P}(n)$ vraie. » Puis démontrer $\mathcal{P}(n+1)$ en utilisant l'hypothèse de récurrence.
    \item \textbf{Conclusion.} Écrire « Par le principe de récurrence, $\mathcal{P}(n)$ est vraie pour tout $n \geq n_0$. »
  \end{enumerate}
}

\exempleRedige{%
  \textbf{Exemple.} Démontrer par récurrence que pour tout $n \in \mathbb{N}$, $3^n \geq 2n + 1$.

  \medskip
  \commentaireMarge{Écrire explicitement la propriété.}%
  Soit $\mathcal{P}(n)$ la propriété : « $3^n \geq 2n + 1$ ».

  \commentaireMarge{Vérifier pour $n = 0$.}%
  \textbf{Initialisation.} $3^0 = 1$ et $2 \times 0 + 1 = 1$. On a $1 \geq 1$, donc $\mathcal{P}(0)$ est vraie.

  \commentaireMarge{Supposer $\mathcal{P}(n)$ et montrer $\mathcal{P}(n+1)$.}%
  \textbf{Heredite.} Soit $n \geq 0$ fixe. Supposons que $3^n \geq 2n + 1$.

  Alors $3^{n+1} = 3 \times 3^n \geq 3(2n+1) = 6n + 3$.

  Or $6n + 3 \geq 2(n+1) + 1 = 2n + 3$ (car $4n \geq 0$).

  Donc $3^{n+1} \geq 2(n+1) + 1$ et $\mathcal{P}(n+1)$ est vraie.

  \textbf{Conclusion.} Par le principe de récurrence, pour tout $n \in \mathbb{N}$, $3^n \geq 2n + 1$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Oublier l'initialisation. Sans elle, la récurrence ne prouve rien.
    \item \textbf{Piège 2.} Dans l'heredite, \emph{démontrer} $\mathcal{P}(n)$ au lieu de la \emph{supposer}. L'hypothèse de récurrence est posee, pas prouvee.
    \item \textbf{Piège 3.} Confondre « montrer $\mathcal{P}(n+1)$ » avec « remplacer $n$ par $n+1$ dans $\mathcal{P}(n)$ ». Il faut partir de l'hypothèse et aboutir a la conclusion.
  \end{itemize}
}

\verifier{%
  Après la redaction, relire en se demandant : « ai-je utilise l'hypothèse de récurrence dans l'heredite ? » Si non, la preuve est probablement fausse ou ne necessite pas de récurrence.
}

\end{fichemethode}
