\begin{corrige}{1SPE-DERGLOBAL-EX-001}
On applique la formule $(x^n)' = nx^{n-1}$ et les dérivées des fonctions constantes et identité.

\textbf{1.} $f'(x) = 5x^4$.

\textbf{2.} $g'(x) = 3x^2$.

\textbf{3.} $h'(x) = 0$ (dérivée d'une constante).

\textbf{4.} $k'(x) = 1$ (dérivée de la fonction identité).
\end{corrige}
