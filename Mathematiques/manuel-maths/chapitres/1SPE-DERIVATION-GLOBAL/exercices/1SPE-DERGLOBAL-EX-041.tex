\begin{exercice}{1SPE-DERGLOBAL-EX-041}{1}{5}
Soit $f(x) = -x^2 + 6x - 5$ définie sur $\mathbb{R}$.
\begin{enumerate}
  \item Calculer $f'(x)$.
  \item Résoudre $f'(x) = 0$ et dresser le tableau de variations de $f$.
  \item En déduire le maximum de $f$ sur $\mathbb{R}$.
\end{enumerate}
\end{exercice}
