\sectionCapacite{Extremums d'une fonction}{Je sais déterminer les extremums à partir des variations et examiner les bornes du domaine.}

% ── Définitions ──────────────────────────────────────────────────────────────

\begin{nxdef}[Extremum local]
Soit $f$ une fonction définie sur un intervalle $I$ et $a \in I$.

\begin{itemize}
  \item $f$ admet un \textbf{maximum local} en $a$ si $f(x) \leq f(a)$ pour tout $x$
        appartenant à $I$ et assez proche de $a$.
  \item $f$ admet un \textbf{minimum local} en $a$ si $f(x) \geq f(a)$ pour tout $x$
        appartenant à $I$ et assez proche de $a$.
\end{itemize}

On dit que $f(a)$ est un \textbf{extremum local} de $f$. Les inégalités sont
larges : ces extremums peuvent être non stricts. Une fonction constante
admet un maximum et un minimum en chacun de ses points.
\end{nxdef}

\begin{nxdef}[Extremum global]
$f$ admet un \textbf{maximum global} en $a$ sur $I$ si $f(x) \leq f(a)$ pour tout
$x \in I$ (et de même pour le minimum global). Un extremum global est aussi un
extremum local, mais la réciproque est fausse en général.
\end{nxdef}

\margeAppui{On distingue : extremum \emph{local} (autour d'un point) et extremum
\emph{global} (sur tout le domaine). En optimisation, c'est l'extremum global
qui répond au problème.}

% ── Théorème principal ───────────────────────────────────────────────────────

\begin{nxthm}[Condition nécessaire d'extremum]
Si $a$ est un point intérieur de l'intervalle de définition, si $f$ est
dérivable en $a$ et y admet un extremum local, alors
\[
  f'(a) = 0.
\]
\end{nxthm}

Au voisinage d'un maximum intérieur, le taux de variation est négatif ou nul
pour un accroissement positif, et positif ou nul pour un accroissement négatif.
Si les deux taux tendent vers un même réel $f'(a)$, ce réel est à la fois
positif ou nul et négatif ou nul : il vaut $0$. Au minimum, les signes sont inversés.

\contreexemple{L'hypothèse de point intérieur est indispensable. Pour
$f(x)=x$ sur $[0\,;\,1]$, le minimum global vaut $f(0)=0$ et le maximum global
vaut $f(1)=1$, aux bornes. Pourtant, la fonction affine a $f'(x)=1$.
Il faut donc examiner les bornes incluses du domaine pour une recherche globale.}

\erreurFrequente{La réciproque est \textbf{fausse} : $f'(a)=0$ ne garantit pas
l'existence d'un extremum en $a$. On étudie les variations de $f$ pour conclure.}

% ── Propriété : critère du changement de signe ───────────────────────────────

\begin{nxprop}[Critère du changement de signe]
Soit $f$ dérivable sur un intervalle $I$ et $a$ un point intérieur tel que
$f'(a)=0$. Les signes stricts ci-dessous sont supposés vérifiés sur un
intervalle immédiatement à gauche et à droite de $a$.

\begin{itemize}
  \item Si $f'$ change de signe de $+$ à $-$ en $a$, alors $f$ admet un
        \textbf{maximum local} en $a$.
  \item Si $f'$ change de signe de $-$ à $+$ en $a$, alors $f$ admet un
        \textbf{minimum local} en $a$.
  \item Si $f'$ garde le même signe \emph{strict} de chaque côté de $a$,
        alors $f$ est strictement monotone dans ce voisinage et n'y a pas d'extremum.
\end{itemize}
Ce sont des critères suffisants. Quand la dérivée s'annule sur tout un
intervalle, $f$ y est constante. Chaque point intérieur à une telle portion
constante a des extremums locaux non stricts.
L'absence de changement de signe, à elle seule, ne permet donc pas d'exclure un extremum.
\end{nxprop}

\margeAppui{Après le choix du domaine : calculer $f'$, étudier son signe,
puis lire les variations. Pour un extremum global, comparer les valeurs aux
points critiques et aux bornes incluses ; une borne exclue n'est pas un point candidat.}

% ── Exemple complet ──────────────────────────────────────────────────────────

\exemple{
Soit $f(x) = 2x^3 - 9x^2 + 12x - 2$, définie sur $\mathbb{R}$.

\textbf{Étape 1 — Calcul de $f'$.}
\[
  f'(x) = 6x^2 - 18x + 12 = 6(x^2 - 3x + 2) = 6(x-1)(x-2).
\]

\textbf{Étape 2 — Résoudre $f'(x) = 0$.}

$f'(x)=0 \Leftrightarrow x=1$ ou $x=2$.

\textbf{Étape 3 — Tableau de signes de $f'$ et tableau de variations.}

\[
\begin{array}{|c|ccccccc|}
\hline
x      & -\infty & & 1 & & 2 & & +\infty \\
\hline
f'(x)  &  & + & 0 & - & 0 & + & \\
\hline
f(x) & & \nearrow & 3 & \searrow & 2 & \nearrow & \\
\hline
\end{array}
\]

\textbf{Conclusions :}
\begin{itemize}
  \item En $x=1$ : $f'$ passe de $+$ à $-$, donc $f(1)=3$ est un
        \textbf{maximum local}.
  \item En $x=2$ : $f'$ passe de $-$ à $+$, donc $f(2)=2$ est un
        \textbf{minimum local}.
\end{itemize}

Ces extremums ne sont pas globaux : $f(0)=-2<2$ et $f(3)=7>3$.
Un extremum global sur $\mathbb R$ serait aussi local et sa dérivée serait nulle ;
les seuls candidats étant $1$ et $2$, il n'y a donc pas d'extremum global.
}

% ── Contre-exemple : annulation sans extremum ─────────────────────────────────

\contreexemple{
Pour $f(x) = x^3$, on a $f'(x) = 3x^2$, donc $f'(0) = 0$.

Pour tout $h>0$, on a $f(-h)=-h^3<0=f(0)<h^3=f(h)$.
Il y a donc des valeurs plus petites et plus grandes que $f(0)$, aussi près
de $0$ que l'on veut : aucun extremum n'y est atteint.
L'annulation de la dérivée est nécessaire en un extremum intérieur d'une
fonction dérivable, mais pas suffisante.
}

\margeAppui{Le point $(0\,;\,0)$ de la courbe de $x^3$ est un \emph{point
d'inflexion} : la tangente (horizontale) traverse la courbe. Ce concept est
hors programme en première ; on retient uniquement que $f'(a)=0$ seul ne suffit
pas.}

\erreurFrequente{Conclure « $f'(1)=0$ donc $f$ a un extremum en $1$ » sans
étudier les variations. Un tableau de signes justifié permet de distinguer
les cas strictement monotones, les changements de sens et les portions constantes.}

\approfondissement{
Pour une fonction deux fois dérivable sur un intervalle, en un point intérieur
$a$, le signe de $f''(a)$ donne
l'information manquante : si $f'(a)=0$ et $f''(a)>0$, alors $f(a)$ est un
minimum local ; si $f''(a)<0$, c'est un maximum local. Si $f''(a)=0$, on ne
peut pas conclure directement (cas de $x^3$ en $0$). Ce critère utilise la dérivée
seconde et dépasse le programme de Première.
}
