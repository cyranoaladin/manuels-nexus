\begin{exercice}{TEXP-CTP-EX-009}{3}{18}
\begin{enumerate}
  \item Démontrer que pour tout entier $n\geq1$ et tous complexes $z$ et $a$,
        $z^n-a^n=(z-a)\sum_{k=0}^{n-1}z^{k}a^{n-1-k}$.
  \item En déduire que si $P$ est un polynôme et $P(a)=0$, alors $z-a$ divise
        $P(z)$.
  \item En déduire qu'un polynôme non nul de degré $n$ a au plus $n$ racines.
\end{enumerate}
\end{exercice}
