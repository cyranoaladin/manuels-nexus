\begin{exercice}{TEXP-CTP-EX-003}{2}{20}
Soit $P(z)=z^3-z^2+z-1$.
\begin{enumerate}
  \item Vérifier que $1$ est racine de $P$.
  \item Déterminer $Q(z)$ tel que $P(z)=(z-1)Q(z)$.
  \item Résoudre $P(z)=0$ dans $\mathbb{C}$ (on résoudra $Q(z)=0$).
\end{enumerate}
\end{exercice}
