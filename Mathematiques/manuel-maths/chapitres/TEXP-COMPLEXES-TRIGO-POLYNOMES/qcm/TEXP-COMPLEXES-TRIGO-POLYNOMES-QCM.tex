\section*{\textcolor{chapcolor}{\MakeUppercase{Complexes : trigonometrie et polynomes}}}

\begin{center}
\textit{Pour chaque question, une seule reponse est exacte.}

\begin{enumerate}

\item[] \textbf{Capacite C1}

\item \textbf{[Q1]} La forme exponentielle de $z = 1 + i\sqrt{3}$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $2e^{i\pi/3}$
    \item $2e^{i\pi/6}$
    \item $e^{i\pi/3}$
    \item $2e^{-i\pi/3}$
  \end{enumerate}

\item \textbf{[Q2]} D'apres la formule de Euler, $\cos(\theta) =$
  \begin{enumerate}[label=\Alph*.]
    \item $\frac{e^{i\theta} - e^{-i\theta}}{2}$
    \item $\frac{e^{i\theta} + e^{-i\theta}}{2i}$
    \item $\frac{e^{i\theta} + e^{-i\theta}}{2}$
    \item $e^{i\theta} + e^{-i\theta}$
  \end{enumerate}

\item[] \textbf{Capacite C2}

\item \textbf{[Q3]} La formule de Moivre donne $(\cos\theta + i\sin\theta)^n =$
  \begin{enumerate}[label=\Alph*.]
    \item $\cos^n\theta + i\sin^n\theta$
    \item $\cos(n\theta) + i\sin(n\theta)$
    \item $n\cos\theta + in\sin\theta$
    \item $\cos(n\theta) - i\sin(n\theta)$
  \end{enumerate}

\item \textbf{[Q4]} Les racines n-iemes de l'unite sont au nombre de :
  \begin{enumerate}[label=\Alph*.]
    \item 1
    \item 2
    \item n-1
    \item n
  \end{enumerate}

\item[] \textbf{Capacite C3}

\item \textbf{[Q5]} Un polynome de degre n a coefficients complexes admet dans $\mathbb{C}$ :
  \begin{enumerate}[label=\Alph*.]
    \item Exactement n racines comptees avec multiplicite
    \item Au plus n-1 racines
    \item Toujours 0 racine
    \item Infini de racines
  \end{enumerate}

\end{enumerate}
