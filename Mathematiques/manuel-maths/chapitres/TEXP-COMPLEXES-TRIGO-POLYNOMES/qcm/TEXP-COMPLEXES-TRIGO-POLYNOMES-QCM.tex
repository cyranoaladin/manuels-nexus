\subsection*{C1/C2 --- Forme exponentielle, Euler, Moivre}

\begin{enumerate}

\item \textbf{[Q1]} La forme exponentielle de $z=r(\cos\theta+i\sin\theta)$ s'écrit :
  \begin{enumerate}[label=\Alph*.]
    \item $z=r\theta$.
    \item $z=r\,\mathrm{e}^{i\theta}$.
    \item $z=r+\mathrm{e}^{i\theta}$.
    \item $z=\theta\,\mathrm{e}^{ir}$.
  \end{enumerate}

\bigskip

\item \textbf{[Q2]} La formule d'Euler pour $\cos\theta$ est :
  \begin{enumerate}[label=\Alph*.]
    \item $\cos\theta = \mathrm{e}^{i\theta}+\mathrm{e}^{-i\theta}$.
    \item $\cos\theta = \dfrac{\mathrm{e}^{i\theta}+\mathrm{e}^{-i\theta}}{2}$.
    \item $\cos\theta = \dfrac{\mathrm{e}^{i\theta}-\mathrm{e}^{-i\theta}}{2i}$.
    \item $\cos\theta = \mathrm{e}^{i\theta}\times\mathrm{e}^{-i\theta}$.
  \end{enumerate}

\subsection*{C4/C5 --- Équations polynomiales}

\item \textbf{[Q3]} Si $P(a)=0$ pour un polynôme $P$, alors $P(z)$ se factorise par :
  \begin{enumerate}[label=\Alph*.]
    \item $z+a$.
    \item $z-a$.
    \item $az$.
    \item $z/a$.
  \end{enumerate}

\bigskip

\item \textbf{[Q4]} Un polynôme de degré $n$ admet :
  \begin{enumerate}[label=\Alph*.]
    \item exactement $n$ racines réelles.
    \item au plus $n$ racines.
    \item toujours $n+1$ racines.
    \item une seule racine.
  \end{enumerate}
