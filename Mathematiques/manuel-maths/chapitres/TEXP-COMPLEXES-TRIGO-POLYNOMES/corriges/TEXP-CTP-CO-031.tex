\begin{corrige}{TEXP-CTP-EX-031}

\textbf{Question 1.} En integrant deux fois avec $f(0)=0$ et $f'(0)=0$ :
\[ f'(x) = \int_0^x 6(t-1)(t-3)\,dt = 2x^3 - 12x^2 + 18x, \]
\[ f(x) = \int_0^x (2t^3 - 12t^2 + 18t)\,dt = \frac{x^4}{2} - 4x^3 + 9x^2. \]

\commentaireMarge{Integrer $f''$ deux fois avec $f(0)=f'(0)=0$.}

\textbf{Question 2.} $f'' > 0$ sur $\left]-\infty, 1\right[ \cup \left]3, +\infty\right[$ (convexe) et $f'' < 0$ sur $\left]1, 3\right[$ (concave).

Points d'inflexion en $x = 1$ et $x = 3$.
