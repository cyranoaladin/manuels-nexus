\begin{corrige}{TEXP-CTP-EX-004}
\[
  \frac{c-a}{b-a} = \frac{(5+5i)-(1+i)}{(3+3i)-(1+i)} = \frac{4+4i}{2+2i} = \frac{4(1+i)}{2(1+i)} = 2.
\]
