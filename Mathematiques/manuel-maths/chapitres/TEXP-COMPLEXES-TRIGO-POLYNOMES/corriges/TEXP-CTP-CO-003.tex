\begin{corrige}{TEXP-CTP-EX-003}
\textbf{1.} $P(1) = 1-1+1-1=0$ : $1$ est bien racine de $P$.

\textbf{2.} Par identification (ou division euclidienne) : $Q(z)=z^2+1$, et on vérifie $(z-1)(z^2+1) = z^3+z-z^2-1 = z^3-z^2+z-1 = P(z)$.

\textbf{3.} $P(z)=0 \iff z=1$ ou $Q(z)=0$. L'équation $z^2+1=0$ donne $z^2=-1$, soit $z=i$ ou $z=-i$. L'ensemble des solutions est $\{1,i,-i\}$.
\end{corrige}
