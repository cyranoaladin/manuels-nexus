\begin{corrige}{TEXP-CTP-EX-006}
\textbf{1.} $\cos\theta=\dfrac{\mathrm{e}^{\mathrm{i}\theta}
+\mathrm{e}^{-\mathrm{i}\theta}}{2}$ et
$\sin\theta=\dfrac{\mathrm{e}^{\mathrm{i}\theta}
-\mathrm{e}^{-\mathrm{i}\theta}}{2\mathrm{i}}$.

\textbf{2.} $\cos^3\theta
=\left(\dfrac{\mathrm{e}^{\mathrm{i}\theta}+\mathrm{e}^{-\mathrm{i}\theta}}{2}
\right)^3=\dfrac{1}{8}\left(\mathrm{e}^{3\mathrm{i}\theta}
+3\mathrm{e}^{\mathrm{i}\theta}+3\mathrm{e}^{-\mathrm{i}\theta}
+\mathrm{e}^{-3\mathrm{i}\theta}\right)
=\dfrac{\cos3\theta+3\cos\theta}{4}$.

\textbf{3.} $1+\mathrm{i}=\sqrt2\,\mathrm{e}^{\mathrm{i}\pi/4}$, donc
$(1+\mathrm{i})^8=(\sqrt2)^8\mathrm{e}^{2\mathrm{i}\pi}=16$.
\end{corrige}
