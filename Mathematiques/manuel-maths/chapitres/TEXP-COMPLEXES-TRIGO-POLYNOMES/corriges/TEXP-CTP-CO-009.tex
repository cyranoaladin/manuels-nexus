\begin{corrige}{TEXP-CTP-EX-009}
\textbf{1.} En développant, $(z-a)\sum_{k=0}^{n-1}z^ka^{n-1-k}
=\sum_{k=0}^{n-1}z^{k+1}a^{n-1-k}-\sum_{k=0}^{n-1}z^ka^{n-k}$. Les deux sommes
se télescopent : il ne reste que $z^n$ dans la première et $-a^n$ dans la
seconde.

\textbf{2.} Écrivons $P(z)=\sum_{j}c_jz^j$. Alors
$P(z)=P(z)-P(a)=\sum_j c_j(z^j-a^j)$, et chaque $z^j-a^j$ est divisible par
$z-a$ d'après 1. Donc $z-a$ divise $P(z)$.

\textbf{3.} Par récurrence sur $n$. Un polynôme de degré $0$ non nul n'a pas
de racine. Si $P$ de degré $n$ admet une racine $a$, alors $P=(z-a)Q$ avec
$\deg Q=n-1$ ; toute autre racine de $P$ est racine de $Q$, qui en a au plus
$n-1$. Donc $P$ en a au plus $n$.
\end{corrige}
