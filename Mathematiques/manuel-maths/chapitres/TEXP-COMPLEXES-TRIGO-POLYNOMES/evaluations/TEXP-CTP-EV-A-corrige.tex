\begin{corrige}{TEXP-CTP-EV-A-EX1}
$\cos\theta = \dfrac{\mathrm{e}^{i\theta}+\mathrm{e}^{-i\theta}}{2}$, donc $\cos^3\theta = \dfrac{(\mathrm{e}^{i\theta}+\mathrm{e}^{-i\theta})^3}{8}$. En développant $(A+B)^3=A^3+3A^2B+3AB^2+B^3$ avec $A=\mathrm{e}^{i\theta}$, $B=\mathrm{e}^{-i\theta}$ (et $AB=1$) :
\[
  (\mathrm{e}^{i\theta}+\mathrm{e}^{-i\theta})^3 = \mathrm{e}^{3i\theta}+3\mathrm{e}^{i\theta}+3\mathrm{e}^{-i\theta}+\mathrm{e}^{-3i\theta} = (\mathrm{e}^{3i\theta}+\mathrm{e}^{-3i\theta}) + 3(\mathrm{e}^{i\theta}+\mathrm{e}^{-i\theta}) = 2\cos(3\theta)+6\cos\theta.
\]
D'où $\cos^3\theta = \dfrac{2\cos(3\theta)+6\cos\theta}{8} = \dfrac{1}{4}\cos(3\theta)+\dfrac{3}{4}\cos\theta$.
\end{corrige}

\begin{corrige}{TEXP-CTP-EV-A-EX2}
\textbf{1.} $P(1)=1-3+4-2=0$ : $1$ est racine.

\textbf{2.} $Q(z)=z^2-2z+2$, et on vérifie $(z-1)(z^2-2z+2) = z^3-2z^2+2z-z^2+2z-2 = z^3-3z^2+4z-2=P(z)$.

\textbf{3.} $P(z)=0 \iff z=1$ ou $z^2-2z+2=0$. Pour cette dernière : $\Delta=4-8=-4<0$, solutions $z=\dfrac{2\pm i2}{2}=1\pm i$. Donc $S=\{1,\ 1-i,\ 1+i\}$.
\end{corrige}

\begin{corrige}{TEXP-CTP-EV-A-EX3}
\textbf{1.} $|z|=\sqrt{1^2+(\sqrt3)^2}=\sqrt4=2$. On factorise :
$z=2\left(\dfrac12+i\dfrac{\sqrt3}{2}\right)=2\left(\cos\dfrac{\pi}{3}+i\sin\dfrac{\pi}{3}\right)=2\mathrm{e}^{i\pi/3}$.
Réciproquement, $2\mathrm{e}^{i\pi/3}=2\cos\dfrac{\pi}{3}+2i\sin\dfrac{\pi}{3}=1+i\sqrt3$ : on retrouve bien la forme algébrique.

\textbf{2.} Par les coordonnées, $\vec u\cdot\vec v=\cos a\cos b+\sin a\sin b$.
Par la définition géométrique, $\vec u\cdot\vec v=\|\vec u\|\,\|\vec v\|\cos(\vec u\,;\vec v)$ ;
les deux vecteurs étant unitaires et l'angle orienté valant $a-b$, cela donne $\cos(a-b)$.
Le produit scalaire ne dépendant pas de la façon de le calculer, les deux résultats sont égaux :
\[
  \cos(a-b)=\cos a\cos b+\sin a\sin b .
\]
\end{corrige}
