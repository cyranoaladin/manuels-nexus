\begin{exercice}{TEXP-CTP-EV-A-EX1}{2}{20}
\textbf{Exercice 1} (6 points) --- \textit{Capacité C2}

\medskip

En utilisant la formule d'Euler pour $\cos\theta$, linéariser $\cos^3\theta$ (l'écrire comme combinaison linéaire de $\cos(k\theta)$).
\end{exercice}

\begin{exercice}{TEXP-CTP-EV-A-EX2}{2}{30}
\textbf{Exercice 2} (9 points) --- \textit{Capacité C4}

\medskip

Soit $P(z)=z^3-3z^2+4z-2$.
\begin{enumerate}
  \item (2 pts) Vérifier que $1$ est racine de $P$.
  \item (3 pts) Déterminer $Q(z)$ tel que $P(z)=(z-1)Q(z)$.
  \item (4 pts) Résoudre $P(z)=0$ dans $\mathbb{C}$.
\end{enumerate}
\end{exercice}

\begin{exercice}{TEXP-CTP-EV-A-EX3}{2}{20}
\textbf{Exercice 3} (5 points) --- \textit{Capacités C1, C3}

\medskip

\begin{enumerate}
  \item (2 pts) Écrire $z=1+i\sqrt3$ sous forme trigonométrique puis exponentielle. Retrouver ensuite la forme algébrique à partir de la forme exponentielle obtenue.
  \item (3 pts) Soient $\vec u(\cos a\,;\sin a)$ et $\vec v(\cos b\,;\sin b)$ deux vecteurs unitaires. Calculer $\vec u\cdot\vec v$ de deux façons, et en déduire la formule
  $\cos(a-b)=\cos a\cos b+\sin a\sin b$.
\end{enumerate}
\end{exercice}
