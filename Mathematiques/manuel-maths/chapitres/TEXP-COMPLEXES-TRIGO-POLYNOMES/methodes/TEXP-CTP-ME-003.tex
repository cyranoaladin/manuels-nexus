\begin{fichemethode}{M3}{Démontrer une formule d'addition par le produit scalaire}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande de DEMONTRER une formule
  d'addition (typiquement $\cos(a - b) = \cos a\cos b + \sin a \sin b$) a
  partir du produit scalaire de deux vecteurs unitaires.
}

\pasApas{%
  \begin{enumerate}
    \item Introduire les vecteurs unitaires
      $\vec{u}(\cos a\,;\,\sin a)$ et $\vec{v}(\cos b\,;\,\sin b)$ portes par
      les angles $a$ et $b$ sur le cercle trigonométrique.
    \item Calculer le produit scalaire en COORDONNEES :
      $\vec{u}\cdot\vec{v} = \cos a\cos b + \sin a\sin b$.
    \item Calculer le même produit scalaire avec la formule GEOMETRIQUE :
      $\vec{u}\cdot\vec{v} = \|\vec{u}\|\,\|\vec{v}\|
      \cos(\vec{u}\,;\,\vec{v}) = 1 \times 1 \times \cos(a - b)$.
    \item Egaler les deux expressions :
      $\cos(a - b) = \cos a\cos b + \sin a\sin b$.
    \item En déduire les autres formules : $b \to -b$ donne $\cos(a+b)$ ;
      $a \to \frac{\pi}{2} - a$ donne les formules en sinus.
  \end{enumerate}
}

\exempleRedige{%
  Démontrer $\cos(a - b) = \cos a\cos b + \sin a\sin b$, puis en déduire
  $\cos(a + b)$.

  \medskip
  \commentaireMarge{Deux calculs du MÊME produit scalaire.}%
  Avec $\vec{u}(\cos a\,;\,\sin a)$ et $\vec{v}(\cos b\,;\,\sin b)$, unitaires :
  en coordonnées, $\vec{u}\cdot\vec{v} = \cos a\cos b + \sin a\sin b$ ;
  geometriquement, $\vec{u}\cdot\vec{v} = \cos(a - b)$ (angle entre les deux
  vecteurs). D'ou l'égalité. \hfill$\square$

  \medskip
  \commentaireMarge{Substitution $b \to -b$ et parite.}%
  En remplacant $b$ par $-b$ : $\cos(a + b) = \cos a\cos(-b) + \sin a\sin(-b)
  = \cos a\cos b - \sin a\sin b$ ($\cos$ paire, $\sin$ impaire).
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Oublier de preciser que les vecteurs sont
      UNITAIRES : sans cela, la forme géométrique contient les normes.
    \item \textbf{Piège 2.} Erreur de signe en passant a $\cos(a+b)$ : c'est la
      parite de $\cos$ et l'imparite de $\sin$ qui donnent le $-$.
    \item \textbf{Piège 3.} Utiliser la formule a démontrer (via une identite
      dérivée d'elle) dans la preuve : seul le double calcul du produit
      scalaire est permis.
  \end{itemize}
}

\verifier{%
  Tester la formule sur des angles connus ($a = b$ : $\cos 0 = 1 =
  \cos^2 a + \sin^2 a$ ; $a = \frac{\pi}{2}$, $b = 0$...), et contrôler les
  signes des variantes.
}
