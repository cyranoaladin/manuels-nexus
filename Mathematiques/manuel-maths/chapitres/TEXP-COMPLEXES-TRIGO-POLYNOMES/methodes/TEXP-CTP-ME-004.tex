\begin{fichemethode}{M4}{Résoudre une équation de degré 2 ou 3 (racine connue) et factoriser}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande de résoudre dans $\mathbb{C}$
  une équation du second degré a coefficients réels, ou une équation de degré 3
  dont une racine est connue (donnee ou evidente), avec factorisation.
}

\pasApas{%
  \begin{enumerate}
    \item Degré 2 : calculer $\Delta = b^2 - 4ac$. Si $\Delta < 0$, les
      racines sont complexes conjuguees :
      $z = \dfrac{-b \pm i\sqrt{-\Delta}}{2a}$.
    \item Degré 3 avec racine connue $a$ : VERIFIER $P(a) = 0$ par le calcul.
    \item Factoriser $P(z) = (z - a)(\alpha z^2 + \beta z + \gamma)$ : trouver
      $\alpha, \beta, \gamma$ par identification des coefficients (developper
      et egaler degré par degré).
    \item Résoudre le facteur du second degré (étape 1).
    \item Conclure : l'ensemble des solutions reunit la racine connue et
      celles du trinôme ; l'écrire completement.
  \end{enumerate}
}

\exempleRedige{%
  Résoudre $z^3 - 3z^2 + 7z - 5 = 0$ sachant que 1 est racine.

  \medskip
  \commentaireMarge{Contrôle prealable de la racine donnee.}%
  $P(1) = 1 - 3 + 7 - 5 = 0$ : 1 est bien racine.

  \medskip
  \commentaireMarge{Identification des coefficients.}%
  On cherche $P(z) = (z - 1)(z^2 + \beta z + \gamma)$. Le developpement donne
  $z^3 + (\beta - 1)z^2 + (\gamma - \beta)z - \gamma$ ; par identification :
  $\beta - 1 = -3$, $-\gamma = -5$, d'ou $\beta = -2$, $\gamma = 5$
  (et $\gamma - \beta = 7$ est coherent).

  \medskip
  \commentaireMarge{$\Delta < 0$ : racines conjuguees.}%
  $z^2 - 2z + 5 = 0$ : $\Delta = 4 - 20 = -16$, donc
  $z = \dfrac{2 \pm 4i}{2} = 1 \pm 2i$.
  $\mathcal{S} = \{1\,;\,1 + 2i\,;\,1 - 2i\}$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Écrire $\sqrt{\Delta}$ pour $\Delta < 0$ : la bonne
      ecriture est $i\sqrt{-\Delta}$ ($\sqrt{16} = 4$, facteur $i$ a part).
    \item \textbf{Piège 2.} Erreur d'identification : vérifier TOUS les
      coefficients, y compris celui qui sert de contrôle.
    \item \textbf{Piège 3.} Oublier la racine connue dans l'ensemble final des
      solutions.
  \end{itemize}
}

\verifier{%
  Redevelopper la factorisation pour retrouver $P$, substituer chaque racine
  dans l'équation initiale, et contrôler que les racines non réelles sont
  conjuguees (coefficients réels).
}

\sEntrainer{\refExos{M4}}

\end{fichemethode}
