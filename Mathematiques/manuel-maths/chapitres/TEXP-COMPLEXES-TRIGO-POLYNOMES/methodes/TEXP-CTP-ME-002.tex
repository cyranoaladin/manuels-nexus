\begin{fichemethode}{M2}{Utiliser Euler et Moivre : linearisation et puissances}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande de LINEARISER une puissance
  de $\cos$ ou $\sin$ (formules d'Euler), d'exprimer $\cos(nt)$ ou $\sin(nt)$
  (formule de Moivre), ou de calculer une puissance $z^n$ via la forme
  exponentielle.
}

\pasApas{%
  \begin{enumerate}
    \item Formules d'Euler :
      $\cos t = \dfrac{e^{it} + e^{-it}}{2}$,
      $\sin t = \dfrac{e^{it} - e^{-it}}{2i}$.
    \item Linearisation : substituer, developper par le binome, puis regrouper
      les termes conjugues $e^{ikt} + e^{-ikt} = 2\cos(kt)$.
    \item Formule de Moivre :
      $(\cos t + i\sin t)^n = \cos(nt) + i\sin(nt)$ — developper le membre de
      gauche par le binome et identifier parties réelle et imaginaire pour
      obtenir $\cos(nt)$, $\sin(nt)$.
    \item Puissance de $z$ : mettre $z$ sous forme exponentielle (fiche M1)
      puis $z^n = r^n e^{in\theta}$.
    \item Reduire l'argument modulo $2\pi$ et revenir a la forme demandee.
  \end{enumerate}
}

\exempleRedige{%
  1) Lineariser $\cos^3 t$. \quad 2) Calculer $(1 + i)^8$.

  \medskip
  \commentaireMarge{Binome sur $\left(\frac{e^{it}+e^{-it}}{2}\right)^3$.}%
  1) $\cos^3 t = \dfrac{1}{8}\left(e^{3it} + 3e^{it} + 3e^{-it} +
  e^{-3it}\right) = \dfrac{1}{8}\left(2\cos(3t) + 6\cos t\right)
  = \dfrac{\cos(3t) + 3\cos t}{4}$.

  \medskip
  \commentaireMarge{Forme exponentielle puis puissance.}%
  2) $1 + i = \sqrt{2}\,e^{i\pi/4}$, donc
  $(1+i)^8 = \left(\sqrt{2}\right)^8 e^{2i\pi} = 16 \times 1 = 16$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Oublier le facteur $\frac{1}{2i}$ (et non
      $\frac{1}{2}$) dans la formule d'Euler du sinus.
    \item \textbf{Piège 2.} Croire que $(\cos t + i \sin t)^n$ se developpe en
      $\cos^n t + i\sin^n t$ : il faut le binome complet.
    \item \textbf{Piège 3.} Dans $z^n = r^n e^{in\theta}$, oublier d'elever le
      MODULE a la puissance $n$.
  \end{itemize}
}

\verifier{%
  Tester la formule linearisee pour une valeur simple ($t = 0$ :
  $\cos^3 0 = 1$ et $\frac{1 + 3}{4} = 1$), et contrôler une puissance par le
  module ($|1+i|^8 = 2^4 = 16$).
}

\sEntrainer{\refExos{M2}}

\end{fichemethode}
