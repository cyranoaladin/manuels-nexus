\begin{fichemethode}{M1}{Passer de la forme algebrique aux formes trigonometrique et exponentielle}

\quandUtiliser{%
  Utiliser cette methode lorsque l'enonce demande de convertir un complexe
  entre ses trois ecritures : algebrique $a + ib$, trigonometrique
  $r(\cos\theta + i\sin\theta)$, exponentielle $r\,e^{i\theta}$.
}

\pasApas{%
  \begin{enumerate}
    \item Algebrique $\to$ trigonometrique : calculer $r = |z| =
      \sqrt{a^2 + b^2}$, puis chercher $\theta$ avec
      $\cos\theta = \frac{a}{r}$ ET $\sin\theta = \frac{b}{r}$.
    \item Reconnaitre un angle usuel a partir du COUPLE de valeurs (sinon,
      valeur approchee).
    \item Ecrire $z = r(\cos\theta + i\sin\theta) = r\,e^{i\theta}$ — la forme
      exponentielle n'est qu'une notation de la forme trigonometrique.
    \item Exponentielle $\to$ algebrique : developper
      $r\,e^{i\theta} = r\cos\theta + i\,r\sin\theta$ avec les valeurs
      exactes.
    \item Controler : $r > 0$ obligatoirement, et le point doit tomber dans le
      bon quadrant.
  \end{enumerate}
}

\exempleRedige{%
  Mettre $z = -1 + i$ sous formes trigonometrique et exponentielle.

  \medskip
  \commentaireMarge{Module d'abord.}%
  $r = \sqrt{(-1)^2 + 1^2} = \sqrt{2}$.

  \medskip
  \commentaireMarge{Deux conditions pour $\theta$ : cosinus ET sinus.}%
  On cherche $\theta$ tel que $\cos\theta = -\dfrac{1}{\sqrt{2}}$ et
  $\sin\theta = \dfrac{1}{\sqrt{2}}$ : $\theta = \dfrac{3\pi}{4}$ (deuxieme
  quadrant).

  \medskip
  Donc $z = \sqrt{2}\left(\cos\dfrac{3\pi}{4} + i\sin\dfrac{3\pi}{4}\right)
  = \sqrt{2}\,e^{3i\pi/4}$.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piege 1.} Ecrire une « forme trigonometrique » avec $r < 0$
      ou avec un signe MOINS entre les deux termes : le format est
      $r(\cos\theta + i\sin\theta)$, $r > 0$.
    \item \textbf{Piege 2.} Deduire $\theta$ du seul cosinus : $\frac{3\pi}{4}$
      et $-\frac{3\pi}{4}$ ont le meme cosinus — le sinus tranche.
    \item \textbf{Piege 3.} Confondre $e^{i\theta}$ (module 1) et
      $r\,e^{i\theta}$ : ne pas oublier le module en revenant a la forme
      algebrique.
  \end{itemize}
}

\verifier{%
  Redevelopper la forme trouvee pour retrouver $a + ib$ exactement, et
  controler le quadrant du point sur une figure rapide.
}

\sEntrainer{\refExos{M1}}

\end{fichemethode}
