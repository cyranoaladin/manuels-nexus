\begin{fichemethode}{M7}{Determiner les racines n-iemes de l'unite et etudier les polygones reguliers}

\quandUtiliser{%
  Utiliser cette methode lorsque l'enonce demande de resoudre $z^n = 1$, de
  decrire l'ensemble $\mathbb{U}_n$ des racines $n$-iemes de l'unite, ou
  d'etudier un polygone regulier a l'aide de leurs affixes.
}

\pasApas{%
  \begin{enumerate}
    \item Chercher $z = r\,e^{i\theta}$ : $z^n = 1$ donne $r^n = 1$ (donc
      $r = 1$, module positif) et $n\theta \equiv 0\ [2\pi]$.
    \item Resoudre l'equation d'angles :
      $\theta = \dfrac{2k\pi}{n}$, $k \in \mathbb{Z}$.
    \item Compter : $k = 0, 1, \dots, n-1$ donnent $n$ valeurs DISTINCTES
      (au-dela, on retombe sur les memes) :
      $\mathbb{U}_n = \left\{e^{2ik\pi/n},\ k = 0, \dots, n-1\right\}$.
    \item Geometrie : ces $n$ points, tous de module 1 et d'arguments
      regulierement espaces de $\frac{2\pi}{n}$, sont les sommets d'un
      polygone regulier a $n$ cotes inscrit dans le cercle unite (un sommet
      en 1).
    \item Exploiter les proprietes : somme des racines nulle (pour
      $n \geq 2$), stabilite par produit, sommets consecutifs relies par la
      rotation d'angle $\frac{2\pi}{n}$.
  \end{enumerate}
}

\exempleRedige{%
  Resoudre $z^5 = 1$ et interpreter geometriquement.

  \medskip
  \commentaireMarge{Module puis argument.}%
  Avec $z = r\,e^{i\theta}$ : $r^5 = 1$ donne $r = 1$, et
  $5\theta \equiv 0\ [2\pi]$ donne $\theta = \dfrac{2k\pi}{5}$.

  \medskip
  Les solutions sont $z_k = e^{2ik\pi/5}$ pour $k = 0, 1, 2, 3, 4$ : cinq
  points du cercle unite, d'arguments espaces de $\dfrac{2\pi}{5}$ — les
  sommets d'un pentagone regulier dont un sommet est le point d'affixe 1.

  \medskip
  De plus $z_0 + z_1 + z_2 + z_3 + z_4 = 0$ : le centre de gravite du
  pentagone est l'origine.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piege 1.} Oublier des solutions (ne donner que $z = 1$) ou en
      compter trop : il y en a EXACTEMENT $n$ distinctes.
    \item \textbf{Piege 2.} Ecrire $\theta = \frac{2\pi}{n}$ seulement : c'est
      $\frac{2k\pi}{n}$ pour $k = 0, \dots, n-1$.
    \item \textbf{Piege 3.} Croire que la somme des racines vaut 1 : elle est
      NULLE pour $n \geq 2$ (serie geometrique de raison $e^{2i\pi/n} \neq 1$).
  \end{itemize}
}

\verifier{%
  Elever chaque solution a la puissance $n$ (on doit trouver 1), controler
  qu'elles sont toutes distinctes et de module 1, et verifier la somme nulle.
}

\end{fichemethode}
