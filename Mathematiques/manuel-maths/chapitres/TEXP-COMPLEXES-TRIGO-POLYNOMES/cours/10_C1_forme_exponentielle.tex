\section{Forme exponentielle, formules d'Euler et de Moivre}

% ============================================================
% STRATE 1 — Forme trigonometrique et exponentielle
% ============================================================

\definition[1]{
Pour $z
eq0$ de module $r=|z|$ et d'argument $\theta$, la \textbf{forme trigonométrique} de $z$ est $z=r(\cos\theta+i\sin\theta)$. On note $\mathrm{e}^{i\theta}=\cos\theta+i\sin\theta$ (\textbf{exponentielle imaginaire}) : la \textbf{forme exponentielle} de $z$ est $z=r\,\mathrm{e}^{i\theta}$.
}

\propriete[Relation fonctionnelle]{
Pour tous réels $\theta,\varphi$ : $\mathrm{e}^{i\theta}\times\mathrm{e}^{i\varphi} = \mathrm{e}^{i(\theta+\varphi)}$.
}

\exemple{
Passons $z=1-i$ de la forme algébrique à la forme exponentielle : $|z|=\sqrt{1+1}=\sqrt2$, $\cos\theta=\dfrac{1}{\sqrt2}$, $\sin\theta=-\dfrac{1}{\sqrt2}$, donc $\theta=-\dfrac{\pi}{4}$. Ainsi $z=\sqrt2\,\mathrm{e}^{-i\pi/4}$.
}

% ============================================================
% STRATE 2 — Formules d'Euler et de Moivre
% ============================================================

\theoreme[Formules d'Euler]{
Pour tout réel $\theta$ :
\[
  \cos\theta = \frac{\mathrm{e}^{i\theta}+\mathrm{e}^{-i\theta}}{2}, \qquad \sin\theta = \frac{\mathrm{e}^{i\theta}-\mathrm{e}^{-i\theta}}{2i}.
\]
}

\margeAppui{
Ces formules s'obtiennent directement en additionnant (ou soustrayant) $\mathrm{e}^{i\theta}=\cos\theta+i\sin\theta$ et $\mathrm{e}^{-i\theta}=\cos\theta-i\sin\theta$.
}

\theoreme[Formule de Moivre]{
Pour tout réel $\theta$ et tout $n\in\mathbb{N}$ : $\cos(n\theta)+i\sin(n\theta) = (\cos\theta+i\sin\theta)^n$.
}

\exemple{
Utilisons Euler pour linéariser $\cos^2\theta$ :
\[
  \cos^2\theta = \left(\frac{\mathrm{e}^{i\theta}+\mathrm{e}^{-i\theta}}{2}\right)^2 = \frac{\mathrm{e}^{2i\theta}+2+\mathrm{e}^{-2i\theta}}{4} = \frac{1}{2} + \frac{\mathrm{e}^{2i\theta}+\mathrm{e}^{-2i\theta}}{4} = \frac{1}{2}+\frac{\cos(2\theta)}{2}.
\]
}

\erreurFrequente{
\textbf{Confondre la formule de Moivre et les formules d'Euler.} La formule de \textbf{Moivre} relie une puissance de $(\cos\theta+i\sin\theta)$ à $\cos(n\theta)+i\sin(n\theta)$ (utile pour calculer une puissance) ; les formules d'\textbf{Euler} expriment $\cos\theta$ et $\sin\theta$ à partir de $\mathrm{e}^{i\theta}$ (utiles pour linéariser une expression trigonométrique).
}
