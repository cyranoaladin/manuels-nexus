\begin{exercice}{TCOMPL-CORR-EX-001}{1}{15}
On mesure, pour $4$ semaines, le budget publicitaire $x_i$ (en milliers d'euros) et le nombre $y_i$ de nouveaux clients :
\[
  (0,5),\ (1,8),\ (2,10),\ (3,15).
\]
\begin{enumerate}
  \item Calculer les coordonnées du point moyen $G$.
  \item À l'aide d'un logiciel ou d'une calculatrice, déterminer l'équation de la droite des moindres carrés $y=ax+b$.
\end{enumerate}
\end{exercice}
