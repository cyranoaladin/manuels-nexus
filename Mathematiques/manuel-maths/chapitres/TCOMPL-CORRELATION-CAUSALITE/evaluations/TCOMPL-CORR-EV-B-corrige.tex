\begin{corrige}{TCOMPL-CORR-EV-B-EX1}
\textbf{1.} On pose $Y=\ln(y)$ : $Y = \ln(A) + bt$, une relation affine en $t$.

\textbf{2.} $Y_0=\ln5\approx1{,}609$, $Y_1=\ln10\approx2{,}303$, $Y_2=\ln20\approx2{,}996$, $Y_3=\ln40\approx3{,}689$.

\textbf{3.} Les écarts successifs entre les $Y_i$ valent tous $\approx0{,}693=\ln2$ : le nuage est parfaitement aligné, de pente $b=\ln2\approx0{,}693$ et d'ordonnée à l'origine $\ln(A)=\ln5$, donc $A=5$. On retrouve $y=5\,\mathrm{e}^{t\ln2}=5\times2^t$.
\end{corrige}
