\begin{corrige}{TCOMPL-CORR-EX-003}
\textbf{1.} Si $y=Ax^b$, alors $\ln(y) = \ln(A) + b\ln(x)$ (équation fonctionnelle du logarithme). En posant $X=\ln(x)$ et $Y=\ln(y)$, on obtient $Y = bX + \ln(A)$ : une relation \textbf{affine} entre $X$ et $Y$.

\textbf{2.} Les points $(X_i,Y_i) = (\ln x_i, \ln y_i)$ sont exactement alignés, de pente $b=2$ et d'ordonnée à l'origine $\ln(A)=\ln(3)$. On en déduit $A=3$ et $b=2$, soit $y=3x^2$ (on vérifie : $3\times1^2=3$, $3\times2^2=12$, $3\times3^2=27$, $3\times4^2=48$, conforme aux données).
\end{corrige}
