\begin{fichemethode}{M1}{Représenter un nuage de points et calculer le point moyen}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé fournit une serie statistique a deux
  variables $(x_i\,;\,y_i)$ et demande de représenter le nuage de points ou de
  calculer son point moyen $G$.
}

\pasApas{%
  \begin{enumerate}
    \item Choisir des echelles adaptees sur les deux axes (le nuage doit occuper
      l'essentiel de la feuille) ; les axes ne partent pas forcement de 0.
    \item Placer chaque couple $(x_i\,;\,y_i)$ par une croix, sans les relier.
    \item Calculer $\bar{x} = \dfrac{x_1 + \dots + x_n}{n}$ et
      $\bar{y} = \dfrac{y_1 + \dots + y_n}{n}$.
    \item Placer le point moyen $G(\bar{x}\,;\,\bar{y})$ sur le graphique.
    \item Commenter la forme du nuage : allonge (ajustement affine envisageable),
      courbe, ou sans structure.
  \end{enumerate}
}

\exempleRedige{%
  Serie : $(1\,;\,2)$, $(2\,;\,3)$, $(3\,;\,5)$, $(4\,;\,6)$. Calculer $G$.

  \medskip
  \commentaireMarge{Moyenne des abscisses puis des ordonnées.}%
  $\bar{x} = \dfrac{1+2+3+4}{4} = \dfrac{10}{4} = 2{,}5$ et
  $\bar{y} = \dfrac{2+3+5+6}{4} = \dfrac{16}{4} = 4$.

  \medskip
  Le point moyen est $G(2{,}5\,;\,4)$ — il se place au « centre de gravite » du
  nuage, sans être necessairement un point de la serie.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Relier les points du nuage : un nuage statistique
      n'est pas une courbe de fonction.
    \item \textbf{Piège 2.} Croire que $G$ est toujours un point observe de la
      serie : c'est faux en general.
    \item \textbf{Piège 3.} Melanger les roles des variables (echanger $x$ et $y$
      change le nuage et le point moyen).
  \end{itemize}
}

\verifier{%
  Contrôler que $\bar{x}$ est entre le minimum et le maximum des $x_i$ (idem pour
  $\bar{y}$), et que $G$ tombe visuellement au centre du nuage.
}

\sEntrainer{\refExos{M1}}

\end{fichemethode}
