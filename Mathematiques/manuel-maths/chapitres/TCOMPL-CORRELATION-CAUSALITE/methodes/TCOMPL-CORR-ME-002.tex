\begin{fichemethode}{M2}{Déterminer la droite des moindres carres et le coefficient de correlation}

\quandUtiliser{%
  Utiliser cette méthode lorsque l'énoncé demande l'équation $y = ax + b$ de la
  droite d'ajustement par la méthode des moindres carres, ou le coefficient de
  correlation $r$, generalement a la calculatrice.
}

\pasApas{%
  \begin{enumerate}
    \item Entrer les listes $x_i$ et $y_i$ dans la calculatrice (mode statistique
      a deux variables) et lancer la regression linéaire.
    \item Lire la pente $a$ et l'ordonnée a l'origine $b$ ; écrire $y = ax + b$
      en arrondissant comme demande.
    \item Lire le coefficient de correlation $r$ (ou $r^2$ selon le modèle).
    \item Contrôle cle : la droite des moindres carres passe TOUJOURS par le
      point moyen $G(\bar{x}\,;\,\bar{y})$ — vérifier $a\bar{x} + b = \bar{y}$.
    \item Interpréter $r$ : proche de $\pm 1$, l'ajustement affine est pertinent ;
      proche de 0, il ne l'est pas. Le signe de $r$ est celui de la pente.
  \end{enumerate}
}

\exempleRedige{%
  Serie : $(1\,;\,2)$, $(2\,;\,3)$, $(3\,;\,5)$, $(4\,;\,6)$.

  \medskip
  \commentaireMarge{Regression linéaire a la calculatrice.}%
  La calculatrice donne $a = 1{,}4$ et $b = 0{,}5$ : la droite des moindres
  carres a pour équation $y = 1{,}4x + 0{,}5$.

  \medskip
  \commentaireMarge{Contrôle : la droite passe par $G$.}%
  Avec $G(2{,}5\,;\,4)$ : $1{,}4 \times 2{,}5 + 0{,}5 = 3{,}5 + 0{,}5 = 4 = \bar{y}$.
  Le coefficient de correlation vaut $r \approx 0{,}99$ : très proche de 1,
  l'ajustement affine est excellent.
}

\pieges{%
  \begin{itemize}
    \item \textbf{Piège 1.} Echanger les listes $x$ et $y$ : la regression de $y$
      en $x$ n'est pas celle de $x$ en $y$.
    \item \textbf{Piège 2.} Confondre $r$ et $r^2$ affiches par la calculatrice.
    \item \textbf{Piège 3.} Conclure d'un $r$ proche de 1 a une CAUSALITE :
      $r$ mesure seulement la qualite de l'ajustement affine (voir fiche M5).
  \end{itemize}
}

\verifier{%
  Vérifier que $a\bar{x} + b = \bar{y}$ (passage par $G$), que le signe de $a$
  correspond a la tendance visible du nuage, et que $-1 \leq r \leq 1$.
}

\end{fichemethode}
