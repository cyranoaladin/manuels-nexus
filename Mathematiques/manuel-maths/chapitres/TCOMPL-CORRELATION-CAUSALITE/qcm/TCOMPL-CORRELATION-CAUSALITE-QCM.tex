\begin{enumerate}

\item[] \textbf{Capacité C2}

\item \textbf{[Q1]} La droite des moindres carres passe toujours par :
  \begin{enumerate}[label=\Alph*.]
    \item l'origine du repere.
    \item le point moyen G.
    \item le premier point du nuage.
    \item aucun point particulier.
  \end{enumerate}

\item \textbf{[Q2]} Un coefficient de correlation r=0,98 indique :
  \begin{enumerate}[label=\Alph*.]
    \item une absence totale de lien.
    \item un ajustement affine de bonne qualite.
    \item une relation de causalite certaine.
    \item une erreur de calcul.
  \end{enumerate}

\item[] \textbf{Capacité C3}

\item \textbf{[Q3]} Pour lineariser une relation $y=Ax^b$ avec $A>0$ et $x>0$, on pose :
  \begin{enumerate}[label=\Alph*.]
    \item X = x\textasciicircum{}2, Y = y\textasciicircum{}2
    \item X = ln(x), Y = ln(y)
    \item X = 1/x, Y = 1/y
    \item Aucun changement n'est possible.
  \end{enumerate}

\item[] \textbf{Capacité C4}

\item \textbf{[Q4]} Estimer une valeur en dehors de la plage des donnees observees, c'est :
  \begin{enumerate}[label=\Alph*.]
    \item interpoler.
    \item extrapoler.
    \item calculer le point moyen.
    \item impossible.
  \end{enumerate}

\item[] \textbf{Capacité C5}

\item \textbf{[Q5]} Une forte correlation entre deux variables X et Y :
  \begin{enumerate}[label=\Alph*.]
    \item prouve toujours que X cause Y.
    \item ne prouve pas, seule, une relation de cause a effet.
    \item prouve que Y cause X.
    \item est toujours due au hasard.
  \end{enumerate}

\item \textbf{[Q6]} Une variable cachee Z qui cause a la fois X et Y peut expliquer :
  \begin{enumerate}[label=\Alph*.]
    \item une correlation entre X et Y sans lien de causalite direct entre elles.
    \item que X cause necessairement Y.
    \item que la correlation entre X et Y est fausse.
    \item rien du tout.
  \end{enumerate}

\item[] \textbf{Capacité C1}

\item \textbf{[Q7]} Pour les couples (1 ; 2), (3 ; 4) et (5 ; 0), quelle proposition représente correctement le nuage et son point moyen ?
  \begin{enumerate}[label=\Alph*.]
    \item On place A(1 ; 2), B(3 ; 4), C(5 ; 0) ; le point moyen est $G(3 ; 2)$.
    \item On place A(1 ; 2), B(3 ; 4), C(5 ; 0) ; le point moyen est $G(9 ; 6)$.
    \item On place A(2 ; 1), B(4 ; 3), C(0 ; 5) ; le point moyen est $G(2 ; 3)$.
    \item On place A(1 ; 2), B(3 ; 0), C(5 ; 4) ; le point moyen est $G(3 ; 2)$.
  \end{enumerate}

\end{enumerate}

\subsection*{Diagnostic des réponses erronées}

\begin{itemize}
  \item \textbf{Q1}
  \begin{itemize}
    \item \textbf{A} — La droite ne passe par l'origine que si l'ordonnée moyenne est compatible avec une ordonnée à l'origine nulle. Les équations normales des moindres carrés imposent à la droite de régression de passer par le point moyen G. \emph{Renvoi : C2.}
    \item \textbf{C} — Le premier point n'a aucun rôle privilégié dans la minimisation de la somme des carrés des résidus. Les équations normales des moindres carrés imposent à la droite de régression de passer par le point moyen G. \emph{Renvoi : C2.}
    \item \textbf{D} — Les équations normales imposent au contraire le passage par un point déterminé du nuage, le point moyen. Les équations normales des moindres carrés imposent à la droite de régression de passer par le point moyen G. \emph{Renvoi : C2.}
  \end{itemize}
  \item \textbf{Q2}
  \begin{itemize}
    \item \textbf{A} — Une valeur absolue proche de 1 indique une forte liaison linéaire, pas une absence de lien. La valeur r=0,98, proche de 1, indique une forte liaison linéaire et donc un ajustement affine de bonne qualité. \emph{Renvoi : C2.}
    \item \textbf{C} — Le coefficient mesure une association linéaire et ne permet pas d'identifier un mécanisme causal certain. La valeur r=0,98, proche de 1, indique une forte liaison linéaire et donc un ajustement affine de bonne qualité. \emph{Renvoi : C2.}
    \item \textbf{D} — La valeur 0,98 appartient bien à [-1,1] et n'indique donc aucune erreur de calcul. La valeur r=0,98, proche de 1, indique une forte liaison linéaire et donc un ajustement affine de bonne qualité. \emph{Renvoi : C2.}
  \end{itemize}
  \item \textbf{Q3}
  \begin{itemize}
    \item \textbf{A} — Élever x et y au carré transforme la loi de puissance sans la rendre affine en général. Prendre les logarithmes donne ln(y)=ln(A)+b ln(x), relation affine entre X=ln(x) et Y=ln(y). \emph{Renvoi : C3.}
    \item \textbf{C} — Prendre les inverses conserve une puissance non linéaire de X et ne sépare pas l'exposant b. Prendre les logarithmes donne ln(y)=ln(A)+b ln(x), relation affine entre X=ln(x) et Y=ln(y). \emph{Renvoi : C3.}
    \item \textbf{D} — Le logarithme transforme effectivement le produit et la puissance en somme et coefficient, donc un changement est possible. Prendre les logarithmes donne ln(y)=ln(A)+b ln(x), relation affine entre X=ln(x) et Y=ln(y). \emph{Renvoi : C3.}
  \end{itemize}
  \item \textbf{Q4}
  \begin{itemize}
    \item \textbf{A} — L'interpolation concerne une valeur située à l'intérieur de la plage des observations, contrairement à l'énoncé. Une estimation hors de l'intervalle couvert par les données observées est une extrapolation. \emph{Renvoi : C4.}
    \item \textbf{C} — Calculer le point moyen résume le nuage mais ne produit pas l'estimation extérieure demandée. Une estimation hors de l'intervalle couvert par les données observées est une extrapolation. \emph{Renvoi : C4.}
    \item \textbf{D} — Une extrapolation est calculable, même si sa fiabilité doit être discutée et décroît hors de la plage observée. Une estimation hors de l'intervalle couvert par les données observées est une extrapolation. \emph{Renvoi : C4.}
  \end{itemize}
  \item \textbf{Q5}
  \begin{itemize}
    \item \textbf{A} — Une variable cachée ou une causalité inverse peut produire la corrélation sans que X cause Y. Une corrélation, même forte, n'établit pas à elle seule un mécanisme causal entre les deux variables. \emph{Renvoi : C5.}
    \item \textbf{C} — Le même coefficient est compatible avec X causant Y, Y causant X ou une cause commune ; il ne tranche pas. Une corrélation, même forte, n'établit pas à elle seule un mécanisme causal entre les deux variables. \emph{Renvoi : C5.}
    \item \textbf{D} — Une corrélation peut refléter une relation réelle sans être causale et n'est donc pas toujours due au hasard. Une corrélation, même forte, n'établit pas à elle seule un mécanisme causal entre les deux variables. \emph{Renvoi : C5.}
  \end{itemize}
  \item \textbf{Q6}
  \begin{itemize}
    \item \textbf{B} — Une cause commune explique la variation conjointe sans rendre nécessaire une action directe de X sur Y. Une cause commune Z peut faire varier X et Y ensemble et créer une corrélation sans causalité directe entre X et Y. \emph{Renvoi : C5.}
    \item \textbf{C} — La corrélation observée peut être statistiquement réelle même si son interprétation causale directe est fausse. Une cause commune Z peut faire varier X et Y ensemble et créer une corrélation sans causalité directe entre X et Y. \emph{Renvoi : C5.}
    \item \textbf{D} — La variable cachée fournit précisément un mécanisme explicatif à l'association mesurée entre X et Y. Une cause commune Z peut faire varier X et Y ensemble et créer une corrélation sans causalité directe entre X et Y. \emph{Renvoi : C5.}
  \end{itemize}
  \item \textbf{Q7}
  \begin{itemize}
    \item \textbf{B} — Les coordonnées du point moyen sont les moyennes, pas les sommes : $x_G=(1+3+5)/3=3$ et $y_G=(2+4+0)/3=2$. \emph{Renvoi : C1.}
    \item \textbf{C} — Les coordonnées de chaque point et les deux moyennes ont été interverties : le nuage correct utilise (1 ; 2), (3 ; 4), (5 ; 0), puis $G(3 ; 2)$. \emph{Renvoi : C1.}
    \item \textbf{D} — Le point moyen est correct, mais le nuage ne respecte pas les couples donnés : B doit avoir pour ordonnée 4 et C pour ordonnée 0. \emph{Renvoi : C1.}
  \end{itemize}
\end{itemize}
