\subsection*{C1/C2 --- Ajustement affine}

\begin{enumerate}

\item \textbf{[Q1]} La droite des moindres carrés passe toujours par :
  \begin{enumerate}[label=\Alph*.]
    \item l'origine du repère.
    \item le point moyen $G$.
    \item le premier point du nuage.
    \item aucun point particulier.
  \end{enumerate}

\bigskip

\item \textbf{[Q2]} Un coefficient de corrélation $r=0{,}98$ indique :
  \begin{enumerate}[label=\Alph*.]
    \item une absence totale de lien.
    \item un ajustement affine de bonne qualité.
    \item une relation de causalité certaine.
    \item une erreur de calcul.
  \end{enumerate}

\subsection*{C3 --- Changement de variable}

\item \textbf{[Q3]} Pour linéariser une relation $y=Ax^b$, on pose :
  \begin{enumerate}[label=\Alph*.]
    \item $X=x^2$, $Y=y^2$.
    \item $X=\ln(x)$, $Y=\ln(y)$.
    \item $X=1/x$, $Y=1/y$.
    \item aucun changement n'est possible.
  \end{enumerate}

\subsection*{C5 --- Corrélation et causalité}

\item \textbf{[Q5]} Une forte corrélation entre deux variables $X$ et $Y$ :
  \begin{enumerate}[label=\Alph*.]
    \item prouve toujours que $X$ cause $Y$.
    \item ne prouve pas, à elle seule, une relation de cause à effet.
    \item prouve que $Y$ cause $X$.
    \item est toujours due au hasard.
  \end{enumerate}

\bigskip

\item \textbf{[Q6]} Une variable cachée $Z$ qui cause à la fois $X$ et $Y$ peut expliquer :
  \begin{enumerate}[label=\Alph*.]
    \item une corrélation entre $X$ et $Y$ sans lien de causalité direct entre elles.
    \item que $X$ cause nécessairement $Y$.
    \item que la corrélation entre $X$ et $Y$ est fausse.
    \item rien du tout.
  \end{enumerate}

\end{enumerate}
