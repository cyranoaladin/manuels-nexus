\chapter*{Plan du manuel et navigation}
\addcontentsline{toc}{chapter}{Plan du manuel}

\begin{center}
\begin{minipage}{0.88\textwidth}
\itshape Ce manuel peut se lire de façon linéaire, mais il offre aussi des
repères utiles pour retrouver rapidement un chapitre, une méthode ou un type
d'exercice.
\end{minipage}
\end{center}

\bigskip

\section*{Parcours de lecture}

\begin{itemize}
  \item \textbf{Comprendre} : ouvrir le chapitre, lire la situation d'ouverture,
puis le cours.
  \item \textbf{Apprendre} : travailler les méthodes et les exemples résolus.
  \item \textbf{S'entraîner} : résoudre les exercices du parcours guidé, puis
les exercices de niveau baccalauréat.
  \item \textbf{Vérifier} : utiliser l'auto-évaluation et les évaluations A/B.
  \item \textbf{Corriger} : revenir aux méthodes et aux ressources de remédiation
si nécessaire.
\end{itemize}

\section*{Les chapitres du manuel}

\begin{description}
  \item[1. Suites] Étude des suites arithmétiques et géométriques, des
  raisonnements récurrents et des raisonnements de convergence.
  \item[2. Second degré] Résolution d'équations et d'inéquations, étude du
  signe et mise en forme canonique.
  \item[3. Dérivation locale] Nombre dérivé, tangente et variation locale.
  \item[4. Dérivation globale] Étude des fonctions, extremum, convexité et
  comportement global.
  \item[5. Exponentielle] Propriétés de la fonction exponentielle et de ses
  applications.
  \item[6. Trigonométrie] Cercle trigonométrique, valeurs remarquables et
  fonctions trigonométriques.
  \item[7. Produit scalaire] Géométrie vectorielle, orthogonalité et
  calculs de longueurs.
  \item[8. Géométrie repérée] Coordonnées, droites, parallélisme et
  orthogonalité dans le plan.
  \item[9. Probabilités conditionnelles] Probabilités, conditionnement,
  indépendance et tableaux.
  \item[10. Variables aléatoires] Lois, espérance, variance et applications.
\end{description}

\section*{Repères utiles}

\begin{itemize}
  \item Le \textbf{formulaire} récapitule les résultats essentiels à connaître.
  \item Les \textbf{méthodes} proposent des démarches pas à pas.
  \item Les \textbf{coups de pouce} accompagnent les exercices guidés.
  \item Les \textbf{évaluations A/B} permettent de vérifier les acquis.
\end{itemize}
